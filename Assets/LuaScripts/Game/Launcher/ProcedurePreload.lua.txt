-- filename:  ProcedurePreload.lua
-- descrip:   游戏预加载流程（启动流程）
--            职责：游戏启动初始化、时间同步、账号初始化、Lua预加载、UI/地图预加载、流程跳转
--=====================================================================================================

---@class ProcedurePreload : ProcedureSuper @游戏预加载流程
local ProcedurePreload = class('ProcedurePreload', import('ProcedureSuper'))

---构造函数
---@type fun(args:table):void
---@param args table @自定义参数
function ProcedurePreload:ctor(args)
    ProcedurePreload.super.ctor(self, args)

    self._preLoadLuaFiles = nil                                         ---@type table                          @预加载的Lua文件列表
    self._loadCountPerFrameOfLow = 3                                    ---@type number                         @低端机每帧加载数量
    self._loadCountPerFrameOfMedium = 5                                 ---@type number                         @中端机每帧加载数量
    self._loadCountPerFrameOfHigh = 10                                  ---@type number                         @高端机每帧加载数量

    --- 初始化预加载Lua文件
    self:InitPreLoadLuaFiles()
    --- 初始化启动时间数据（必须最先执行）
    self:InitLaunchTimeData()
end

---创建流程实例
---@type fun(args:table):ProcedurePreload
---@param args table @自定义参数
---@return ProcedurePreload @流程实例
function ProcedurePreload:Create(args)
    local obj = ProcedurePreload.new(args)
    return obj
end

---初始化需要预加载的核心Lua文件
---@type fun():void
function ProcedurePreload:InitPreLoadLuaFiles()
    self._preLoadLuaFiles = {
        "LuaBehaviourSuper",
        "ViewSuper",
        "GameMapBase",
        "UIMapBase",
    }
end

---初始化游戏启动时间数据（上次启动/当前启动）
---@type fun():void
function ProcedurePreload:InitLaunchTimeData()
    -- 首次启动时，同步时间到存档
    if self.cs:IsLaunch() == true then
        PbRootData.UserData.launchTime.last.utc0 = PbRootData.UserData.launchTime.current.utc0
        PbRootData.UserData.launchTime.last.utc0String = PbRootData.UserData.launchTime.current.utc0String
        PbRootData.UserData.launchTime.last.locality = PbRootData.UserData.launchTime.current.locality
        PbRootData.UserData.launchTime.last.localString = PbRootData.UserData.launchTime.current.localString

        -- 更新本次启动时间
        PbRootData.UserData.launchTime.current.utc0 = HonorTimerHelper:GetUTC0TimeStamp()
        PbRootData.UserData.launchTime.current.utc0String = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", HonorTimerHelper:GetUTC0TimeStamp())
        PbRootData.UserData.launchTime.current.locality = HonorTimerHelper:GetLocalTimeStamp()
        PbRootData.UserData.launchTime.current.localString = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", HonorTimerHelper:GetLocalTimeStamp())

        --- 新账号初始化时间
        if not PbRootData.GameData.isInit then
            PbRootData.UserData.launchTime.last.utc0 = PbRootData.UserData.launchTime.current.utc0
            PbRootData.UserData.launchTime.last.utc0String = PbRootData.UserData.launchTime.current.utc0String
            PbRootData.UserData.launchTime.last.locality = PbRootData.UserData.launchTime.current.locality
            PbRootData.UserData.launchTime.last.localString = PbRootData.UserData.launchTime.current.localString
        end

        -- 同步最近时间
        PbRootData.UserData.latestTime.utc0 = PbRootData.UserData.launchTime.last.utc0
        PbRootData.UserData.latestTime.utc0String = PbRootData.UserData.launchTime.last.utc0String
        PbRootData.UserData.latestTime.locality = PbRootData.UserData.launchTime.last.locality
        PbRootData.UserData.latestTime.localString = PbRootData.UserData.launchTime.last.localString

        -- 保存时间数据
        PbIO.UserData.Save_launchTime()
        PbIO.UserData.Save_latestTime()

        -- 初始化游戏时间
        GameTimeHelper:SetupFutureTimeStamp()
    end
end

---进入流程
---@type fun(ownerMachine: userdata):void
---@param ownerMachine userdata @流程状态机
function ProcedurePreload:OnEnter(ownerMachine)
    ProcedurePreload.super.OnEnter(self, ownerMachine)

    -- 预加载流程不清理资源，避免影响下一个流程
    self.cs.RemoveAllContentsOnProcedureTransition = false

    -- 先不开始加载，等待初始化完成
    self.cs.StartLoading = false

    --- 初始化游戏基础设置
    self:InitGameSetting()
    --- 初始化新账号数据
    self:InitNewAccountData()
    --- 初始化首次注册时间
    self:InitGameDataFirstLaunchTime()
    --- 执行分步加载
    self:LoadAddSteps()
end

---尝试显示隐私政策弹窗
---@type fun(finishCallback:function):void
---@param finishCallback function @显示完成回调
function ProcedurePreload:TryShowTermsPrivacy(finishCallback)
    if not PbRootData.GameData.playerInfo.isShowPrivacy then
        OpenUISync(UIs.UITermsPrivacy, { callback = finishCallback })
        PbRootData.GameData.playerInfo.isShowPrivacy = true
        PbIO.GameData.Save_playerInfo_isShowPrivacy()
    else
        if finishCallback then
            xpcall(finishCallback, __G__TRACK_BACK__)
        end
    end
end

    ProcedurePreload.super.OnUpdate(self, ownerMachine)
end

---离开流程
---@type fun(ownerMachine: userdata):void

function ProcedurePreload:OnLeave(ownerMachine)
    ProcedurePreload.super.OnLeave(self, ownerMachine)
end

---初始化游戏基础设置
---@type fun():void
function ProcedurePreload:InitGameSetting()
    GameLib.Util.SetMultiTouchEnabled(false) -- 关闭多点触摸
end

---初始化新账号数据（仅第一次进入游戏执行）
---@type fun():void
function ProcedurePreload:InitNewAccountData()
    if PbRootData.GameData.isInit == false then
        PrintInfo("初始化新账号数据")

        -- 初始化管理器
        ItemManager:Init()
        ChapterListManager:Init()

        -- 初始化道具
        ItemManager:InitNewAccountItemData()
        -- 解锁第一章
        ChapterListManager:UnlockChapter("1")

        -- 标记已初始化
        PbRootData.GameData.isInit = true
        PbIO.GameData.Save_isInit()

        -- 清理临时数据
        ItemManager:ClearData()
        ChapterListManager:ClearData()

        -- 清空事件
        HonorEventHelper:Cleanup()
    end
end

---执行分步加载流程
---@type fun():void
function ProcedurePreload:LoadAddSteps()
    local timeStart = Unity.Time.realtimeSinceStartup

    -- 初始化所有单例
    SingletonManager:InitAllSingleForStep(self)
    -- 预加载核心Lua
    self:PreLoadLuaFiles()

    -- 加载完成后跳转到游戏主流程
    self.cs:AddStep(function()
        self.cs:PrepareToNextProcedure(typeof(Honor.ProcedurePlaying), self.cs.ArgsFromChanging)
    end)

    self.cs.StartLoading = true
end

---预加载单个Lua文件
---@type fun(luaFileName:string):void
---@param luaFileName string @Lua文件名
function ProcedurePreload:PreLoadLuaFile(luaFileName)
    if not string.isNullOrEmpty(luaFileName) then
        require(luaFileName)
    end
end

---批量预加载Lua文件（按设备性能分帧加载）
---@type fun():void
function ProcedurePreload:PreLoadLuaFiles()
    if self._preLoadLuaFiles then
        BatchTraverseFunc(self._preLoadLuaFiles, self:LoadCountPreFrame(), function(batchData)
            if batchData and #batchData > 0 then
                local resultList = CopyTab(batchData)
                self.cs:AddStep(function()
                    for _, v in ipairs(resultList) do
                        self:PreLoadLuaFile(v)
                    end
                end)
            end
        end)
    end
end

---预加载UI配置文件
---@type fun():void
function ProcedurePreload:PreLoadUIConfig()
    if UIPreloadConfig then
        BatchTraverseFunc(UIPreloadConfig, self:LoadCountPreFrame(), function(batchData)
            if batchData and #batchData > 0 then
                local resultList = CopyTab(batchData)
                self.cs:AddStep(function()
                    for _, v in ipairs(resultList) do
                        self:PreLoadLuaFile(v)
                    end
                end)
            end
        end)
    end
end

---根据设备性能获取每帧加载数量
---@type fun():number
---@return number @每帧加载数量
function ProcedurePreload:LoadCountPreFrame()
    local deviceLevel = GameManager:GetDevicePerformanceLevel()
    PrintInfo("当前设备性能等级：" .. tostring(deviceLevel))

    if deviceLevel == Honor.DevicePerformanceLevel.High then
        return self._loadCountPerFrameOfHigh
    end
    if deviceLevel == Honor.DevicePerformanceLevel.Mid then
        return self._loadCountPerFrameOfMedium
    end
    if deviceLevel == Honor.DevicePerformanceLevel.Low then
        return self._loadCountPerFrameOfLow
    end

    return self._loadCountPerFrameOfMedium
end

---预加载地图资源（异步）
---@type fun(callback:function):void
---@param callback function @加载完成回调
function ProcedurePreload:PreLoadAssetAsyncOfMap(callback)
    local chapterId = ChapterListMgr:GetLatestChapterId()
    local hasDownGroupChapterId = IncreaserHotfixHelper:GetLastHasDownGroupChapterId(chapterId)
    local mapName = MapManager:GetLoadMapResName(hasDownGroupChapterId)

    if mapName then
        PrintInfo("预加载地图资源：" .. mapName)
        local mapRes = RES["GameObject_" .. mapName]
        GameMainRoot.Asset:PreLoadAssetAsync(mapRes.TypeName, mapRes.ABPath, mapName, function(assetObject, asset)
            PrintInfo("预加载地图资源完成：" .. mapName)
            if asset then
                GameMainRoot.Asset:UnloadAsset(asset)
            end
            if callback then
                xpcall(callback, __G__TRACK_BACK__)
            end
        end, true)
    end
end

---加载地图
---@type fun(callback:function):void
---@param callback function @加载完成回调
function ProcedurePreload:LoadMap(callback)
    local chapterId = ChapterListManager:GetLatestChapterId()
    local hasDownGroupChapterId = IncreaserHotfixHelper:GetLastHasDownGroupChapterId(chapterId)

    -- 设置运行时章节
    ChapterListManager:SetRuntimeChapter(hasDownGroupChapterId)

    -- 加载地图
    local isSuccess = MapManager:LoadMapById(hasDownGroupChapterId, nil)
    if callback then
        xpcall(callback, __G__TRACK_BACK__, isSuccess)
    end
end

---检查游戏是否异常退出（上次未正常结束关卡）
---@type fun():void
function ProcedurePreload:CheckIsUnusualExitLevel()
    local levelTag = GameCoreManager:GetPlayingLevelTag()
    if levelTag ~= nil and not string.isNullOrEmpty(levelTag) then
        local array = string.split(levelTag, "_")
        if #array ~= 2 then
            PrintError("退出关卡标签格式错误：" .. levelTag)
            return
        end

        -- 触发异常退出逻辑
        HonorEventHelper:Fire(EventCmd.OnLevelUnusualExit, array[1], tonumber(array[2]))
        GameCoreManager:SetPlayingLevelTag('')
    end
end

---初始化玩家首次注册时间
---@type fun():void
function ProcedurePreload:InitGameDataFirstLaunchTime()
    local regUserTime = PbRootData.GameData.playerInfo.regUserTime
    if regUserTime == nil or regUserTime == 0 then
        PbRootData.GameData.playerInfo.regUserTime = HonorTimerHelper:GetUTC0TimeStamp()
        PbIO.GameData.Save_playerInfo_regUserTime()
    end
end

return ProcedurePreload
