---@class ProcedurePreload : ProcedureSuper 游戏预加载流程（游戏入口状态）
local ProcedurePreload = class('ProcedurePreload', import('ProcedureSuper'))

---@brief 构造函数
---@param args table 外部传入参数
function ProcedurePreload:ctor(args)
    ProcedurePreload.super.ctor(self, args)

    --- 预加载Lua文件列表
    self._preLoadLuaFiles = nil
    --- 低端机每帧加载数量
    self._loadCountPerFrameOfLow = 3
    --- 中端机每帧加载数量
    self._loadCountPerFrameOfMedium = 5
    --- 高端机每帧加载数量
    self._loadCountPerFrameOfHigh = 10
    
    --- 初始化启动时间数据
    self:InitLaunchTimeData()
end

---@brief 创建流程实例
---@param args table 外部参数
---@return ProcedurePreload 实例对象
function ProcedurePreload:Create(args)
    return ProcedurePreload.new(args)
end

---@brief 初始化游戏启动时间数据（上次启动时间 / 当前启动时间）
function ProcedurePreload:InitLaunchTimeData()
    if not self.cs or not PbRootData or not PbRootData.UserData then
        return
    end

    if self.cs:IsLaunch() == true then
        -- 安全赋值时间数据
        local last = PbRootData.UserData.launchTime.last
        local curr = PbRootData.UserData.launchTime.current
        local latest = PbRootData.UserData.latestTime

        last.utc0 = curr.utc0
        last.utc0String = curr.utc0String
        last.locality = curr.locality
        last.localString = curr.localString

        -- 更新当前启动时间
        local utc0 = HonorTimerHelper:GetUTC0TimeStamp()
        local localStamp = HonorTimerHelper:GetLocalTimeStamp()

        curr.utc0 = utc0
        curr.utc0String = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", utc0)
        curr.locality = localStamp
        curr.localString = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", localStamp)

        -- 新账号初始化
        if not PbRootData.GameData.isInit then
            last.utc0 = curr.utc0
            last.utc0String = curr.utc0String
            last.locality = curr.locality
            last.localString = curr.localString
        end

        -- 同步最近时间
        latest.utc0 = last.utc0
        latest.utc0String = last.utc0String
        latest.locality = last.locality
        latest.localString = last.localString

        -- 安全保存
        xpcall(function()
            PbIO.UserData.Save_launchTime()
            PbIO.UserData.Save_latestTime()
        end, function() end)

        -- 初始化游戏时间
        if GameTimeHelper then
            GameTimeHelper:SetupFutureTimeStamp()
        end
    end
end

---@brief 进入流程生命周期
---@param ownerMachine userdata 流程状态机
function ProcedurePreload:OnEnter(ownerMachine)
    ProcedurePreload.super.OnEnter(self, ownerMachine)

    if self.cs then
        self.cs.RemoveAllContentsOnProcedureTransition = false
        self.cs.StartLoading = false
    end

    --- 初始化游戏基础设置
    self:InitGameSetting()
    --- 初始化新账号数据
    self:InitNewAccountData()
    --- 初始化首次注册时间
    self:InitGameDataFirstLaunchTime()
    --- 执行分步加载
    self:LoadAddSteps()
end

---@brief 尝试显示隐私政策弹窗
---@param finishCallback function 显示完成回调
function ProcedurePreload:TryShowTermsPrivacy(finishCallback)
    if not PbRootData or not PbRootData.GameData or not PbRootData.GameData.playerInfo then
        if finishCallback then xpcall(finishCallback, __G__TRACK_BACK__) end
        return
    end

    if not PbRootData.GameData.playerInfo.isShowPrivacy then
        OpenUISync(UIs.UITermsPrivacy, { callback = finishCallback })
        PbRootData.GameData.playerInfo.isShowPrivacy = true
        xpcall(function()
            PbIO.GameData.Save_playerInfo_isShowPrivacy()
        end, function() end)
    else
        if finishCallback then
            xpcall(finishCallback, __G__TRACK_BACK__)
        end
    end
end

    ProcedurePreload.super.OnUpdate(self, ownerMachine)

---@brief 初始化游戏基础设置（多点触摸等）
function ProcedurePreload:InitGameSetting()
    if GameLib and GameLib.Util and GameLib.Util.SetMultiTouchEnabled then
        GameLib.Util.SetMultiTouchEnabled(false)
    end
end

---@brief 新账号初始化（仅首次进入游戏执行）
function ProcedurePreload:InitNewAccountData()
    if not PbRootData or not PbRootData.GameData or PbRootData.GameData.isInit ~= false then
        return
    end

    PrintInfo("=== 初始化新账号数据 ===")

    -- 安全初始化管理器
    xpcall(function() ItemManager:Init() end, function() end)
    xpcall(function() ChapterListManager:Init() end, function() end)
    xpcall(function() ItemManager:InitNewAccountItemData() end, function() end)
    xpcall(function() ChapterListManager:UnlockChapter("1") end, function() end)

    -- 标记已初始化
    PbRootData.GameData.isInit = true
    xpcall(function() PbIO.GameData.Save_isInit() end, function() end)

    -- 清理临时数据
    xpcall(function() ItemManager:ClearData() end, function() end)
    xpcall(function() ChapterListManager:ClearData() end, function() end)
    xpcall(function() HonorEventHelper:Cleanup() end, function() end)
end

---@brief 执行分步加载逻辑（单例、Lua、资源、跳转）
function ProcedurePreload:LoadAddSteps()
    -- 初始化所有单例
    if SingletonManager then
        SingletonManager:InitAllSingleForStep(self)
    end
    
    -- 加载完成跳转到主流程
    self.cs:AddStep(function()
        if self.cs and typeof(Honor.ProcedurePlaying) then
            self.cs:PrepareToNextProcedure(typeof(Honor.ProcedurePlaying), self.cs.ArgsFromChanging)
        end
    end)

    if self.cs then
        self.cs.StartLoading = true
    end
end

---@brief 预加载单个Lua文件
---@param luaFileName string 文件名
function ProcedurePreload:PreLoadLuaFile(luaFileName)
    if not string.isNullOrEmpty(luaFileName) then
        xpcall(function() require(luaFileName) end, function() end)
    end
end

---@brief 根据设备性能等级获取每帧加载数量
---@return number 每帧加载个数
function ProcedurePreload:LoadCountPreFrame()
    if not GameManager or not GameManager.GetDevicePerformanceLevel then
        return self._loadCountPerFrameOfMedium
    end

    local level = GameManager:GetDevicePerformanceLevel()
    if level == Honor.DevicePerformanceLevel.High then
        return self._loadCountPerFrameOfHigh
    elseif level == Honor.DevicePerformanceLevel.Mid then
        return self._loadCountPerFrameOfMedium
    elseif level == Honor.DevicePerformanceLevel.Low then
        return self._loadCountPerFrameOfLow
    end
    return self._loadCountPerFrameOfMedium
end

---@brief 检查游戏是否异常退出（未正常结束关卡）
function ProcedurePreload:CheckIsUnusualExitLevel()
    if not GameCoreManager then return end

    local levelTag = GameCoreManager:GetPlayingLevelTag()
    if not levelTag or string.isNullOrEmpty(levelTag) then
        return
    end

    local arr = string.split(levelTag, "_")
    if #arr ~= 2 then
        PrintError("[CheckIsUnusualExitLevel] 格式错误：" .. levelTag)
        GameCoreManager:SetPlayingLevelTag('')
        return
    end

    -- 触发异常退出事件
    HonorEventHelper:Fire(EventCmd.OnLevelUnusualExit, arr[1], tonumber(arr[2]))
    GameCoreManager:SetPlayingLevelTag('')
end

---@brief 初始化玩家首次注册时间（时间戳）
function ProcedurePreload:InitGameDataFirstLaunchTime()
    if not PbRootData or not PbRootData.GameData or not PbRootData.GameData.playerInfo then
        return
    end

    local info = PbRootData.GameData.playerInfo
    if not info.regUserTime or info.regUserTime <= 0 then
        info.regUserTime = HonorTimerHelper:GetUTC0TimeStamp()
        xpcall(function()
            PbIO.GameData.Save_playerInfo_regUserTime()
        end, function() end)
    end
end

return ProcedurePreload
