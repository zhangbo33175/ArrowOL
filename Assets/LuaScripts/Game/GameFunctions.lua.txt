--- 创建一个新的对象池子节点
---@type fun():UnityEngine.GameObject
---@return UnityEngine.GameObject 新创建的对象池子节点
function CreateNewPoolRoot()
    local poolRootGo = GetPoolRoot()
    if IsNull(poolRootGo) then
        poolRootGo = Unity.GameObject("PoolRoot")
    end

    local newRoot = Unity.GameObject("PoolNew")
    newRoot.transform:SetParent(poolRootGo.transform)
    newRoot.transform.localScale = Unity.Vector3.one
    newRoot.transform.localPosition = Unity.Vector3.zero

--region 字符串工具
--=========================================================================
-- 字符串工具
--=========================================================================
--- 判断字符串是否以指定内容开头
---@type fun(str:string, startStr:string):boolean
---@param str string 原字符串
---@param startStr string 开头匹配串
---@return boolean 是否匹配
function StringStartsWith(str, startStr)
    if not str or not startStr then return false end
    return string.sub(str, 1, string.len(startStr)) == startStr
end

--- 判断字符串是否以指定内容结尾
---@type fun(str:string, endStr:string):boolean
---@param str string 原字符串
---@param endStr string 结尾匹配串
---@return boolean 是否匹配
function StringEndsWith(str, endStr)
    if not str or not endStr then return false end
    return string.sub(str, -string.len(endStr)) == endStr
end

--region Spine 动画工具
--=========================================================================
-- Spine 动画工具
--=========================================================================
--- 获取 Spine.SkeletonAnimation 动画时长
---@type fun(spineAnim:Spine.Unity.SkeletonAnimation, animName:string):number
---@param spineAnim Spine.Unity.SkeletonAnimation Spine动画组件
---@param animName string 动画名称
---@return number 动画时长（秒）
function GetSpineAnimationDuration(spineAnim, animName)
    if IsNull(spineAnim) or IsNull(spineAnim.skeletonDataAsset) then return 0 end
    local animData = spineAnim.skeletonDataAsset:GetAnimationStateData().SkeletonData:FindAnimation(animName)
    return animData and animData.Duration or 0
end

--- 获取 Spine.SkeletonGraphic 动画时长
---@type fun(skeletonGraphic:Spine.Unity.SkeletonGraphic, animName:string):number
---@param skeletonGraphic Spine.Unity.SkeletonGraphic UI Spine组件
---@param animName string 动画名称
---@return number 动画时长（秒）
function GetSpineGraphicAnimationDuration(skeletonGraphic, animName)
    if IsNull(skeletonGraphic) or IsNull(skeletonGraphic.AnimationState) then return 0 end
    local anim = skeletonGraphic.AnimationState.Data.SkeletonData:FindAnimation(animName)
    return anim and anim.Duration or 0
end

--- 判断是否存在指定 Spine 动画
---@type fun(skeletonGraphic:Spine.Unity.SkeletonGraphic, animName:string):boolean
---@param skeletonGraphic Spine.Unity.SkeletonGraphic UI Spine组件
---@param animName string 动画名称
---@return boolean 是否存在该动画
function HasSpineAnimation(skeletonGraphic, animName)
    if IsNull(skeletonGraphic) or IsNull(skeletonGraphic.AnimationState) then return false end
    return skeletonGraphic.AnimationState.Data.SkeletonData:FindAnimation(animName) ~= nil
end

--- 播放 Spine 动画
---@type fun(skeletonGraphic:Spine.Unity.SkeletonGraphic, animName:string, loop:boolean):Spine.TrackEntry
---@param skeletonGraphic Spine.Unity.SkeletonGraphic UI Spine组件
---@param animName string 动画名称
---@param loop boolean 是否循环
---@return Spine.TrackEntry 动画轨道信息
function PlaySpineAnimation(skeletonGraphic, animName, loop)
    if IsNull(skeletonGraphic) or IsNull(skeletonGraphic.AnimationState) then return nil end
    local state = skeletonGraphic.AnimationState
    local entry = state:SetAnimation(0, animName, loop)
    state:Update(0)
    state:Apply(skeletonGraphic.Skeleton)
    return entry
end

--- 延迟播放 Spine 动画
---@type fun(skeletonGraphic:Spine.Unity.SkeletonGraphic, animName:string, delay:number, loop:boolean):Spine.TrackEntry
---@param skeletonGraphic Spine.Unity.SkeletonGraphic UI Spine组件
---@param animName string 动画名称
---@param delay number 延迟时间（秒）
---@param loop boolean 是否循环
---@return Spine.TrackEntry 动画轨道信息
function DelayPlaySpineAnimation(skeletonGraphic, animName, delay, loop)
    if IsNull(skeletonGraphic) or IsNull(skeletonGraphic.AnimationState) then return nil end
    return skeletonGraphic.AnimationState:AddAnimation(0, animName, loop, delay)
end

--- 贝塞尔曲线飞行动画（金币/道具飞入）
---@type fun(starFly:RGame.StarFly, parentTransform:UnityEngine.Transform, count:number, startPos:UnityEngine.Vector3, targetPos:UnityEngine.Vector3, duration:number, delay:number, ease:DG.Tweening.Ease, tweenId:string, realTime:boolean, onComplete:function, reverseCtrlX:boolean, minOffsetX:number, maxOffsetX:number):void
---@param starFly RGame.StarFly 飞行动画组件
---@param parentTransform UnityEngine.Transform 父节点
---@param count number 数量
---@param startPos UnityEngine.Vector3 起点世界坐标
---@param targetPos UnityEngine.Vector3 终点世界坐标
---@param duration number 动画时长
---@param delay number 延迟时间
---@param ease DG.Tweening.Ease 缓动类型
---@param tweenId string 动画ID
---@param realTime boolean 是否使用真实时间
---@param onComplete function 完成回调
---@param reverseCtrlX boolean 是否反转X轴控制点
---@param minOffsetX number X轴最小偏移
---@param maxOffsetX number X轴最大偏移
function StarFlyAnimation(starFly, parentTransform, count, startPos, targetPos, duration, delay, ease, tweenId, realTime, onComplete, reverseCtrlX, minOffsetX, maxOffsetX)
    if IsNull(starFly) then
        if onComplete then onComplete(nil) end
        return
    end

    -- 默认参数
    realTime = realTime or false
    tweenId = tweenId or ECustomTweenID.ON_COLLECT_GOAL_ANIM_FLY .. starFly.gameObject:GetInstanceID()
    ease = ease or DG.Tweening.Ease.Linear
    delay = delay or 0
    reverseCtrlX = reverseCtrlX or false
    minOffsetX = minOffsetX or 0
    maxOffsetX = maxOffsetX or 0

    -- 坐标转换
    local localStart = parentTransform:InverseTransformPoint(startPos)
    local localTarget = parentTransform:InverseTransformPoint(targetPos)
    starFly.startPoint = Vector2(localStart.x, localStart.y)
    starFly.endPoint = Vector2(localTarget.x, localTarget.y)
    starFly:ResetCtrlPoints()
    starFly:SetProgress(0)

    -- 控制点偏移
    if starFly.ctrlPoints then
        local offsetX = (minOffsetX ~= 0 or maxOffsetX ~= 0) and math.random(minOffsetX, maxOffsetX) or 0
        for i = 0, starFly.ctrlPoints.Count - 1 do
            local p = starFly.ctrlPoints[i]
            p.x = p.x + offsetX
            if reverseCtrlX then p.x = -p.x end

            local worldP = starFly.transform:TransformPoint(Vector3(p.x, p.y, 0))
            local localP = parentTransform:InverseTransformPoint(worldP)
            starFly.ctrlPoints[i] = Vector2(localP.x, localP.y)
        end
    end

    -- 执行Tween
    DG.Tweening.DOTween.Kill(tweenId, true)
    DG.Tweening.DOTween.To(
            function() return starFly:GetProgress() end,
            function(v) starFly:SetProgress(v) end,
            1, duration
    ):SetEase(ease)
      :SetDelay(delay)
      :SetUpdate(realTime)
      :OnComplete(function()
        if onComplete then onComplete(starFly.gameObject) end
    end).stringId = tweenId
end

--- 物体按路径点曲线运动
---@type fun(go:UnityEngine.GameObject, pathPoints:UnityEngine.Vector3[], preDelay:number, duration:number, ease:DG.Tweening.Ease):DG.Tweening.Sequence
---@param go UnityEngine.GameObject 运动物体
---@param pathPoints UnityEngine.Vector3[] 路径点数组
---@param preDelay number 前置延迟
---@param duration number 时长
---@param ease DG.Tweening.Ease 缓动类型
---@return DG.Tweening.Sequence 动画序列
function DOPathAndWait(go, pathPoints, preDelay, duration, ease)
    if IsNull(go) or not pathPoints or #pathPoints < 2 then return nil end

    go.transform.position = pathPoints[1]
    local sequence = DG.Tweening.DOTween.Sequence()
                       :AppendInterval(preDelay)
                       :AppendCallback(function() go:SetActive(true) end)
                       :Append(go.transform:DOPath(pathPoints, duration, DG.Tweening.PathType.CatmullRom))
                       :SetEase(ease)
                       :SetUpdate(true)

    return sequence
end

--- 设计宽度转当前屏幕宽度
---@type fun(designWidth:number):number
---@param designWidth number 设计尺寸宽度
---@return number 实际屏幕宽度
function GetRealScreenWidthByDesign(designWidth)
    local designRes = GameMainRoot.UI.ScreenDesignedResolution
    local canvas = GameMainRoot.UI.ScreenUICanvas
    local canvasWidth = designRes.x

    local rect = canvas.transform:GetComponent(typeof(UnityEngine.RectTransform))
    if not IsNull(rect) then
        canvasWidth = rect.rect.size.x

    return (designWidth * UnityEngine.Screen.width) / canvasWidth
end

--- 判断 table 是否为空
---@type fun(t:table):boolean
---@param t table 目标表
---@return boolean 是否为空
function IsTableEmpty(t)
    if not t then return true end
    return next(t) == nil
end

--- 数组映射转换
---@type fun(source:table, converter:function):table
---@param source table 原数组
---@param converter function 转换函数
---@return table 转换后数组
function ArrayConvert(source, converter)
    if not source or not converter then return {} end
    local result = {}
    for _, v in ipairs(source) do
        table.insert(result, converter(v))
    end
    return result
end

--- 清空数组（高效）
---@type fun(array:table):void
---@param array table 目标数组
function ClearArray(array)
    if not array then return end
    while #array > 0 do
        table.remove(array)
    end
end

--- 清空字典
---@type fun(dict:table):void
---@param dict table 目标字典
function ClearDictionary(dict)
    if not dict then return end
    for k, _ in pairs(dict) do
        dict[k] = nil
    end
end

--- Fisher-Yates 打乱数组
---@type fun(array:table):void
---@param array table 目标数组
function ShuffleArray(array)
    if not array or #array <= 1 then return end
    local i = #array
    while i > 1 do
        local j = math.random(i)
        array[i], array[j] = array[j], array[i]
        i = i - 1
    end
end

--- 二分法插入有序数组
---@type fun(array:table, item:table, sortField:string):void
---@param array table 有序数组
---@param item table 插入项
---@param sortField string 排序字段名
function BinaryInsertSortedArray(array, item, sortField)
    if not array or not item or not sortField then return end
    local count = #array
    local min = 1
    local max = count

    while count > 0 and max > min do
        local mid = min + math.floor((max - min) / 2)
        local midVal = tonumber(array[mid][sortField])
        local itemVal = tonumber(item[sortField])

        if midVal > itemVal then
            max = math.clamp(mid - 1, min, max)
        elseif midVal < itemVal then
            min = math.clamp(mid + 1, min, max)
        else
            while mid > 1 and tonumber(array[mid-1][sortField]) == itemVal do
                mid = mid - 1
            end
            min = mid
            break
        end
    end

    if array[min] == nil or tonumber(array[min][sortField]) >= tonumber(item[sortField]) then
        table.insert(array, min, item)
    else
        table.insert(array, min + 1, item)
    end
end

--- 获取 UI 中心点世界坐标
---@type fun(rectTransform:UnityEngine.RectTransform):UnityEngine.Vector3
---@param rectTransform UnityEngine.RectTransform UI矩形变换
---@return UnityEngine.Vector3 中心点世界坐标
function GetRectTransformCenterWorldPos(rectTransform)
    if IsNull(rectTransform) then return Vector3.zero end

    local pivot = rectTransform.pivot
    if math.abs(pivot.x - 0.5) < 0.001 and math.abs(pivot.y - 0.5) < 0.001 then
        return rectTransform.position
    end

    local offsetPivot = Vector2(0.5, 0.5) - pivot
    local size = rectTransform.sizeDelta
    local offset = Vector2(size.x * offsetPivot.x, size.y * offsetPivot.y)
    return rectTransform:TransformPoint(Vector3(offset.x, offset.y, 0))
end

--- 等比缩放限制尺寸
---@type fun(origin:Vector2, maxWidth:number, maxHeight:number):Vector2
---@param origin Vector2 原始尺寸
---@param maxWidth number 最大宽度
---@param maxHeight number 最大高度
---@return Vector2 缩放后尺寸
function ScaleToFitSize(origin, maxWidth, maxHeight)
    if origin.y > maxHeight then
        local scale = maxHeight / origin.y
        origin.x = origin.x * scale
        origin.y = maxHeight
    end
    if origin.x > maxWidth then
        local scale = maxWidth / origin.x
        origin.x = maxWidth
        origin.y = origin.y * scale
    end
    return Vector2(origin.x, origin.y)
end

--- 检查父节点是否隐藏（最多查几层）
---@type fun(transform:UnityEngine.Transform, maxDepth:number):boolean
---@param transform UnityEngine.Transform 目标变换
---@param maxDepth number 检查深度
---@return boolean 是否有父节点隐藏
function IsAnyParentInactive(transform, maxDepth)
    if IsNull(transform) then return true end
    maxDepth = math.min(maxDepth or 3, 5)
    local parent = transform.parent

    while not IsNull(parent) and maxDepth > 0 do
        maxDepth = maxDepth - 1
        if not parent.gameObject.activeSelf then return true end
        parent = parent.parent
    end
    return false
end

--- 复制物体并设为子节点
---@type fun(parent:UnityEngine.GameObject, original:UnityEngine.GameObject):UnityEngine.GameObject
---@param parent UnityEngine.GameObject 父物体
---@param original UnityEngine.GameObject 原始物体
---@return UnityEngine.GameObject 复制后的物体
function InstantiateChild(parent, original)
    if IsNull(parent) or IsNull(original) then return nil end
    local go = UnityEngine.Object.Instantiate(original)
    go:SetActive(true)
    go.transform:SetParent(parent.transform)
    go.transform.localScale = Vector3.one
    go.transform.localPosition = Vector3.zero
    go.transform.localEulerAngles = Vector3.zero
    return go
end

    child.transform:SetParent(parent.transform)
    child.transform.localScale = Vector3.one
    child.transform.localEulerAngles = Vector3.zero
    child.transform.localPosition = Vector3.zero
end

--region 包围盒工具
--=========================================================================
-- 包围盒工具
--=========================================================================
--- 获取模型屏幕包围盒
---@type fun(meshFilter:UnityEngine.MeshFilter):table
---@param meshFilter UnityEngine.MeshFilter 网格过滤器
---@return table 包围盒信息（center + size）
function GetMeshScreenBounds(meshFilter)
    if IsNull(meshFilter) then return nil end
    local mesh = meshFilter.mesh or meshFilter.sharedMesh
    if IsNull(mesh) then return nil end

    local pos = meshFilter.transform.position
    local bounds = mesh.bounds
    local minWorld = bounds.min + pos
    local maxWorld = bounds.max + pos

    local minScreen = SceneToScreenPoint(minWorld)
    local maxScreen = SceneToScreenPoint(maxWorld)

    return {
        center = (minScreen + maxScreen) / 2,
        size = maxScreen - minScreen
    }
end

--- 获取2D碰撞盒屏幕包围盒
---@type fun(collider2D:UnityEngine.BoxCollider2D):table
---@param collider2D UnityEngine.BoxCollider2D 2D碰撞器
---@return table 包围盒信息（center + size）
function GetBox2DScreenBounds(collider2D)
    if IsNull(collider2D) then return nil end
    local b = collider2D.bounds
    local minScreen = SceneToScreenPoint(b.min)
    local maxScreen = SceneToScreenPoint(b.max)
    return {
        center = (minScreen + maxScreen) / 2,
        size = maxScreen - minScreen
    }
end

--- 获取UI屏幕包围盒
---@type fun(rectTransform:UnityEngine.RectTransform, camera:UnityEngine.Camera):table
---@param rectTransform UnityEngine.RectTransform UI矩形变换
---@param camera UnityEngine.Camera 渲染相机
---@return table 包围盒信息（center + size）
function GetRectTransformScreenBounds(rectTransform, camera)
    if IsNull(rectTransform) then return nil end
    local size = rectTransform.rect.size
    local half = size / 2
    local centerLocal = rectTransform.localPosition

    local minLocal = centerLocal - Vector3(half.x, half.y, 0)
    local maxLocal = centerLocal + Vector3(half.x, half.y, 0)

    local minWorld = rectTransform:TransformPoint(minLocal)
    local maxWorld = rectTransform:TransformPoint(maxLocal)

    local minScreen = UIToScreenPoint(minWorld, camera)
    local maxScreen = UIToScreenPoint(maxWorld, camera)
    local centerScreen = UIToScreenPoint(GetRectTransformCenterWorldPos(rectTransform), camera)

    return {
        center = centerScreen,
        size = maxScreen - minScreen
    }
end

--=========================================================================
-- 安全调用 / 延迟工具
--=========================================================================
--- 安全执行函数（捕获异常）
---@type fun(func:function, errorMsg:string):void
---@param func function 执行函数
---@param errorMsg string 错误提示信息
function SafeInvoke(func, errorMsg)
    if not func then return end
    xpcall(func, function(msg)
        PrintError(string.format("[SafeInvoke] %s => %s", errorMsg or "Error", debug.traceback(msg, 4)))
    end)
end

--- DOTween 延迟执行
---@type fun(delay:number, realTime:boolean, callback:function, tweenId:string):DG.Tweening.Sequence
---@param delay number 延迟时间
---@param realTime boolean 是否真实时间
---@param callback function 回调函数
---@param tweenId string 动画ID
---@return DG.Tweening.Sequence 动画序列
function DelayInvoke(delay, realTime, callback, tweenId)
    local seq = DG.Tweening.DOTween.Sequence()
                  :AppendInterval(delay)
                  :AppendCallback(function()
        SafeInvoke(callback, "DelayInvoke Error")
    end)
                  :SetUpdate(realTime)

    if tweenId and not string.isNullOrEmpty(tweenId) then
        seq.stringId = tweenId
    end
    return seq
end

--- 协程延迟执行
---@type fun(self:table, delay:number, realTime:boolean, callback:function):UnityEngine.Coroutine
---@param self table 执行主体
---@param delay number 延迟时间
---@param realTime boolean 是否真实时间
---@param callback function 回调函数
---@return UnityEngine.Coroutine 协程对象
function CoroutineDelayInvoke(self, delay, realTime, callback)
    if delay <= 0 then
        SafeInvoke(callback)
        return
    end

    local action = function()
        if realTime then
            HonorCoroutineHelper:WaitForSecondsRealtime(delay)
        else
            HonorCoroutineHelper:WaitForSeconds(delay)
        end
        SafeInvoke(callback)
    end

    return HonorCoroutineHelper:Start(handler(self, action))
end

--- 协程延迟N帧执行
---@type fun(self:table, frameCount:number, callback:function):UnityEngine.Coroutine
---@param self table 执行主体
---@param frameCount number 帧数
---@param callback function 回调函数
---@return UnityEngine.Coroutine 协程对象
function CoroutineDelayFrame(self, frameCount, callback)
    if frameCount <= 0 then
        SafeInvoke(callback)
        return
    end

    return HonorCoroutineHelper:Start(handler(self, function()
        for i = 1, frameCount do
            coroutine.yield(HonorCoroutineHelper.waitForEndOfFrame)
        end
        SafeInvoke(callback)
    end))
end

--- 独立延迟帧调用
---@type fun(frameCount:number, callback:function, param:any):void
---@param frameCount number 帧数
---@param callback function 回调函数
---@param param any 回调参数
function DelayFrameInvoke(frameCount, callback, param)
    if frameCount <= 0 then
        callback(param)
        return
    end

    HonorCoroutineHelper:StartAlone(function()
        for i = 1, frameCount do
            coroutine.yield(HonorCoroutineHelper.waitForEndOfFrame)
        end
        callback(param)
    end)
end

--- 分批遍历（防卡顿）
---@type fun(data:table, batchCount:number, callback:function):void
---@param data table 数据列表
---@param batchCount number 每批数量
---@param callback function 处理回调
function BatchProcess(data, batchCount, callback)
    if not data or not callback or batchCount <= 0 then return end
    local batch = {}
    for _, v in ipairs(data) do
        table.insert(batch, v)
        if #batch >= batchCount then
            callback(batch)
            ClearArray(batch)
        end
    end
    if #batch > 0 then
        callback(batch)
        ClearArray(batch)
    end
end

--- 播放 Spine 显示 → Idle → 隐藏 整套动画
---@type fun(spine:Spine.Unity.SkeletonGraphic, idleDuration:number, showAnim:string, idleAnim:string, hideAnim:string, onHideStart:function, onHideComplete:function):void
---@param spine Spine.Unity.SkeletonGraphic UI Spine组件
---@param idleDuration number Idle停留时长
---@param showAnim string 显示动画
---@param idleAnim string 循环Idle动画
---@param hideAnim string 隐藏动画
---@param onHideStart function 隐藏开始回调
---@param onHideComplete function 隐藏完成回调
function PlaySpineShowIdleHide(spine, idleDuration, showAnim, idleAnim, hideAnim, onHideStart, onHideComplete)
    if IsNull(spine) then return end

    spine.Skeleton:SetSlotsToSetupPose()
    spine.AnimationState:ClearTracks()

    local showEntry = spine.AnimationState:SetAnimation(0, showAnim, false)
    showEntry:AddComplete(function()
        if IsNull(spine.gameObject) then return end
        spine.AnimationState:SetAnimation(0, idleAnim, true)

        -- Idle 延迟后隐藏
        DG.Tweening.DOTween.Sequence()
          :SetUpdate(true)
          :AppendInterval(idleDuration)
          :AppendCallback(function()
            if IsNull(spine.gameObject) then return end
            if onHideStart then SafeInvoke(onHideStart) end

            local hideEntry = spine.AnimationState:SetAnimation(0, hideAnim, false)
            hideEntry.TimeScale = 2
            hideEntry:AddComplete(function()
                if onHideComplete then SafeInvoke(onHideComplete) end
            end)
        end)
    end)
end

--===================== 字符串转Vector3 =====================--
---将字符串格式的坐标转换为Unity Vector3
---支持格式: (1,2,3) 、1,2,3、(-1.5, 2.0, 3.7)
---@param str string 坐标字符串
---@return UnityEngine.Vector3 转换后的三维坐标
function StringToVector3(str)
    if not str or str == "" then
        return CS.UnityEngine.Vector3.zero
    end

    -- 去除括号
    str = string.gsub(str, "[()]", "")

    -- 正则匹配 x,y,z
    local x, y, z = string.match(str, "([%d%.%-]+),%s*([%d%.%-]+),%s*([%d%.%-]+)")

    -- 转换为数字，空值则为0
    x = tonumber(x) or 0
    y = tonumber(y) or 0
    z = tonumber(z) or 0

    -- 返回Unity Vector3
    return CS.UnityEngine.Vector3(x, y, z)
end

--===================== 通用表长度计算 =====================--
---安全获取表长度（支持数组、哈希表、空表、不连续表）
---@param _table table 目标表
---@return number 表中元素数量
function GetTableCount(_table)
    local count = 0
    if _table ~= nil then
        for k, v in pairs(_table) do
            count = count + 1
        end
    end
    return count
end

--=========================================================================
--- 检查表中是否包含指定Key（通用字典包含判断）
---@param _table table 目标表
---@param key any 要判断的Key
---@return boolean 是否包含
--=========================================================================
function ContainsKey(_table, key)
    if type(_table) ~= "table" or key == nil then
        return false
    end
    if _table[key] ~= nil then
        return true
    end
    return  false
end
