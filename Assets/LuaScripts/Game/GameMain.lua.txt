---
--- 应用挂起/恢复回调（切后台/切回）
--- @param pause boolean  true=挂起  false=恢复
---
function GameApplicationPauseCallback(pause)
    -- 如果正在删除用户，不处理
    if LoginManager:IsDelUser() then
        return
    end
    -- 从后台切回 → 尝试重新登录
    if pause == false then
        LoginManager:TryReLogin()
    end
end
