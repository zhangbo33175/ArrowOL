--- 应用挂起/恢复回调（切后台 / 切回前台）
--- @param isPaused boolean true=挂起 false=恢复
function GameApplicationPauseCallback(isPaused)
    -- 安全判断
    if IsNull(LoginManager) or LoginManager:IsDelUser() then
        return
    end

    -- 从后台切回 → 自动尝试重登
    if not isPaused then
        PrintInfo("[LuaMain] 应用从后台恢复，尝试重新登录")
        LoginManager:TryReLogin()
    end
end
