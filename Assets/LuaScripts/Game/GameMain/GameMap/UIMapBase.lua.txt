--region 【地图卸载 & 资源清理】

---@brief 卸载地图前清理（MapManager:UnloadOldMap 驱动，核心资源释放）
---@desc 结束 GameMapMotion、遍历 Kill 所有 DOTween 延时调用并清空列表
---@type fun():void
function UIMapBase:UnloadMapBeforeClear()
    UIMapBase.super.UnloadMapBeforeClear(self)

    -- 结束地图控制
    GameMapMotion:OnMapEnd()

    -- 清理所有延迟调用
    for id, _ in pairs(self._delayCallIds) do
        DOTween.Kill(id)
    end
    ClearDictionary(self._delayCallIds)
end

---@brief 地图卸载后清理（MapManager 驱动）
---@type fun():void
function UIMapBase:UnloadMapAfterClear()
    UIMapBase.super.UnloadMapAfterClear(self)
end
