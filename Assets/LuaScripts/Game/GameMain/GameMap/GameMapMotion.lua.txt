---@brief 根据当前相机 aspect/ortho 与"已解锁分区聚合边界(Limit)"，算出相机可移动的 X/Y 边界
---@type fun():number,number,number,number
---@return number minX, number maxX, number minY, number maxY
---@desc 缩放自适应：hw/hh 随 orthographicSize 实时变化，边界自动扣除视口半宽/半高，
---      视口比聚合区域还大时（min>max）收敛到区域中心，避免 clamp 反向把相机吸到角落。
function GameMapMotion:GetCameraBounds()
    local aspect = self.ScreenCamera.aspect
    local ortho  = self.ScreenCamera.orthographicSize
    local hw = aspect * ortho
    local hh = ortho
    local minX = self.LimitMinX + hw
    local maxX = self.LimitMaxX - hw
    local minY = self.LimitMinY + hh
    local maxY = self.LimitMaxY - hh
    -- 视口大于聚合区域：边界收敛到区域中心（保持旧公式在 Limit==整章时的等价语义）
    if minX > maxX then
        local c = (self.LimitMinX + self.LimitMaxX) / 2
        minX, maxX = c, c
    end
    if minY > maxY then
        local c = (self.LimitMinY + self.LimitMaxY) / 2
        minY, maxY = c, c
    end
    return minX, maxX, minY, maxY
end

---@brief 把单个分区数据内所有实体位置并入聚合边界 bounds
---@type fun(bounds:table,sectionData:table):void
---@param bounds table 聚合边界 {minX,maxX,minY,maxY}
---@param sectionData table 分区数据（MapData/Characters/Buildings/Objects 各数组 Position）
function GameMapMotion:_UnionSectionPositions(bounds, sectionData)
    if not sectionData then return end
    ---@param arr table 实体数组
    local function walk(arr)
        if not arr then return end
        for _, v in ipairs(arr) do
            local pos = v and v.Position
            if pos then
                ---TODO [坐标变换] entry.Position 为编辑器左下原点4096场景坐标，
                ---     相机边界用 Root 局部世界系（中心原点±1024），聚合前统一变换，否则 clamp 窗口错倍→拖动露黑边
                local ex = tonumber(pos[1])
                local ey = tonumber(pos[2])
                if ex and ey then
                    local x, y = EditorPointToWorld(ex, ey)
                    if x < bounds.minX then bounds.minX = x end
                    if x > bounds.maxX then bounds.maxX = x end
                    if y < bounds.minY then bounds.minY = y end
                    if y > bounds.maxY then bounds.maxY = y end
                end
            end
        end
    end
    walk(sectionData.MapData)
    walk(sectionData.Characters)
    walk(sectionData.Buildings)
    walk(sectionData.Objects)
end

---@brief 按已解锁分区聚合 AABB 重算相机 clamp 边界（未解锁分区不可达）
---@desc 枚举章节已解锁分区(1..lv)，把各分区实体位置求并集作为聚合 AABB；
---      数据缺失/退化（合并后过小）时回退整章预制体边界 BaseBounds，保证不回归。
---      地图启动与分区解锁后均调用本方法。
---@type fun():void
function GameMapMotion:RefreshUnlockedSectionBounds()
    if not self.IsMapWork or not self.ScreenCamera then return end

    ---TODO: [多分区] 相机边界按已解锁分区聚合；以存档进度 lv 为解锁权威（编辑器 Unlocked 仅初始态）
    local chapterId = ChapterListManager:GetRuntimeChapterId()
    local lv = GamerDataManager:GetLevelLvByChapterId(chapterId)
    local datas = GamePlayHelper:GetUnlockedLevelDatas(chapterId, lv)

    local bounds = { minX = math.huge, maxX = -math.huge, minY = math.huge, maxY = -math.huge }
    for _, entry in ipairs(datas) do
        self:_UnionSectionPositions(bounds, entry.data)
    end

    -- 退化保护：无数据或合并范围过小 → 回退整章预制体边界（旧行为）
    local valid = bounds.maxX - bounds.minX > 1 and bounds.maxY - bounds.minY > 1
    if not valid then
        self.LimitMinX, self.LimitMaxX = self.BaseBoundsMinX, self.BaseBoundsMaxX
        self.LimitMinY, self.LimitMaxY = self.BaseBoundsMinY, self.BaseBoundsMaxY
    else
        self.LimitMinX, self.LimitMaxX = bounds.minX, bounds.maxX
        self.LimitMinY, self.LimitMaxY = bounds.minY, bounds.maxY
    end

    self:ClampFollowPosition()
end

---TODO: [相机聚焦] 按已解锁分区几何矩形聚焦（row-major 前缀，cover 无黑边）
function GameMapMotion:FocusUnlockedSections(unlockedCount)
    ---TODO [代码质量] ScreenCamera nil 惰性取（Awake 期早于 OnMapStart 赋值）
    if not self.ScreenCamera then
        self.ScreenCamera = GameLib.MapCamHelper.MapCamera
    end
    if not self.ScreenCamera then return end
    unlockedCount = tonumber(unlockedCount) or 1
    ---TODO [优化] 分区布局从 MapLayoutConfig 读取，不再硬编码 cols/rows/2048
    local cols = tonumber(EMapLayoutConfig.regionCols) or 1
    local rows = tonumber(EMapLayoutConfig.regionRows) or 1
    local zoneW = tonumber(EMapLayoutConfig.zoneWidth) or 2048
    local zoneH = tonumber(EMapLayoutConfig.zoneHeight) or 2048

    local minX, maxX = math.huge, -math.huge
    local minY, maxY = math.huge, -math.huge
    ---TODO [优化] 分区尺寸从 EMapLayoutConfig.zoneWidth/zoneHeight 读取
    for sid = 1, unlockedCount do
        local idx = sid - 1
        local c = idx % cols
        local r = math.floor(idx / cols)
        local x0 = c * zoneW
        local x1 = c * zoneW + zoneW
        -- ---TODO [坐标] top-left anchor，Y 向下为负
        local yTop = -r * zoneH
        local yBottom = -(r * zoneH + zoneH)
        minX = math.min(minX, x0); maxX = math.max(maxX, x1)
        minY = math.min(minY, yBottom); maxY = math.max(maxY, yTop)
    end

    local cx = (minX + maxX) / 2
    local cy = (minY + maxY) / 2
    -- ---TODO [修复] ZoneRoot pivot=center，地图以原点为中心，需要减去半区偏移
    cx = cx - zoneW / 2
    cy = cy + zoneH / 2
    -- ---TODO [修复] 用红框宽高比计算 ortho，而非全屏比例
    local aspect = self._RedBoxAspect or (UnityEngine.Screen.width / UnityEngine.Screen.height)
    -- ---TODO [相机适配] 竖屏游戏：填满宽度无左右黑边，垂直方向自动扩展
    local ortho = (maxX - minX) / 2 / aspect
    ortho = math.max(ortho, self.MinOrthoSize)

    self.ScreenCamera.orthographicSize = ortho
    local cfg = self:_GetMotionConfig()
    local minZoomRatio = cfg and cfg.m_MinZoomRatio or 0.5
    self.MinOrthoSize = ortho * minZoomRatio
    self.MaxOrthoSize = ortho
    self.FollowPosition = UnityEngine.Vector3(cx, cy, 0)
    self.ScreenCamera.transform.position = UnityEngine.Vector3(cx, cy, self.ScreenCamera.transform.position.z)

    -- ---TODO [修复] ZoneRoot pivot=center，边界需要减去半区偏移
    self.LimitMinX = minX - zoneW / 2
    self.LimitMaxX = maxX - zoneW / 2
    self.LimitMinY = minY + zoneH / 2
    self.LimitMaxY = maxY + zoneH / 2
end

---@brief 把跟随目标位置 clamp 到边界内
---@type fun():void
function GameMapMotion:ClampFollowPosition()
    local minX, maxX, minY, maxY = self:GetCameraBounds()
    self.FollowPosition.x = math.clamp(self.FollowPosition.x, minX, maxX)
    self.FollowPosition.y = math.clamp(self.FollowPosition.y, minY, maxY)
    self.FollowPosition.x = math.floor(self.FollowPosition.x * 1000) / 1000
    self.FollowPosition.y = math.floor(self.FollowPosition.y * 1000) / 1000
end

---@brief 应用新的正交尺寸：clamp 到 [Min,Max]，赋值后重算边界
---@type fun(newOrtho:number):void
---@param newOrtho number 期望的正交尺寸
function GameMapMotion:ApplyZoom(newOrtho)
    local clamped = math.clamp(newOrtho, self.MinOrthoSize, self.MaxOrthoSize)
    if clamped <= 0 then
        clamped = (self.MinOrthoSize + self.MaxOrthoSize) / 2
    end
    self.ScreenCamera.orthographicSize = clamped
    self:ClampFollowPosition()
    -- 缩放是即时操作：直接把相机 transform 吸到 FollowPosition，
    -- 否则 OnUpdate 里的 SmoothStep 会让相机滞后，从角上缩小瞬间露出地图外黑边并抖动
    if self.FollowPosition then
        self.ScreenCamera.transform:SetPositionX(self.FollowPosition.x)
        self.ScreenCamera.transform:SetPositionY(self.FollowPosition.y)
    end
end

---@brief 屏幕两点位移 → 世界位移（拖动地图用）
---@type fun(curX:number,curY:number,lastX:number,lastY:number):number,number
---@return number dx, number dy 世界空间位移（已取反：手指往哪拖，地图跟着走）
function GameMapMotion:ScreenDeltaToWorldDelta(curX, curY, lastX, lastY)
    self.Vector3Num.x = curX;  self.Vector3Num.y = curY
    GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer1)
    self.Vector3Num.x = lastX; self.Vector3Num.y = lastY
    GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer2)
    return -(self.VectorContainer1.x - self.VectorContainer2.x),
           -(self.VectorContainer1.y - self.VectorContainer2.y)
end

---@brief 判断两点屏幕距离是否小于拖动阈值（小于=单击）
---@type fun(x0:number,y0:number,x1:number,y1:number):boolean
function GameMapMotion:IsTapDistance(x0, y0, x1, y1)
    local dx = x1 - x0
    local dy = y1 - y0
    local dragThreshold = self:_GetMotionConfig() and self:_GetMotionConfig().m_DragThreshold or 12
    return math.sqrt(dx * dx + dy * dy) < dragThreshold
end

---@brief 判断屏幕坐标是否在合法屏幕范围内
---@type fun(x:number,y:number):boolean
function GameMapMotion:IsInsideScreen(x, y)
    return x >= 0 and y >= 0 and x <= UnityEngine.Screen.width and y <= UnityEngine.Screen.height
end

---@brief 重置滑动相关状态（惯性系数/平滑值/临时向量归零）
---@type fun():void
function GameMapMotion:ResetSwipeState()
    self.InertialSlidingFactor = 0
    self.SwipeSmoothStepNum = 0
    self.Vector3Num.x = 0
    self.Vector3Num.y = 0
end

---@brief 应用一次世界位移：写入临时向量、移动跟随目标并递增平滑系数
---@type fun(dx:number, dy:number):void
---@param dx number 世界X位移
---@param dy number 世界Y位移
function GameMapMotion:ApplyWorldDelta(dx, dy)
    self.Vector3Num.x = dx
    self.Vector3Num.y = dy
    self:MovingFollowPosition(self.Vector3Num)
    local cfg = self:_GetMotionConfig()
    local smoothMin = cfg and cfg.m_SwipeSmoothMin or 0.3
    local smoothMax = cfg and cfg.m_SwipeSmoothMax or 1.0
    self.SwipeSmoothStepNum = math.clamp(self.SwipeSmoothStepNum + UnityEngine.Time.deltaTime * self.StepAddFactor, smoothMin, smoothMax)
end

--region 【核心更新逻辑】每帧驱动

---@brief 地图控制主更新函数
---@type fun():void
function GameMapMotion:OnUpdate()
    if not self.IsMapWork then
        return
    end

    -- 安全矫正相机尺寸
    if self.ScreenCamera.orthographicSize <= 0 then
        local unlockedCount = (MapLoadManager and MapLoadManager.UnlockedCount) or 1
        self:FocusUnlockedSections(unlockedCount)
    else
        self.ScreenCamera.orthographicSize = math.clamp(
                self.ScreenCamera.orthographicSize,
                math.max(self.MinOrthoSize, 200),
                self.MaxOrthoSize
        )
    end

    -- 自动扫描时直接匀速移动相机，跳过平滑跟随（避免 waypoint 切换时顿挫）
    if not self.AutoScanActive then
        -- X轴平滑跟随
        local currCameraX = self.ScreenCamera.transform:GetPositionX()
        if not math.Approximately(currCameraX, self.FollowPosition.x) then
            local smoothX = math.SmoothStep(currCameraX, self.FollowPosition.x, self.SwipeSmoothStepNum)
            self.ScreenCamera.transform:SetPositionX(smoothX)
        end

        -- Y轴平滑跟随
        local currCameraY = self.ScreenCamera.transform:GetPositionY()
        if not math.Approximately(currCameraY, self.FollowPosition.y) then
            local smoothY = math.SmoothStep(currCameraY, self.FollowPosition.y, self.SwipeSmoothStepNum)
            self.ScreenCamera.transform:SetPositionY(smoothY)
        end
    end

    -- 惯性滑动
    if self.InertialSlidingFactor > 0 then
        self:OnInertialSliding(self.InertialSlidingFactor)
    end

    -- 自动扫描（蛇形扫图找预制体）
    self:UpdateAutoScan()

    -- 输入控制：用户拖拽/双指时自动停止扫描
    -- 注意：启动扫描的那一下点击本身也会让 GetMouseButton(0)=true，
    -- 用一个帧数宽限避免刚启动就被自己的点击停掉
    if self.AutoScanActive then
        if self.AutoScanGraceFrames and self.AutoScanGraceFrames > 0 then
            self.AutoScanGraceFrames = self.AutoScanGraceFrames - 1
        elseif UnityEngine.Input.GetMouseButton(0) or UnityEngine.Input.touchCount > 0 then
            self:StopAutoScan()
        end
    end

    self:CheckPinchZoom()
    self:UpdateMouseScrollZoom()
    self:UpdateMouseDrag()
    self:UpdateClickAndDrag()
end

--endregion

--region 【PC端鼠标拖动】

---@brief PC鼠标拖拽移动地图
---@type fun():void
function GameMapMotion:UpdateMouseDrag()
    -- 按下
    if UnityEngine.Input.GetMouseButtonDown(0) then
        self.PointStartFromUI = self:IsPointerOverGameObject()
        local mp = UnityEngine.Input.mousePosition

        -- UI区域 / 屏幕外 不响应
        if self.PointStartFromUI or not self:IsInsideScreen(mp.x, mp.y) then
            self.IsMouseDown = false
            return
        end

        self.IsMouseDown = true
        self.MouseLastScreenPos.x = mp.x
        self.MouseLastScreenPos.y = mp.y

        -- 重置状态
        self:ResetSwipeState()
    end

    -- 拖拽中
    if UnityEngine.Input.GetMouseButton(0) and self.IsMouseDown and not self.PointStartFromUI then
        local curr = UnityEngine.Input.mousePosition
        if not self:IsInsideScreen(curr.x, curr.y) then return end

        local dx, dy = self:ScreenDeltaToWorldDelta(
            curr.x, curr.y,
            self.MouseLastScreenPos.x, self.MouseLastScreenPos.y
        )
        self:ApplyWorldDelta(dx, dy)
        self.MouseLastScreenPos.x = curr.x
        self.MouseLastScreenPos.y = curr.y
    end

    -- 抬起
    if UnityEngine.Input.GetMouseButtonUp(0) and self.IsMouseDown then
        self.IsMouseDown = false
        self.PointStartFromUI = false
        self.InertialSlidingFactor = 1 - self.SwipeSmoothStepNum
        self.SwipeSmoothStepNum = 0.3
    end
end

---@brief 判断当前是否触摸在UI上
---@type fun():boolean
---@return boolean 是否在UI上
function GameMapMotion:IsPointerOverGameObject()
    return GameLib.RSetupLuaTableValue.IsPointerOverUIObject()
end

--region 【核心：严格区分 单击 / 拖动】

---@brief 单击/拖动分离逻辑，超过阈值=拖动，不触发单击
---@type fun():void
function GameMapMotion:UpdateClickAndDrag()
    if not self.IsMapWork or not self.ClickEnable or self.PointStartFromUI then
        return
    end

    -- 鼠标按压开始
    if UnityEngine.Input.GetMouseButtonDown(0) then
        local mp = UnityEngine.Input.mousePosition
        self.IsPressing = true
        self.ClickStartX = mp.x
        self.ClickStartY = mp.y
    end

    -- 鼠标按压结束
    if UnityEngine.Input.GetMouseButtonUp(0) then
        if self.IsPressing then
            self.IsPressing = false
            local mp = UnityEngine.Input.mousePosition
            if self:IsTapDistance(self.ClickStartX, self.ClickStartY, mp.x, mp.y) then
                self:TriggerClick(mp)
            end
        end
    end

    -- 触摸单击判断
    local tc = UnityEngine.Input.touchCount
    if tc == 1 then
        local touch = UnityEngine.Input.GetTouch(0)
        if touch.phase == UnityEngine.TouchPhase.Began then
            self.IsPressing = true
            self.ClickStartX = touch.position.x
            self.ClickStartY = touch.position.y
        elseif touch.phase == UnityEngine.TouchPhase.Ended or touch.phase == UnityEngine.TouchPhase.Canceled then
            if self.IsPressing then
                self.IsPressing = false
                if self:IsTapDistance(self.ClickStartX, self.ClickStartY, touch.position.x, touch.position.y) then
                    self:TriggerClick(touch.position)
                end
            end
        end
    end
end

---@brief 触发单击回调（仅真正单击时调用）
---@type fun(screenPos:UnityEngine.Vector2):void
---@param screenPos UnityEngine.Vector2 点击屏幕坐标
function GameMapMotion:TriggerClick(screenPos)
    if not self.ScreenCamera then return end
    if not self:IsInsideScreen(screenPos.x, screenPos.y) then return end

    local isHit = false
    local hitTarget = nil
    local characterLayer = UnityEngine.LayerMask.NameToLayer("Character")

    -- 遍历Character层碰撞体进行点击检测（无射线、全平台兼容）
    local allColliders = UnityEngine.Object.FindObjectsOfType(typeof(UnityEngine.BoxCollider2D))
    if allColliders ~= nil then
        for i = 0, allColliders.Length - 1 do
            local collider = allColliders[i]
            if collider ~= nil and collider.gameObject ~= nil then
                local go = collider.gameObject
                if go.layer == characterLayer then
                    local screenPoint = self.ScreenCamera:WorldToScreenPoint(collider.transform.position)
                    local bounds = collider.bounds
                    local screenRect = UnityEngine.Rect(
                            screenPoint.x - bounds.extents.x,
                            screenPoint.y - bounds.extents.y,
                            bounds.size.x,
                            bounds.size.y
                    )
                    if screenRect:Contains(UnityEngine.Vector2(screenPos.x, screenPos.y)) then
                        isHit = true
                        hitTarget = go
                        break
                    end
                end
            end
        end
    end

    if not (isHit and hitTarget) then
        HonorEventHelper:Fire(EventCmd.OnClickCharacter, nil)
    end

    -- 外部回调
    if self.OnMapClick then
        self.OnMapClick(nil, isHit, hitTarget)
    end
end

--region 【移动端EasyTouch滑动】

---@brief 滑动开始
---@type fun(gesture:HedgehogTeam.EasyTouch.Gesture):void
---@param gesture HedgehogTeam.EasyTouch.Gesture 滑动手势
function GameMapMotion:OnSwipeBegin(gesture)
    if GameManager:IsGamePaused() then return end
    self.PointStartFromUI = self:IsPointerOverGameObject()
    self.IsPressing = false
    if self.PointStartFromUI then return end

    self:ResetSwipeState()
end

---@brief 滑动中
---@type fun(gesture:HedgehogTeam.EasyTouch.Gesture):void
---@param gesture HedgehogTeam.EasyTouch.Gesture 滑动手势
function GameMapMotion:OnSwipe(gesture)
    if self.PointStartFromUI or not self.IsMapWork then return end
    if gesture.touchCount ~= 1 then return end

    local tx = gesture:GetPositionX()
    local ty = gesture:GetPositionY()
    if not self:IsInsideScreen(tx, ty) then return end

    local dx, dy = self:ScreenDeltaToWorldDelta(
        tx, ty,
        tx - gesture.GetDeltaPositionX(),
        ty - gesture.GetDeltaPositionY()
    )
    self:ApplyWorldDelta(dx, dy)
end

---@brief 滑动结束
---@type fun(gesture:HedgehogTeam.EasyTouch.Gesture):void
---@param gesture HedgehogTeam.EasyTouch.Gesture 滑动手势
function GameMapMotion:OnSwipeEnd(gesture)
    if GameManager:IsGamePaused() then return end
    if self.PointStartFromUI then
        self:ResetSwipeState()
    end
    self.PointStartFromUI = false
    self.IsPinching = false
    self.InertialSlidingFactor = 1 - self.SwipeSmoothStepNum
    self.SwipeSmoothStepNum = 0.3
end

--endregion

--region 【惯性滑动】

---@brief 拖拽结束后的惯性移动
---@type fun(factor:number):void
---@param factor number 惯性系数
function GameMapMotion:OnInertialSliding(factor)
    self.InertialSlidingVector.x = self.Vector3Num.x * self.InertialSlidingFactor
    self:MovingFollowPosition(self.InertialSlidingVector)
    local cfg = self:_GetMotionConfig()
    local decay = cfg and cfg.m_InertiaDecay or 0.025
    local smoothFloor = cfg and cfg.m_InertiaSmoothFloor or 0.75
    self.InertialSlidingFactor = math.clamp(self.InertialSlidingFactor - decay, 0, 1)
    self.SwipeSmoothStepNum = math.clamp(self.SwipeSmoothStepNum + GameLib.RSetupLuaTableValue.GetTimeDeltaTime(), smoothFloor, 1)
end

--region 【移动到目标点】

---@brief 终止并清理自动扫描 DOTween 序列（幂等）
---@type fun():void
function GameMapMotion:KillAutoScanSeq()
    if self.AutoScanSeq then
        self.AutoScanSeq:Kill()
        self.AutoScanSeq = nil
    end
end

---@brief 自动扫描地图：蛇形路径从左上扫到右下，找预制体用
---@type fun(orthoSize:number):void
---@param orthoSize number 扫描时的相机尺寸，默认 300
function GameMapMotion:StartAutoScan(orthoSize)
    if not self.ScreenCamera or not self.FollowPosition then return end
    local cfg = self:_GetMotionConfig()
    local defaultOrtho = cfg and cfg.m_AutoScanDefaultOrtho or 300
    local overlap = cfg and cfg.m_AutoScanOverlap or 0.8
    local segDur = cfg and cfg.m_AutoScanSegmentDuration or 1.2
    local graceFrames = cfg and cfg.m_AutoScanGraceFrames or 15
    orthoSize = orthoSize or defaultOrtho

    -- 切到扫描缩放
    self:ApplyZoom(orthoSize)

    -- 重新算边界（缩放后 hw/hh 变了）
    local minX, maxX, minY, maxY = self:GetCameraBounds()
    if minX > maxX or minY > maxY then
        return
    end

    -- 每次横向/纵向步进 = 视野的 overlap（留重叠避免漏看）
    local aspect = self.ScreenCamera.aspect
    local stepX = aspect * orthoSize * 2 * overlap
    local stepY = orthoSize * 2 * overlap

    -- 生成蛇形路径点
    self.AutoScanWaypoints = {}
    local y = maxY
    local row = 0
    while y >= minY - 0.01 do
        if row % 2 == 0 then
            -- 从左到右
            local x = minX
            while x <= maxX + 0.01 do
                table.insert(self.AutoScanWaypoints, { x = x, y = y })
                x = x + stepX
            end
            -- 最后一个点对齐 maxX
            self.AutoScanWaypoints[#self.AutoScanWaypoints] = { x = maxX, y = y }
        else
            -- 从右到左
            local x = maxX
            while x >= minX - 0.01 do
                table.insert(self.AutoScanWaypoints, { x = x, y = y })
                x = x - stepX
            end
            self.AutoScanWaypoints[#self.AutoScanWaypoints] = { x = minX, y = y }
        end
        y = y - stepY
        row = row + 1
    end

    -- 用 DOTween Sequence 串起所有 waypoint，丝滑补间
    local camTransform = self.ScreenCamera.transform
    local camZ = camTransform:GetPositionZ()

    -- 先杀掉旧 tween
    self:KillAutoScanSeq()

    local seq = CS.DG.Tweening.DOTween.Sequence()
    for i, wp in ipairs(self.AutoScanWaypoints) do
        local target = UnityEngine.Vector3(wp.x, wp.y, camZ)
        local tween = camTransform:DOMove(target, segDur)
        tween:SetEase(CS.DG.Tweening.Ease.InOutSine)
        seq:Append(tween)
    end

    -- 完成回调
    seq:OnComplete(CS.DG.Tweening.TweenCallback(function()
        self:StopAutoScan()
    end))

    self.AutoScanSeq = seq
    self.AutoScanActive = true
    self.AutoScanGraceFrames = graceFrames
end

---@brief 停止自动扫描
function GameMapMotion:StopAutoScan()
    self.AutoScanActive = false
    self.AutoScanWaypoints = nil
    self.AutoScanIndex = 0
    self:KillAutoScanSeq()
    -- 同步 FollowPosition 到相机当前位置，防止退出后跳回
    if self.ScreenCamera then
        self.FollowPosition.x = self.ScreenCamera.transform:GetPositionX()
        self.FollowPosition.y = self.ScreenCamera.transform:GetPositionY()
    end
end

---@brief 自动扫描由 DOTween 驱动，这里只做输入检测（用户拖拽即停止）
function GameMapMotion:UpdateAutoScan()
    -- 空实现，移动由 DOTween Sequence 处理
end

--region 【双指缩放 + 鼠标滚轮缩放】

---@brief 移动端双指缩放检测
---@type fun():void
function GameMapMotion:CheckPinchZoom()
    if not self.ScreenCamera or not self.IsMapWork or self.PointStartFromUI then return end
    local tc = UnityEngine.Input.touchCount
    if tc ~= 2 then
        self.IsPinching = false
        return
    end

    local t1 = UnityEngine.Input.GetTouch(0)
    local t2 = UnityEngine.Input.GetTouch(1)
    if t1.phase == UnityEngine.TouchPhase.Began or t2.phase == UnityEngine.TouchPhase.Began then
        self.LastPinchDistance = self:GetTwoTouchDistance(t1, t2)
        self.IsPinching = true
        self.InertialSlidingFactor = 0
        return
    end
    if t1.phase ~= UnityEngine.TouchPhase.Moved and t2.phase ~= UnityEngine.TouchPhase.Moved then
        return
    end

    local curr = self:GetTwoTouchDistance(t1, t2)
    local diff = curr - self.LastPinchDistance
    if math.abs(diff) <= 1 then return end

    local ortho = self.ScreenCamera.orthographicSize
    local zoomSpeed = self:_GetMotionConfig() and self:_GetMotionConfig().m_ZoomSpeed or 1.1
    ortho = diff > 0 and (ortho / zoomSpeed) or (ortho * zoomSpeed)
    self:ApplyZoom(ortho)
    self.LastPinchDistance = curr
end

---@brief 获取两个触摸点距离
---@type fun(t1:UnityEngine.Touch,t2:UnityEngine.Touch):number
---@param t1 UnityEngine.Touch 触摸点1
---@param t2 UnityEngine.Touch 触摸点2
---@return number 两点距离
function GameMapMotion:GetTwoTouchDistance(t1, t2)
    local dx = t1.position.x - t2.position.x
    local dy = t1.position.y - t2.position.y
    return math.sqrt(dx * dx + dy * dy)
end

---@brief 鼠标滚轮缩放
---@type fun():void
function GameMapMotion:UpdateMouseScrollZoom()
    if not self.ScreenCamera or not self.IsMapWork then return end
    local scroll = UnityEngine.Input.GetAxis("Mouse ScrollWheel")
    if math.abs(scroll) < 0.001 then return end

    local ortho = self.ScreenCamera.orthographicSize
    local zoomSpeed = self:_GetMotionConfig() and self:_GetMotionConfig().m_ZoomSpeed or 1.1
    ortho = scroll > 0 and (ortho / zoomSpeed) or (ortho * zoomSpeed)
    self:ApplyZoom(ortho)
end

---@brief 注册/反注册触摸事件
---@type fun(action:string):void
---@param action string "+"注册 / "-"注销
function GameMapMotion:RegisterTouchEvents(action)
    if action == "+" then
        self.SwipeBegin = self.SwipeBegin or function(g) self:OnSwipeBegin(g) end
        self.Swipe = self.Swipe or function(g) self:OnSwipe(g) end
        self.SwipeEnd = self.SwipeEnd or function(g) self:OnSwipeEnd(g) end

        CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragStart("+", self.SwipeBegin)
        CS.HedgehogTeam.EasyTouch.EasyTouch.On_Drag("+", self.Swipe)
        CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragEnd("+", self.SwipeEnd)
    else
        if self.SwipeBegin then
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragStart("-", self.SwipeBegin)
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_Drag("-", self.Swipe)
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragEnd("-", self.SwipeEnd)
        end
    end
end

--region 【初始化与销毁】

---@brief 地图启动初始化
---@type fun(mapTrans:UnityEngine.Transform):void
---@param mapTrans UnityEngine.Transform 地图根节点
function GameMapMotion:OnMapStart(mapTrans)
    if not mapTrans then return end
    self.MapMotionLayer = mapTrans:GetComponent("GameLib.MapMotionLayer")
    if not self.MapMotionLayer then return end

    -- 运行时根据子物体自动重算包围盒（此时地图内容已加载）
    self.MapMotionLayer:RecalculateBounds()

    -- 基础配置
    self.ScreenCamera = GameLib.MapCamHelper.MapCamera
    self.IsMapWork = true
    self.SwipeSmoothStepNum = 0
    self.IsMouseDown = false
    self.IsPressing = false

    -- 预守卫：绑定相机后立即矫正 ortho size
    if self.ScreenCamera ~= nil and self.ScreenCamera.orthographicSize <= 0 then
        local unlockedCount = (MapLoadManager and MapLoadManager.UnlockedCount) or 1
        self:FocusUnlockedSections(unlockedCount)
    end

    -- 相机渲染区域
    self:SetCameraRect(nil, nil, self:_GetCanvasHeight())

    -- 设置负 near 裁剪面，消除 z=0 坐标转换的视锥外告警
    if self.ScreenCamera ~= nil then
        self._MapCamNearBackup = self.ScreenCamera.nearClipPlane
        GameLib.MapCamHelper.SetAllCamerasNear(-10)
    end

    -- 禁用物理射线检测（点击由 TriggerClick 手动遍历处理）
    local raycaster2D = self.ScreenCamera:GetComponent(typeof(UnityEngine.EventSystems.Physics2DRaycaster))
    if raycaster2D then raycaster2D.enabled = false end
    local raycaster3D = self.ScreenCamera:GetComponent(typeof(UnityEngine.EventSystems.PhysicsRaycaster))
    if raycaster3D then raycaster3D.enabled = false end

    -- 地图边界数据
    self.MapHalfWidth  = self.MapMotionLayer.m_AreaBounds.extents.x
    self.MapHalfHeigh  = self.MapMotionLayer.m_AreaBounds.extents.y
    self.MapCenterX    = self.MapMotionLayer.m_AreaBounds.center.x
    self.MapCenterY    = self.MapMotionLayer.m_AreaBounds.center.y
    self.MapBounds     = self.MapMotionLayer.m_AreaBounds

    -- 基础边界（整章预制体兜底）与 Limit（实际 clamp）
    self.BaseBoundsMinX = self.MapCenterX - self.MapHalfWidth
    self.BaseBoundsMaxX = self.MapCenterX + self.MapHalfWidth
    self.BaseBoundsMinY = self.MapCenterY - self.MapHalfHeigh
    self.BaseBoundsMaxY = self.MapCenterY + self.MapHalfHeigh
    self.LimitMinX, self.LimitMaxX = self.BaseBoundsMinX, self.BaseBoundsMaxX
    self.LimitMinY, self.LimitMaxY = self.BaseBoundsMinY, self.BaseBoundsMaxY

    -- 初始位置（必须先建好 FollowPosition，再调 ApplyZoom/ClampFollowPosition）
    self.FollowPosition = {
        x = self.MapCenterX,
        y = self.MapCenterY,
        z = self.ScreenCamera.transform:GetPositionZ()
    }
    self.PositionWhenPaused = { x = self.FollowPosition.x, y = self.FollowPosition.y, z = self.FollowPosition.z }

    -- 最大缩放兜底值，FocusUnlockedSections 会按实际分区覆盖
    self.MaxOrthoSize = math.max(self.MapHalfHeigh, self.MinOrthoSize)
    self:ApplyZoom(self.ScreenCamera.orthographicSize)

    -- 聚焦已解锁分区
    local unlockedCount = (MapLoadManager and MapLoadManager.UnlockedCount) or 1
    self:FocusUnlockedSections(unlockedCount)

    -- 临时向量
    self.InertialSlidingVector = { x = 0, y = 0, z = 0 }
    self.Vector3Num = { x = 0, y = 0, z = 0 }

    -- 相机信息
    self.ScreenCameraAspect = self.ScreenCamera.aspect
    self.ScreenCameraOrthographicSize = self.ScreenCamera.orthographicSize

    -- 注册触摸
    self:RegisterTouchEvents("+")
    -- 按已解锁分区聚合 AABB 收敛相机边界
    self:RefreshUnlockedSectionBounds()
    self:ClampFollowPosition()
end

---@brief 获取屏幕顶部/底部安全区偏移（设计分辨率单位，自动适配刘海屏与底部手势条）
---@type fun(canvasH:number):number,number
---@param canvasH number Canvas设计高度（默认 1334）
---@return number topInset 顶部安全区偏移（刘海/挖孔避让，设计分辨率单位）
---@return number bottomInset 底部安全区偏移（手势条/导航栏避让，设计分辨率单位）
function GameMapMotion:GetSafeAreaInsets(canvasH)
    canvasH = canvasH or self:_GetCanvasHeight()
    local safeArea = UnityEngine.Screen.safeArea
    local screenH = UnityEngine.Screen.height
    if screenH <= 0 then return 0, 0 end
    local scale = canvasH / screenH
    -- Unity 屏幕坐标原点在左下角：
    -- safeArea.yMin = 底部非安全区高度（底部手势条/导航栏），safeArea.yMax 以上 = 顶部非安全区（刘海/挖孔）
    local topInset    = math.max(0, screenH - safeArea.yMax) * scale
    local bottomInset = math.max(0, safeArea.yMin) * scale
    return topInset, bottomInset
end

---@brief 像素高度 → 设计分辨率高度（单源换算，相机rect统一使用设计单位）
---@type fun(px:number, canvasH:number):number
---@param px number 像素高度
---@param canvasH number Canvas设计高度（默认 1334）
---@return number 设计分辨率单位高度
function GameMapMotion:PixelsToDesign(px, canvasH)
    canvasH = canvasH or self:_GetCanvasHeight()
    local screenH = UnityEngine.Screen.height
    if screenH <= 0 then return px or 0 end
    return (px or 0) * canvasH / screenH
end

---@brief 从当前地图数据读取顶部/底部HUD设计高度（MapData缺失时回退硬编码设计值）
---@type fun(canvasH:number):number,number
---@param canvasH number Canvas设计高度（默认 1334）
---@return number topHudDesign 顶部HUD设计高度
---@return number bottomHudDesign 底部HUD设计高度
---@desc 与 MapData.cs 的 m_TopHudPixelHeight / m_BottomHudPixelHeight（像素）联动，消除 SetCameraRect 硬编码双源
function GameMapMotion:GetHudDesignHeights(canvasH)
    canvasH = canvasH or self:_GetCanvasHeight()
    -- 回退值（设计分辨率单位，MapData 缺失时使用）
    local topDesign, bottomDesign = 116.7621, 199.237
    if MapManager ~= nil and MapManager.CurMapData ~= nil then
        local topPx    = MapManager.CurMapData.m_TopHudPixelHeight
        local bottomPx = MapManager.CurMapData.m_BottomHudPixelHeight
        if topPx ~= nil and topPx > 0 then
            topDesign = self:PixelsToDesign(topPx, canvasH)
        end
        if bottomPx ~= nil and bottomPx > 0 then
            bottomDesign = self:PixelsToDesign(bottomPx, canvasH)
        end
    end
    return topDesign, bottomDesign
end

---@brief 设置相机渲染区域（顶部留 topHud、底部留 botHud，中间渲染地图）
---@type fun(botHud:number, topHud:number, canvasH:number):void
---@param botHud number 底部UI高度（设计分辨率单位，nil=取MapData）
---@param topHud number 顶部UI高度（设计分辨率单位，nil=取MapData）
---@param canvasH number Canvas高度，默认 1334
---@desc 未显式传值时从 MapData 读取 HUD 高度（单源）；并自动叠加顶部/底部安全区偏移，刘海屏/手势条设备上场景不会顶到不安全区
function GameMapMotion:SetCameraRect(botHud, topHud, canvasH)
    canvasH = canvasH or self:_GetCanvasHeight()
    -- 未显式传值时，从 MapData 读取 HUD 设计高度（替代原硬编码 116.7621/199.237）
    if topHud == nil or botHud == nil then
        local topDesign, bottomDesign = self:GetHudDesignHeights(canvasH)
        topHud = topHud or topDesign
        botHud = botHud or bottomDesign
    end

    -- 安全区联动：顶部刘海 + 底部手势条避让（设计分辨率单位）
    local topInset, bottomInset = self:GetSafeAreaInsets(canvasH)
    topHud = topHud + topInset
    botHud = botHud + bottomInset

    local rectY = botHud / canvasH
    local rectH = math.max(0, (canvasH - topHud - botHud) / canvasH)
    -- ---TODO [修复] 相机rect设为红框区域（顶部留topHud、底部留botHud），地图只渲染在红框内
    self.ScreenCamera.rect = UnityEngine.Rect(0, rectY, 1, rectH)
    -- ---TODO [修复] 缓存红框宽高比，供 ortho 计算使用
    self._RedBoxAspect = rectH > 0 and (1.0 / rectH) or (UnityEngine.Screen.width / UnityEngine.Screen.height)
end

---@brief 更新相机渲染区域（底部UI隐藏/显示时调用）
---@type fun(bottomHudHeight:number):void
---@param bottomHudHeight number 底部UI高度（设计分辨率单位，nil=取MapData），0=隐藏底部UI
---@desc 顶部HUD始终取 MapData，底部由调用方控制（0=隐藏/延展地图）
function GameMapMotion:UpdateCameraRect(bottomHudHeight)
    self:SetCameraRect(bottomHudHeight, nil, self:_GetCanvasHeight())
end

---@brief 地图结束，清理
---@type fun():void
function GameMapMotion:OnMapEnd()
    self:RegisterTouchEvents("-")
    self:PauseMapMotion()

    -- 玩法结束：恢复地图相机全屏视口，避免裁剪残留(rect非全屏)导致后续界面(登录/主界面)渲染区域异常
    --TODO: [修复] 玩法结束恢复相机rect全屏，与 SetCameraRect 的裁剪形成完整闭环
    if self.ScreenCamera ~= nil then
        self.ScreenCamera.rect = UnityEngine.Rect(0, 0, 1, 1)
        -- 方案B：还原近裁剪面，避免负near残留影响后续其他相机/界面使用
        if self._MapCamNearBackup ~= nil then
            self.ScreenCamera.nearClipPlane = self._MapCamNearBackup
            self._MapCamNearBackup = nil
        end
    end
end

--endregion

return GameMapMotion
