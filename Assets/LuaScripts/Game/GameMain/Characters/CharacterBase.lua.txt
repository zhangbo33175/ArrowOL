---@class CharacterBase : LuaBehaviourSuper
---@field cs Honor.Runtime.LuaBehaviour @LuaBehaviour
---@field _CharacterIcon UnityEngine.Transform @角色
---@field _CharData Tables.Characters_Item @角色配置数据
---@field _GameObject UnityEngine.GameObject @角色根节点GameObject（CharacterGenerationManager 生成时赋值）
---@field _WaitFrame GameLib.WaitForFrames @帧等待对象
---@field _ClickItem GameLib.AorClickItem @点击组件
---@field _LastSoundSerialID number @最后播放的音效ID
---@field _name string @角色名称
---@field _Destroying boolean @销毁中标志（防销毁 GameObject 触发 C# OnDestroy 递归）
local CharacterBase = class('CharacterBase', import('LuaBehaviourSuper'))

--region 【构造 & 创建】

---@brief 构造函数
---@type fun(args:table):void
---@param args table @自定义参数
function CharacterBase:ctor(args)
    CharacterBase.super.ctor(self, args)
    self._CharData          = nil
    self._GameObject        = nil
    self._ClickItem         = nil
    self._CharacterIcon     = self.cs.transform:Find("CharacterIcon")
    ---@type GameLib.WaitForFrames @ 公共的
    self._WaitFrame         = GameLib.WaitForFrames(1)
end

---@brief 创建函数
---@type fun(args:table):CharacterBase
---@param args table @自定义参数
---@return CharacterBase @CharacterBase实例
function CharacterBase:Create(args)
    return CharacterBase.new(args)
end

--endregion

--region 【生命周期】

---@brief 注册监听代码处（注：所有成员函数的注册监听代码都写在这里，成员函数作为监听回调时不需要手动注销，父类会自动辅助完成事件的注销操作。）
---@type fun():void
function CharacterBase:OnAddListeners()
    CharacterBase.super.OnAddListeners(self)
    if self._ClickItem then
        self._ClickItem.onClick  = function()
            handler(self, self:OnClickCharacter())
        end
    end
end

---@brief 唤醒
---@type fun():void
function CharacterBase:Awake()
    if self._CharacterIcon then
        self._ClickItem = self._CharacterIcon.gameObject:GetOrAddComponent(typeof(GameLib.AorClickItem))
        self._ClickItem.clickInterval = 0.1
    end
    CharacterBase.super.Awake(self)
end

---@brief 开始
---@type fun():void
function CharacterBase:Start()
    CharacterBase.super.Start(self)
end

---@brief 销毁
---@type fun():void
---@desc 销毁角色实体根节点 GameObject，使角色图标从场上消失；原用 _GameObject 隐藏图标的逻辑在
---      OnDestroy 时因字段被清理置 nil 而报错，故改用 self.cs 根节点（与 _GameObject 同对象）可靠销毁。
---      销毁会触发 C# LuaBehaviour.OnDestroy 再次进入本方法，用 _Destroying 标志防递归
--TODO: [修复] OnDestroy 改用 self.cs 根节点销毁角色，保证点击收集后角色图标消失；防递归
function CharacterBase:OnDestroy()
    if self._Destroying then return end -- 防递归：销毁 GameObject 会再次触发本方法
    self._Destroying = true
    CharacterBase.super.OnDestroy(self)
    -- 销毁角色根节点 GameObject，使图标从场上消失
    if self.cs ~= nil and not IsNull(self.cs) and self.cs.gameObject ~= nil and not IsNull(self.cs.gameObject) then
        UnityEngine.Object.Destroy(self.cs.gameObject)
    end
end

--endregion

--region 【点击交互】

---@brief 点击了角色
---@type fun():void
function CharacterBase:OnClickCharacter()
    if GameStateMachine:CurrentStateType() == GameStateCmd.PlayingState then
        --主界面允许点击角色
        PrintInfo("CharacterBase:OnClickCharacter name:{0}", self._CharData.ID)
        HonorEventHelper:Fire(EventCmd.OnClickCharacter, self._CharData)
    end
end

--endregion

--region 【数据与清理】

---@brief 清理数据
---@type fun():void
function CharacterBase:Clear()
    self._CharData = nil
    self._ClickItem = nil
    self._CharacterIcon = nil
    self._GameObject   = nil
end

--endregion

--region 【音效播放】

---@brief 播放音效
---@type fun(name:string)
---@param name string @ Tables.Characters_Item 里面的字段
function CharacterBase:PlaySound(name)
    if self._CharData ~= nil and self._CharData[name] ~= nil then
        if self._LastSoundSerialID ~= nil then
            SoundManager:StopSound(self._LastSoundSerialID)
        end
        self._LastSoundSerialID = SoundManager:PlaySound(self._CharData[name])
    end
end

--endregion

--region 【公共获取接口】

---@brief 获取角色ID
---@type fun():string
---@return string @ 角色ID
function CharacterBase:GetCharID()
    if not self._CharData then
        PrintWarning(string.format("%s without CharData", self.cs.gameObject.name))
        return '-1'
    end
    return self._CharData.ID
end

---@brief 获取角色类型
---@type fun():ECharacterType
---@return number @ 角色类型
function CharacterBase:GetCharType()
    return self._CharData.CharType
end
