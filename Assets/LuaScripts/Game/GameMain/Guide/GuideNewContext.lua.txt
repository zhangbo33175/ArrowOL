---@class GuideNewContext:Singleton @引导环境
---@field _levelListener GuideLevelListener @关卡监听
---@field _conditions table<Tables.GuideList_Item,GuideCondition> @ 一个容器，保存当前的触发器
---@field _setter table<Tables.GuideList_Item,GuideForceSetup> @ 一个容器，保存当前的设置器
---@field _preconditioning GuidePreconditioningExecuteEnv @ 引导执行前预处理一些东西
---@field _executing table<Tables.GuideContent_Item[],GuideCondition> @ 引导步骤和执行条件绑定一下
---@field _dispatcher GuideEvent @ 封装一下引导的事件派发
---@field _stepWorker GuideStepExecuteContext @ 引导步骤的执行者
---@field _pauseLevelTime boolean @ 引导暂停关内计时
---@field _stepInWaittingState boolean @ 当前步骤是否是等待类型
---@field _pauseableTweenCollection  table  @ 收集允许被新手引导暂停的动画
---@field _coroutine thread @ 统计一帧的触发数据
---@field _waittingExecute GuideCondition @ 等待执行的那个触发器
---@field _waittingExecuteParam table @ 参数
---@field _envChecker TutorialExecuteEnvCheck @ 用于检查执行环境
local GuideNewContext = class("GuideNewContext", import("Singleton"))

--region 【构造 & 单例】

---@brief 构造函数
---@type fun(args:table):void
---@param args table @自定义参数
function GuideNewContext:ctor(args)
    GuideNewContext.super.ctor(self, args)
    self._envChecker = require('TutorialExecuteEnvCheck'):Create()
end

---@brief 创建函数
---@type fun(args:table):GuideNewContext
---@param args table @自定义参数
---@return GuideNewContext @引导环境
function GuideNewContext:Create(args)
    local obj = GuideNewContext.new(args)
    return obj
end

--endregion

--region 【清理 & 释放】

---@brief 清理数据
---@type fun():void
function GuideNewContext:ClearData()
    GuideNewContext.super.ClearData(self)
    xpcall(function()
        self:Dispose()
    end, function(msg)
        PrintError(debug.traceback(msg, 5))
    end)
end

---@brief 释放所有资源
---@type fun():void
function GuideNewContext:Dispose()
    if self._levelListener then
        self._levelListener:Dispose()
    end

    if self._conditions then
        for key, condition in pairs(self._conditions) do
            condition:Dispose()
            condition:CloseTriggerListener()
            self._conditions[key] = nil
        end
    end

    if self._setter then
        for key, setter in pairs(self._setter) do
            setter:Dispose()
            setter:CloseListener()
            self._setter[key] = nil
        end
    end

    if self._dispatcher then
        self._dispatcher:Dispose()
    end

    if self._stepWorker then
        self._stepWorker:Dispose()
    end

    self._executing = nil

    if self._coroutine then
        HonorCoroutineHelper:StopAlone(self._coroutine)
        self._coroutine = nil
    end

    self._waittingExecute = nil
    self._waittingExecuteParam = nil
end

--endregion

--region 【GM & 强制关闭】

---@brief GM指令直接完成当前正在执行的引导
---@type fun():void
function GuideNewContext:GMCloseRunning()
    self:ForceCloseTutorial()
end

---@brief 如果某引导执行步骤执行失败，强制跳过该引导，防止卡死
---@type fun():void
function GuideNewContext:ForceCloseTutorial()
    self:TryForceCloseTutorialByID()
    if GameStateMachine:CurrentStateType() == GameStateCmd.MainState then
        HonorEventHelper:Fire(EventCmd.OnTryResumeGameMainPushUI, "ForceCloseTutorial")
    end
end

---@brief 强制关闭某个引导
---@type fun(id:string):void
---@param id string @ 引导ID（可空）
function GuideNewContext:TryForceCloseTutorialByID(id)
    if not (self._stepWorker and self._stepWorker._cmds) then
        return
    end

    -- 检查是否是所需ID
    if id and self._stepWorker._current and self._stepWorker._current.GuideID ~= id then
        return
    end

    -- 清理事件监听
    if self._dispatcher and self._dispatcher._listener then
        for eventCmd, handles in pairs(self._dispatcher._listener) do
            for idx, handler in ipairs(handles) do
                if handler.target == self._stepWorker._waitter or handler.target == self._stepWorker._instant then
                    HonorEventHelper:RemoveListener(eventCmd, handler)
                    table.remove(handles, idx)
                end
            end
        end
    end

    self:GuideStepsExecutedCompleteTagStep(self._stepWorker._cmds)
    self._stepWorker:ExecuteComplete()
    self._stepWorker:Dispose()
end

--endregion

--region 【初始化 & 存档】

---@brief 初始化引导进度
---@type fun():void
function GuideNewContext:Init()
    GuideNewContext.super.Init(self)

    if SaveRootData.GameData.guidesCompleted and string.match(SaveRootData.GameData.guidesCompleted, "Jump") then
        return
    end

    self:CheckDataFormat()
    self._preconditioning = require("GuidePreconditioningExecuteEnv"):Create()
    self._pauseLevelTime = false
    self._stepInWaittingState = false

    local guidesTable = Tables.GuideList
    for id, guide in pairs(guidesTable) do
        self:CreateTutorialTrigger(guide)
    end

    if self._levelListener then
        self._levelListener:Init()
    end
end

---@brief 检查存档格式，给存档加前后下划线
---@type fun():void
function GuideNewContext:CheckDataFormat()
    if not StringStart(SaveRootData.GameData.guidesCompleted, "_") then
        SaveRootData.GameData.guidesCompleted = string.format("_%s", SaveRootData.GameData.guidesCompleted)
    end
    if not StringEnd(SaveRootData.GameData.guidesCompleted, "_") then
        SaveRootData.GameData.guidesCompleted = string.format("%s_", SaveRootData.GameData.guidesCompleted)
    end
end

--endregion

---@brief 获取当前引导步骤执行者
---@type fun():GuideStepExecuteContext
---@return GuideStepExecuteContext
function GuideNewContext:GetCurrentGuideStepWorker()
    return self._stepWorker
end

---@brief 是否有执行中的引导
---@type fun():boolean
---@return boolean
function GuideNewContext:IsTutorialExecute()
    return self._waittingExecute or (self._executing ~= nil and not TableIsEmpty(self._executing))
end

---@brief 指定引导是否正在执行
---@type fun(guideId:string):boolean
---@param guideId string
---@return boolean
function GuideNewContext:IsTutorialRunning(guideId)
    if TableIsEmpty(self._executing) then
        return false
    end

    for _, condition in pairs(self._executing) do
        if condition.guideItem and condition.guideItem.ID == guideId then
            return true
        end
    end
    return false
end

---@type fun():number
---@return number

end

--region 【游戏暂停/恢复（关内逻辑）】

---@brief 步骤开始执行时，暂停关卡内计时器
---@type fun():void
function GuideNewContext:TryPauseLevelTime()
    self._pauseLevelTime = true

    PauseableCoroutineHelper:Pause(PauseableCoroutineHelper.Any)
    PauseableCoroutineHelper:Pause(PauseableCoroutineHelper.Level)

    if self._pauseableTweenCollection then
        for tween, _ in pairs(self._pauseableTweenCollection) do
            local t = tween
            if t:IsComplete() then
                self._pauseableTweenCollection[tween] = nil
            elseif t:IsPlaying() then
                t:Pause()
            end
        end
    end
end

---@brief 引导结束后恢复游戏
---@type fun():void
function GuideNewContext:TryResumeLevelTime()
    self._pauseLevelTime = false

    PauseableCoroutineHelper:Resume(PauseableCoroutineHelper.Any)
    PauseableCoroutineHelper:Resume(PauseableCoroutineHelper.Level)

    if self._pauseableTweenCollection then
        for tween, _ in pairs(self._pauseableTweenCollection) do
            local t = tween
            if t:IsComplete() then
                self._pauseableTweenCollection[tween] = nil
            elseif not t:IsPlaying() then
                t:Play()
            end
        end
    end
end

---@brief 等待事件中恢复/暂停关卡逻辑
---@type fun(isResume:boolean):void
---@param isResume boolean
function GuideNewContext:TryResumeOrPauseInWaitEvent(isResume)
    self._stepInWaittingState = isResume

    if isResume then
        PauseableCoroutineHelper:Resume(PauseableCoroutineHelper.Level)
    else
        PauseableCoroutineHelper:Pause(PauseableCoroutineHelper.Level)
    end

    if not self._pauseableTweenCollection then
        return
    end

    for tween, resumeInWaitState in pairs(self._pauseableTweenCollection) do
        local t = tween
        if t:IsComplete() then
            self._pauseableTweenCollection[tween] = nil
        elseif (isResume and resumeInWaitState and self._stepInWaittingState) and not t:IsPlaying() then
            t:Play()
        elseif not isResume and t:IsPlaying() then
            t:Pause()
        end
    end
end

--endregion

--region 【动画管理】

---@brief 注册可暂停动画
---@type fun(sq:DG.Tweening.Sequence,resumeInWaitState:boolean):void
---@param sq DG.Tweening.Sequence 待注册的动画序列
---@param resumeInWaitState boolean 是否在等待状态下恢复
function GuideNewContext:InsertTweenToPauseableTable(sq, resumeInWaitState)
    sq:AppendCallback(function()
        self:RemoveTweenFromPauseableTable(sq)
    end)

    if self._pauseLevelTime and not (resumeInWaitState and self._stepInWaittingState) then
        sq:Pause()
    end

    self._pauseableTweenCollection = self._pauseableTweenCollection or {}
    self._pauseableTweenCollection[sq] = resumeInWaitState
end

---@brief 移除非活动动画
---@type fun(sq:DG.Tweening.Sequence):void
---@param sq DG.Tweening.Sequence 待移除的动画序列
function GuideNewContext:RemoveTweenFromPauseableTable(sq)
    if not self._pauseableTweenCollection then
        return
    end
    self._pauseableTweenCollection[sq] = nil
end

---@brief 按名称移除动画
---@type fun(name:string):void
---@param name string 动画名称
function GuideNewContext:RemoveTweenFromPauseableTableByName(name)
    if not self._pauseableTweenCollection then
        return
    end
    for tween, _ in pairs(self._pauseableTweenCollection) do
        if tween.stringId == name then
            self._pauseableTweenCollection[tween] = nil
            break
        end
    end
end

--endregion

--region 【触发器创建 & 状态解析】

---@brief 创建引导触发器
---@type fun(guide:Tables.GuideList_Item):void
---@param guide Tables.GuideList_Item 引导配置
function GuideNewContext:CreateTutorialTrigger(guide)
    local state = self:ParseGuideState(guide)

    if state == EGuideState.ForceTrigger then
        self:CreateGuideForceSetup(guide)
        self:CreateGuideCondition(guide)
    elseif state == EGuideState.WaittingTrigger then
        self:CreateGuideCondition(guide)
    end

    if state == EGuideState.Lock or state == EGuideState.WaittingTrigger then
        self._levelListener = self._levelListener or require("GuideLevelListener")
        self._levelListener:AddGuideItem(guide, state)
    end
end

---@brief 创建条件触发器
---@type fun(guide:Tables.GuideList_Item):void
---@param guide Tables.GuideList_Item 引导配置
function GuideNewContext:CreateGuideCondition(guide)
    if self._conditions and self._conditions[guide] then
        PrintInfo("不能重复创建触发器：" .. guide.ID)
        return
    end
    self._conditions = self._conditions or {}

    local status, result = pcall(function()
        return require(string.format('GuideCondition%s', guide.TriggerType))
    end)

    if type(result) == 'table' then
        local condition = result:Create()
        condition:SetupGuideItem(guide)
        condition:StartTriggerListener()
        self._conditions[guide] = condition
        PrintInfo(string.format("Tutorial->Condition %s: %s is waitting ...", guide.ID, guide.TriggerType))
    else
        PrintInfo(string.format("%s 找不到合适的触发装置", guide.TriggerType))
    end
end

---@brief 创建强制设置器
---@type fun(guide:Tables.GuideList_Item):void
---@param guide Tables.GuideList_Item 引导配置
function GuideNewContext:CreateGuideForceSetup(guide)
    if not guide.GuidePreparation or guide.GuidePreparation == "" then
        return
    end
    if self._setter and self._setter[guide] then
        PrintInfo("不能重复创建设置器：" .. guide.ID)
        return
    end
    self._setter = self._setter or {}

    local setup = nil
    if guide.GuidePreparation[EGuidePreparationType.Food] then
        setup = require('GuideForceSetupFood'):Create()
    elseif guide.GuidePreparation[EGuidePreparationType.Patience] then
        setup = require('GuideForceSetupGuest'):Create()
    end

    if setup then
        setup:SetupConfig(guide)
        setup:StartListener()
        self._setter[guide] = setup
    end
end

---@brief 引导是否已完成
---@type fun(guideID:string):boolean
---@param guideID string 引导ID
---@return boolean 是否已完成
function GuideNewContext:IsTutorialComplete(guideID)
    local completed = SaveRootData.GameData.guidesCompleted
    local cfg = Tables.GuideList[guideID]

    if cfg.GuideConnect then
        for _, connect in ipairs(cfg.GuideConnect) do
            if string.match(completed, connect) then
                return true
            end
        end
    end
    return string.match(completed, guideID) ~= nil
end

---@brief 解析引导状态
---@type fun(guideItem:Tables.GuideList_Item):EGuideState
---@param guideItem Tables.GuideList_Item 引导配置
---@return EGuideState 引导状态枚举
function GuideNewContext:ParseGuideState(guideItem)
    if self:IsTutorialComplete(guideItem.ID) then
        return EGuideState.Executed
    end

    local chapterLv = GamerDataManager:GetLatestChapterLevel()
    local forceLv = guideItem.GuidePreparationLevel

    if forceLv and forceLv.Chapter == chapterLv.Chapter and forceLv.Level == chapterLv.Level then
        return EGuideState.ForceTrigger
    end

    if forceLv and (forceLv.Chapter < chapterLv.Chapter or
            (forceLv.Chapter == chapterLv.Chapter and forceLv.Level < chapterLv.Level)) then
        return EGuideState.GiveUp
    end

    local unlockLv = guideItem.UnlockLevel
    if unlockLv and (unlockLv.Chapter < chapterLv.Chapter or
            (unlockLv.Chapter == chapterLv.Chapter and unlockLv.Level <= chapterLv.Level)) then
        return EGuideState.WaittingTrigger
    end

    return EGuideState.Lock
end

--region 【步骤执行 & 遮罩】

---@brief 生成引导步骤列表
---@type fun(guideItem:Tables.GuideList_Item):Tables.GuideContent_Item[]
---@param guideItem Tables.GuideList_Item 引导配置
---@return Tables.GuideContent_Item[] 引导步骤列表
function GuideNewContext:GenerateGuideSteps(guideItem)
    local steps = {}
    for _, step in pairs(Tables.GuideContent) do
        if step.GuideID == guideItem.ID then
            EasyInsert(steps, step, "StepID")
        end
    end
    return steps
end

---@brief 引导是否允许点击检测（如收金币）
---@type fun():boolean
---@return boolean
function GuideNewContext:IsTutorialAllowCheck()
    if not self._stepWorker then return false end
    local currStep = self._stepWorker:GetCurrent()
    return currStep and currStep.Type == EGuideOneStepBehavioutType.Coin
end

---@brief 引导遮罩屏蔽点击
---@type fun(screenSpacePosition:UnityEngine.Vector3):boolean
---@param screenSpacePosition UnityEngine.Vector3 屏幕空间坐标
---@return boolean 是否屏蔽点击
function GuideNewContext:IsTutorialBlockClicks(screenSpacePosition)
    if not self._stepWorker then return false end

    local currMask = self._stepWorker:GetScreenSpaceMaskBounds()
    if not currMask then return false end
    if math.abs(currMask.size.x) < 0.0001 and math.abs(currMask.size.y) < 0.0001 then
        return false
    end

    local step = self._stepWorker:GetCurrent()
    if step and step.Type == EGuideOneStepBehavioutType.Info then
        return true
    end

    local halfSize = currMask.size / 2
    local min = currMask.center - halfSize
    local max = currMask.center + halfSize

    return screenSpacePosition.x < min.x or screenSpacePosition.y < min.y
            or screenSpacePosition.x > max.x or screenSpacePosition.y > max.y
end

--region 【触发 & 完成流程】

---@brief 强制设置完成
---@type fun(setter:GuideForceSetup):void
---@param setter GuideForceSetup 设置器实例
function GuideNewContext:ForceSetupComplete(setter)
    if self._setter and self._setter[setter.guideItem] then
        self._setter[setter.guideItem]:CloseListener()
    end
end

---@brief 执行一帧内优先级最高的引导
---@type fun():void
function GuideNewContext:ExecuteTriggedTutorial()
    local condition = self._waittingExecute
    local param = self._waittingExecuteParam

    PrintInfo(string.format("Tutorial->Condition: %s %s is triggering ...", condition.guideItem.ID, condition.guideItem.TriggerType))

    if self._levelListener then
        self._levelListener:RemoveGuideItem(condition.guideItem)
    end
    if self._conditions and self._conditions[condition.guideItem] then
        self._conditions[condition.guideItem]:CloseTriggerListener()
    end
    if self._setter and self._setter[condition.guideItem] then
        self._setter[condition.guideItem]:CloseListener()
    end

    self._preconditioning:Preconditioning(condition.guideItem)

    self._stepWorker = self._stepWorker or require("GuideStepExecuteContext"):Create()
    local steps = self:GenerateGuideSteps(condition.guideItem)
    self._executing = self._executing or {}
    self._executing[steps] = condition
    self._stepWorker:ExecuteGuideSteps(steps, param)

    self._coroutine = nil
    self._waittingExecute = nil
    self._waittingExecuteParam = nil
end

---@brief 触发引导
---@type fun(condition:GuideCondition,param:table):void
---@param condition GuideCondition 触发条件
---@param param table 触发参数
function GuideNewContext:TriggerTriggering(condition, param)
    if self._executing and not TableIsEmpty(self._executing) then
        PrintWarning(string.format("Tutorial->Repeated triggering: %s", condition.guideItem.ID))
        condition:OnGiveupTriggering(false)
        return
    end

    if condition.guideItem.EnvCheck and not self._envChecker:EnvCheck(condition.guideItem.EnvCheck) then
        PrintWarning(string.format("Tutorial->EvnCheck failed: %s", condition.guideItem.ID))
        condition:OnGiveupTriggering(true)
        return
    end

    local currOrder = condition.guideItem.GuideOrder or 0
    local prevOrder = self._waittingExecute and self._waittingExecute.guideItem.GuideOrder or 0

    if not self._waittingExecute or prevOrder < currOrder then
        self._waittingExecute = condition
        self._waittingExecuteParam = param
    end

    if not self._coroutine then
        self._coroutine = HonorCoroutineHelper:StartAlone(function()
            coroutine.yield(CoroutineHelper.waitForEndOfFrame)
            self:ExecuteTriggedTutorial()
        end)
    end
end

---@brief 步骤执行完成
---@type fun(steps:Tables.GuideContent_Item[]):void
---@param steps Tables.GuideContent_Item[] 引导步骤列表
function GuideNewContext:GuideStepsExecuteComplete(steps)
    local condition = self._executing[steps]
    if condition then
        local guideItem = condition.guideItem
        PrintInfo(string.format("Tutorial->Condition %s execute over", guideItem.ID))

        if self._conditions and self._conditions[guideItem] then
            self._conditions[guideItem]:Dispose()
            self._conditions[guideItem] = nil
        end
        if self._setter and self._setter[guideItem] then
            self._setter[guideItem]:Dispose()
            self._setter[guideItem] = nil
        end

        self._preconditioning:RollbackPreconditioningConfig(guideItem)
        self._executing[steps] = nil
    end

    self._pauseLevelTime = false
    self._stepInWaittingState = false

    if self._conditions and TableIsEmpty(self._conditions) then
        self._conditions = nil
    end
    if self._setter and TableIsEmpty(self._setter) then
        self._setter = nil
    end
end

---@brief 标记引导完成并写入存档
---@type fun(steps:Tables.GuideContent_Item[]):void
---@param steps Tables.GuideContent_Item[] 引导步骤列表
function GuideNewContext:GuideStepsExecutedCompleteTagStep(steps)
    local condition = self._executing[steps]
    if not condition then return end

    local completed = SaveRootData.GameData.guidesCompleted or ""
    if string.match(completed, condition.guideItem.ID) then
        return
    end

    if condition.guideItem.GuideConnect and #condition.guideItem.GuideConnect > 0 then
        SaveRootData.GameData.guidesCompleted = completed .. table.concat(condition.guideItem.GuideConnect, '_')
        self:CloseConnectTutorial(condition.guideItem.GuideConnect, condition.guideItem.ID)
    else
        SaveRootData.GameData.guidesCompleted = completed .. condition.guideItem.ID
    end

    self:CheckDataFormat()
    SaveIO.GameData.Save_guidesCompleted()
end

---@brief 关闭关联引导
---@type fun(tutorialStrs:string[],currTutorial:string):void
---@param tutorialStrs string[] 关联引导ID数组
---@param currTutorial string 当前引导ID
function GuideNewContext:CloseConnectTutorial(tutorialStrs, currTutorial)
    for _, str in ipairs(tutorialStrs) do
        if str ~= currTutorial then
            for key, condition in pairs(self._conditions or {}) do
                if condition.guideItem.ID == str then
                    condition:Dispose()
                    condition:CloseTriggerListener()
                    self._conditions[key] = nil
                    PrintInfo(string.format("Tutorial->%s 同步关闭 %s", currTutorial, str))
                end
            end
        end
    end
end

---@brief 移除已完成引导（GM/调试）
---@type fun(removeId:string):void
---@param removeId string 待移除的引导ID
function GuideNewContext:RemoveCompletedTutorial(removeId)
    self:CheckDataFormat()
    local completed = SaveRootData.GameData.guidesCompleted
    local remove = string.format("_%s_", removeId)
    local result = completed:gsub(remove, "_")

    if string.len(completed) - string.len(result) == string.len(remove) - 1 then
        SaveRootData.GameData.guidesCompleted = result
        SaveIO.GameData.Save_guidesCompleted()
    end

    local guide = Tables.GuideList[removeId]
    if guide and (TableIsEmpty(self._conditions or {}) or not self._conditions[guide]) then
        self:CreateTutorialTrigger(guide)
    end
end
