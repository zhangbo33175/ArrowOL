---@class GameMainState : GameStateBase @游戏主界面状态
local GameStateBase = require("GameStateBase")
local GameMainState = class('GameMainState', GameStateBase)

--region 【构造 & 实例化】

---=========================================================================
--- 构造函数
---@brief 状态实例构造
---@param stateType GameStateCmd 状态枚举类型
---=========================================================================
function GameMainState:ctor(stateType)
    GameMainState.super.ctor(self, stateType)
end

---=========================================================================
--- 创建状态实例对外工厂入口
---@brief 状态工厂创建接口
---@param stateType GameStateCmd 状态枚举类型
---@return GameMainState 状态实例对象
---=========================================================================
function GameMainState:Create(stateType)
    return GameMainState.new(stateType)
end

--endregion

--region 【状态进入】

---=========================================================================
--- 进入主界面状态（状态机入口）
---@brief 状态机进入回调，初始化触摸、遮罩、背景音乐，调用内部初始化
---@param finishCallback function|nil 进入完成回调函数
---=========================================================================
function GameMainState:OnStateEnter(finishCallback)
    PrintInfo("OnStateEnter --> GameState: {0} ", tostring(self:GetStateType()))

    -- 关闭多点触摸
    GameLib.Util.SetMultiTouchEnabled(false)
    -- 添加UI遮罩，防止未初始化完成时用户操作
    GameMainRoot.UI:AddUIMaskLayerRef("GameMainState:OnStateEnter")

    -- 本地活动热更资源状态更新
    --LocalActivityManager:UpdateActivityHotfixResStates()
    -- 尝试开启本地活动
    --LocalActivityManager:TryOpenActivities()

    -- 播放章节主界面背景音乐
    local chapterId = MapManager.CurLoadChapterId
    if chapterId then
        local chapterData = ChapterDataManager:GetChapterDataByChapterType(chapterId)
        if chapterData and chapterData.MapMusic and chapterData.MapMusic.outgame then
            SoundManager:PlayBgm(chapterData.MapMusic.outgame)
        end
    end

    -- 执行主界面内部初始化逻辑
    self:DoStateEnter(finishCallback)
end

--endregion

--region 【主界面初始化】

---=========================================================================
--- 主界面真正初始化逻辑
---@brief 打开主UI、解除遮罩、登录逻辑、弹窗处理、关卡返回特殊分支
---@param finishCallback function|nil 完成回调函数
---=========================================================================
function GameMainState:DoStateEnter(finishCallback)
    -- 打开主界面HUD
    OpenUISync(UIs.UIMainStateView)
    -- 移除UI遮罩，允许用户交互
    GameMainRoot.UI:SubUIMaskLayerRef()
    -- 主HUD内执行登录流程
    LoginManager:TryLoginInMainHud()
    -- 弹出本地活动弹窗
    LocalActivityManager:TryOpenPopupActivityUI()

    -- UI推送队列切换为主界面模式
    GameMainUIPushManager:OnEnterMainState()
    -- 关闭场景物体自动点击选择
    GameManager:SetupEasyTouchAutoSelect(false)

    -- 安全执行完成回调
    if finishCallback then
        xpcall(finishCallback, __G__TRACK_BACK__)
    end
end

--endregion

--region 【状态退出 & 更新】

---=========================================================================
--- 离开主界面状态
---@brief 状态机退出回调，清理特效、音乐、推送队列、关闭主UI
---@param finishCallback function|nil 退出完成回调函数
---=========================================================================
function GameMainState:OnStateExit(finishCallback)
    PrintInfo("OnStateExit --> GameState: {0} ", tostring(self:GetStateType()))

    -- 清理全局特效
    GameEffectManager:ClearData()
    -- 停止当前背景音乐
    SoundManager:StopCurrentBgm()
    -- UI推送队列离开主界面状态
    GameMainUIPushManager:OnLeaveMainState()
    -- 关闭主界面HUD
    CloseUI(UIs.UIMainStateView)

    -- 安全执行回调，使用xpcall携带堆栈捕获，统一项目规范
    if finishCallback then
        xpcall(finishCallback, __G__TRACK_BACK__)
    end
end

--endregion

return GameMainState
