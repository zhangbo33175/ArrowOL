---@class GameMainState : GameStateBase 游戏主界面状态
local GameStateBase = require("GameStateBase")
local GameMainState = class('GameMainState', GameStateBase)

---构造函数
---@type fun(stateType:GameStateCmd):void
---@param stateType GameStateCmd 状态类型
function GameMainState:ctor(stateType)
    GameMainState.super.ctor(self, stateType)
end

---创建状态实例
---@type fun(stateType:GameStateCmd):GameMainState
---@param stateType GameStateCmd 状态类型
---@return GameMainState 状态实例
function GameMainState:Create(stateType)
    return GameMainState.new(stateType)
end

---进入主界面状态（入口）
---@type fun(finishCallback:function):void
---@param finishCallback function 进入完成回调
function GameMainState:OnStateEnter(finishCallback)
    PrintInfo("OnStateEnter --> GameState: {0} ", tostring(self:GetStateType()))

    --- 关闭多点触摸
    GameLib.Util.SetMultiTouchEnabled(false)
    --- 添加UI遮罩，防止操作
    GameMainRoot.UI:AddUIMaskLayerRef("GameMainState:OnStateEnter")

    --- 本地活动热更资源状态更新
    LocalActivityManager:UpdateActivityHotfixResStates()
    --- 尝试开启本地活动
    LocalActivityManager:TryOpenActivities()

    --- 播放章节主界面背景音乐
    local chapterData = ChapterDataManager:GetChapterDataByChapterId(MapManager.CurLoadChapterId)
    if chapterData and chapterData.MapMusic and chapterData.MapMusic.outgame then
        SoundManager:PlayBGM(chapterData.MapMusic.outgame)
    end

    --- 相机定位到主界面
    self:_LocateCamera(finishCallback)
end

---相机定位总入口
---@type fun(finishCallback:function):void
---@param finishCallback function 回调函数
function GameMainState:_LocateCamera(finishCallback)
    self:_LocateNormalShowChapterMap(finishCallback)
end

---首次进入非第一章地图（备用逻辑）
---@type fun(finishCallback:function):void
---@param finishCallback function 回调函数
function GameMainState:_LocateFirstShowChapterMap(finishCallback)
    local mapData = MapManager.CurMapData
    GameMapMotion:PauseMapMotion()
end

---正常回到章节地图（主逻辑）
---@type fun(finishCallback:function):void
---@param finishCallback function 回调函数
function GameMainState:_LocateNormalShowChapterMap(finishCallback)
    local mapData = MapManager.CurMapData

    --- 如果相机已经在正确位置，直接进入
    if GameLib.MapCamHelper.IsEqualMapCamData(mapData.normalCamData) then
        GameMainRoot.Effect.efCamera.orthographicSize = mapData.normalCamData.size
        self:DoStateEnter(finishCallback)
    else

        --    GameLib.MapCamHelper.LookMap(mapData.normalCamData, Tables.Constants.CAM_LOOK_MAP_ANIM_TIME.Value.Main, function()
        --        self:DoStateEnter(finishCallback)
        --    end)
        --end)
    end
end

---主界面真正初始化逻辑（相机到位后执行）
---@type fun(finishCallback:function):void
---@param finishCallback function 回调函数
function GameMainState:DoStateEnter(finishCallback)
    --- 打开主界面HUD
    OpenUISync(UIs.UIMainHud)
    --- 移除遮罩
    GameMainRoot.UI:SubUIMaskLayerRef()

    --- 登录流程
    LoginManager:TryLoginInMainHud()
    --- 每日签到
    DailyRewardManager:TrySignin()
    --- 道具促销
    BoosterSaleManager:TryOpenBoosterSaleUI()
    --- 刷新升级时间
    UpgradeMapManager:RefreshTimeStamp()

    --- 本地活动弹窗
    LocalActivityManager:TryOpenPopupActivityUI()

    --- 如果从关卡胜利回来，自动打开升级界面
    local preStateType = GameStateMachine:PreStateType()
    if preStateType and
            preStateType == GameStateCmd.PlayingState and
            GameCoreManager:GetGameOverType() ~= EGameOverType.PlayerGiveUp then
        UpgradeMapMgr:CheckAndAutoOpenUpgradeUI(ChapterListManager:GetRuntimeChapterId())
    end

    --- UI推送队列进入主界面状态
    GameMainUIPushManager:OnEnterMainState()
    --- 关闭自动点击选择
    GameManager:SetupEasyTouchAutoSelect(false)
    --- 创建地图自治实体（客人、NPC等）
    MapManager.CurGameMapBase:CreateAutonomousEntitys()

    if finishCallback then
        xpcall(finishCallback, __G__TRACK_BACK__)
    end
end

---离开主界面状态
---@type fun(finishCallback:function):void
---@param finishCallback function 退出完成回调
function GameMainState:OnStateExit(finishCallback)
    PrintInfo("OnStateExit --> GameState: {0} ", tostring(self:GetStateType()))

    --- 清理特效
    EffectManager:ClearData()
    --- 停止BGM
    SoundManager:StopCurBGM()
    --- 停止升级计时
    UpgradeMapManager:StopTimeStamp()
    --- 推送队列离开主界面
    GameMainUIPushManager:OnLeaveMainState()
    --- 关闭主界面HUD
    CloseUI(UIs.UIMainHud)

    pcall(finishCallback)
end

---帧更新
---@type fun():void
function GameMainState:OnUpdate()
    GameMainState.super.OnUpdate(self)
    --- 地图移动更新
    GameMapMotion:OnUpdate()
end

return GameMainState
