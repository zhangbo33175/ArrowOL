---@class GamePlayingState : GameStateBase 游戏运行状态
local GamePlayingState = class('GamePlayingState', import('GameStateBase'))

--region 【构造 & 实例化】
---@brief 构造函数
---@type fun(stateType:GameStateCmd):void
---@param stateType GameStateCmd 状态类型
function GamePlayingState:ctor(stateType)
    GamePlayingState.super.ctor(self, stateType)
end

---@brief 创建状态实例
---@type fun(stateType:GameStateCmd):GamePlayingState
---@param stateType GameStateCmd 状态类型
---@return GamePlayingState 状态实例
function GamePlayingState:Create(stateType)
    local obj = GamePlayingState.new(stateType)
    return obj
end
--endregion

--region 【状态进入 - 关卡正式开始】
---@brief 进入游戏运行状态
---@type fun(finishCallback:function):void
---@param finishCallback function 进入完成回调
function GamePlayingState:OnStateEnter(finishCallback)
    -- 开启多点触摸
    GameLib.Util.SetMultiTouchEnabled(true)
    -- 增加UI遮罩，防止操作
    GameMainRoot.UI:AddUIMaskLayerRef("GamePlayingState:OnStateEnter")
    -- 清空UI推送队列
    GameMainUIPushManager:ClearWorkQueue()
    -- 开启点击检测
    GameManager:SetupEasyTouchAutoSelect(true)
    -- 暂停地图拖动
    GameMapMotion:PauseMapMotion()

    -- 相机移动到关卡战斗位置
    local mapData = MapManager.CurMapData
    GameLib.MapCamHelper.LookMap(
            mapData.m_PlayCamData,
            1.3,
            function()
                -- 相机动画完成后的回调
                local closeUICallback = function()
                    -- 触发引导事件
                    GuideNewContext:TryDispatchEvent(EventCmd.OnLevelShowGoalComplete)
                end

                -- 移除UI遮罩
                GameMainRoot.UI:SubUIMaskLayerRef()
                -- 恢复地图交互
                GameMapMotion:ResumeMapMotion()
                -- 打开游戏内核心UI
                OpenUISync(UIs.UIMainPlaying)
            end
    )

    -- 执行完成回调
    if finishCallback then
        xpcall(finishCallback, __G__TRACK_BACK__)
    end
end
--endregion

--region 【状态退出 - 退出关卡】
---@brief 离开游戏运行状态（退出关卡）
---@type fun(finishCallback:function):void
---@param finishCallback function 退出完成回调
function GamePlayingState:OnStateExit(finishCallback)
    -- 清空核心管理器数据
    GameCoreManager._GameCoreManagerEffect:ClearData()
    -- 清理特效、停止音乐
    GameEffectManager:ClearData()
    SoundManager:StopCurBGM()

    -- 关闭战斗UI
    CloseUI(UIs.UIPlayingHud)
    CloseUI(UIs.UIPlayingHudCollect)
    CloseUI(UIs.UILevelSuccessBg)

    -- 退出游戏逻辑
    PlayHudCycle:OnAction(EPlayHudCycleAction.ExitGame)

    -- 执行完成回调
    if finishCallback then
        xpcall(finishCallback, __G__TRACK_BACK__)
    end
end
--endregion

return GamePlayingState
