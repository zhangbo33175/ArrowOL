---@class GameLoadMapState : GameLoadMapStateBase @游戏加载地图状态
local GameLoadMapState = class('GameLoadMapState', import('GameLoadMapStateBase'))

--region 【构造 & 实例化】

---=========================================================================
---@brief 构造函数
---@param stateType GameStateCmd 状态枚举类型
---=========================================================================
function GameLoadMapState:ctor(stateType)
    GameLoadMapState.super.ctor(self, stateType)
end

---=========================================================================
---@brief 创建状态实例对外工厂入口
---@param stateType GameStateCmd 状态枚举类型
---@return GameLoadMapState 状态实例对象
---=========================================================================
function GameLoadMapState:Create(stateType)
    local obj = GameLoadMapState.new(stateType)
    return obj
end

--endregion

--region 【状态生命周期】

---=========================================================================
---@brief 进入加载地图状态
---@param finishCallback function|nil 进入完成回调函数
---=========================================================================
function GameLoadMapState:OnStateEnter(finishCallback)
    GameLoadMapState.super.OnStateEnter(self, finishCallback)
end

---=========================================================================
---@brief 离开加载地图状态
---@param finishCallback function|nil 退出完成回调函数
---=========================================================================
function GameLoadMapState:OnStateExit(finishCallback)
    GameLoadMapState.super.OnStateExit(self, finishCallback)
end

--endregion

--region 【加载界面控制】

---=========================================================================
---@brief 打开地图加载UI
---=========================================================================
function GameLoadMapState:OpenLoadingUI()
    OpenUISync(UIs.UILoadMapView)
end

---=========================================================================
---@brief 关闭地图加载UI
---=========================================================================
function GameLoadMapState:CloseLoadingUI()
    CloseUI(UIs.UILoadMapView)
end

--endregion

--region 【地图加载完成逻辑】

---=========================================================================
---@brief 地图加载完毕入口，分流新手首次游戏流程 / 普通正常流程
---@param finishCallback function|nil 加载完成回调函数
---=========================================================================
function GameLoadMapState:LoadMapOver(finishCallback)
    local chapterId = ChapterListManager:GetRuntimeChapterId()

    -- 新手首次游戏流程
    if GamePlayHelper:IsPlayFirstLevel() then
        local levelId = GamerDataManager:GetLevelLvByChapterId(chapterId)
        -- 补满体力
        ItemManager:FullLife()
        -- 加载关卡数据
        local levelData = GamePlayHelper:LoadLevelData(chapterId, levelId)
        if GamePlayHelper:PlayLevel(levelData) then
            -- 切换至战斗游玩状态
            GameStateMachine:ChangeState(GameStateCmd.PlayingState, finishCallback)
        else
            -- 失败保护：回退到主界面状态
            GameStateMachine:ChangeState(GameStateCmd.MainState, finishCallback)
            PrintError("[GameLoadMapState] 默认进入第一关失败，改为进入主页面")
        end
    else
        -- 正常流程：直接进入战斗游玩状态
        GameStateMachine:ChangeState(GameStateCmd.PlayingState, finishCallback)
        --GameStateMachine:ChangeState(GameStateCmd.MainState, finishCallback)
    end
end

--endregion

--region 【准备加载地图】

---=========================================================================
---@brief 准备加载地图，设置运行时章节ID
---@param chapterId string 要加载的章节ID
---=========================================================================
function GameLoadMapState:OnReadyToLoadMap(chapterId)
    ChapterListManager:SetRuntimeChapter(chapterId)
end

--endregion

return GameLoadMapState
