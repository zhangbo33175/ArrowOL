---@class UILoginView : LuaBehaviourSuper
---@field cs Honor.Runtime.LuaBehaviour LuaBehaviour 组件
---@field LoginBtn UnityEngine.UI.Button 登录按钮
local UILoginView = class('UILoginView', import('LuaBehaviourSuper'))

---@brief 构造函数
---@param args table 自定义参数
function UILoginView:ctor(args)
    UILoginView.super.ctor(self, args)
end

---@brief 创建 UI 实例
---@param args table 自定义参数
---@return UILoginView
function UILoginView:Create(args)
    return UILoginView.new(args)
end

---@brief 注册事件监听
function UILoginView:OnAddListeners()
    UILoginView.super.OnAddListeners(self)
end

---@brief 唤醒
function UILoginView:Awake()
    UILoginView.super.Awake(self)
    AddUIListenerFunction(self.LoginBtn, 'Clicked', handler(self, self.OnLoginBtnClicked))
end

---@brief 开始
function UILoginView:Start()
    UILoginView.super.Start(self)
end

---@brief 销毁
function UILoginView:OnDestroy()
    OnRemoveListener(self.LoginBtn, 'Clicked', handler(self, self.OnLoginBtnClicked))
    UILoginView.super.OnDestroy(self)
end

---@brief 播放打开动画
function UILoginView:OnPlayOpenAnimation()
    UILoginView.super.OnPlayOpenAnimation(self)
end

---@brief 播放关闭动画
function UILoginView:OnPlayCloseAnimation()
    UILoginView.super.OnPlayCloseAnimation(self)
end

---@brief 子UI销毁回调
---@param luaClass XLua.LuaTable
function UILoginView:OnAddedUIDestroyed(luaClass)
    UILoginView.super.OnAddedUIDestroyed(self, luaClass)
end

---@brief 登录按钮点击回调
---@param args any 事件参数
function UILoginView:OnLoginBtnClicked(args)
    GameStateMachine:ChangeState(GameStateCmd.ChooseLevel, nil)
end

return UILoginView
