--region 【枚举定义】

---@class ESelectType 存档选择类型
---@field None number 无
---@field Local number 使用本地存档
---@field Server number 使用云端存档
ESelectType = {
    None   = 0,
    Local  = 1,
    Server = 2,
}

---@class ELoginState 登录流程状态
---@field None number 无状态
---@field Working number 登录中
---@field Finish number 登录完成
---@field MultiLogin number 异地登录冲突
---@field SelectData number 选择存档（本地/云端）
ELoginState = {
    None       = 0,
    Working    = 1,
    Finish     = 2,
    MultiLogin = 3,
    SelectData = 4,
}

--endregion

---@class LoginManager : Singleton 登录管理器（全局单例）
---@field _loginState ELoginState @当前登录流程状态
---@field _isOnline boolean @是否在线（登录成功后置 true）
---@field _loginFinishCallback function|nil @登录结束外部回调（登录完成后一次性触发并置空）
---@field _isNewUser boolean @是否新用户
---@field _isDelUser boolean @是否已注销/删除账号
local LoginManager = class("LoginManager", import("Singleton"))

--region 【构造 & 初始化】

---@brief 构造函数
---@desc 框架驱动；初始化登录状态与各标志位默认值
---@param args table 自定义参数
function LoginManager:ctor(args)
    LoginManager.super.ctor(self, args)

    self._loginState         = ELoginState.None    ---@type ELoginState 当前登录状态
    self._isOnline           = false               ---@type boolean 是否在线
    self._loginFinishCallback = nil               ---@type function 登录结束回调
    self._isNewUser          = false               ---@type boolean 是否新用户
    self._isDelUser          = false               ---@type boolean 是否已删除账号
end

---@brief 创建单例
---@desc 框架驱动，由单例系统实例化
---@param args table 自定义参数
---@return LoginManager
function LoginManager:Create(args)
    return LoginManager.new(args)
end

---@brief 初始化（注册事件）
---@desc 框架驱动；注册第三方账号绑定/解绑结果事件
function LoginManager:Init()
    LoginManager.super.Init(self)
    HonorEventHelper:AddListener(EventCmd.OnAidBindingResult, handler(self, self.OnAidBindingResult))
    HonorEventHelper:AddListener(EventCmd.OnAidUnBindingResult, handler(self, self.OnAidUnBindingResult))
end

---@brief 清理数据
---@desc 登录开始前/切号时重置登录状态与标志位（保留 _isDelUser 语义由外部维护）
function LoginManager:ClearData()
    LoginManager.super.ClearData(self)

    self._loginState   = ELoginState.None
    self._isOnline     = false
    self._isNewUser    = false
    self._isDelUser    = false
end

--endregion

--region 【登录流程】

---@brief 登录完成统一回调
---@desc 由 LoginBackFun/ErrorCodeDiagnosis 内部调用；置在线与完成状态、派发 OnLoginResult 事件并触发外部回调
---@param isSuccess boolean 是否成功
function LoginManager:LoginFinishCallback(isSuccess)
    PrintInfo("Login 登陆结束: LoginFinish --> isSuccess: {0} ", isSuccess)

    self._isOnline   = isSuccess
    self._loginState = ELoginState.Finish

    HonorEventHelper:FireNow(EventCmd.OnLoginResult, isSuccess)

    if self._loginFinishCallback then
        xpcall(self._loginFinishCallback, __G__TRACK_BACK__)
    end

    self._loginFinishCallback = nil
end

---@brief 登录入口
---@desc 外部/网络流程调用；Working 状态下非强制登录被拒，随后清空状态并置 Working
---@param force boolean 强制登录
---@param finishCallback function 结束回调
function LoginManager:Login(force, finishCallback)
    PrintInfo("Login 登陆开始: LoginStart --> force: {0} ", force)

    if not force and self._loginState == ELoginState.Working then
        PrintInfo("Login 正在登陆中,不允许重复登陆")
        return
    end

    self:ClearData()
    self._loginState         = ELoginState.Working
    self._loginFinishCallback = finishCallback
end

---@brief 登录网络回调（核心逻辑）
---@desc 网络登录响应回调；按 error_code 分支处理新用户/老用户、存档冲突、异地登录与数据同步
---@param req table 请求对象
---@param resp table 响应对象
---@param cmd table 命令对象
---@param returnData table 返回数据
function LoginManager:LoginBackFun(req, resp, cmd, returnData)
    if returnData ~= nil and cmd ~= nil then
        local errorCode = returnData["error_code"]

        if errorCode == NetCode.Success[1] then
            self._isOnline = true
            SaveDataSyncHelper:SaveToken(returnData)
            HearBeatMgr:BeginHeartBeat()

            local isNewUser = cmd.Request.Body.Content.is_new_user == 1 or returnData.is_new_user == true

            if isNewUser then
                self._isNewUser = true

                SaveDataSyncHelper:SaveLocalDataAndAutoSyncData(returnData, function()
                    self:LogLoginBack(req, resp, cmd, returnData)
                    self:LoginFinishCallback(true)
                end)
            else
                if returnData.ServerUserInfo ~= nil and
                        not string.isNullOrEmpty(returnData.ServerUserInfo.last_sync_device) and
                        not string.isNullOrEmpty(returnData.ServerUserInfo.server_game_data)
                then
                    local serverGameData = JsonDecode(returnData.ServerUserInfo.server_game_data)
                    local isSameDevice   = returnData.ServerUserInfo.last_sync_device == HonorRoot.SDK.TGAHelper:GetDeviceID()
                    local isSameUid      = returnData.ServerUserInfo.server_user_id == SaveRootData.UserData.uid

                    if (not isSameUid or not isSameDevice) and serverGameData then
                        self:OpenSelectData(returnData.ServerUserInfo.last_sync_time, serverGameData, returnData.ServerUserInfo.server_user_id, function(selectType)
                            self._loginState = ELoginState.SelectData

                            if selectType == ESelectType.Local then
                                if SaveRootData.UserData.uid == 0 then
                                    HintMarkerHelper:OnSaveChoose(selectType)
                                    self:Login(true, self._loginFinishCallback)
                                else
                                    SaveDataSyncHelper:SaveLocalDataAndAutoSyncData(returnData, function()
                                        HintMarkerHelper:OnSaveChoose(selectType)
                                        self:LogLoginBack(req, resp, cmd, returnData)
                                        self:LoginFinishCallback(true)
                                    end)
                                end
                            elseif selectType == ESelectType.Server then
                                local localUid  = SaveRootData.UserData.uid
                                local uid       = returnData.ServerUserInfo.server_user_id
                                local aid       = returnData.ServerUserInfo.server_account
                                local platform  = returnData.channel_type

                                SaveDataSyncHelper:SaveServerUserInfoToLocalByGameData(serverGameData, uid, aid, platform)
                                HintMarkerHelper:OnSaveChoose(selectType)

                                if isSameUid then
                                    UploadingGameDataHelper:UploadAllGameData(function()
                                        local procedure = HonorRoot.Procedure.CurrentProcedure
                                        if not self:TurnToProLoadStateByCurProcedure(procedure, localUid, uid) then
                                            self:Login(false, self._loginFinishCallback)
                                        end
                                    end)
                                else
                                    self:SyncOtherUidDeviceId(localUid, uid, function()
                                        local procedure = HonorRoot.Procedure.CurrentProcedure
                                        if not self:TurnToProLoadStateByCurProcedure(procedure, localUid, uid) then
                                            self:Login(false, self._loginFinishCallback)
                                        end
                                    end)
                                end
                            end
                        end)
                    else
                        SaveDataSyncHelper:SaveLocalDataAndAutoSyncData(returnData)
                        self:LogLoginBack(req, resp, cmd, returnData)
                        self:LoginFinishCallback(true)
                    end
                else
                    self._isNewUser = true

                    if SaveRootData.UserData.uid <= 0 then
                        SaveDataSyncHelper:SaveLocalDataAndAutoSyncData(returnData, function()
                            self:LogLoginBack(req, resp, cmd, returnData)
                            self:LoginFinishCallback(true)
                        end)
                    else
                        PrintInfo("Login 服务端没存档，error_code ==0 ,且本地有uid > 0，服务器有对应的uid但是没有存档数据 " .. SaveRootData.UserData.uid)

                        SaveDataSyncHelper:SaveLocalDataAndAutoSyncData(returnData, function()
                            self:LogLoginBack(req, resp, cmd, returnData)
                            self:LoginFinishCallback(true)
                        end)
                    end
                end
            end
        else
            self:ErrorCodeDiagnosis(errorCode, req, resp, cmd, returnData)
        end
    else
        self:LogLoginBack(req, resp, cmd, returnData)
        self:LoginFinishCallback(false)
    end
end

---@brief 登录错误码处理
---@desc 由 LoginBackFun 内部调用；按错误码区分无效UID重登、异地登录冲突弹窗、其余失败直接回调
---@param errorCode number 错误码
---@param req table 请求对象
---@param resp table 响应对象
---@param cmd table 命令对象
---@param returnData table 返回数据
function LoginManager:ErrorCodeDiagnosis(errorCode, req, resp, cmd, returnData)
    if errorCode == NetCode.InvalidUID[1] or errorCode == NetCode.NewInvalidUID[1] then
        self._loginState = ELoginState.None
        self:Login(true, self._loginFinishCallback)

    elseif errorCode == NetCode.UIDSignToOtherDevices[1] then
        if not GamePlayHelper:IsInLevelState() then
            self._loginState = ELoginState.MultiLogin
            self:OpenLoginConflict()
        else
            self:LoginFinishCallback(false)
        end
    else
        self:LogLoginBack(req, resp, cmd, returnData)
        self:LoginFinishCallback(false)
    end
end

---@brief 登录/注册打点
---@desc 由 LoginBackFun/ErrorCodeDiagnosis 内部调用；按错误码与是否新用户上报注册/登录埋点
---@param req table 请求对象
---@param resp table 响应对象
---@param cmd table 命令对象
---@param returnData table 返回数据
function LoginManager:LogLoginBack(req, resp, cmd, returnData)
    local statusCode = -1
    local isReg = false

    if cmd then
        isReg = cmd.Request.Body.Content.is_new_user == 1 or returnData.is_new_user == true
    end

    if resp ~= nil then
        statusCode = resp.StatusCode
    end

    if returnData ~= nil then
        if returnData.error_code == NetCode.Success[1] then
            if isReg then
                HintMarkerHelper:OnAccountReg(1, statusCode)
            end
            HintMarkerHelper:OnAccountLogin(1, statusCode)
        else
            if isReg then
                HintMarkerHelper:OnAccountReg(0, statusCode)
            else
                HintMarkerHelper:OnAccountLogin(0, statusCode)
            end
        end
    else
        if isReg then
            HintMarkerHelper:OnAccountReg(0, statusCode)
        else
            HintMarkerHelper:OnAccountLogin(0, statusCode)
        end
    end
end

---@brief 打开异地登录弹窗
---@desc 由 ErrorCodeDiagnosis 在异地登录冲突且非关卡内时调用

end

---@brief 打开存档选择UI
---@desc 由 LoginBackFun 在检测到跨设备/跨UID云端存档时调用，本地/云端存档选择结果经回调回传
---@param lastSaveTime number 上次存档时间
---@param serverGameData table 云端存档数据
---@param serverUid number 云端用户ID
---@param callback fun(selectType:number):void 选择回调
function LoginManager:OpenSelectData(lastSaveTime, serverGameData, serverUid, callback)
    local localData = SaveRootData.GameData

    OpenUISync(UIs.UISelectData, {
        localData             = localData,
        serverData            = serverGameData,
        serverUid             = serverUid,
        lastServerDataSaveTime = lastSaveTime,
        selectCallback        = callback
    })
end

---@brief 根据当前流程判断是否需要重启
---@desc 由 LoginBackFun/SyncOtherUidDeviceId 在切号后调用；非预加载流程则切到预加载状态并清单例，否则仅清单例
---@param procedure table 当前流程对象
---@param oldUid number 原UID
---@param newUid number 新UID
---@return boolean @是否触发了流程切换
function LoginManager:TurnToProLoadStateByCurProcedure(procedure, oldUid, newUid)
    if procedure and procedure.Name ~= "ProcedurePreload" then
        GameStateMachine:ChangeState(GameStateCmd.TurnToPreloadState, function()
            HonorEventHelper:FireNow(EventCmd.OnReloadGameBefore, oldUid, newUid)
            SingletonManager:ClearAllSingleData()
            procedure:PrepareToNextProcedure(typeof(Honor.ProcedurePreload))
        end)
        return true

    elseif procedure and procedure.Name == "ProcedurePreload" then
        HonorEventHelper:FireNow(EventCmd.OnReloadGameBefore, oldUid, newUid)
        SingletonManager:ClearAllSingleData()
        return false
    end

    return false
end

--region 【账号绑定】

---@brief 绑定结果回调
---@desc 事件注册回调（Init 中 OnAidBindingResult 注册）；按平台与成败打开对应成功/失败弹窗并上报埋点
---@param success boolean 是否成功
---@param platform number 平台
---@param aid string 账号ID
---@param position number 埋点位置
function LoginManager:OnAidBindingResult(success, platform, aid, position)
    if success then
        if platform == EAidPlatform.Apple then
            OpenUISync(UIs.UIAppleConnectSuccess)
        else
            OpenUISync(UIs.UIFBConnectSuccess)
        end
    else
        if platform == EAidPlatform.Apple then
            OpenUISync(UIs.UIAppleConnectFailure)
        else
            OpenUISync(UIs.UIFBConnectFailure)
        end
    end

    HintMarkerHelper:OnThirdPartyAccountLogin(TrackProperty.LoginTime.login, success, position, aid, platform)
end

---@brief 解绑结果回调
---@desc 事件注册回调（Init 中 OnAidUnBindingResult 注册）；上报第三方账号登出埋点
---@param success boolean 是否成功
---@param platform number 平台
---@param aid string 账号ID
---@param position number 埋点位置
function LoginManager:OnAidUnBindingResult(success, platform, aid, position)
    HintMarkerHelper:OnThirdPartyAccountLogin(TrackProperty.LoginTime.logout, success, position, aid, platform)
end

---@brief 同步其他UID设备ID
---@desc 由 LoginBackFun 在跨UID登录分支调用；发送同步设备ID网络请求，结果经回调回传
---@param uid number 本机UID
---@param otherUid number 对方UID
---@param callback fun(success:boolean):void 完成回调
function LoginManager:SyncOtherUidDeviceId(uid, otherUid, callback)
    local netCallback = function(req, resp, cmd, returnData)
        if returnData and returnData.error_code == NetCode.Success[1] and returnData.status == NetCode.Success[1] then
            pcall(callback, true)
        else
            pcall(callback, false)
        end
    end

    if self:CanSendNetMsg(false) then
        SaveDataSyncHelper:CmdSyncOtherUidDeviceId(uid, nil, otherUid, netCallback)
    else
        pcall(callback, false)
    end
end
