---播放"左右晃动 + 碎裂消失"动画（每个 FishItem 独立播放）
---@brief 用 DOTween 序列实现：本地旋转左右往复摆动数下 -> 缩放至 0 碎裂 -> 回调
---@desc FishItem 父节点为 HorizontalLayoutGroup（会覆盖子节点位置），故左右晃动改用旋转摆动实现，布局不会干预；
---      target 显式传入该 FishItem 自身的 Transform，保证每只鱼独立作用于自己的节点，互不干扰
---@param fishTransform UnityEngine.Transform 该 FishItem 自身的 Transform
---@param startDelay number 启动延迟（秒），用于多只鱼错峰逐只播放动画
---@param onComplete function|nil 动画完成后回调（由调用方决定是否销毁）
--TODO: [修复] PlayBreakAnimation 改为显式传入目标 Transform 并支持错峰延迟，确保每只鱼各自播放摇晃+碎裂动画
function FishItem:PlayBreakAnimation(fishTransform, startDelay, onComplete)
    if IsNull(fishTransform) then
        if onComplete then onComplete(self) end
        return
    end

    --TODO: [诊断] 打印每只鱼动画开始，用于确认"每只鱼是否都实际播放碎裂动画"
    PrintInfo("[FishItem] 播放碎裂动画 name=" .. fishTransform.name .. " delay=" .. tostring(startDelay))

    -- 清除目标上可能残留的旧补间，避免与打开动画叠加
    DOTween.Kill(fishTransform.gameObject, true)

    local baseZ      = fishTransform.localEulerAngles.z -- 初始旋转
    local swingDeg   = 14                                 -- 左右摆动角度（度）
    local swingDur   = 0.06                               -- 单次摆动时长
    local shatterDur = 0.25                               -- 碎裂缩放时长

    local sequence = DOTween.Sequence()
    -- 错峰延迟：多只鱼时逐只错开启动，形成"每只鱼各自晃动碎裂消失"的独立效果
    if startDelay and startDelay > 0 then
        sequence:AppendInterval(startDelay)
    end
    -- 左右往复摆动 3 个来回（旋转式晃动）
    for i = 1, 3 do
        sequence:Append(fishTransform:DOLocalRotate(CS.UnityEngine.Vector3(0, 0, baseZ + swingDeg), swingDur):SetEase(Ease.InOutSine))
        sequence:Append(fishTransform:DOLocalRotate(CS.UnityEngine.Vector3(0, 0, baseZ - swingDeg), swingDur):SetEase(Ease.InOutSine))
    end
    -- 摆回初始角度
    sequence:Append(fishTransform:DOLocalRotate(CS.UnityEngine.Vector3(0, 0, baseZ), swingDur):SetEase(Ease.InOutSine))
    -- 碎裂：缩放到 0（InBack 带轻微过冲，读作碎裂消失）
    sequence:Append(fishTransform:DOScale(0, shatterDur):SetEase(Ease.InBack))
    sequence:OnComplete(function()
        --TODO: [诊断] 打印动画完成，用于确认每只鱼动画正常结束
        PrintInfo("[FishItem] 碎裂动画完成 name=" .. fishTransform.name)
        if onComplete then onComplete(self) end
    end)
end
