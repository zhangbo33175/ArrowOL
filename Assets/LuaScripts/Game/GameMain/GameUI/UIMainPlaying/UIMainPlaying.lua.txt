---唤醒
---@brief 框架驱动；逐条注册按钮交互监听到 Manager 业务动作（业务逻辑统一在 Manager 中处理）；
---@desc 监听经 self:AddUIListener 记录，界面销毁时由 LuaBehaviourSuper:OnDestroy 自动注销
---@desc 回调统一使用 Bind(self, method) 绑定（对齐 HonorUI `self:AddUIListener(eventCode, Bind(self, _OnX))` 结构）
--TODO: [优化] 按钮监听由 AddUIListenerBatch 批量改为 self:AddUIListener 逐条（自动注销，对齐 UIAnnalsView 模式）
function UIMainPlaying:Awake()
    UIMainPlaying.super.Awake(self)
    -- 绑定按钮 → Manager 业务动作（业务逻辑统一在 Manager 中处理）
    --TODO: [音效] 按钮点击统一播放开始音效（保持 Bind 结构表 {target,method,func}，仅替换 func：
    --       先播放 UI_SeasonStart 再执行业务逻辑；不可返回裸函数，否则 AddListener/RemoveListener 访问
    --       handler.target/method/func 时会报 "attempt to index a function value"）
    local function withSeasonStart(handler)
        return {
            target = handler.target,
            method = handler.method,
            func = function(...)
                SoundManager:PlaySound("UI_SeasonStart")
                return handler.func(...)
            end,
        }
    end
    self:AddUIListener(self.SettingButton, 'Clicked', withSeasonStart(Bind(UIMainPlayingManager, UIMainPlayingManager.OpenSettingView)))
    self:AddUIListener(self.SearchButton, 'Clicked', withSeasonStart(Bind(UIMainPlayingManager, UIMainPlayingManager.StartAutoScan)))
    self:AddUIListener(self.xianshi, 'Clicked', withSeasonStart(Bind(UIMainPlayingManager, UIMainPlayingManager.ShowBackground)))
    self:AddUIListener(self.yinchang, 'Clicked', withSeasonStart(Bind(UIMainPlayingManager, UIMainPlayingManager.HideBackground)))

    -- 给Top和Down节点添加透明Image拦截点击穿透
    self:AddRaycastBlocker("Top")
    self:AddRaycastBlocker("Down")

    -- 挂载 Manager：注册业务事件监听并初始化界面数据
    UIMainPlayingManager:OnViewOpened(self)
end

---获取 FishItem 挂载父节点（LifeTransform）
---@brief 兼容注入字段为 Transform 或 GameObject 两种形态（预制体存在同名双注入），
---      注入为空时按名称递归查找，最终兜底到自身节点
---@desc 由 UIMainPlayingManager 创建/清除 FishItem 时调用
---@return UnityEngine.Transform 挂载父节点（Transform）
function UIMainPlaying:GetFishParent()
    local parent = self.LifeTransform
    -- GameObject 注入时取其 transform；Transform 注入时 transform 为其自身，统一为 Transform
    if not IsNull(parent) and parent.transform then
        parent = parent.transform
    end
    -- 注入为空时按名称递归查找兜底
    if IsNull(parent) then
        parent = FindChildTransform(self.cs.transform, "LifeTransform")
    end
    -- 最终兜底：挂载到自身节点
    if IsNull(parent) then
        PrintWarning("[UIMainPlaying] 未找到 LifeTransform 挂载节点，FishItem 将挂载到自身节点下")
        parent = self.cs.transform
    end
    return parent
end

---给指定节点添加透明Image拦截点击穿透
---@brief 优先使用注入字段，其次在自身节点、根节点下按名称查找，找到后添加透明 Image 拦截点击
---@desc 由 Awake 调用，为 Top/Down 节点加透明 Image 拦截点击穿透，避免穿透到下层地图
---@param nodeName string @节点名
function UIMainPlaying:AddRaycastBlocker(nodeName)
    local node = nil

    -- 1. 优先使用框架自动绑定字段（如 self.Top）
    if self[nodeName] ~= nil then
        local bound = self[nodeName]
        -- 绑定字段可能是 Transform 或 GameObject，统一取 transform
        if bound.gameObject then
            node = bound.transform
        else
            node = bound
        end
    end

    -- 2. 在 self.cs.transform 下直接查找
    if not node then
        node = self.cs.transform:Find(nodeName)
    end

    -- 3. 向上遍历到根节点，再从根查找该子节点（Top/Down 是根节点的直接子节点）
    if not node then
        local root = self.cs.transform
        while root.parent do
            root = root.parent
        end
        node = root:Find(nodeName)
    end

    if not node then
        print("[RaycastBlocker] not found: "..nodeName)
        return
    end

    -- 添加透明 Image 拦截点击（如果已有 Image 则只确保 raycastTarget=true）
    local img = node:GetComponent(typeof(UnityEngine.UI.Image))
    if not img then
        img = node.gameObject:AddComponent(typeof(UnityEngine.UI.Image))
        img.color = UnityEngine.Color(1, 1, 1, 0)
    end
    img.raycastTarget = true
    print("[RaycastBlocker] OK on "..nodeName.." ("..node.name..")")
end
