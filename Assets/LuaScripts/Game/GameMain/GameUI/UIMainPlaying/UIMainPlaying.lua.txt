---@class UIMainPlaying : LuaBehaviourSuper
---@field cs Honor.Runtime.LuaBehaviour @LuaBehaviour
---@field Top                                UnityEngine.RectTransform     √                        Top
---@field Right                              UnityEngine.RectTransform     √                        Right
---@field CompleteNumber                     TMPro.TextMeshProUGUI         √                        目标完成显示
---@field LifeTransform                      UnityEngine.Transform         √                        当前体力显示
---@field UsedNumber                         TMPro.TextMeshProUGUI         √                        显示道具次数
---@field SettingButton                      UnityEngine.UI.Button         √         Clicked        设置按钮
---@field SearchButton                       UnityEngine.UI.Button         √         Clicked        搜索按钮
---@field Bg                                 UnityEngine.UI.Image          √                        界面bg图
local UIMainPlaying = class('UIMainPlaying', import('LuaBehaviourSuper'))

---构造函数
---@type fun(args:table):void
---@param args table @自定义参数
function UIMainPlaying:ctor(args)
    UIMainPlaying.super.ctor(self, args)
    self:Init()
end

---创建函数
---@type fun(args:table):UIMainPlaying
---@param args table @自定义参数
---@return UIMainPlaying @实例
function UIMainPlaying:Create(args)
    local obj = UIMainPlaying.new(args)
    return obj
end

---注册监听（自动注销）
---@type fun():void
function UIMainPlaying:OnAddListeners()
    UIMainPlaying.super.OnAddListeners(self)

end

---唤醒
---@type fun():void
function UIMainPlaying:Awake()
    UIMainPlaying.super.Awake(self)
    -- 2=====================================================================================================
    AddUIListenerFunction(self.SettingButton, 'Clicked', handler(self, self.OnSettingButtonClicked))
    AddUIListenerFunction(self.SearchButton, 'Clicked', handler(self, self.OnSearchButtonClicked))
-- ====================================================================================================2
    HonorEventHelper:AddListener(EventCmd.LifeTransformation, handler(self, self.OnSelectedResult))
end

function UIMainPlaying:OnSelectedResult(data)
    if data then
        self:OnSetCompleteNumber()
    else
        if  GameCoreManager:IsLifeEnough() then  
            self.LifeNumber.text = GameCoreManager:GetLifeNumber()
        else
            OpenUISync(UIs.UILevelFailedView)
        end
    end

    for i = 1, MainPlayingManager:GetLifeNumber() do
        
        
    end
    
    
    
end


---初始化逻辑
---@type fun():void
function UIMainPlaying:Init()
    self:OnSetCompleteNumber()
    self:OnSetUsedNumber()
  
end

function UIMainPlaying:OnSetCompleteNumber()
    self.CompleteNumber.text =  MainPlayingManager:GetCharacterAndChooseListCount()
end

function UIMainPlaying:OnSetUsedNumber()
    self.UsedNumber.text = MainPlayingManager:GetNumberOfPropsUsed()
end

function UIMainPlaying:OnSetLifeNumber()
    self.LifeNumber.text = MainPlayingManager:GetLifeNumber()
end


---开始
---@type fun():void
function UIMainPlaying:Start()
    UIMainPlaying.super.Start(self)
    self:Init()
end

---销毁
---@type fun():void
function UIMainPlaying:OnDestroy()
    UIMainPlaying.super.OnDestroy(self)
    -- 3=====================================================================================================
    OnRemoveListener(self.SettingButton, 'Clicked', handler(self, self.OnSettingButtonClicked))
    OnRemoveListener(self.SearchButton, 'Clicked', handler(self, self.OnSearchButtonClicked))
-- ====================================================================================================3
    HonorEventHelper:RemoveListener(EventCmd.LifeTransformation, handler(self, self.OnSelectedResult))
end

---播放打开动画（可重写）
---@type fun():void
function UIMainPlaying:OnPlayOpenAnimation()
    UIMainPlaying.super.OnPlayOpenAnimation(self)
end

---播放关闭动画（可重写）
---@type fun():void
function UIMainPlaying:OnPlayCloseAnimation()
    UIMainPlaying.super.OnPlayCloseAnimation(self)
end

---子UI销毁回调
---@type fun(luaClass:XLua.LuaTable):void
---@param luaClass XLua.LuaTable
function UIMainPlaying:OnAddedUIDestroyed(luaClass)
    UIMainPlaying.super.OnAddedUIDestroyed(self, luaClass)
end

---设置按钮-交互回调
---@type fun(args:any):void
function UIMainPlaying:OnSettingButtonClicked(args)
    OpenUIAsync(UIs.UISettingView)
end

---搜索按钮-交互回调
---@type fun(args:any):void
function UIMainPlaying:OnSearchButtonClicked(args)
    GameMapMotion:MoveToCenter(MainPlayingManager:GetCharacterLocalPosition(),0.02)
end

return UIMainPlaying
