---@class UIRankView : LuaBehaviourSuper
---@field cs Honor.Runtime.LuaBehaviour @LuaBehaviour
---@field CloseButton                          UnityEngine.UI.Button         √         Clicked        关闭排行榜
---@field RankRoot                             UnityEngine.Transform         √                        榜单列表容器（RankItem 的父节点）
---@field PlayerRankText                       TMPro.TextMeshProUGUI         √                        我的排名
---@field PlayerScoreText                      TMPro.TextMeshProUGUI         √                        我的分数
local UIRankView = class('UIRankView', import('LuaBehaviourSuper'))

---构造函数
---@type fun(args:table):void
---@param args table @自定义参数
function UIRankView:ctor(args)
    UIRankView.super.ctor(self, args)

end

---创建函数
---@type fun(args:table):UIRankView
---@param args table @自定义参数
---@return UIRankView @实例
function UIRankView:Create(args)
    local obj = UIRankView.new(args)
    return obj
end

---注册监听（自动注销）
---@type fun():void
function UIRankView:OnAddListeners()
    UIRankView.super.OnAddListeners(self)

end

---唤醒
---@type fun():void
function UIRankView:Awake()
    UIRankView.super.Awake(self)
-- 2=====================================================================================================
    AddUIListenerFunction(self.CloseButton, 'Clicked', handler(self, self.OnCloseButtonClicked))
-- ====================================================================================================2

end

---初始化逻辑（刷新排行榜数据展示）
---@type fun():void
function UIRankView:Init()
    self:RefreshPlayerInfo()
    self:RefreshRankList()
end

---开始
---@type fun():void
function UIRankView:Start()
    UIRankView.super.Start(self)
    self:Init()
end

---销毁
---@type fun():void
function UIRankView:OnDestroy()
    UIRankView.super.OnDestroy(self)
-- 3=====================================================================================================
    OnRemoveListener(self.CloseButton, 'Clicked', handler(self, self.OnCloseButtonClicked))
-- ====================================================================================================3

end

---播放打开动画（可重写）
---@type fun():void
function UIRankView:OnPlayOpenAnimation()
    UIRankView.super.OnPlayOpenAnimation(self)
end

---播放关闭动画（可重写）
---@type fun():void
function UIRankView:OnPlayCloseAnimation()
    UIRankView.super.OnPlayCloseAnimation(self)
end

---子UI销毁回调
---@type fun(luaClass:XLua.LuaTable):void
---@param luaClass XLua.LuaTable
function UIRankView:OnAddedUIDestroyed(luaClass)
    UIRankView.super.OnAddedUIDestroyed(self, luaClass)
end

---关闭排行榜-交互回调
---@type fun(args:any):void
---@param args any @UI交互内容
function UIRankView:OnCloseButtonClicked(args)
    self:Close()
end

---刷新玩家自身排名与分数展示
---@type fun():void
function UIRankView:RefreshPlayerInfo()
    -- 我的排名
    if self.PlayerRankText then
        local rank = RankDataManager:GetPlayerRank()
        self.PlayerRankText.text = rank > 0 and tostring(rank) or "--"
    end
    -- 我的分数
    if self.PlayerScoreText then
        self.PlayerScoreText.text = tostring(RankDataManager:GetPlayerScore())
    end
end

---刷新排行榜列表（清空旧项后按榜单数据创建 RankItem 子项）
---@type fun():void
function UIRankView:RefreshRankList()
    self:ClearRankRoot()

    if not UIs or not UIs.RankItem then
        PrintError("[UIRankView] 缺少排行榜子项配置 UIs.RankItem，请通过资源导出工具生成")
        return
    end

    local rankList = RankDataManager:GetRankList()
    for _, rankData in ipairs(rankList) do
        self:AddUISync(UIs.RankItem, self.RankRoot, { data = rankData })
    end
end

---清空榜单容器下所有旧子项
---@type fun():void
function UIRankView:ClearRankRoot()
    if not self.RankRoot then
        return
    end
    local childCount = self.RankRoot.childCount
    for i = childCount - 1, 0, -1 do
        local child = self.RankRoot:GetChild(i)
        if child and child.gameObject then
            UnityEngine.Object.Destroy(child.gameObject)
        end
    end
end

return UIRankView
