---@type fun():void

--TODO: [优化] 按钮监听由 AddUIListenerBatch 改为 self:AddUIListener 逐条（自动注销）

        HonorCoroutineHelper:WaitForSecondsRealtime(0.05)
        if GameMainRoot.HonorUIManager and GameMainRoot.HonorUIManager.FadeOutLoadingViewThenClose then
            GameMainRoot.HonorUIManager:FadeOutLoadingViewThenClose()
        end
    end)

--region 【重置章节/等级按钮（运行时动态创建，调试用）】

---创建"重置章节/等级为1"按钮（运行时动态挂载到UI根节点右上角）
---@brief 运行时动态创建调试用"重置章节"按钮（幂等，Start 中调用）
---@type fun():void
---@desc 点击后将存档最新章节/当前章节/所有章节等级统一重置为1，便于联调从第一章重新开始。
---      按钮为纯代码动态创建（GameObject+RectTransform+Image+Button+Text），无需改动Prefab。
function UIMainStateViewView:CreateResetButton()
    if not self.cs or IsNull(self.cs.gameObject) then
        return
    end
    local parent = self.cs.gameObject.transform
    -- 幂等：已存在同名按钮则跳过，避免界面多次打开重复创建
    if parent:Find("AorResetLevelBtn") then
        return
    end

    -- 按钮根节点
    local go = UnityEngine.GameObject("AorResetLevelBtn")
    go.transform:SetParent(parent, false)
    local rect = go:AddComponent(typeof(UnityEngine.RectTransform))
    -- 背景（必须带Graphic才能参与UI射线拾取）
    local img = go:AddComponent(typeof(UnityEngine.UI.Image))
    img.color = UnityEngine.Color(0.2, 0.5, 0.8, 0.85)
    -- 按钮组件
    local btn = go:AddComponent(typeof(UnityEngine.UI.Button))
    btn.targetGraphic = img
    btn.onClick:AddListener(function()
        self:OnResetLevelButtonClicked()
    end)

    -- 按钮文字
    local textGo = UnityEngine.GameObject("Text")
    textGo.transform:SetParent(go.transform, false)
    local text = textGo:AddComponent(typeof(UnityEngine.UI.Text))
    text.text = "重置"
    text.fontSize = 22
    text.alignment = UnityEngine.TextAnchor.MiddleCenter
    text.color = UnityEngine.Color.white
    text.font = UnityEngine.Resources.GetBuiltinResource(typeof(UnityEngine.Font), "LegacyRuntime.ttf")
    local textRect = text.rectTransform
    textRect.anchorMin = UnityEngine.Vector2.zero
    textRect.anchorMax = UnityEngine.Vector2.one
    textRect.offsetMin = UnityEngine.Vector2.zero
    textRect.offsetMax = UnityEngine.Vector2.zero

    -- 布局：贴右上角（设置按钮下方），避免遮挡现有UI
    rect.anchorMin = UnityEngine.Vector2(1, 1)
    rect.anchorMax = UnityEngine.Vector2(1, 1)
    rect.pivot = UnityEngine.Vector2(1, 1)
    rect.anchoredPosition = UnityEngine.Vector2(-12, -70)
    rect.sizeDelta = UnityEngine.Vector2(120, 46)
end

---重置按钮-交互回调：将存档章节与等级全部重置为1
---@brief 调试按钮点击回调：重置存档章节与等级为 1 并刷新界面文案
---@type fun():void
---@desc 更新 latestChapterId/curChapterId/全部章节 lv=1 并持久化，运行时章节回退为"1"，刷新主界面当前关卡文案
function UIMainStateViewView:OnResetLevelButtonClicked()
    SoundManager:PlaySound("UI_SeasonStart")
    -- 最新解锁章节 → 1
    SaveRootData.GameData.latestChapterId = "1"
    -- 当前游玩章节 → 1
    SaveRootData.GameData.curChapterId = "1"
    -- 全部章节等级 → 1
    local chapterInfos = SaveRootData.GameData.chapterInfos
    if chapterInfos then
        for _, info in pairs(chapterInfos) do
            if type(info) == "table" then
                info.lv = 1
            end
        end
    end
    -- 持久化存档
    SaveIO.GameData.SaveAll()
    -- 运行时章节回退为第一章节
    ChapterListManager:SetRuntimeChapterId("1")
    -- 刷新主界面当前关卡文案（LevelManager:GetChapterLevel）
    if not IsNull(self.LevelTxt) and self:GetViewModel() then
        self.LevelTxt.text = self:GetViewModel():GetChapterLevel()
    end
    PrintInfo("[UIMainStateView] 重置成功：章节与等级已重置为 1")
end
