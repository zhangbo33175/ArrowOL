    self:Init()

---@brief 初始化拼图布局：显示单图节点、隐藏四图节点
---@type fun():void
function SectionItem:Init()
    if self.OnePicturePuzzle then self.OnePicturePuzzle:SetActive(true) end
    if self.FourPicturePuzzle then self.FourPicturePuzzle:SetActive(false) end
end

---开始
---@brief 开始时批量生成 16 个收集子项（框架驱动，勿删）
---@type fun():void
function SectionItem:Start()
    SectionItem.super.Start(self)

    -- 按传输的数字加载对应个数的 CollectItem 并挂载到 Content
    self:CreateCollectItems(16)
end

---@brief 销毁回调（框架驱动，勿删）
---@type fun():void

--region 【子项创建】

---按数量循环加载 CollectItem 预制体并挂载到 Content 下
---@brief 在 Content 节点下实例化多个 CollectItem
---@desc 根据传入数量 count，循环在 Content 下创建 count 个 CollectItem；
---      第 i 个 CollectItem 通过 params.data.number = i 传入拼图序号，由其加载对应拼图。
---      若注入的 Content 未赋值，回退挂载到自身节点下并打印提示。
---@param count number @CollectItem 实例数量
function SectionItem:CreateCollectItems(count)
    local assetInfo = LoadRes["GameObject_CollectItem"]
    if assetInfo == nil then
        PrintError("[SectionItem] 缺少资源定义 GameObject_CollectItem")
        return
    end

    -- 解析挂载父节点：注入 → 固定路径 → 递归查找名为 Content 的子节点，最后回退自身节点
    local parent = self.ContentTransform
    for i = 1, count do
        -- 挂载到父节点下，并携带当前拼图序号
        local params = { data = { number = i } }
        CreatePrefabLuaClassSync(assetInfo, parent, params)
    end
end

--endregion
