-- @brief     Spine 动画控制器工具类（收拢所有 spine 动画播放方法）
-- @desc      统一管理 Spine 动画播放、延迟播放、动画时长查询、显示-待机-隐藏整套流程
-- @core      支持 SkeletonAnimation（场景内）和 SkeletonGraphic（UI 内）两种组件
-- @scope     全局通用，业务代码统一通过此工具类调用 Spine 动画
-- @dependency SpineUtil（底层实现）, UnityUtil（空值判断）
--=====================================================================================================

---@class SpineAnimationHelper @Spine 动画控制器工具类
local SpineAnimationHelper = {}

--region 依赖引入
local SpineUtil = require('SpineUtil')
local UnityUtil = require('UnityUtil')
--endregion

--region 动画时长查询

---@brief 获取 Spine.SkeletonAnimation 动画时长（场景内 Spine）
---@param spineAnim Spine.UnityEngine.SkeletonAnimation Spine动画组件
---@param animName string 动画名称
---@return number 动画时长（秒）
function SpineAnimationHelper.GetAnimationDuration(spineAnim, animName)
    return SpineUtil.GetSpineAnimationDuration(spineAnim, animName)
end

---@brief 获取 Spine.SkeletonGraphic 动画时长（UI Spine）
---@param skeletonGraphic Spine.UnityEngine.SkeletonGraphic UI Spine组件
---@param animName string 动画名称
---@return number 动画时长（秒）
function SpineAnimationHelper.GetGraphicAnimationDuration(skeletonGraphic, animName)
    return SpineUtil.GetSpineGraphicAnimationDuration(skeletonGraphic, animName)
end

---@brief 判断是否存在指定 Spine 动画
---@param skeletonGraphic Spine.UnityEngine.SkeletonGraphic UI Spine组件
---@param animName string 动画名称
---@return boolean 是否存在该动画
function SpineAnimationHelper.HasAnimation(skeletonGraphic, animName)
    return SpineUtil.HasSpineAnimation(skeletonGraphic, animName)
end

--endregion

--region 动画播放控制

---@brief 播放 Spine 动画（UI Spine）
---@param skeletonGraphic Spine.UnityEngine.SkeletonGraphic UI Spine组件
---@param animName string 动画名称
---@param loop boolean 是否循环
---@return Spine.TrackEntry 动画轨道信息
function SpineAnimationHelper.Play(skeletonGraphic, animName, loop)
    return SpineUtil.PlaySpineAnimation(skeletonGraphic, animName, loop)
end

---@brief 延迟播放 Spine 动画（UI Spine）
---@param skeletonGraphic Spine.UnityEngine.SkeletonGraphic UI Spine组件
---@param animName string 动画名称
---@param delay number 延迟时间（秒）
---@param loop boolean 是否循环
---@return Spine.TrackEntry 动画轨道信息
function SpineAnimationHelper.PlayDelayed(skeletonGraphic, animName, delay, loop)
    return SpineUtil.DelayPlaySpineAnimation(skeletonGraphic, animName, delay, loop)
end

---@brief 播放场景内 Spine 动画（SkeletonAnimation）
---@param spineAnim Spine.UnityEngine.SkeletonAnimation Spine动画组件
---@param animName string 动画名称
---@param loop boolean 是否循环
---@param trackIndex number 轨道索引，默认 0
---@return Spine.TrackEntry 动画轨道信息
function SpineAnimationHelper.PlayScene(spineAnim, animName, loop, trackIndex)
    if UnityUtil.IsNull(spineAnim) then return nil end
    trackIndex = trackIndex or 0
    return spineAnim.AnimationState:SetAnimation(trackIndex, animName, loop)
end

---@brief 延迟播放场景内 Spine 动画（SkeletonAnimation）
---@param spineAnim Spine.UnityEngine.SkeletonAnimation Spine动画组件
---@param animName string 动画名称
---@param delay number 延迟时间（秒）
---@param loop boolean 是否循环
---@param trackIndex number 轨道索引，默认 0
---@return Spine.TrackEntry 动画轨道信息
function SpineAnimationHelper.PlaySceneDelayed(spineAnim, animName, delay, loop, trackIndex)
    if UnityUtil.IsNull(spineAnim) then return nil end
    trackIndex = trackIndex or 0
    return spineAnim.AnimationState:AddAnimation(trackIndex, animName, loop, delay)
end

---@brief 清空所有轨道动画
---@param skeletonGraphic Spine.UnityEngine.SkeletonGraphic UI Spine组件
function SpineAnimationHelper.ClearTracks(skeletonGraphic)
    if UnityUtil.IsNull(skeletonGraphic) or UnityUtil.IsNull(skeletonGraphic.AnimationState) then return end
    skeletonGraphic.AnimationState:ClearTracks()
end

---@brief 设置动画播放速度
---@param skeletonGraphic Spine.UnityEngine.SkeletonGraphic UI Spine组件
---@param timeScale number 播放速度（1=正常，2=两倍速，0.5=半速）
function SpineAnimationHelper.SetTimeScale(skeletonGraphic, timeScale)
    if UnityUtil.IsNull(skeletonGraphic) then return end
    skeletonGraphic.timeScale = timeScale
end

--endregion

--region 整套动画流程（显示 → 待机 → 隐藏）

---@brief 播放 Spine 显示 → Idle → 隐藏 整套动画
---@param spine Spine.UnityEngine.SkeletonGraphic UI Spine组件
---@param idleDuration number Idle停留时长（秒）
---@param showAnim string 显示动画名称
---@param idleAnim string 循环Idle动画名称
---@param hideAnim string 隐藏动画名称
---@param onHideStart function 隐藏开始回调（可选）
---@param onHideComplete function 隐藏完成回调（可选）
function SpineAnimationHelper.PlayShowIdleHide(spine, idleDuration, showAnim, idleAnim, hideAnim, onHideStart, onHideComplete)
    SpineUtil.PlaySpineShowIdleHide(spine, idleDuration, showAnim, idleAnim, hideAnim, onHideStart, onHideComplete)
end

---@brief 播放单次动画，完成后回调
---@param skeletonGraphic Spine.UnityEngine.SkeletonGraphic UI Spine组件
---@param animName string 动画名称
---@param onComplete function 动画完成回调（可选）
function SpineAnimationHelper.PlayOnce(skeletonGraphic, animName, onComplete)
    local entry = SpineUtil.PlaySpineAnimation(skeletonGraphic, animName, false)
    if entry and onComplete then
        entry:AddComplete(onComplete)
    end
    return entry
end

--endregion

--region 静态工具（不需要实例）

return SpineAnimationHelper
