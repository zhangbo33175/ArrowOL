--            核心作用：本地存档 ↔ 云端存档 双向同步，账号体系入口
-- @core      登录返回信息落盘（UID/注册时间/登录时间/国家）、鉴权 Token 缓存、登录后自动同步
-- @scope     登录流程中需要把服务端账号数据写入本地存档的环节（GameRequires 实例别名为 SaveDataSyncHelper）
-- @规范      Unity Lua 商业项目编码标准 / 注释完整 / 健壮性 / 性能优化

---@class SaveDataHelper 存档&登录&云端同步核心单例
---@field _loginToken string @ 登录鉴权 Token（登录成功后缓存，待后续网络请求使用）
local SaveDataHelper = class("SaveDataHelper")

-- 内部方法前向声明（仅本文件内部调用，见「内部方法·工具」region）
local SaveSignInInfos

--region 构造 & 实例化

---=========================================================================
---@brief 构造函数
---=========================================================================
function SaveDataHelper:ctor(args)
    ---@type string 登录鉴权 Token
    self._loginToken = ""
end

---=========================================================================
---@brief 创建单例实例
---@return SaveDataHelper
---=========================================================================
function SaveDataHelper:Create(args)
    return SaveDataHelper.new(args)
end

--endregion

--region 登录信息本地保存

---=========================================================================
---@brief 仅保存登录 Token
---@param returnData table 登录数据
---=========================================================================
function SaveDataHelper:SaveToken(returnData)
    if returnData and returnData.token then
        self._loginToken = returnData.token
    end
end

--endregion

--region 云端存档 → 本地覆盖

---=========================================================================
---@brief 登录成功后：保存信息 + 自动全量同步
---@param returnData table 登录数据
---@param completeCallback function 完成回调
---=========================================================================
function SaveDataHelper:SaveLocalDataAndAutoSyncData(returnData, completeCallback)
    SaveSignInInfos(returnData)
    if completeCallback then
        completeCallback()
    end
end

--endregion

--region 内部方法·工具

---=========================================================================
---@brief 保存登录返回的用户信息（UID、注册时间、登录时间、绑定时间）
---@desc 有副作用：按字段逐项写入 SaveRootData 并触发对应 SaveIO 落盘；登录成功后由 SaveLocalDataAndAutoSyncData 调用
---@param returnData table 登录返回数据
---=========================================================================
function SaveSignInInfos(returnData)
    if not returnData then return end

    -- 保存 UID
    if returnData.user_id and returnData.user_id > 0 then
        SaveRootData.UserData.uid = returnData.user_id
        SaveIO.UserData.Save_uid()
    end

    -- 保存注册时间
    if returnData.create_time and returnData.create_time > 0 then
        local time = returnData.create_time
        SaveRootData.UserData.regTime.utc0 = time
        SaveRootData.UserData.regTime.utc0String =
            HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", time)
        SaveRootData.UserData.regTime.locality = HonorTimerHelper:GetLocalTimeStamp(time)
        SaveRootData.UserData.regTime.localString =
            HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", SaveRootData.UserData.regTime.locality)
        SaveIO.UserData.Save_regTime()
    end

    -- 保存登录时间（上次 & 当前）
    if returnData.login_time and returnData.login_time > 0 then
        local lastLogin = SaveRootData.UserData.loginTime.current
        SaveRootData.UserData.loginTime.last.utc0 = lastLogin.utc0
        SaveRootData.UserData.loginTime.last.utc0String = lastLogin.utc0String
        SaveRootData.UserData.loginTime.last.locality = lastLogin.locality
        SaveRootData.UserData.loginTime.last.localString = lastLogin.localString

        local currLoginTime = returnData.login_time
        SaveRootData.UserData.loginTime.current.utc0 = currLoginTime
        SaveRootData.UserData.loginTime.current.utc0String =
            HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", currLoginTime)
        SaveRootData.UserData.loginTime.current.locality = HonorTimerHelper:GetLocalTimeStamp(currLoginTime)
        SaveRootData.UserData.loginTime.current.localString =
            HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", SaveRootData.UserData.loginTime.current.locality)
        SaveIO.UserData.Save_loginTime()
    end

    -- 保存国家地区
    if returnData.country then
        UserTypeMgr:TrySaveUserCountry(returnData.country)
    end
end

--endregion

return SaveDataHelper
