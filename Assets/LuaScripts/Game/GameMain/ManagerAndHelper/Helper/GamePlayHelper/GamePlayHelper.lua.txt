function GamePlayHelper:LoadNextLevel()
    local curChapterId = ChapterListManager:GetRuntimeChapterId()
    if string.isNullOrEmpty(curChapterId) then
        PrintError("[GamePlayHelper] 无法获取当前章节ID，加载下一关失败")
        return
    end

    -- 本章节内部关卡全部通关：进入下一章节第一关
    if ChapterListManager:IsChapterCompleted(curChapterId) then
        self:LoadNextChapter(curChapterId)
        return
    end

    -- 本章节还有下一内部关卡：先卸载旧图清空 CurLoadChapterId，
    -- 使 GameLoadMapStateBase:OnStateEnter 走完整重载分支，地图重建时按新 lv 更新 _CurLevelData
    MapManager:UnloadOldMap()
    MapManager:ReadyToLoadMap(curChapterId, function()
    end)
end

function GamePlayHelper:LoadNextChapter(curChapterId)
    -- 获取下一章节ID（无下一章节配置 = 已是最后一章）
    local nextChapterId = ChapterListManager:GetNextChapterId(curChapterId)
    if not nextChapterId then
        -- 已是最后一章：返回主界面关卡列表
        self:GameExitPlayHud()
        return
    end

    -- 下一章未解锁才执行解锁初始化（成功结算已推进或历史已解锁时跳过重复初始化），
    -- 保证 GameLoadMapState:LoadMapOver 的 IsLatestChapter 判定通过并进入游玩
    if not ChapterListManager:IsUnlockedChapter(nextChapterId) then
        ChapterListManager:UnlockChapter(nextChapterId)
    end
    -- 设置运行时章节（内部带配置有效性校验）
    ChapterListManager:SetRuntimeChapter(nextChapterId)
    -- 加载下一章节第一关（refuseAd=true 跳过广告）
    self:LoadMainMap(nextChapterId, function()
    end, true)
end
