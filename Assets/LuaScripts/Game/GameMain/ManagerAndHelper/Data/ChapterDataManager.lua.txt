---@class ChapterDataManager : Singleton @章节管理器
---@field Levels table|nil 主线关卡配置表（ctor 缓存，兼容旧引用）
local ChapterDataManager = class("ChapterDataManager", import("Singleton"))

--region 内部方法·工具

---@brief 获取默认最大章节ID（保底2章，仅内部使用）
---@type fun():number
---@return number 默认最大章节ID
local function GetDefaultMaxChapterID()
    return 2
end

--endregion

--region 构造 & 实例化

---@brief 构造函数
---@desc 调用父类 Singleton 构造；缓存主线关卡表引用（Init 会重建，双保险）
---@type fun():void
function ChapterDataManager:ctor()
    ChapterDataManager.super.ctor(self)
    self.Levels = Tables and Tables.MainLevels
end

---@brief 创建单例
---@type fun():ChapterDataManager
---@return ChapterDataManager 章节管理器实例
function ChapterDataManager:Create()
    return ChapterDataManager.new()
end

--endregion

--region 初始化 & 生命周期

---@brief 初始化
---@desc 构建配置表缓存引用（安全缓存：Init 时固定引用，查询走缓存避免反复访问全局 Tables；
---      配置表缺失时置 nil，由各查询做空值防护）
---@type fun():void
function ChapterDataManager:Init()
    ChapterDataManager.super.Init(self)
    self._MainLevels = Tables and Tables.MainLevels
    self._Chapters  = Tables and Tables.Chapters
end

---@brief 清理数据
---@desc 切号/重载/重置时调用，清空缓存引用（下一轮 Init 会重建）
---@type fun():void
function ChapterDataManager:ClearData()
    ChapterDataManager.super.ClearData(self)
    self._MainLevels = nil
    self._Chapters  = nil
end

--endregion

--region 内部访问·安全缓存

---@brief 获取主线关卡配置表缓存（安全缓存，优先 Init 缓存，回退 ctor Levels，空值返回 nil）
---@type fun():table|nil
---@return table|nil 主线关卡配置表
function ChapterDataManager:_GetMainLevels()
    return self._MainLevels or self.Levels
end

---@brief 获取章节配置表缓存（安全缓存，空值返回 nil）
---@type fun():table|nil
---@return table|nil 章节配置表
function ChapterDataManager:_GetChapters()
    return self._Chapters
end

--endregion

--region 关卡模式获取对应章节配置

---=========================================================================
---@brief  根据当前关卡模式获取对应章节配置
---@desc 关卡模式到章节配置映射：小节ID→章节配置（透传 GetChapterDataBySectionId）
---@type fun(SectionId:string):Tables.Chapters_Item|nil
---@param SectionId string 小节Id
---@return Tables.Chapters_Item|nil 章节配置数据
---=========================================================================
function ChapterDataManager:GetChapterDataByChapterType(SectionId)
    return self:GetChapterDataBySectionId(SectionId)
end

---=========================================================================
---@brief  根据章节ID获取主线章节配置
---@desc 走安全缓存（_MainLevels），空值防护
---@type fun(SectionId:string):Tables.Chapters_Item|nil
---@param SectionId string 小节Id
---@return Tables.Chapters_Item|nil 章节配置数据
---=========================================================================
function ChapterDataManager:GetChapterDataBySectionId(SectionId)
    if not SectionId then
        return nil
    end
    local levels = self:_GetMainLevels()
    if not levels then
        return nil
    end
    return levels[tostring(SectionId)]
end

---=========================================================================
---@brief  根据章节ID获取主线章节配置
---@desc 走安全缓存（_Chapters），空值防护
---@type fun(chapterId:string):Tables.Chapters_Item|nil
---@param chapterId string 章节Id
---@return Tables.Chapters_Item|nil 章节配置数据
---=========================================================================
function ChapterDataManager:GetChaptersDataByChapterId(chapterId)
    if not chapterId then
        return nil
    end
    local chapters = self:_GetChapters()
    if not chapters then
        return nil
    end
    return chapters[tostring(chapterId)]
end

--endregion

--region 章节ID & 区域配置

---=========================================================================
---@brief  获取配置表中最大章节ID
---@desc 遍历主线关卡表取数值 ID 最大值（保底 GetDefaultMaxChapterID）
---@type fun():number
---@return number 最大章节ID
---=========================================================================
function ChapterDataManager:GetConfigMaxChapterID()
    local max = GetDefaultMaxChapterID()
    local levels = self:_GetMainLevels()
    if not levels then
        return max
    end
    for _, chapter in pairs(levels) do
        local id = tonumber(chapter and chapter.ID)
        if id then
            max = math.max(max, id)
        end
    end
    return max
end

---=========================================================================
---@brief  获取按 ID 数值升序排列的主线关卡ID数组
---@desc 排序稳定：显式转数值后 table.sort 升序，保证章节顺序确定（供章节列表/进度推进遍历）
---@type fun():number[]
---@return number[] 升序关卡ID数组
---=========================================================================
function ChapterDataManager:GetSortedLevelIds()
    local levels = self:_GetMainLevels()
    local ids = {}
    if not levels then
        return ids
    end
    for k in pairs(levels) do
        local id = tonumber(k)
        if id then
            table.insert(ids, id)
        end
    end
    table.sort(ids)
    return ids
end

---=========================================================================
---@brief  获取按 ID 数值升序排列的章节ID数组
---@desc 排序稳定：显式转数值后升序，供章节列表稳定排序
---@type fun():number[]
---@return number[] 升序章节ID数组
---=========================================================================
function ChapterDataManager:GetSortedChapterIds()
    local chapters = self:_GetChapters()
    local ids = {}
    if not chapters then
        return ids
    end
    for k in pairs(chapters) do
        local id = tonumber(k)
        if id then
            table.insert(ids, id)
        end
    end
    table.sort(ids)
    return ids
end

---=========================================================================
---@brief  获取章节配置的解锁下一关卡号
---@desc 解锁状态读取：Chapters.UnlockNextLevel（空值防护）
---@type fun(chapterId:string):number|nil
---@param chapterId string 章节ID
---@return number|nil 可解锁下一章关卡号，未配置返回 nil
---=========================================================================
function ChapterDataManager:GetChapterUnlockNextLevel(chapterId)
    local data = self:GetChaptersDataByChapterId(chapterId)
    if not data then
        return nil
    end
    return tonumber(data.UnlockNextLevel)
end

--endregion

return ChapterDataManager
