---@class ChapterDataManager : Singleton 章节管理器
---@field Levels table 主线关卡配置表
local ChapterDataManager = class("ChapterDataManager", import("Singleton"))

--region 内部方法·工具

---@brief 获取默认最大章节ID（保底2章，仅内部使用）
---@type fun():number
---@return number 默认最大章节ID
local function GetDefaultMaxChapterID()
    return 2
end

--endregion

--region 构造 & 实例化

---@brief 构造函数
---@type fun():void
function ChapterDataManager:ctor()
    ChapterDataManager.super.ctor(self)
    self.Levels = Tables.MainLevels
end

---@brief 创建单例
---@type fun():ChapterDataManager
---@return ChapterDataManager 章节管理器实例
function ChapterDataManager:Create()
    return ChapterDataManager.new()
end

--endregion

--region 初始化 & 生命周期

---@brief 初始化
---@type fun():void
function ChapterDataManager:Init()
    ChapterDataManager.super.Init(self)
end

---@brief 清理数据
---@type fun():void
---@desc 切号/重载/重置时调用，清空缓存数据
function ChapterDataManager:ClearData()
    ChapterDataManager.super.ClearData(self)
end

--endregion

--region 关卡模式获取对应章节配置

---=========================================================================
---@brief  根据当前关卡模式获取对应章节配置
---@type fun(SectionId:string):Tables.Chapters_Item|nil
---@param SectionId string 小节Id
---@return Tables.Chapters_Item|nil 章节配置数据
---=========================================================================
function ChapterDataManager:GetChapterDataByChapterType(SectionId)
    return self:GetChapterDataBySectionId(SectionId)
end

---=========================================================================
---@brief  根据章节ID获取主线章节配置
---@type fun(SectionId:string):Tables.Chapters_Item|nil
---@param SectionId string 小节Id
---@return Tables.Chapters_Item|nil 章节配置数据
---=========================================================================
function ChapterDataManager:GetChapterDataBySectionId(SectionId)
    if not SectionId or not Tables or not Tables.MainLevels then
        return nil
    end
    return Tables.MainLevels[tostring(SectionId)]
end

---=========================================================================
---@brief  根据章节ID获取主线章节配置
---@type fun(chapterId:string):Tables.Chapters_Item|nil
---@param chapterId string 章节Id
---@return Tables.Chapters_Item|nil 章节配置数据
---=========================================================================
function ChapterDataManager:GetChaptersDataByChapterId(chapterId)
    if not chapterId or not Tables or not Tables.Chapters then
        return nil
    end
    return Tables.Chapters[tostring(chapterId)]
end

--endregion

--region 章节ID & 区域配置

---=========================================================================
---@brief  获取配置表中最大章节ID
---@type fun():number
---@return number 最大章节ID
---=========================================================================
function ChapterDataManager:GetConfigMaxChapterID()
    local max = GetDefaultMaxChapterID()
    local chapters = Tables.MainLevels
    if not chapters then return max end

    for _, chapter in pairs(chapters) do
        local id = tonumber(chapter.ID)
        if id then
            max = math.max(max, id)
        end
    end
    return max
end

--endregion

return ChapterDataManager
