--region 目标配置获取

---=========================================================================
---@brief  【核心接口】根据目标ID获取目标配置表数据
---@type fun(goalId:string|number):Tables.Goals_Item|nil
---@param goalId string | number 目标ID
---@return Tables.Goals_Item | nil 目标配置（类型、条件、奖励、参数等）
---=========================================================================
function GoalDataManager:GetGoalDataById(goalId)
    -- 安全校验
    if not goalId then
        PrintError("[GoalDataManager] 查询失败：goalId 为空")
        return nil
    end

    -- 从全局配置表查询（正式项目启用）
    local goalData = nil
    if Tables and Tables.Goals then
        goalData = Tables.Goals[tostring(goalId)]
    end

    -- 调试日志
    if not goalData then
        PrintDebug("[GoalDataManager] 未找到目标配置 ID: " .. tostring(goalId))
    end

    return goalData
end

---=========================================================================
---@brief  【扩展接口】批量获取一组目标的配置
---@type fun(goalIdList:table):table
---@param goalIdList table 目标ID数组
---@return table 配置数组
---=========================================================================
function GoalDataManager:GetGoalDataList(goalIdList)
    local result = {}
    if not goalIdList or #goalIdList == 0 then
        return result
    end

    for _, id in ipairs(goalIdList) do
        local data = self:GetGoalDataById(id)
        if data then
            table.insert(result, data)
        end
    end

    return result
end
