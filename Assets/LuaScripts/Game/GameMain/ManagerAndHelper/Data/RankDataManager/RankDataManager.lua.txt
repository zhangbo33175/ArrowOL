---@class RankDataManager : Singleton 排行榜数据管理器
local RankDataManager = class("RankDataManager", import("Singleton"))

--region 常量

---@brief 榜单最大展示条数
RankDataManager.MaxRankCount = 20

---@brief 玩家自身昵称兜底值（PlayerInfoDataManager 未设置昵称时使用）
RankDataManager.DefaultPlayerName = "我"

---@brief 玩家分数持久化标识（独立于 GameDataPB，避免侵入自动生成存档结构）
RankDataManager.PersistClassify = "RankDataManager"
RankDataManager.PersistScoreKey = "RankDataManager.playerScore"

--endregion

--region 构造 & 实例化

---@brief 构造函数
---@type fun():void
function RankDataManager:ctor()
    RankDataManager.super.ctor(self)
    self.m_RankList = {}
    self.m_PlayerScore = 0
    self.m_PlayerName = nil
end

---@brief 创建单例
---@type fun():RankDataManager
---@return RankDataManager
function RankDataManager:Create()
    return RankDataManager.new()
end

--endregion

--region 初始化 & 生命周期

---@brief 初始化（加载玩家分数、同步昵称、生成本地模拟榜单）
---@type fun():void
function RankDataManager:Init()
    RankDataManager.super.Init(self)
    self:LoadPlayerScore()
    self:UpdatePlayerName()
    self:InitMockRankList()
end

---@brief 清理数据（清空榜单与玩家分数缓存）
---@type fun():void
function RankDataManager:ClearData()
    RankDataManager.super.ClearData(self)
    self.m_RankList = {}
    self.m_PlayerScore = 0
    self.m_PlayerName = nil
end

--endregion

---@brief 获取完整榜单（模拟玩家 + 玩家自身，按分数降序排列并附加排名）
---@type fun():table[]
---@return table[] 榜单条目数组，元素结构：{rank:number, name:string, score:number, isSelf:boolean}
function RankDataManager:GetRankList()
    -- 合并玩家自身进入榜单
    local fullList = {}
    for _, item in ipairs(self.m_RankList) do
        table.insert(fullList, { name = item.name, score = item.score, isSelf = false })
    end
    table.insert(fullList, { name = self:GetPlayerName(), score = self.m_PlayerScore, isSelf = true })

    -- 按分数降序排序（同分时玩家自身优先展示）
    table.sort(fullList, function(a, b)
        if a.score ~= b.score then
            return a.score > b.score
        end
        return a.isSelf and not b.isSelf
    end)

    -- 截断榜单并附加排名
    local list = {}
    local count = math.min(#fullList, RankDataManager.MaxRankCount)
    for i = 1, count do
        list[i] = fullList[i]
        list[i].rank = i
    end
    return list
end

---@brief 获取玩家自身排名（1 起；未进榜返回 -1）
---@type fun():number
---@return number 玩家排名，-1 表示未进榜
function RankDataManager:GetPlayerRank()
    local rankList = self:GetRankList()
    for _, item in ipairs(rankList) do
        if item.isSelf then
            return item.rank
        end
    end
    return -1
end

---@brief 获取玩家自身分数
---@type fun():number
---@return number 玩家分数
function RankDataManager:GetPlayerScore()
    return self.m_PlayerScore
end

---@brief 更新玩家分数并持久化
---@type fun(score:number):void
---@param score number 最新分数（非负数）
function RankDataManager:UpdatePlayerScore(score)
    if type(score) ~= "number" or score < 0 then
        PrintError("[RankDataManager] 更新分数失败：参数无效 score=" .. tostring(score))
        return
    end
    self.m_PlayerScore = score
    self:SavePlayerScore()
end

--endregion

--region 玩家昵称

---@brief 获取玩家自身昵称（无昵称时自动同步并兜底）
---@type fun():string
---@return string 玩家昵称
function RankDataManager:GetPlayerName()
    if self.m_PlayerName == nil or self.m_PlayerName == "" then
        self:UpdatePlayerName()
    end
    return self.m_PlayerName or RankDataManager.DefaultPlayerName
end

---@brief 同步玩家昵称（优先取 PlayerInfoDataManager，异常时兜底默认名）
---@type fun():void
function RankDataManager:UpdatePlayerName()
    local name = nil
    xpcall(function()
        if PlayerInfoDataManager and PlayerInfoDataManager.GetPlayerInfoUserName then
            name = PlayerInfoDataManager:GetPlayerInfoUserName()
        end
    end, function() end)
    self.m_PlayerName = (name ~= nil and name ~= "") and name or RankDataManager.DefaultPlayerName
end

--endregion

--region 分数持久化

---@brief 从本地存档加载玩家分数
---@type fun():void
function RankDataManager:LoadPlayerScore()
    self.m_PlayerScore = 0
    if not GameMainRoot or not GameMainRoot.Persist then
        return
    end
    xpcall(function()
        if GameMainRoot.Persist:HasItem(Honor.PersistWayType.PlayerPrefs, RankDataManager.PersistClassify, RankDataManager.PersistScoreKey) then
            local scoreStr = GameMainRoot.Persist:GetString(Honor.PersistWayType.PlayerPrefs, RankDataManager.PersistClassify, RankDataManager.PersistScoreKey, "0")
            self.m_PlayerScore = tonumber(scoreStr) or 0
        end
    end, function(err)
        PrintError("[RankDataManager] 加载玩家分数失败：" .. tostring(err))
    end)
end

---@brief 持久化玩家分数到本地存档
---@type fun():void
function RankDataManager:SavePlayerScore()
    if not GameMainRoot or not GameMainRoot.Persist then
        return
    end
    xpcall(function()
        GameMainRoot.Persist:SetString(Honor.PersistWayType.PlayerPrefs, RankDataManager.PersistClassify, RankDataManager.PersistScoreKey, tostring(self.m_PlayerScore))
    end, function(err)
        PrintError("[RankDataManager] 保存玩家分数失败：" .. tostring(err))
    end)
end

--endregion

return RankDataManager
