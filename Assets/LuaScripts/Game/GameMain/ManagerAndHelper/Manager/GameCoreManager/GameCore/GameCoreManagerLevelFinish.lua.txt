local GameCoreManagerLevelFinish = class("GameCoreManagerLevelFinish")

--region 构造 & 初始化

---=========================================================================
--- 构造函数
---@brief 初始化结算流程协程容器
---@param args table 外部参数
---=========================================================================
function GameCoreManagerLevelFinish:ctor(args)
    --- 关卡结束流程协程
    self._LevelFinishProcessCoroutine = nil ---@type UnityEngine.Coroutine
end

end

---=========================================================================
--- 清空数据 & 停止协程
---@brief 切换关卡/重置时安全清理资源
---=========================================================================
function GameCoreManagerLevelFinish:ClearData()
    self:StopAloneLevelFinishProcess()
end

--endregion

--region 游戏结束后统一逻辑

---=========================================================================
--- 游戏结束后统一逻辑（胜利/失败）
---@brief 负责：进度保存、奖励发放、挑战关卡处理、数据持久化
---=========================================================================
function GameCoreManagerLevelFinish:OnGameOverAfter()
    local isSuccess = GameCoreManager._IsSuccessFinishGame
    local curChapterId = GamerDataManager:GetCurChapterId()

    if isSuccess then
        local coreData = GameCoreManager._GameCoreManagerData
        -- 升级关卡进度
        GamerDataManager:AddLevelLvByChapterId(curChapterId)
        -- 保存核心数据
        coreData:SaveData()
    else
        -- 失败 → 清空连胜
       -- SaveRootData.GameData.playerInfo.winStreak = 0
       -- SaveIO.GameData.Save_playerInfo_winStreak()
        -- 挑战关卡失败 → 清空挑战进度
        if GameCoreManager:IsInChallengeLevel() then
            ChallengeLevelHelper:ClearCurChallengeLevel()
        end
        -- 派发关卡失败事件
        HonorEventHelper:FireNow(
                EventCmd.OnLevelFailure,
                GameCoreManager._CurLevelData.ChapterId,
                GameCoreManager._CurLevelData.LevelId
        )
    end
end

--endregion

--region 结算流程协程管理

---=========================================================================
--- 启动独立协程执行关卡结束流程
---@brief 安全启动，自动停止旧协程
---=========================================================================
function GameCoreManagerLevelFinish:StartAloneLevelFinishProcess()
    self:StopAloneLevelFinishProcess()
    self._LevelFinishProcessCoroutine = HonorCoroutineHelper:StartAlone(function()
        self:StartLevelFinishProcessCoroutine()
    end)
end

---=========================================================================
--- 安全停止关卡结束协程，防止内存泄漏
---@brief 强制停止协程，避免流程重复执行
---=========================================================================
function GameCoreManagerLevelFinish:StopAloneLevelFinishProcess()
    if self._LevelFinishProcessCoroutine then
        HonorCoroutineHelper:StopAlone(self._LevelFinishProcessCoroutine)
    end
    self._LevelFinishProcessCoroutine = nil
end

---=========================================================================
--- 关卡结束协程主体
---@brief 流程：添加UI遮罩 → 等待配置延时 → 移除遮罩 → 打开结算界面
---=========================================================================
function GameCoreManagerLevelFinish:StartLevelFinishProcessCoroutine()
    -- 等待 FishItem 碎裂动画播完再盖遮罩，保证"晃动+碎裂消失"动画对玩家可见
    --TODO: [新增] 结算遮罩等待 FishItem 碎裂动画完成（GameCoreManager.FishBreakAnimationDone 由 UIMainPlaying 设置）
    --TODO: [修复] 兜底超时 1.5→3.0：FishItem 错峰间隔加大后多鱼总时长可能超过 1.5s，联动提高防止门控提前放行
    local maxBreakWait = 3.0 -- 兜底超时，防止信号异常导致结算卡死
    local waited = 0
    while GameCoreManager.FishBreakAnimationDone == false and waited < maxBreakWait do
        HonorCoroutineHelper:WaitForSeconds(0.05)
        waited = waited + 0.05
    end

    -- UI遮罩，防止操作穿透
    GameMainRoot.UI:AddUIMaskLayerRef("GamePlayManagerLevelFinish:StartLevelFinishProcessCoroutine")
    local startTime = Unity.Time.realtimeSinceStartup
    local delayTime = Tables.Constants.GameEndTime.Value[1]
    local remainTime = delayTime - (Unity.Time.realtimeSinceStartup - startTime)

    if remainTime > 0 then
        HonorCoroutineHelper:WaitForSeconds(remainTime)
    end

    -- 移除遮罩
    GameMainRoot.UI:SubUIMaskLayerRef()
    -- 显示结算界面
    self:ShowLevelFinishUI()
end

--endregion

--region 结算界面展示

---=========================================================================
--- 显示关卡结算界面（胜利/失败）
---@brief 统一入口：停止音效 → 区分胜负 → 打开对应UI
---=========================================================================
function GameCoreManagerLevelFinish:ShowLevelFinishUI()
    -- 停止所有音效
    SoundManager:StopAllSound()
    if GameCoreManager._IsSuccessFinishGame then
        OpenUISync(UIs.UILevelVictoryView, self:GetLevelSuccessInfo())
    else
        -- 失败流程：直接打开失败结算界面（原前置道具推荐管理器 PreBoosterRecommendMgr 未定义，避免 nil 崩溃）
        --TODO: [修复] 移除未定义的 PreBoosterRecommendMgr 依赖，失败直接打开 UILevelFailedView
        self:OpenLevelFailureUI()
    end
end

---=========================================================================
---@brief 打开关卡失败界面
---@desc 统一打开失败结算界面 UILevelFailedView（当前仅存在该失败结算视图，
---       UIChallengeLevelFailure/UILevelFailure 未在 UIs 定义，已统一收敛）
---@param preBoosterId string 推荐的前置道具ID（可选）
---=========================================================================
function GameCoreManagerLevelFinish:OpenLevelFailureUI(preBoosterId)
    --TODO: [修复] 原引用未定义的 UIs.UILevelFailure / UIs.UIChallengeLevelFailure，统一收敛到唯一存在的失败视图 UILevelFailedView
    OpenUISync(UIs.UILevelFailedView, {
        InfoDatas = self:GetLevelFailureInfo(),
        RecommendPreboosterId = preBoosterId
    })
end

--endregion
