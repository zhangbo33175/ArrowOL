-- @brief     关卡内特效管理类
-- @desc      负责关卡运行时所有特效的创建、播放、回收、清理等生命周期管理
-- @core      关卡内特效缓存容器管理、特效创建/播放/回收/清理生命周期骨架（接线全局 GameEffectManager）
-- @scope     GameCoreManager 的特效子模块，由主模块 ctor 持有并在 ClearData 时统一清理
-- @author    标准规范版 | 空安全 | 可维护 | 无崩溃风险
-- @规范      Unity Lua 商业项目编码标准 / 注释完整 / 健壮性 / 性能优化
-- @dependency Singleton, EffectManager, PoolManager, CoroutineHelper, GlobalUtility

---@class GameCoreManagerEffect @关卡内特效管理核心类
---@field _EffectList table 本模块创建的特效对象缓存列表
local GameCoreManagerEffect = class("GameCoreManagerEffect")

--region 构造 & 初始化

---=========================================================================
---@brief 构造函数
---@param args table @外部传入的初始化参数
---=========================================================================
function GameCoreManagerEffect:ctor(args)
    -- 初始化特效缓存容器，记录本模块创建的特效对象（供统一回收/清理）
    self._EffectList = {}       ---@type table 关卡内特效对象缓存列表
end

---=========================================================================
---@brief 静态创建实例
---@param args table 初始化参数
---@return GameCoreManagerEffect 特效管理器实例对象
---=========================================================================
function GameCoreManagerEffect:Create(args)
    return GameCoreManagerEffect.new(args)
end

--endregion

--region 特效创建/播放

---=========================================================================
---@brief 创建并播放一个特效，登记到本模块缓存
---@desc 转发全局 GameEffectManager:AddEffect，常用参数子集（位置本地坐标、自动回收）；返回特效对象
---@param effectName string 特效名称
---@param parent UnityEngine.Transform 父节点
---@param layerName ELayerName 层级
---@param pos UnityEngine.Vector3 位置
---@param durationTime number|nil 持续时间（缺省0=无限，>0自动回收）
---@return UnityEngine.GameObject|nil 特效对象，创建失败返回 nil
---=========================================================================
function GameCoreManagerEffect:PlayEffect(effectName, parent, layerName, pos, durationTime)
    if not GameEffectManager or not effectName then
        return nil
    end
    local efGo = GameEffectManager:AddEffect(effectName, parent, layerName, pos or UnityEngine.Vector3.zero,
            EPos.Local, durationTime or 0, nil, false, true)
    if not IsNull(efGo) then
        table.insert(self._EffectList, efGo)
    end
    return efGo
end

--endregion

--region 特效回收 & 清理

---=========================================================================
---@brief 回收单个特效并从本模块缓存移除
---@param go UnityEngine.GameObject 特效对象
---=========================================================================
function GameCoreManagerEffect:RecycleEffect(go)
    if IsNull(go) or not GameEffectManager or not self._EffectList then
        return
    end
    GameEffectManager:RecycleEffect(go)
    -- 同一特效对象只会登记一次，命中后直接跳出
    for i = #self._EffectList, 1, -1 do
        if self._EffectList[i] == go then
            table.remove(self._EffectList, i)
            break
        end
    end
end

---=========================================================================
---@brief 清理当前管理器所有数据与特效资源
---@desc 释放特效对象、停止播放、清空缓存、解除引用，用于场景切换/重置时调用
---=========================================================================
function GameCoreManagerEffect:ClearData()
    -- 统一回收本模块全部特效并清空缓存
    if GameEffectManager then
        for _, v in ipairs(self._EffectList) do
            if not IsNull(v) then
                GameEffectManager:RecycleEffect(v)
            end
        end
    end
    self._EffectList = {}
end

--endregion

return GameCoreManagerEffect
