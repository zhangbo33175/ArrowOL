---@field _mapObjectTable HonorList @地图物件/建筑实例列表（Buildings/Objects 实体，回收用）
local CharacterGenerationManager = class("CharacterGenerationManager")

    self._mapObjectTable = HonorList:Create()

function CharacterGenerationManager:Create()
    local obj = CharacterGenerationManager.new()
    return obj
end

--region 【角色生成/销毁接口】

---=========================================================================
---@brief 生成单个角色（按角色ID查配置后生成）
---@desc 由地图角色批量生成 / 关卡逻辑调用；从角色静态表取配置后委托 SpawnCharacterByData 生成实例
---@type fun(characterId:string,newParent:UnityEngine.Transform):CharacterBase|nil
---@param characterId string @角色ID
---@param newParent UnityEngine.Transform @挂载父节点
---@return CharacterBase|nil @角色实例，配置或资源缺失返回nil
---=========================================================================
function CharacterGenerationManager:SpawnCharacter(characterId, newParent)
    local data = CharacterGenerationDataManager:GetDataById(characterId)
    return self:SpawnCharacterByData(data, newParent)
end

---=========================================================================
---@brief 根据角色配置数据生成角色实例（对象池复用）
---@desc 校验资源名、从对象池取 GameObject、绑定 LuaBehaviour 与角色数据；
---      非闲置角色自动登记进全局角色列表，供生命周期统一广播
---@type fun(characterData:Tables.Characters_Item,newParent:UnityEngine.Transform,isIdle:boolean):CharacterBase|nil
---@param characterData Tables.Characters_Item @角色配置数据
---@param newParent UnityEngine.Transform @挂载父节点
---@param isIdle boolean @是否闲置角色（缺省false：非闲置角色加入全局角色列表）
---@return CharacterBase|nil @角色实例，数据或资源异常返回nil
---=========================================================================
function CharacterGenerationManager:SpawnCharacterByData(characterData, newParent, isIdle)
    -- 数据空安全：无配置直接返回
    if not characterData then
        return nil
    end
    -- 资源名校验：无资源名无法生成
    if characterData.ResName == nil or characterData.ResName == '' then
        PrintError("角色id:{0},没有配置资源名字", characterData.ID)
        return nil
    end

    -- TODO: 兜底保护 - 对象池未命中或预制体资源缺失时 PoolManager:Get 返回 nil/已销毁对象，
    --       后续 go:GetComponent 会抛 MissingReferenceException，必须先判空再取组件
    local go = PoolManager:Get(characterData.ResName, newParent)
    if not go or IsNull(go) then
        PrintError("角色对象池获取失败 ResName:{0}, ID:{1}", characterData.ResName, tostring(characterData.ID))
        return nil
    end
    ---@type Honor.Runtime.LuaBehaviour
    local luaBehaviour = go:GetComponent("LuaBehaviour")
    ---@type CharacterBase
    -- TODO: 兜底保护 - 预制体未挂 LuaBehaviour 或 luaClass 绑定脚本缺失时 luaClass 为 nil，后续调用角色方法会崩
    local characterBase = luaBehaviour and luaBehaviour.luaClass or nil
    if not characterBase then
        PrintError("角色 LuaBehaviour/luaClass 为空 ResName:{0}, ID:{1}", characterData.ResName, tostring(characterData.ID))
        return nil
    end

    -- 复位角色实例并绑定配置数据与游戏对象
    characterBase:Clear()
    characterBase._CharData = characterData
    characterBase.ID = characterData.ID
    characterBase._GameObject = go

    -- 非闲置角色登记进全局角色列表（闲置角色由外部独立管理）
    if not isIdle then
        self._characterBaseTable:Add(characterBase)
    end
    return characterBase
end

---=========================================================================
---@brief 实体回收回池（角色/物件通用）
---@desc 遍历实体列表逐个放回对象池并复位；组件已销毁时跳过，避免 MissingReferenceException
---@type fun(list:HonorList):void
---@param list HonorList @实体列表（_characterBaseTable / _mapObjectTable）
---=========================================================================
function CharacterGenerationManager:_DespawnEntityList(list)
    if not list then return end
    for i = 1, list:Count() do
        local entity = list.items[i]
        if entity ~= nil and entity.cs ~= nil and not IsNull(entity.cs) then
            PoolManager:Put(entity.cs.gameObject)
            entity:Clear()
        end
    end
    list:Clear()
end

---=========================================================================
---@brief 销毁所有角色（放回对象池）
---@desc 关卡退出/地图卸载时调用，角色与建筑/物件同批回池，最后清空列表
---@type fun():void
---=========================================================================
function CharacterGenerationManager:DespawnAllCharacters()
    self:_DespawnEntityList(self._characterBaseTable)
    self:_DespawnEntityList(self._mapObjectTable)
end

--region 【地图角色批量生成】

---=========================================================================
---@brief 清空全局角色列表（防重载/重玩残留）
---@desc 遍历清空 GameCoreManager 全局角色列表的全部键，并返回该列表供后续写入
---@type fun():table|nil
---@return table|nil @清空后的全局角色列表（空表或原表，可能为nil）
---=========================================================================
function CharacterGenerationManager:_ClearCharacterList()
    local characterList = GameCoreManager:GetCharacterList()
    if characterList ~= nil then
        for k in pairs(characterList) do
            characterList[k] = nil
        end
    end
    return characterList
end

---=========================================================================
---@brief 解析角色图标别名
---@desc 优先用地图配置 sprite（每个物件各自的猫图），否则回退 fallbackIcon（角色表 Icon）；
---      sprite 可能是完整路径（Assets/Res/.../sc_2_cat_1.png）或纯名（sc_2_cat_1），
---      RefreshImageSprite 需要别名：含路径时提取文件名(去扩展名)，纯名时直接使用
---@type fun(sprite:string,fallbackIcon:string):string|nil
---@param sprite string @地图配置的图标路径或别名
---@param fallbackIcon string @角色静态表回退图标名
---@return string|nil @可用的图标别名，无可解析时返回nil
---=========================================================================
function CharacterGenerationManager:_ResolveSpriteName(sprite, fallbackIcon)
    local spriteName = nil
    if sprite and sprite ~= '' then
        if string.find(sprite, '/') then
            spriteName = string.match(sprite, "([^/]+)%.%w+$") or sprite
        else
            spriteName = sprite
        end
    end
    if (not spriteName) and fallbackIcon then
        spriteName = fallbackIcon
    end
    return spriteName
end

---=========================================================================
---@brief 构造实例唯一收集键（角色/物件通用）
---@desc 同分区同 Id 多实例时靠 seq 保证键唯一；prefix 区分类型（物件带 "Obj_" 前缀）
---@type fun(prefix:string?,sectionId:number,id:any,seq:number):string
---@param prefix string? @类型前缀（物件传 "Obj_"，角色传 nil）
---@param sectionId number @分区ID
---@param id any @条目ID
---@param seq number @实例序号（同 Id 多实例区分）
---@return string @收集键字符串
---=========================================================================
function CharacterGenerationManager:_MakeCollectKey(prefix, sectionId, id, seq)
    local p = prefix and (prefix .. "_") or ""
    return p .. tostring(sectionId or 1) .. "_" .. tostring(id) .. "_" .. tostring(seq or 1)
end

---=========================================================================
---@brief 读取格子像素尺寸（编辑器/运行时统一为 32）
---@desc 角色与物件坐标换算共用：pixel = Grid × gridSize；统一读取点，避免散落解析与魔数
---@type fun():number
---@return number @格子像素尺寸（配置缺失时回退 32）
---=========================================================================
function CharacterGenerationManager:_GetGridSize()
    return tonumber(EMapLayoutConfig.gridSize) or 32
end

---=========================================================================
---@brief 编辑器格子坐标 → 实体 localPosition（公共换算）
---@desc CoordAdapter 坐标系：左下角原点、Y 向上为正；编辑器 GridY 从上往下数，需 Y 轴翻转
---@type fun(gridX:number,gridY:number):UnityEngine.Vector3
---@param gridX number @编辑器格子 X（从左向右）
---@param gridY number @编辑器格子 Y（从上向下）
---@return UnityEngine.Vector3 @localPosition（x=gridX*gridSize, y=mapH-gridY*gridSize）
---=========================================================================
function CharacterGenerationManager:_GridToLocalPos(gridX, gridY)
    local gs = self:_GetGridSize()
    -- 【修复】不用 EMapLayoutConfig.height（值是错的2048），直接用 gridRows * gridSize 算真实高度
    local mapH = (EMapLayoutConfig.gridRows or 128) * gs
    -- GridY 从上往下 → 运行时 Y 从下往上（翻转）
    local x = (gridX or 0) * gs
    local y = mapH - (gridY or 0) * gs
    return UnityEngine.Vector3(x, y, 0)
end

---=========================================================================
---@brief 刷新单个角色图标
---@desc 查找角色身上的 CharacterIcon 图标组件，解析别名后刷新显示；组件缺失则静默跳过
---@type fun(characterBase:CharacterBase,sprite:string,fallbackIcon:string):void
---@param characterBase CharacterBase @角色实例
---@param sprite string @地图配置的图标路径或别名
---@param fallbackIcon string @角色静态表回退图标名
---=========================================================================
function CharacterGenerationManager:_RefreshCharacterIcon(characterBase, sprite, fallbackIcon)
    -- 查找角色图标UI组件，缺失则跳过
    local characterIcon = characterBase._GameObject.transform:Find("CharacterIcon")
    if not characterIcon then
        return
    end
    local uiImage = characterIcon.transform:GetComponent(typeof(UnityEngine.UI.Image))
    if not uiImage then
        return
    end

    -- 解析图标别名并刷新显示
    local spriteName = self:_ResolveSpriteName(sprite, fallbackIcon)
    if spriteName then
        GamePlayHelper:RefreshImageSprite(characterBase, uiImage, spriteName)
    end
end

---=========================================================================
---@brief 注册角色到全局角色列表
---@desc 以角色ID为主键写入 {ID, Character} 结构，供其他模块按ID查询角色实例
---@type fun(characterList:table,id:string,characterBase:CharacterBase):void
---@param characterList table @全局角色列表
---@param id string @角色ID
---@param characterBase CharacterBase @角色实例
---=========================================================================
function CharacterGenerationManager:_RegisterCharacterToList(characterList, id, characterBase)
    if characterList ~= nil then
        characterList[id] = {}
        characterList[id].ID = id
        characterList[id].Character = characterBase
    end
end

---=========================================================================
---@brief 根据地图配置批量生成所有角色实体，初始化图标、坐标、全局列表缓存
---@type fun(gameMapRoot:UnityEngine.Transform):void
---@param gameMapRoot UnityEngine.Transform @地图根节点（挂载生成的角色实体）
---@desc 由 GameMapBase:LoadCharacterByMapData 在 Awake 相机/HUD 就绪后转调（勿在 ctor 中调用，
---      此时地图物体尚未激活、相机未就绪，易导致角色显示异常）。
---      流程：读取关卡根配置 → 挂载当前关卡数据 → 清空旧角色列表防重载残留 → 单分区整图生成
---      角色（对象池复用）→ 设置本地坐标 → 刷新角色图标 → 写入全局角色列表。
---      【取消分区】章节地图为单分区整图，一次全量生成、一次全量收集，无分区解锁补生成。
---=========================================================================
function CharacterGenerationManager:LoadCharacterByMapData(gameMapRoot)
    PrintInfo("[CGM] LoadCharacterByMapData 入口, gameMapRoot=" .. tostring(gameMapRoot))

    -- 地图根节点空安全保护
    if gameMapRoot == nil then
        PrintWarning("[CGM] gameMapRoot 为 nil，跳过")
        return
    end

    -- 获取关卡根配置数据并挂载当前关卡（单分区：整图数据）
    local mapData = GameManager:GetMapRootData()
    if not mapData then
        PrintWarning("[CGM] GetMapRootData 返回 nil，跳过（章节ID/关卡配置表未就绪？）")
        return
    end
    PrintInfo("[CGM] 关卡就绪: " .. tostring(mapData.ChapterName)
        .. " | 角色=" .. tostring(mapData.Characters and #mapData.Characters or 0)
        .. " 建筑=" .. tostring(mapData.Buildings and #mapData.Buildings or 0)
        .. " 物件=" .. tostring(mapData.Objects and #mapData.Objects or 0))
    GameCoreManager:SetCurLevelData(mapData)

    -- 清空旧角色列表，防止重载/重玩时残留脏数据
    local characterList = self:_ClearCharacterList()

    -- MapLoadManager 未就绪硬守卫（骨架已由 MapLoadManager:LoadMap 建立，分区数据存于 LevelData）
    if MapLoadManager.ZoneRoot == nil or MapLoadManager.LevelData == nil then
        PrintWarning("[CGM] 骨架未就绪（ZoneRoot/LevelData 为 nil），跳过实体生成")
        return
    end

    -- 实体挂载点：ZoneRoot → ZoneItem → Root → CoordAdapter（编辑器坐标系层，pivot=左下角）
    local zoneRootRef = MapLoadManager.ZoneRoot
    local rootT = zoneRootRef:Find("ZoneItem/Root/CoordAdapter")
    if not rootT then rootT = zoneRootRef:Find("ZoneItem/Root") end  -- 兜底：兼容旧预制体
    if not rootT then
        PrintError("[CGM] 找不到实体挂载点 ZoneItem/Root/CoordAdapter")
        return
    end

    local _okE, _errE = pcall(function()
        self:_GenerateEntitiesForSection(mapData, rootT, characterList)
    end)
    if not _okE then
        PrintError("[CGM] 实体生成失败: " .. tostring(_errE))
    end
end

---=========================================================================
---@brief 合并本次加载所有分区的 OccupiedCells
---@type fun(unlockedDatas:table):table
function CharacterGenerationManager:_CollectOccupiedCells(unlockedDatas)
    local merged = {}
    for _, entry in ipairs(unlockedDatas) do
        local occ = entry.data and entry.data.OccupiedCells
        if occ then
            for _, cell in ipairs(occ) do
                merged[#merged + 1] = cell
            end
        end
    end
    return merged
end

---=========================================================================
---@brief 按需加载子表（带缓存，避免重复 require）
---@desc 子表数据挂在 Tables.xxx 全局表上，require 只负责触发文件加载
---@type fun(mapData:table, subKey:string):table|nil
function CharacterGenerationManager:_LoadSubTable(mapData, subKey)
    if type(mapData) ~= "table" then
        return nil
    end
    -- 缓存键：关卡ID + 子表名
    local levelId = "unknown"
    local mainTableName = "LevelData"
    if type(mapData.Meta) == "table" then
        levelId = tostring(mapData.Meta.LevelId or "unknown")
    end
    -- 推导主表名（Tables.LevelData_X_Y）
    if type(mapData.Meta) == "table" and mapData.Meta.ChapterId and mapData.Meta.LevelId then
        mainTableName = "LevelData_" .. tostring(mapData.Meta.ChapterId) .. "_" .. tostring(mapData.Meta.LevelId)
    end
    local cacheKey = levelId .. "_" .. subKey
    if self._SubTableCache and self._SubTableCache[cacheKey] then
        return self._SubTableCache[cacheKey]
    end
    -- 读子表 require 路径（空值保护）
    local subPath = nil
    if type(mapData.SubTables) == "table" then
        subPath = mapData.SubTables[subKey]
    end
    PrintInfo("[CGM] 加载子表: subKey=" .. tostring(subKey) .. " subPath=" .. tostring(subPath))
    if not subPath then
        PrintWarning("[CGM] 子表路径为空: " .. tostring(subKey))
        return nil
    end
    -- require 触发文件加载（文件里定义 Tables.xxx = {...}）
    local ok, err = pcall(require, subPath)
    PrintInfo("[CGM] require 结果: ok=" .. tostring(ok) .. " err=" .. tostring(err))
    if not ok then
        PrintWarning("[CGM] 子表加载失败: " .. tostring(subPath) .. " 错误: " .. tostring(err))
        return nil
    end
    -- 从 Tables 全局表里取对应子表
    -- 表名规律：主表名 + "_" + subKey（例：LevelData_1_1 + "_" + Characters = LevelData_1_1_Characters）
    local tableName = mainTableName .. "_" .. subKey
    local data = Tables[tableName]
    PrintInfo("[CGM] 取表: Tables." .. tableName .. " type=" .. type(data))
    if type(data) ~= "table" then
        PrintWarning("[CGM] 子表加载后数据为空: Tables." .. tableName)
        -- 兜底：直接用 subPath 对应的表名试试（带 Table 前缀）
        local altTableName = subPath  -- TableLevelData_1_1_Characters
        if altTableName:startswith("Table") then
            altTableName = altTableName:sub(6)  -- LevelData_1_1_Characters
        end
        PrintInfo("[CGM] 兜底尝试: Tables." .. altTableName .. " type=" .. type(Tables[altTableName]))
        return nil
    end
    self._SubTableCache = self._SubTableCache or {}
    self._SubTableCache[cacheKey] = data
    return data
end

---=========================================================================
---@brief 生成单个分区数据内的全部角色（多分区复用）
---@desc 优化格式：从主表 SubTables.Characters 子表读角色列表，合并 Defaults 默认配置，
---      从 Paths 子表按 PathId 拿路径点
---@type fun(mapSectionData:table,gameMapRoot:UnityEngine.Transform,characterList:table):void
---@param mapSectionData table 分区数据（主表 Tables.LevelData_Item）
---@param gameMapRoot UnityEngine.Transform 地图根节点
---@param characterList table 全局角色列表（登记用）
---=========================================================================
function CharacterGenerationManager:_SpawnCharactersForSection(mapSectionData, gameMapRoot, characterList)
    if not mapSectionData then
        return
    end

    -- 加载角色子表（优化格式）
    local charList = self:_LoadSubTable(mapSectionData, "Characters")
    if not charList then
        -- 兜底：兼容旧格式（主表直接带 Characters）
        charList = mapSectionData.Characters or mapSectionData.MapData
    end
    if not charList then
        return
    end
    local isNewFormat = mapSectionData.SubTables ~= nil
    local sectionId = tonumber(mapSectionData.Meta and mapSectionData.Meta.LevelId) or 1

    for i, v in ipairs(charList) do
        self:_ProcessSingleCharacter(v, i, gameMapRoot, characterList, sectionId, isNewFormat, mapSectionData)
    end
end

---=========================================================================
---@brief 处理单个分区角色条目（生成/定位/图标/路径/登记）
---@desc 优化格式：字段用 TemplateId/GridX/GridY/Image/Anim/PathId，公共字段走 Defaults 合并
---@type fun(v:table,index:number,gameMapRoot:UnityEngine.Transform,characterList:table,sectionId:number,isNewFormat:boolean,mapSectionData:table):void
---@param v table @分区角色条目
---@param index number @循环序号（同分区同ID多实例收集键唯一用）
---@param gameMapRoot UnityEngine.Transform @地图根节点（实体挂载点）
---@param characterList table @全局角色列表（登记用）
---@param sectionId number @当前分区ID
---@param isNewFormat boolean @是否新地图数据格式
---@param mapSectionData table @主表数据（用于读 Defaults / Paths 子表）
---=========================================================================
function CharacterGenerationManager:_ProcessSingleCharacter(v, index, gameMapRoot, characterList, sectionId, isNewFormat, mapSectionData)
    -- 无效条目直接跳过
    if not v then
        print("[CharacterGenerationManager] 无效角色配置数据，跳过")
        return
    end

    -- 字段兼容：新格式 TemplateId，旧格式 Id
    local templateId = v.TemplateId or v.Id
    if not templateId then
        print("[CharacterGenerationManager] 角色缺少 TemplateId/Id，跳过")
        return
    end
    -- 字符串转数字：模板表 Tables.Characters 的 key 是数字 [1001]，不是字符串 "1001"
    templateId = tonumber(templateId) or templateId

    -- 合并默认配置（新格式：Defaults + 条目自身字段）
    local defaults = (isNewFormat and mapSectionData and mapSectionData.Defaults) or {}
    local charConfig = setmetatable({}, { __index = function(_, k)
        if v[k] ~= nil then return v[k] end
        return defaults[k]
    end })

    -- 根据ID生成角色实体挂载到地图根节点
    local characterBase = self:SpawnCharacter(templateId, gameMapRoot)
    if not characterBase then
        print("[CharacterGenerationManager] 角色生成失败 ID:" .. tostring(templateId))
        return
    end

    -- 实例唯一收集键：分区ID_角色ID_序号
    local collectKey = self:_MakeCollectKey(nil, sectionId, templateId, index)
    characterBase.ID = templateId
    characterBase._CollectKey = collectKey

    -- 【坐标系】实体挂到 CoordAdapter（pivot=左下角）下，Y 轴翻转
    local startPos = self:_GridToLocalPos(v.GridX, v.GridY)
    characterBase._GameObject.transform.localPosition = startPos
    PrintInfo("[坐标调试] 角色ID=" .. tostring(templateId)
        .. " GridX=" .. tostring(v.GridX) .. " GridY=" .. tostring(v.GridY)
        .. " → PosX=" .. string.format("%.1f", startPos.x)
        .. " PosY=" .. string.format("%.1f", startPos.y))
    -- 【调试】打印设置后实际的 localPosition，验证是否和预期一致
    local actualPos = characterBase._GameObject.transform.localPosition
    PrintInfo("[坐标调试] 角色ID=" .. tostring(templateId)
        .. " 设置后实际 localPosition = (" .. string.format("%.1f", actualPos.x)
        .. ", " .. string.format("%.1f", actualPos.y) .. ")")

    -- 刷新角色图标
    local catData = CharacterGenerationDataManager:GetDataById(templateId)
    if isNewFormat then
        ---TODO 新格式：角色静态表 Icon 优先，地图条目 Image 仅作兜底
        self:_RefreshCharacterIcon(characterBase, catData and catData.Icon, v.Image)
    else
        self:_RefreshCharacterIcon(characterBase, v.Sprite, catData and catData.Icon)
    end

    -- 路径移动（新格式：从 Paths 子表按 PathId 拿路径点）
    local pathPoints = nil
    if isNewFormat and v.PathId then
        local pathsTable = self:_LoadSubTable(mapSectionData, "Paths")
        if pathsTable then
            pathPoints = pathsTable[v.PathId]
        end
    elseif isNewFormat and v.PathPoints then
        -- 兼容：旧格式直接嵌在条目里
        pathPoints = v.PathPoints
    end

    if pathPoints and #pathPoints > 0 then
        local worldPathPoints = TransformEditorPathPoints(pathPoints, sectionId)
        characterBase._PathPoints = worldPathPoints
        local loop = v.PatrolLoop == "loop" or charConfig.PatrolLoop == "loop"
        characterBase:StartPathMove(worldPathPoints, charConfig.Speed, loop)
    end

    -- 存入全局角色管理列表
    self:_RegisterCharacterToList(characterList, collectKey, characterBase)

    -- 登记角色归属关卡与关卡物品总数
    local charData = characterBase._CharData
    local charMeta = charData and { icon = charData.Icon, isRole = true } or { isRole = true }
    LevelCollectManager:RegisterLevelItem(sectionId, collectKey, charMeta)
end

---=========================================================================
---@brief 分区实体生成统一入口（角色 + 可交互建筑 + 物件）
---@desc 优化格式：角色/建筑/物件分别从子表加载，默认值走主表 Defaults
---@type fun(mapSectionData:table,gameMapRoot:UnityEngine.Transform,characterList:table?,buildingList:table?,objectList:table?):void
---@param mapSectionData table @分区数据（主表 Tables.LevelData_Item）
---@param gameMapRoot UnityEngine.Transform @地图根节点（CoordAdapter）
---@param characterList table? @角色列表（外部传入则复用）
---@param buildingList table? @建筑列表（外部传入则复用）
---@param objectList table? @物件列表（外部传入则复用）
---=========================================================================
function CharacterGenerationManager:_GenerateEntitiesForSection(mapSectionData, gameMapRoot, characterList, buildingList, objectList)
    if not mapSectionData or gameMapRoot == nil then
        return
    end
    -- 分层挂载：角色→Characters，建筑→Buildings，物件→Objects
    local charRoot = gameMapRoot:Find("Characters") or gameMapRoot
    local buildRoot = gameMapRoot:Find("Buildings") or gameMapRoot
    local objRoot = gameMapRoot:Find("Objects") or gameMapRoot
    PrintInfo("[CGM] 分层挂载: char=" .. charRoot.name .. " build=" .. buildRoot.name .. " obj=" .. objRoot.name)

    -- 列表初始化
    characterList = characterList or {}
    buildingList = buildingList or {}
    objectList = objectList or {}
    self._CharacterList = characterList
    self._BuildingList = buildingList
    self._ObjectList = objectList

    -- 角色生成
    self:_SpawnCharactersForSection(mapSectionData, charRoot, characterList)

    -- 建筑生成（从子表加载）
    local objSeq = 0
    objSeq = self:_SpawnBuildingsForSection(mapSectionData, buildRoot, objSeq, buildingList)

    -- 物件生成（从子表加载）
    objSeq = self:_SpawnMapObjectsForSection(mapSectionData, objRoot, objSeq, objectList)
end

---=========================================================================
---@brief 可交互建筑生成（从子表加载）
---@type fun(mapSectionData:table,gameMapRoot:UnityEngine.Transform,objSeq:number,buildingList:table):number
---@param mapSectionData table @主表数据
---@param gameMapRoot UnityEngine.Transform @建筑挂载节点（Buildings 层）
---@param objSeq number @物件全局序号
---@param buildingList table @建筑列表
---@return number @更新后的全局序号
---=========================================================================
function CharacterGenerationManager:_SpawnBuildingsForSection(mapSectionData, gameMapRoot, objSeq, buildingList)
    if not mapSectionData then
        return objSeq or 0
    end
    buildingList = buildingList or {}

    -- 从子表加载建筑列表（优化格式）
    local buildingListData = self:_LoadSubTable(mapSectionData, "Buildings")
    if not buildingListData then
        -- 兜底：兼容旧格式
        buildingListData = mapSectionData.Buildings
    end
    if not buildingListData then
        return objSeq or 0
    end

    local sectionId = tonumber(mapSectionData.Meta and mapSectionData.Meta.LevelId) or tonumber(mapSectionData.SectionId) or 1
    for _, entry in ipairs(buildingListData) do
        objSeq = (objSeq or 0) + 1
        local building = self:_SpawnSingleMapObject(entry, gameMapRoot, "GameObject_BuildingTemplate", sectionId, objSeq)
        if building then
            buildingList[#buildingList + 1] = building
        end
    end
    return objSeq
end

---=========================================================================
---@brief 物件生成（从子表加载）
---@type fun(mapSectionData:table,gameMapRoot:UnityEngine.Transform,objSeq:number,objectList:table):number
---@param mapSectionData table @主表数据
---@param gameMapRoot UnityEngine.Transform @物件挂载节点（Objects 层）
---@param objSeq number @物件全局序号
---@param objectList table @物件列表
---@return number @更新后的全局序号
---=========================================================================
function CharacterGenerationManager:_SpawnMapObjectsForSection(mapSectionData, gameMapRoot, objSeq, objectList)
    if not mapSectionData then
        return objSeq or 0
    end
    objectList = objectList or {}

    -- 从子表加载物件列表（优化格式）
    local objectListData = self:_LoadSubTable(mapSectionData, "Objects")
    if not objectListData then
        -- 兜底：兼容旧格式
        objectListData = mapSectionData.Objects
    end
    if not objectListData then
        return objSeq or 0
    end

    local sectionId = tonumber(mapSectionData.Meta and mapSectionData.Meta.LevelId) or tonumber(mapSectionData.SectionId) or 1

        end
    end
    return objSeq
end

---=========================================================================
---@brief 生成单个地图物件/建筑实体（对象池复用）
---@desc 从对象池取 GameObject_ItemMap 实例，绑定 MapObjectBase 与条目数据：
---      坐标（localPosition=Position，与角色同空间）、BG 图标刷新、显示尺寸（Width/Height 格子数×32）、
---      PathPoints 路径移动启动；气泡由 MapObjectBase:Start 读 BubbleId 自动挂载。
---@type fun(entry:table,gameMapRoot:UnityEngine.Transform,resName:string,sectionId:number):MapObjectBase|nil
---@param entry table @MapObjectData 条目（Buildings/Objects）
---@param gameMapRoot UnityEngine.Transform @地图根节点
---@param resName string @资源注册键（GameObject_BuildingTemplate / GameObject_ObjectTemplate）
---@param sectionId number @所属分区ID（登记收集归属用）
---@param objSeq number @物件全局序号（同分区同 Id 多实例键唯一）
---@return MapObjectBase|nil @物件实例，数据缺失返回nil
---=========================================================================
function CharacterGenerationManager:_SpawnSingleMapObject(entry, gameMapRoot, resName, sectionId, objSeq)
    -- 数据空安全：无条目/无坐标无法生成（编辑器导出字段为 GridX/GridY，非 Position）
    if not entry or (entry.GridX == nil and entry.GridY == nil) then
        return nil
    end

    ---TODO [物件生成] 物件/建筑走 Model 运行时实体模板（BuildingTemplate/ObjectTemplate，
    ---     已在 LoadResDefs_1 手工注册；模板结构=UI RectTransform+LuaBehaviour[MapObjectCtrl]+子节点 BG Image，
    ---     复刻已验证的 ItemMap 模式；条目无 ResName 映射，按数组类型取对应模板资源名）
    -- TODO: 兜底保护 - 物件模板资源未注册或对象池未命中时返回 nil，后续 go:GetComponent 会崩，先判空
    local go = PoolManager:Get(resName, gameMapRoot)
    if not go or IsNull(go) then
        PrintError("地图物件对象池获取失败 ResName:{0}, ID:{1}", resName, tostring(entry.Id))
        return nil
    end
    ---@type Honor.Runtime.LuaBehaviour
    local luaBehaviour = go:GetComponent("LuaBehaviour")
    ---@type MapObjectBase
    -- TODO: 兜底保护 - 物件模板预制体未挂 LuaBehaviour 或绑定脚本缺失时 luaClass 为 nil，后续调用物件方法会崩
    local mapObj = luaBehaviour and luaBehaviour.luaClass or nil
    if not mapObj then
        PrintError("地图物件实例化失败：LuaBehaviour.luaClass 为空 ID:{0}", tostring(entry.Id))
        return nil
    end

    -- 复位物件实例并绑定条目数据；对象池复用时 Start 不重触发，故显式按新 BubbleId 重启气泡定时器
    mapObj:Clear()
    mapObj._MapData = entry
    mapObj._GameObject = go
    mapObj:RefreshBubbleTimer()

    -- 物件唯一收集键（关卡ID_物件ID_全局序号，避免同关卡同 Id 多实例键冲突）
    local collectKey = self:_MakeCollectKey("Obj", sectionId, entry.Id, objSeq)
    mapObj._CollectKey = collectKey
    -- 登记物件归属关卡（收敛到 LevelCollectManager 门面）
    -- 携带展示信息 meta（图标名=entry.Image/是否角色=false），供收集面板列表展示
    LevelCollectManager:RegisterLevelItem(sectionId or 1, collectKey, { icon = entry.Image, isRole = false })

    -- 【坐标系】同角色：挂到 CoordAdapter（左下角原点）下，直接用 Grid×gridSize
    go.transform.localPosition = self:_GridToLocalPos(entry.GridX, entry.GridY)

    -- 图标刷新（子节点 BG Image）与显示尺寸（格子数×32 像素）
    self:_SetupMapObjectIconAndSize(mapObj, go, entry)

    -- 路径移动：PathPoints 非空即实际沿路径移动；循环语义 PatrolLoop 优先，否则取 Loop
    ---TODO [坐标变换] 物件 PathPoints 编辑器坐标 → 世界坐标后再启动移动
    local worldObjPathPoints = entry.PathPoints and #entry.PathPoints > 0 and TransformEditorPathPoints(entry.PathPoints, sectionId) or nil
    local isLoop = entry.PatrolLoop == true or entry.Loop == true
    mapObj:StartPathMove(worldObjPathPoints, entry.Speed, isLoop)

    -- 登记回收列表
    self._mapObjectTable:Add(mapObj)
    return mapObj
end

---=========================================================================
---@brief 配置地图物件图标与显示尺寸（BG 子节点）
---@desc 子节点 BG Image 按 entry.Image 解析别名刷新；Width/Height 格子数换算为像素尺寸写入 RectTransform；
---      BG 节点缺失时静默跳过
---@type fun(mapObj:MapObjectBase,go:UnityEngine.GameObject,entry:table):void
---@param mapObj MapObjectBase @物件实例（图标刷新的上下文）
---@param go UnityEngine.GameObject @物件根节点
---@param entry table @MapObjectData 条目（Image/Width/Height 取数）
---=========================================================================
function CharacterGenerationManager:_SetupMapObjectIconAndSize(mapObj, go, entry)
    local iconTrans = go.transform:Find("BG")
    if not iconTrans then
        return
    end
    local uiImage = iconTrans:GetComponent(typeof(UnityEngine.UI.Image))
    if uiImage then
        local spriteName = self:_ResolveSpriteName(entry.Image, nil)
        if spriteName then
            GamePlayHelper:RefreshImageSprite(mapObj, uiImage, spriteName)
        end
    end
    ---TODO [物件生成] 编辑器矩形像素尺寸=Width*GridSize × Height*GridSize，运行时直接把 BG 的 sizeDelta 设为该像素尺寸
    local gridPixel = self:_GetGridSize()
    local width = tonumber(entry.Width)
    local height = tonumber(entry.Height)
    if width and height then
        local iconRect = iconTrans:GetComponent(typeof(UnityEngine.RectTransform))
        if iconRect then
            iconRect.sizeDelta = UnityEngine.Vector2(width * gridPixel, height * gridPixel)
        end
    end
end

---@brief 广播指定生命周期方法给所有已生成角色（内部公共实现，消除重复遍历）
---@type fun(methodName:string):void
---@param methodName string @角色生命周期方法名（传 ECharacterLifecycle 枚举值）
---@desc 生命周期方法群统一委托本方法，避免 8 处相同遍历；动态调用仅在状态切换时发生一次，无性能负担
function CharacterGenerationManager:_BroadcastToAllCharacters(methodName)
    for _, v in ipairs(self._characterBaseTable.items) do
        if v and v[methodName] then
            v[methodName](v)
        end
    end
end
