local MapLoadManager = {}

--region 【状态】

MapLoadManager.ZoneRoot = nil        ---@type UnityEngine.Transform @ZoneRoot 容器
MapLoadManager.SectionDatas = nil    ---@type table @全部分区数据列表
MapLoadManager.UnlockedCount = 0    ---@type number @已解锁分区数

--endregion

--region 【配置加载】

---加载地图配置，写入 EMapLayoutConfig 缓存
---@param chapterId string @章节ID
---@param unlockedCount number @已解锁分区数
---@return table @已解锁分区数据列表
function MapLoadManager:_LoadConfig(chapterId, unlockedCount)
    local unlockedDatas = GamePlayHelper:GetUnlockedLevelDatas(chapterId, unlockedCount)
    local fd = unlockedDatas[1] and unlockedDatas[1].data or {}
    local n = #unlockedDatas

    if not fd.Width then return unlockedDatas end

    local w = Num(fd.Width, 4096)
    local h = Num(fd.Height, 4096)
    EMapLayoutConfig.centerX = w / 2
    EMapLayoutConfig.centerY = h / 2
    EMapLayoutConfig.width = w
    EMapLayoutConfig.scale = 1.0
    EMapLayoutConfig.regionCols = Num(fd.RegionCols, math.max(1, math.ceil(math.sqrt(n))))
    EMapLayoutConfig.regionRows = Num(fd.RegionRows, math.max(1, math.ceil(n / EMapLayoutConfig.regionCols)))
    EMapLayoutConfig.gridSize = Num(fd.GridSize, 32)
    EMapLayoutConfig.gridCols = Num(fd.GridCols, 128)
    EMapLayoutConfig.gridRows = Num(fd.GridRows, 128)
    EMapLayoutConfig.zoneWidth = w / EMapLayoutConfig.regionCols
    EMapLayoutConfig.zoneHeight = h / EMapLayoutConfig.regionRows

    return unlockedDatas
end

---加载分区表数据（全部分区，含未解锁）
---@param chapterId string @章节ID
---@param unlockedCount number @已解锁分区数
---@param unlockedDatas table @已解锁分区数据
---@return table @全部分区数据列表
function MapLoadManager:_LoadSections(chapterId, unlockedCount, unlockedDatas)
    local fd = unlockedDatas[1] and unlockedDatas[1].data or {}
    local totalSec = Num(fd.SectionCount, #unlockedDatas)

    local allSectionDatas = GamePlayHelper:GetLevelDatasByCount(chapterId, totalSec)
    self.SectionDatas = allSectionDatas
    self.UnlockedCount = unlockedCount
    return allSectionDatas
end

--endregion

--region 【网格系统】

---初始化 GridManager
---@param mainGo UnityEngine.GameObject @GameMainMap 父节点
---@param unlockedDatas table @已解锁分区数据
function MapLoadManager:_SetupGrid(mainGo, unlockedDatas)
    if not mainGo then return end

    -- 收集占用格
    local occupied = {}
    for _, e in ipairs(unlockedDatas or {}) do
        local occ = e.data and e.data.OccupiedCells
        if occ then
            for _, c in ipairs(occ) do
                occupied[#occupied + 1] = Num(c.X, 0)
                occupied[#occupied + 1] = Num(c.Y, 0)
            end
        end
    end

    CallCS(function()
        local gridMgr = mainGo:GetComponent(typeof(CS.Honor.Runtime.GridManager))
        if not gridMgr then
            gridMgr = mainGo:AddComponent(typeof(CS.Honor.Runtime.GridManager))
        end
        gridMgr:Setup(
            EMapLayoutConfig.gridSize, EMapLayoutConfig.gridCols, EMapLayoutConfig.gridRows,
            EMapLayoutConfig.regionCols, EMapLayoutConfig.regionRows,
            EMapLayoutConfig.scale, self.UnlockedCount, self.ZoneRoot
        )
        gridMgr:SetOccupiedCells(occupied)
        gridMgr:BuildOverlay()
    end)
end

--endregion

--region 【分区挂载】

---创建 ZoneRoot 容器
---@param gameMapRoot UnityEngine.GameObject @GameMainMap 根节点
---@return UnityEngine.Transform @ZoneRoot Transform
function MapLoadManager:_CreateZoneRoot(gameMapRoot)
    local ok, zoneRoot = CallCS(function()
        return CS.GameLib.MapLoadManager.CreateZoneRoot(gameMapRoot)
    end)
    if not ok or not zoneRoot then return nil end
    self.ZoneRoot = zoneRoot
    return zoneRoot
end

---实例化所有分区节点
---@param allSectionDatas table @全部分区数据
---@param unlockedCount number @已解锁分区数
function MapLoadManager:_PopulateZoneItems(allSectionDatas, unlockedCount)
    if not self.ZoneRoot then return end

    for _, e in ipairs(allSectionDatas or {}) do
        local sid = Num(e.data and e.data.SectionId, 1)

        -- 加载预制体
        local ok, zoneGo = CallCS(function()
            local r = LoadRes["GameObject_ZoneItem"]
            if not r then return nil end
            return GameMainRoot.HonorAssetManager:LoadPrefabSync(r.ABPath, r.AssetName, self.ZoneRoot, nil)
        end)
        if not ok or not zoneGo then
            print(string.format("[MapLoadManager] 分区%d 预制体加载失败", sid))
            goto continue
        end

        -- 加载背景图
        local sprite = nil
        if e.data.BgName then
            local r = LoadSpriteByName(e.data.BgName)
            if r then sprite = LoadAssetSync(r) end
        end

        -- C# 挂载定位
        CallCS(function()
            CS.GameLib.MapLoadManager.SetupZoneItem(
                zoneGo, self.ZoneRoot, sid,
                EMapLayoutConfig.regionCols, EMapLayoutConfig.regionRows,
                EMapLayoutConfig.scale, sprite, sid <= unlockedCount
            )
        end)

        ::continue::
    end
end

--endregion

--region 【包围盒与相机】

---计算地图外接矩形（top-left 坐标系）
---@return number, number, number, number @minX, maxX, minY, maxY
function MapLoadManager:_CalcBoundsRect()
    local cols = EMapLayoutConfig.regionCols
    local rows = EMapLayoutConfig.regionRows
    local zoneW = EMapLayoutConfig.zoneWidth
    local zoneH = EMapLayoutConfig.zoneHeight

    local minX, maxX = math.huge, -math.huge
    local minY, maxY = math.huge, -math.huge

    for sid = 1, self.UnlockedCount do
        local idx = sid - 1
        local c = idx % cols
        local r = math.floor(idx / cols)
        local zx = c * zoneW
        local zy = -r * zoneH
        minX = math.min(minX, zx)
        maxX = math.max(maxX, zx + zoneW)
        minY = math.min(minY, zy - zoneH)
        maxY = math.max(maxY, zy)
    end

    return minX, maxX, minY, maxY
end

---适配包围盒
---@param gameMapRoot UnityEngine.GameObject @GameMainMap 根节点
function MapLoadManager:_SetupBounds(gameMapRoot)
    local minX, maxX, minY, maxY = self:_CalcBoundsRect()
    if minX == math.huge then return end

    local w = maxX - minX
    local h = maxY - minY
    local cx = (minX + maxX) / 2
    local cy = (minY + maxY) / 2

    CallCS(function()
        CS.GameLib.MapLoadManager.SetupMapBounds(gameMapRoot, w, h, cx, cy)
    end)

    local cam = GameLib.MapCamHelper and GameLib.MapCamHelper.MapCamera or UnityEngine.Camera.main
    CallCS(function()
        CS.GameLib.MapLoadManager.RefreshMapBoundsAndCamera(gameMapRoot, EMapLayoutConfig.zoneHeight, cam)
    end)
end

---设置相机聚焦已解锁分区 + 同步 GameMapMotion 参数
function MapLoadManager:_SetupCamera()
    local cam = FindSceneCamera()
    if not cam then return end

    local zoneW = EMapLayoutConfig.zoneWidth
    local zoneH = EMapLayoutConfig.zoneHeight
    local totalW = self.UnlockedCount * zoneW
    local cx = totalW / 2
    local cy = -zoneH / 2

    -- 计算 orthoSize（用红框宽高比而非全屏比例）
    local screenW = CS.UnityEngine.Screen.width
    local screenH = CS.UnityEngine.Screen.height
    local aspect = screenW / screenH
    local ortho = totalW / 2 / aspect

    cam.orthographicSize = ortho
    cam.transform.position = CS.UnityEngine.Vector3(cx, cy, cam.transform.position.z)

    -- 同步 GameMapMotion 缩放/边界参数
    if GameMapMotion then
        GameMapMotion.ScreenCamera = cam
        GameMapMotion.FollowPosition = CS.UnityEngine.Vector3(cx, cy, 0)
        GameMapMotion.MaxOrthoSize = ortho
        GameMapMotion.MinOrthoSize = ortho
        GameMapMotion.LimitMinX = 0
        GameMapMotion.LimitMaxX = totalW
        GameMapMotion.LimitMinY = -zoneH
        GameMapMotion.LimitMaxY = 0
        GameMapMotion.MapCenterX = cx
        GameMapMotion.MapCenterY = cy
        GameMapMotion.MapHalfWidth = totalW / 2
        GameMapMotion.MapHalfHeigh = zoneH / 2
    end
end

--endregion

--region 【编排入口】

---加载地图骨架（编排入口）
---@param mapData table @MapData 实例
---@param chapterId string @章节ID
---@param unlockedCount number @已解锁分区数
---@param gameMapRoot UnityEngine.GameObject @GameMainMap 根节点
function MapLoadManager:LoadMap(mapData, chapterId, unlockedCount, gameMapRoot)
    if not mapData or not gameMapRoot or not chapterId then return end
    unlockedCount = Num(unlockedCount, 0)

    local unlockedDatas = self:_LoadConfig(chapterId, unlockedCount)
    local allSectionDatas = self:_LoadSections(chapterId, unlockedCount, unlockedDatas)
    local mainGo = gameMapRoot.parent and gameMapRoot.parent.gameObject
    self:_SetupGrid(mainGo, unlockedDatas)
    local zoneRoot = self:_CreateZoneRoot(gameMapRoot)
    if not zoneRoot then return end
    self:_PopulateZoneItems(allSectionDatas, unlockedCount)
    self:_SetupBounds(gameMapRoot)
    self:_SetupCamera()
end

---静态工厂（对齐 GameRequires `:Create()` 注册惯例）
---@return table @MapLoadManager 实例
function MapLoadManager.Create()
    return MapLoadManager
end

--endregion

return MapLoadManager
