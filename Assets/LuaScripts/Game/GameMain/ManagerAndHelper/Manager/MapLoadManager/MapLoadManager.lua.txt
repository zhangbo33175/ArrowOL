-- @core      编排私有方法链，C# GameLib.MapLoadManager 执行实际操作

local MapLoadManager = {}

--region 【状态】

MapLoadManager.ZoneRoot = nil        ---@type UnityEngine.Transform @ZoneRoot 容器节点
MapLoadManager.LevelData = nil       ---@type table @当前关卡地图数据

--endregion

--region 【配置加载】

---加载关卡配置，写入 EMapLayoutConfig 缓存
---@param chapterId string @章节ID
---@return table @关卡数据列表
function MapLoadManager:_LoadConfig(chapterId)
    local levelDatas = GamePlayHelper:GetUnlockedLevelDatas(chapterId, 1)
    local fd = levelDatas[1] and levelDatas[1].data or {}

    if not fd.Width then return levelDatas end

    local w = Num(fd.Width, 2048)
    local h = Num(fd.Height, 2048)
    EMapLayoutConfig.centerX = w / 2
    EMapLayoutConfig.centerY = h / 2
    EMapLayoutConfig.width = w
    EMapLayoutConfig.height = h
    EMapLayoutConfig.scale = 1.0
    EMapLayoutConfig.gridSize = Num(fd.GridSize, 32)
    EMapLayoutConfig.gridCols = Num(fd.GridCols, 64)
    EMapLayoutConfig.gridRows = Num(fd.GridRows, 64)
    EMapLayoutConfig.levelWidth = w
    EMapLayoutConfig.levelHeight = h

    return levelDatas
end

---加载关卡地图数据
---@param chapterId string @章节ID
---@param levelDatas table @关卡数据列表
---@return table @关卡数据列表
function MapLoadManager:_LoadLevelData(chapterId, levelDatas)
    local fd = levelDatas[1] and levelDatas[1].data or {}
    local levelCount = Num(fd.LevelCount, 1)

    local allLevelDatas = GamePlayHelper:GetLevelDatasByCount(chapterId, levelCount)
    self.LevelData = allLevelDatas
    return allLevelDatas
end

--endregion

--region 【网格系统】

---初始化 GridManager
---@param mainGo UnityEngine.GameObject @GameMainMap 父节点
---@param levelDatas table @关卡数据列表
function MapLoadManager:_SetupGrid(mainGo, levelDatas)
    if not mainGo then return end

    -- 收集占用格
    local occupied = {}
    for _, e in ipairs(levelDatas or {}) do
        local occ = e.data and e.data.OccupiedCells
        if occ then
            for _, c in ipairs(occ) do
                occupied[#occupied + 1] = Num(c.X, 0)
                occupied[#occupied + 1] = Num(c.Y, 0)
            end
        end
    end

    CallCS(function()
        local gridMgr = mainGo:GetComponent(typeof(CS.Honor.Runtime.GridManager))
        if not gridMgr then
            gridMgr = mainGo:AddComponent(typeof(CS.Honor.Runtime.GridManager))
        end
        gridMgr:Setup(
            EMapLayoutConfig.gridSize, EMapLayoutConfig.gridCols, EMapLayoutConfig.gridRows,
            EMapLayoutConfig.scale, self.ZoneRoot
        )
        gridMgr:SetOccupiedCells(occupied)
        gridMgr:BuildOverlay()
    end)
end

--endregion

--region 【关卡挂载】

---实例化 ZoneItem（单关卡模式：一张地图一个 ZoneItem）
---@param levelDatas table @关卡数据列表
function MapLoadManager:_PopulateZoneItems(levelDatas)
    if not self.ZoneRoot then return end

    local data = levelDatas and levelDatas[1] and levelDatas[1].data or {}

    -- 加载 ZoneItem 预制体
    local ok, zoneGo = CallCS(function()
        local r = LoadRes["GameObject_ZoneItem"]
        if not r then return nil end
        return GameMainRoot.HonorAssetManager:LoadPrefabSync(r.ABPath, r.AssetName, self.ZoneRoot, nil)
    end)
    if not ok or not zoneGo then return end

    CallCS(function()
        -- 节点命名
        zoneGo.name = "ZoneItem"

        -- 设置 RectTransform（中心 pivot，与 GameMainMap 对齐）
        local rt = zoneGo:GetComponent(typeof(UnityEngine.RectTransform))
        if rt then
            rt.pivot = UnityEngine.Vector2(0.5, 0.5)
            rt.anchorMin = UnityEngine.Vector2(0.5, 0.5)
            rt.anchorMax = UnityEngine.Vector2(0.5, 0.5)
            rt.sizeDelta = UnityEngine.Vector2(EMapLayoutConfig.width, EMapLayoutConfig.height)
            rt.anchoredPosition = UnityEngine.Vector2(0, 0)
        end

        -- 设置 RawImage 子节点（确保图像居中）
        local rawImg = zoneGo.transform:Find("RawImage")
        if rawImg then
            local imgRt = rawImg:GetComponent(typeof(UnityEngine.RectTransform))
            if imgRt then
                imgRt.pivot = UnityEngine.Vector2(0.5, 0.5)
                imgRt.anchorMin = UnityEngine.Vector2(0.5, 0.5)
                imgRt.anchorMax = UnityEngine.Vector2(0.5, 0.5)
                imgRt.sizeDelta = UnityEngine.Vector2(EMapLayoutConfig.width, EMapLayoutConfig.height)
                imgRt.anchoredPosition = UnityEngine.Vector2(0, 0)
            end
        end

        -- 设置背景图（兼容新旧格式：新格式在 Meta.BgName，旧格式在根级别 BgName）
        local bgName = (data.Meta and data.Meta.BgName) or data.BgName
        if bgName then
            local img = rawImg and rawImg:GetComponent(typeof(UnityEngine.UI.Image))
            if img then
                local r = LoadSpriteByName(bgName)
                if r then
                    local sprite = LoadAssetSync(r)
                    if sprite then img.sprite = sprite end
                end
            end
        end
    end)
end

--endregion

--region 【包围盒与相机】

---适配包围盒（整图尺寸，中心坐标系：原点在地图中心）
---@param zoneRoot UnityEngine.Transform @ZoneRoot 节点
function MapLoadManager:_SetupBounds(zoneRoot)
    local w = EMapLayoutConfig.width
    local h = EMapLayoutConfig.height
    local cx = 0
    local cy = 0

    -- 传入 GameMainMap 根节点（zoneRoot 的父节点），SetupMapBounds 会在下面找 ZoneRoot 子节点
    local targetGo = zoneRoot and zoneRoot.parent and zoneRoot.parent.gameObject
    if not targetGo then return end

    CallCS(function()
        -- 禁用 GridManager / Grid Layout Group（单关卡模式不需要自动排列）
        local gridMgr = targetGo:GetComponent("GridManager")
        if gridMgr then gridMgr.enabled = false end

        local glg = zoneRoot:GetComponent(typeof(UnityEngine.UI.GridLayoutGroup))
        if glg then glg.enabled = false end

        -- 禁用 MapMotionLayer（自动相机动画），完全手动控制相机
        local motionLayer = zoneRoot:GetComponent("MapMotionLayer")
        if motionLayer then motionLayer.enabled = false end

        -- 只设置包围盒，不自动刷新相机（避免算出错误的 Y 值）
        CS.GameLib.MapLoadManager.SetupMapBounds(targetGo, w, h, cx, cy)
    end)
end

---设置相机：整图视角 + 同步 FollowPosition
---@param zoneRoot UnityEngine.Transform @ZoneRoot 节点
function MapLoadManager:_SetupCamera(zoneRoot)
    -- 兼容初始化顺序：如果 CameraManager 还没 Init，先手动初始化
    if not GameMapCameraManager.IsInitialized then
        local mapData = MapManager.CurMapData
        if mapData then
            GameMapCameraManager:Init(mapData)
        end
    end

    if not GameMapCameraManager.IsInitialized then
        return
    end

    -- 设置地图根节点，相机逻辑全部由 GameMapCameraManager 自动处理
    GameMapCameraManager:SetGameMapRoot(zoneRoot)
end

--endregion

--region 【编排入口】

---加载地图骨架（编排入口）
---@param mapData table @MapData 实例
---@param chapterId string @章节ID
---@param levelId number @关卡ID（单关卡模式恒为 1）
---@param zoneRoot UnityEngine.Transform @ZoneRoot 容器节点（GameMapBase 已 Find 传入）
function MapLoadManager:LoadMap(mapData, chapterId, levelId, zoneRoot)
    if not mapData or not zoneRoot or not chapterId then return end

    local levelDatas = self:_LoadConfig(chapterId)
    local allLevelDatas = self:_LoadLevelData(chapterId, levelDatas)
    local mainGo = zoneRoot.parent and zoneRoot.parent.gameObject

    self.ZoneRoot = zoneRoot
    self:_SetupGrid(mainGo, levelDatas)
    self:_PopulateZoneItems(allLevelDatas)
    self:_SetupBounds(zoneRoot)
    self:_SetupCamera(zoneRoot)
end

---静态工厂（对齐 GameRequires `:Create()` 注册惯例）
---@return table @MapLoadManager 实例
function MapLoadManager.Create()
    return MapLoadManager
end

--endregion

return MapLoadManager
