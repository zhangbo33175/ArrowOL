--========================================================================
-- 主目标相关功能
--========================================================================
---获取当前已完成的主目标数量
---@return number @已完成主目标数量
function GameCoreManagerData:CurRootGoalCount()
    -- 根据当前的目标类型，计算当前完成的目标数量
    return 0
end

---获取当前主目标ID
---@return string @主目标唯一ID
function GameCoreManagerData:CurRootGoalID()
    return self._RootGoalData.ID
end

---获取完成主目标所需的总数量
---@return number @目标总数量
function GameCoreManagerData:TotalRootGoalCount()
    return self._TotalRootGoalCount
end

---判断主目标是否已完成
---@return boolean @true=已完成
function GameCoreManagerData:IsRootGoalReach()
    return self:CurRootGoalCount() >= self._TotalRootGoalCount
end

---判断当前是否为特殊客人关卡
---@return boolean @true=特殊客人关卡
function GameCoreManagerData:IsSpecialGuest()
    return self:CurRootGoalID() == EGoalId.specialGuestCount or self:IsEnableAdditionalGoal(EGoalId.specialGuestCount)
end

--========================================================================
-- 寻找物数量相关
--========================================================================
---减少剩余寻找物数量（触发后派发事件）
function GameCoreManagerData:SubRemainGuestCount()
    self._RemainGuestCount = self._RemainGuestCount - 1
    HonorEventHelper:Fire(EventCmd.ChangeOnTheNumberOfRemainingSearchItems)
end

--========================================================================
-- 道具使用统计相关
--========================================================================
---获取关卡内道具使用明细
---@return table @{id=道具ID, num=使用数量}数组
function GameCoreManagerData:GetBoosterUse()
    if not self._LevelBoosterUsed then
        return nil
    end
    local result = {}
    for id, num in pairs(self._LevelBoosterUsed) do
        table.insert(result, { id = id, num = num })
    end
    return result
end

---检查是否使用过任意关内道具
---@return boolean @true=已使用关内道具
function GameCoreManagerData:IsUsedBooster()
    if not self._LevelBoosterUsed then
        return false
    end

    for id, num in pairs(self._LevelBoosterUsed) do
        if ItemManager:IsBooster(id) then
            return true
        end
    end
    return false
end

---检查是否使用过任意前置道具
---@return boolean @true=已使用前置道具
function GameCoreManagerData:IsUsedPreBooster()
    if not self._LevelBoosterUsed then
        return false
    end

    for id, num in pairs(self._LevelBoosterUsed) do
        if ItemMgr:IsPreBooster(id) then
            return true
        end
    end
    return false
end

---记录关卡内道具使用
---@param id string @道具唯一ID
---@param num number @使用数量
function GameCoreManagerData:RecordLevelBoosterUse(id, num)
    self._LevelBoosterUsed = self._LevelBoosterUsed or {}
    self._LevelBoosterUsed[id] = (self._LevelBoosterUsed[id] or 0) + num
end

return GameCoreManagerData
