---@class GameCoreManagerData @ 关卡核心运行时数据管理类
local GameCoreManagerData = class("GameCoreManagerData")

--- 构造函数：初始化所有运行时数据默认值
function GameCoreManagerData:ctor()
    --- 主目标配置
    self._RootGoalData = nil                    ---@type Tables.Goals_Item              主目标配置
    self._TotalRootGoalCount = 0                ---@type number                         主目标总需求数量
    self._RemainGuestCount = 0                  ---@type number                         剩余寻找物数量

    --- 关卡限制条件
    self._LimitCheckData = nil                  ---@type Tables.LimitCheck_Item[]       关卡限制条件配置
    self._LimitTriggered = nil                  ---@type string[]                      已触发的限制条件
    self._TutorialIgnoreLimitGoal = nil         ---@type table                         引导中忽略的限制条件

    --- 关卡基础信息
    self._ComboCount = 0                        ---@type number                         连击次数
    self._LevelDifficulty = 1                   ---@type number                         关卡难度系数
    self._LevelMode = 0                          ---@type number                         关卡玩法模式
    self._LevelID = 0                           ---@type number                         当前关卡ID

    --- 道具 & 复活 & 体力
    self._ReviveNum = 0                         ---@type number                         复活次数
    self._LevelBoosterUsed = nil                ---@type table<string, number>          道具使用统计
    self._LifeCost = false                      ---@type boolean                        是否消耗体力
    self._UseDoublePoints = false               ---@type boolean                        是否双倍积分

    --- 关卡结果
    self._StarCount = 0                         ---@type number                         获得星星数量
    self._IsLatestChapterLevel = false          ---@type boolean                        是否最新章节关卡
    self._watchReviveAdvertiseCount = 0         ---@type number                         复活广告观看次数
    self._showWatchAdvertiseButtonCount = 0     ---@type number                         复活广告显示次数
end

--- 静态创建实例
---@return GameCoreManagerData
function GameCoreManagerData:Create()
    local instance = GameCoreManagerData.new()
    return instance
end

--- 根据关卡配置初始化核心数据
---@param levelData Tables.LevelData_Item 关卡配置
function GameCoreManagerData:InitData(levelData)
end

--- 清空运行时数据（关卡退出/重置时调用）
function GameCoreManagerData:ClearData()
end

--- 保存玩家游戏数据到本地存档
---@desc 包含星级、货币、道具使用等关键进度数据
function GameCoreManagerData:SaveData()
    self:SaveStarCount()
    self:SaveTotalMoney()
end

--- 动态设置关卡难度系数
function GameCoreManagerData:SetupDifficultyFactor()
end

--========================================================================
-- 主目标相关
--========================================================================

--- 获取当前已完成的主目标数量
---@return number
function GameCoreManagerData:CurRootGoalCount()
    -- 根据目标类型动态计算已完成数量
    return 0
end

--- 获取当前主目标ID
---@return string
function GameCoreManagerData:CurRootGoalID()
    return self._RootGoalData and self._RootGoalData.ID or ""
end

--- 获取主目标总需求数量
---@return number
function GameCoreManagerData:TotalRootGoalCount()
    return self._TotalRootGoalCount
end

--- 主目标是否已完成
---@return boolean
function GameCoreManagerData:IsRootGoalReach()
    return self:CurRootGoalCount() >= self._TotalRootGoalCount
end

--- 是否为特殊客人关卡
---@return boolean
function GameCoreManagerData:IsSpecialGuest()
    local curGoalId = self:CurRootGoalID()
    return curGoalId == EGoalId.specialGuestCount or self:IsEnableAdditionalGoal(EGoalId.specialGuestCount)
end

--========================================================================
-- 寻找物数量相关
--========================================================================

--- 减少剩余寻找物数量，并派发事件
function GameCoreManagerData:SubRemainGuestCount()
    self._RemainGuestCount = math.max(0, self._RemainGuestCount - 1)
    HonorEventHelper:Fire(EventCmd.ChangeOnTheNumberOfRemainingSearchItems)
end

--========================================================================
-- 道具使用统计
--========================================================================

--- 获取关卡内道具使用明细列表
---@return table[]
function GameCoreManagerData:GetBoosterUse()
    if not self._LevelBoosterUsed then
        return nil
    end

    local result = {}
    for id, num in pairs(self._LevelBoosterUsed) do
        table.insert(result, { id = id, num = num })
    end
    return result
end

--- 是否使用过任意【关内道具】
---@return boolean
function GameCoreManagerData:IsUsedBooster()
    if not self._LevelBoosterUsed then
        return false
    end

    for id, num in pairs(self._LevelBoosterUsed) do
        if ItemManager:IsBooster(id) then
            return true
        end
    end
    return false
end

--- 是否使用过任意【前置道具】
---@return boolean
function GameCoreManagerData:IsUsedPreBooster()
    if not self._LevelBoosterUsed then
        return false
    end

    for id, num in pairs(self._LevelBoosterUsed) do
        if ItemManager:IsPreBooster(id) then
            return true
        end
    end
    return false
end

--- 记录关卡内道具使用数量
---@param id string 道具ID
---@param num number 使用数量
function GameCoreManagerData:RecordLevelBoosterUse(id, num)
    self._LevelBoosterUsed = self._LevelBoosterUsed or {}
    self._LevelBoosterUsed[id] = (self._LevelBoosterUsed[id] or 0) + num
end

return GameCoreManagerData
