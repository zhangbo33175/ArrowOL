---@class GameCoreManagerData @关卡核心运行时数据管理类，GameCoreManager内部数据载体
local GameCoreManagerData = class("GameCoreManagerData")

--region 构造 & 实例初始化

---=========================================================================
--- 构造函数
---@brief 实例新建时初始化所有运行时数据默认初始值，统一字段类型
---@type fun(args:table):void
---@param args table 构造自定义入参（预留扩展）
---=========================================================================
function GameCoreManagerData:ctor()
    -- 地图寻找物/角色收集相关集合
    self._CharacterList         = {}        ---@type table<string, table> 本局全部可收集角色/寻找物总集合，ID为Key
    self._CharacterChooseList   = {}        ---@type table<string, table> 本局玩家已收集完成的角色/寻找物集合
    self._NumberOfPropsUsed     = 0         ---@type number 本局累计使用道具总数量
    self._LifeNumber            = 3         ---@type number 本局初始剩余可操作生命值

    -- 关卡通关限制条件配置缓存
    self._LimitCheckData        = nil       ---@type Tables.LimitCheck_Item[] 关卡全部限制规则静态表数据数组
    self._LimitTriggered        = nil       ---@type string[] 本局已经触发过的限制条件ID列表

    -- 关卡基础运行参数
    self._LevelDifficulty       = 1         ---@type number 本局关卡难度系数，影响判定、积分、生成逻辑
    self._LevelMode             = 0         ---@type number 关卡玩法模式ID，区分普通/限时/挑战等分支逻辑
    self._LevelID               = 0         ---@type number 当前加载关卡唯一静态ID
end

---=========================================================================
--- 静态创建实例对外入口
---@brief 统一实例化创建接口，隔离new直接调用
---@type fun(args:table):GameCoreManagerData
---@param args table 自定义构造参数（预留）
---@return GameCoreManagerData 全新空数据实例
---=========================================================================
function GameCoreManagerData:Create()
    local instance = GameCoreManagerData.new()
    return instance
end

---=========================================================================
--- 基于关卡静态配置初始化本局全部运行数据
---@brief 加载关卡预制配置，重置所有临时计数、初始化限制规则、清空收集列表
---@type fun(levelData:Tables.LevelData_Item):void
---@param levelData Tables.LevelData_Item 关卡静态配表行数据
---=========================================================================
function GameCoreManagerData:InitData(levelData)
    -- 空配置拦截，避免空索引报错
    if levelData == nil then
        print("[GameCoreManagerData] InitData 传入关卡配置为空，初始化中断")
        return
    end

    -- 本局临时计数全部重置归零
    self._NumberOfPropsUsed = 0
    self._LifeNumber = 3

    -- 读取关卡基础标识与难度配置
    self._LevelID = levelData.Id
    self._LevelMode = levelData.LevelMode
    self._LevelDifficulty = levelData.Difficulty

    -- 初始化关卡限制条件容器
    self._LimitCheckData = {}
    self._LimitTriggered = {}

    -- 遍历配置内限制ID，加载对应静态表数据存入缓存
    if levelData.LimitCheckIds and #levelData.LimitCheckIds > 0 then
        for _, limitId in ipairs(levelData.LimitCheckIds) do
            local limitItem = Tables.LimitCheck[limitId]
            if limitItem then
                table.insert(self._LimitCheckData, limitItem)
            end
        end
    end

    -- 清空收集物列表，等待地图生成角色填充
    self._CharacterList = {}
    self._CharacterChooseList = {}

    -- 默认非章节最新解锁关卡，业务层按需修改赋值
    self._IsLatestChapterLevel = false
end

---=========================================================================
--- 清空本局全部运行时数据
---@brief 关卡退出、重玩、通关失败/成功统一调用；table容器置空切断引用，防止内存泄漏
---@type fun():void
---=========================================================================
function GameCoreManagerData:ClearData()
    -- 集合容器置空释放引用
    self._CharacterList = {}
    self._CharacterChooseList = {}

    self._LimitCheckData = nil
    self._LimitTriggered = nil

    -- 数值类全部归零恢复初始值
    self._NumberOfPropsUsed = 0
    self._LifeNumber = 3
    self._LevelDifficulty = 1
    self._LevelMode = 0
    self._LevelID = 0

    self._IsLatestChapterLevel = false
end

---=========================================================================
--- 持久化保存本局关卡进度数据至本地存档
---@brief 通关后调用
---@type fun():void
---=========================================================================
function GameCoreManagerData:SaveData()
end

--endregion

--region 角色/寻找物收集列表 对外只读查询接口

---=========================================================================
--- 获取本局全部可收集角色/寻找物总集合
---@type fun():table<string, table>
---@return table<string, table> 角色ID为Key的完整角色字典
---=========================================================================
function GameCoreManagerData:GetCharacterList()
    return self._CharacterList
end

---=========================================================================
--- 获取本局玩家已收集完成的角色/寻找物集合
---@type fun():table<string, table>
---@return table<string, table> 已收集角色字典
---=========================================================================
function GameCoreManagerData:GetCharacterChooseList()
    return self._CharacterChooseList
end

---=========================================================================
--- 获取本局所有可收集角色总数量
---@type fun():number
---@return number 总角色条目计数
---=========================================================================
function GameCoreManagerData:GetCharacterListCount()
    return GetTableCount(self._CharacterList)
end

---=========================================================================
--- 获取本局当前已成功收集的角色数量
---@type fun():number
---@return number 已收集角色条目计数
---=========================================================================
function GameCoreManagerData:GetCharacterChooseListCount()
    return GetTableCount(self._CharacterChooseList)
end

---=========================================================================
--- 获取收集进度文本（已收集/总数量），直接用于UI文本展示
---@type fun():string
---@return string 拼接后的进度字符串，格式 "已收集数/总数"
---=========================================================================
function GameCoreManagerData:GetCharacterAndChooseListCount()
    return GetTableCount(self._CharacterChooseList) .. "/" .. GetTableCount(self._CharacterList)
end

---=========================================================================
--- 获取本局累计使用道具总数量
---@type fun():number
---@return number 全局道具使用合计数值
---=========================================================================
function GameCoreManagerData:GetNumberOfPropsUsed()
    return self._NumberOfPropsUsed
end

---=========================================================================
--- 查询当前剩余可操作生命值
---@type fun():number
---@return number 剩余生命点数，最低为0
---=========================================================================
function GameCoreManagerData:GetLifeNumber()
    return self._LifeNumber
end

---=========================================================================
--- 扣除一点操作生命值，做下限保护最低0，并返回扣除后数值
---@type fun():number
---@return number 扣除后剩余生命值
---=========================================================================
function GameCoreManagerData:SubLifeNumber()
    self._LifeNumber = math.max(0, self._LifeNumber - 1)
    return self._LifeNumber
end

--endregion

--region 关卡道具使用记录 业务接口

---=========================================================================
--- 记录单次道具使用，自动累加对应道具计数与全局道具总数
---@type fun(id:string, num:number):void
---@param id string 道具静态配置唯一ID
---@param num number 本次使用道具数量
---=========================================================================
function GameCoreManagerData:RecordLevelBoosterUse(id, num)
    self._NumberOfPropsUsed = self._NumberOfPropsUsed + num
end

--endregion

return GameCoreManagerData
