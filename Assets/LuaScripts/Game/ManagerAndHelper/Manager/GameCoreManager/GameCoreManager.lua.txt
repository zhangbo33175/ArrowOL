---@class GameCoreManager : Singleton @ 关卡核心管理器：游戏状态、胜负、体力、道具、目标、结束流程总控
---@field _GameCoreManagerWaiter            GameCoreManagerWaiter|nil       指令调度器
---@field _GameCoreManagerData              GameCoreManagerData             核心数据模块
---@field _GameCoreManagerLevelFinish       GameCoreManagerLevelFinish      关卡结束流程
---@field _GameCoreManagerUseProp           GameCoreManagerUseProp|nil      道具使用
---@field _GameCoreManagerEffect            GameCoreManagerEffect           特效管理
---@field _CurLevelData                     Tables.LevelData_Item|nil       当前关卡配置表数据
---@field _IsPlaying                        boolean                         是否游戏进行中
---@field _IsGameOver                       boolean                         是否游戏结束
---@field _IsExitGame                       boolean                         是否退出游戏
---@field _IsSuccessFinishGame              boolean                         是否成功通关
---@field _eGameOverType                    EGameOverType                   游戏结束枚举类型
---@field _HandleGameOverMsgQueue           HonorQueue                      游戏结束消息安全队列
---@field _HandleGameOverMsgQueueCoroutine  table|nil                       结束队列异步协程句柄
---@field _ExitSureTip                      table                           退出二次弹窗提示配置
---@field _ExitSureTipAgain                 table                           多次退出弹窗提示配置
local GameCoreManager = class("GameCoreManager", import("Singleton"))

--region 生命周期 & 构造/初始化

---=========================================================================
--- 构造函数
---@brief 实例初始化，创建子模块、初始化状态标记、消息队列、提示缓存
---@param args table 实例构造自定义参数
---=========================================================================
function GameCoreManager:ctor(args)
    GameCoreManager.super.ctor(self, args)

    -- 游戏运行状态标记
    self._IsPlaying                     = false
    self._IsGameOver                    = false
    self._IsExitGame                    = false
    self._IsSuccessFinishGame           = false
    self._CurLevelData                  = nil
    self._eGameOverType                 = EGameOverType.None

    -- 游戏结束异步消息队列，防止多逻辑同时触发结束冲突
    self._HandleGameOverMsgQueue        = HonorQueue:Create()
    self._HandleGameOverMsgQueueCoroutine = nil

    -- 退出弹窗提示缓存列表
    self._ExitSureTip                   = {}
    self._ExitSureTipAgain              = {}

    -- 子模块初始化（解耦架构，GameCoreManagerSkill 文件缺失，注释屏蔽加载）
    self._GameCoreManagerData           = require("GameCoreManagerData"):Create()
    self._GameCoreManagerLevelFinish    = require("GameCoreManagerLevelFinish"):Create()
    self._GameCoreManagerEffect         = require("GameCoreManagerEffect"):Create()
end

---=========================================================================
--- 静态创建管理器实例
---@brief 对外统一创建入口，打印初始化日志
---@param args table 实例自定义入参
---@return GameCoreManager 核心管理器实例对象
---=========================================================================
function GameCoreManager:Create(args)
    local instance = GameCoreManager.new(args)
    PrintInfo("[GameCoreManager] 核心管理器初始化完成")
    return instance
end

---=========================================================================
--- 清空本局所有数据与状态
---@brief 切换关卡、重玩、退出时调用，重置状态、清空队列、清理所有子模块缓存
---=========================================================================
function GameCoreManager:ClearData()
    -- 重置游戏运行状态标记
    self._IsPlaying = false
    self._IsGameOver = false
    self._IsExitGame = false
    self._IsSuccessFinishGame = false
    self._eGameOverType = EGameOverType.None
    self._CurLevelData = nil

    -- 清空结束消息队列并停止协程
    self._HandleGameOverMsgQueue:Clear()
    self:StopCheckGameOverMsgQueue()

    -- 清空退出弹窗缓存
    self._ExitSureTip = {}
    self._ExitSureTipAgain = {}

    -- 逐层清理子模块缓存（Skill模块注释屏蔽，防止nil调用报错）
    self._GameCoreManagerData:ClearData()
    self._GameCoreManagerEffect:ClearData()
    self._GameCoreManagerLevelFinish:ClearData()

    PrintInfo("[GameCoreManager] 本局运行时数据全部清空")
end

--endregion

--region 游戏胜负逻辑

---=========================================================================
--- 触发游戏胜利流程
---@brief 游戏收集目标完成后统一调用，标记通关状态、推送任务、执行结算表现
---=========================================================================
function GameCoreManager:OnGameVictory()
    -- 防止重复触发通关逻辑
    if self._IsGameOver then
        PrintWarn("[GameCoreManager] 当前已结束，重复触发胜利，拦截")
        return
    end
    self._IsSuccessFinishGame = true
    self._IsGameOver = true
    -- 同步通关任务进度
    self:SendQuestFinishProgress()
    -- 底层游戏结束标记
    GamePlayHelper:GameOver(true)
    -- 结算后置动画、弹窗逻辑
    self._GameCoreManagerLevelFinish:OnGameOverAfter()
    self._GameCoreManagerLevelFinish:StartAloneLevelFinishProcess()

    PrintInfo("[GameCoreManager] 关卡胜利，开启结算流程")
end

---=========================================================================
--- 同步关卡完成任务进度
---@brief 通关后上报关卡星级、完成进度、任务完成状态至服务器/本地存档
---=========================================================================
function GameCoreManager:SendQuestFinishProgress()
    -- 业务拓展：同步关卡完成进度、星级、任务完成标记
    PrintInfo("[GameCoreManager] 同步关卡完成任务进度")
end

---=========================================================================
--- 触发游戏失败流程
---@brief 生命值耗尽、超时等失败场景统一入口
---@param eGameOverType EGameOverType 失败原因枚举
---=========================================================================
function GameCoreManager:DoGameFailed(eGameOverType)
    -- 防止重复触发失败逻辑
    if self._IsGameOver then
        PrintWarn("[GameCoreManager] 游戏已结束，重复触发失败，拦截")
        return
    end
    self._eGameOverType = eGameOverType
    self._IsSuccessFinishGame = false
    self._IsGameOver = true
    GamePlayHelper:GameOver(false)
    self._GameCoreManagerLevelFinish:OnGameOverAfter()
    PrintInfo("[GameCoreManager] 关卡失败，结束类型：", eGameOverType)
end

---=========================================================================
--- 执行复活逻辑
---@brief 玩家观看广告复活，重置结束状态、增加复活计数、恢复游戏交互
---=========================================================================
function GameCoreManager:DoGameRevive()
    -- 未结束时禁止复活调用
    if not self._IsGameOver then
        PrintWarn("[GameCoreManager] 游戏未结束，无法执行复活")
        return
    end
    self._IsGameOver = false
    self._IsSuccessFinishGame = false
    self._eGameOverType = EGameOverType.None
    -- 数据模块记录本局复活次数
    self._GameCoreManagerData:RecordRevive()
    PrintInfo("[GameCoreManager] 执行复活逻辑，恢复游戏运行")
end

---=========================================================================
--- 获取本局游戏结束类型
---@brief 供结算、音效、弹窗分支判断使用
---@return EGameOverType 游戏结束枚举值
---=========================================================================
function GameCoreManager:GetGameOverType()
    return self._eGameOverType
end

--region 游戏结束判定 & 消息队列（核心防重入机制）

---=========================================================================
--- 单条结束消息业务分发逻辑
---@brief 根据结束类型分发胜利/失败分支
---@param eGameOverType EGameOverType 结束枚举
---@param bounds LuaBoundsData 区域数据
---=========================================================================
function GameCoreManager:CheckGameResult(eGameOverType, bounds)
    PrintInfo("[GameCoreManager] 处理结束消息，类型：", eGameOverType)
    -- 业务拓展：判定胜利/失败、目标完成、扣除体力
    self._eGameOverType = eGameOverType
    if eGameOverType == EGameOverType.Success then
        self:OnGameVictory()
    else
        self:DoGameFailed(eGameOverType)
    end
end

---=========================================================================
--- 校验关卡主收集目标是否全部达成
---@brief 触发结束前预校验是否满足通关条件
---@param hasCoins boolean 是否包含金币收集目标
---@param eGameOverType EGameOverType 结束类型
---@param bounds LuaBoundsData 区域坐标数据
---@return boolean true=目标全部完成 false=未集齐
---=========================================================================
function GameCoreManager:CheckGoalReach(hasCoins, eGameOverType, bounds)
    if not self._GameCoreManagerData then
        return false
    end
    return self._GameCoreManagerData:IsRootGoalReach()
end

---=========================================================================
--- 停止并销毁结束消息协程、清空队列
---@brief 关卡切换、重玩、退出时释放协程资源，防止内存泄漏
---=========================================================================
function GameCoreManager:StopCheckGameOverMsgQueue()
    if self._HandleGameOverMsgQueueCoroutine then
        HonorCoroutineHelper:StopAlone(self._HandleGameOverMsgQueueCoroutine)
        self._HandleGameOverMsgQueueCoroutine = nil
    end
    self._HandleGameOverMsgQueue:Clear()
end

end

--region 游戏全生命周期调度

---=========================================================================
--- 初始化本局游戏数据
---@brief 关卡加载完成后一次性执行，重置收集、机器、道具状态
---=========================================================================
function GameCoreManager:OnInitGame()
    PrintInfo("[GameCoreManager] 初始化本局游戏数据")
end

---=========================================================================
--- 进入关卡准备阶段
---@brief 加载资源、预生成实体阶段，屏蔽玩家交互
---=========================================================================
function GameCoreManager:OnEnterGame()
    self._IsPlaying = false
    self._IsExitGame = false
end

---=========================================================================
--- 正式开启游戏交互
---@brief 资源加载完成，允许触摸、操作、道具使用
---=========================================================================
function GameCoreManager:OnBeginGame()
    self._IsPlaying = true
    self._IsGameOver = false
    self._IsSuccessFinishGame = false
    self._eGameOverType = EGameOverType.None
    PrintInfo("[GameCoreManager] 游戏正式开始运行")
end

---=========================================================================
--- 游戏暂停
---@brief 弹窗、剧情、切UI、广告时调用，屏蔽所有操作
---=========================================================================
function GameCoreManager:OnPauseGame()
    if not GamePlayHelper:IsInLevelState() then
        return
    end
    self._IsPlaying = false
    PrintInfo("[GameCoreManager] 游戏暂停")
end

---=========================================================================
--- 解除暂停，恢复玩家交互
---@brief 关闭弹窗、跳过剧情、广告结束后调用
---=========================================================================
function GameCoreManager:OnResumeGame()
    if not GamePlayHelper:IsInLevelState() then
        return
    end
    self._IsPlaying = true
    PrintInfo("[GameCoreManager] 游戏恢复运行")
end

---=========================================================================
--- 游戏结束统一销毁入口
---@brief 胜利、失败、主动退出都会执行，释放所有子模块资源
---=========================================================================
function GameCoreManager:OnGameOver()
    self._IsPlaying = false
    self._IsExitGame = false
    -- 停止结束消息协程队列
    self:StopCheckGameOverMsgQueue()
    -- 停止全局背景音乐
    SoundManager:StopCurrentBgm()
    self._GameCoreManagerData:ClearData()
    PrintInfo("[GameCoreManager] 本局游戏统一结束，释放运行资源")
end

---=========================================================================
--- 主动退出当前关卡
---@brief 返回章节选择、主界面时调用
---=========================================================================
function GameCoreManager:OnExitGame()
    self._IsPlaying = false
    self._IsExitGame = true
    self:OnGameOver()
end

---=========================================================================
--- 重玩当前关卡
---@brief 结算弹窗重玩按钮触发，清空数据并重启游戏流程
---=========================================================================
function GameCoreManager:OnRePlayGame()
    self:ClearData()
    self:OnBeginGame()
end

---=========================================================================

---@return table 弹窗ID与多语言文本映射表
---=========================================================================
function GameCoreManager:GetExitSureTip()
    return self._ExitSureTip
end

---=========================================================================
--- 判断是否存在待弹出的退出提示
---@brief 返回主界面前前置判断是否需要弹窗拦截
---@return boolean true=存在弹窗配置 false=无提示直接退出
---=========================================================================
function GameCoreManager:HasExitSureTip()
    return not TableIsEmpty(self._ExitSureTip)
end

---=========================================================================
--- 注册一条退出拦截弹窗
---@brief 不同关卡、剧情节点添加对应提示文本
---@param id any 弹窗唯一标识ID
---@param strNum string 多语言表文本Key
---=========================================================================
function GameCoreManager:RegisterExitSureTip(id, strNum)
    local tip = self:FindExitSureTip(id)
    if tip then
        tip.strNum = strNum
    else
        table.insert(self._ExitSureTip, { id = id, strNum = strNum })
    end
end

---=========================================================================
--- 移除指定ID的退出弹窗配置
---@brief 剧情完成、关卡切换时清除过期提示
---@param id any 弹窗唯一ID
---=========================================================================
function GameCoreManager:UnRegisterExitSureTip(id)
    for i, t in ipairs(self._ExitSureTip) do
        if t.id == id then
            table.remove(self._ExitSureTip, i)
            break
        end
    end
end

---=========================================================================
--- 根据ID查找已注册弹窗配置
---@brief 更新弹窗文本时查询缓存
---@param id any 弹窗标识ID
---@return table|nil 弹窗配置对象 / nil 不存在
---=========================================================================
function GameCoreManager:FindExitSureTip(id)
    for _, t in ipairs(self._ExitSureTip) do
        if t.id == id then
            return t
        end
    end
    return nil
end

--endregion

--region 关卡目标进度对外接口

---=========================================================================
--- 获取主收集目标基础信息
---@brief 进度UI、结算面板展示
---@return string 目标ID, number 总需求数量, number 当前已完成数量
---=========================================================================
function GameCoreManager:GetMainGoalCfg()
    if not self._GameCoreManagerData then
        return "", 0, 0
    end
    local mainId = self._GameCoreManagerData:CurRootGoalID()
    local mainNum = self._GameCoreManagerData:TotalRootGoalCount()
    local mainProgress = self._GameCoreManagerData:CurRootGoalCount()
    return mainId, mainNum, mainProgress
end

---=========================================================================
--- 根据副目标ID查询完成进度
---@brief 多支线收集玩法进度查询
---@param goalId string 支线目标唯一ID
---@return number|nil 总数量, number|nil 当前完成数量
---=========================================================================
function GameCoreManager:GetAdditionalGoalProgressById(goalId)
    if not self._GameCoreManagerData then
        return nil, nil
    end
    if not self._GameCoreManagerData:IsEnableAdditionalGoal(goalId) then
        return nil, nil
    end

    local total = self._GameCoreManagerData:TotalAdditionalGoalCount(goalId)
    local progress = self._GameCoreManagerData:CurAdditionalGoalCount(goalId)
    return total, progress
end

---=========================================================================
--- 角色点击交互统一业务处理
---@brief 地图点击角色实体回调入口，处理收集、通关判定、扣生命逻辑
---@param character CharacterBase 点击的角色实体对象
---=========================================================================
function GameCoreManager:SetOnGameplay(character)
    if character then
        -- 读取全局角色列表缓存
        local characterChooseList = self:GetCharacterChooseList()
        local characterList = self:GetCharacterList()

        -- 未收集则加入已收集列表，并销毁原候选实体
        if not ContainsKey(characterChooseList, character.ID) then
            if characterChooseList then
                characterChooseList[character.ID] = {}
                characterChooseList[character.ID].ID = character.ID
                characterChooseList[character.ID].Character = character
            end
            if characterList[character.ID] and characterList[character.ID].Character then
                characterList[character.ID].Character:OnDestroy()
            end
        end

        -- 已收集数量等于总数量则直接通关
        if self:GetCharacterChooseListCount() == self:GetCharacterListCount() then
            self:OnGameVictory()
        end
    else
        -- 点击空白无匹配角色，扣除生命值
        self:SubLifeNumber()
        HonorEventHelper:Fire(EventCmd.OnClickCharacter)

        -- 生命值清零触发失败流程
        local remainLife = self:GetLifeNumber()
        if remainLife == 0 then
            --self:FindExitSureTip(EGameOverType.Failure)
        end
    end
end
