---@field _HandleGameOverMsgQueueCoroutine  table                结束队列协程

local GameCoreManager = class("GameCoreManager")

--- 构造函数
function GameCoreManager:ctor(args)
    -- 游戏状态
    self._IsPlaying = false
    self._IsGameOver = false
    self._IsExitGame = false
    self._IsSuccessFinishGame = false
    self._CurLevelData = nil
    self._eGameOverType = EGameOverType.None

    -- 结束消息队列
    self._HandleGameOverMsgQueue = HonorQueue:Create()
    self._HandleGameOverMsgQueueCoroutine = nil

    -- 退出提示
    self._ExitSureTip = {}
    self._ExitSureTipAgain = {}

    -- 子模块初始化
    self._GameCoreManagerData = require("GameCoreManagerData"):Create()
    self._GameCoreManagerLevelFinish = require("GameCoreManagerLevelFinish"):Create()
    self._GameCoreManagerUseItem = require("GameCoreManagerUseItem"):Create()
    self._GameCoreManagerEffect = require("GameCoreManagerEffect"):Create()
end

--- 创建实例
---@return GameCoreManager
function GameCoreManager:Create(args)
    local instance = GameCoreManager.new(args)
    PrintInfo("[GameCoreManager] 初始化完成")
    return instance
end

--- 清空数据（切换关卡/重置时调用）
function GameCoreManager:ClearData()
    -- 状态重置
    self._IsPlaying = false
    self._IsGameOver = false
    self._IsExitGame = false
    self._IsSuccessFinishGame = false
    self._eGameOverType = EGameOverType.None
    self._CurLevelData = nil

    -- 队列清空
    self._HandleGameOverMsgQueue:Clear()
    self:StopCheckGameOverMsgQueue()

    -- 提示清空
    self._ExitSureTip = {}

    -- 子模块清空
    self._GameCoreManagerData:ClearData()
    self._GameCoreManagerEffect:ClearData()
    self._GameCoreManagerLevelFinish:ClearData()

    PrintInfo("[GameCoreManager] 数据已清空")
end

--- 等待金币收集动画完成
---@param finishCallback function 完成回调
function GameCoreManager:WaitCollectCoinsAnim(finishCallback)
    -- 显示遮罩防止操作
    GameMainRoot.UI:AddUIMaskLayerRef("WaitCollectCoinsAnim")

    -- 等待动画时间
    local waitTime = Tables.Constants.GoalsCollectTime.Value[1] + Tables.Constants.GoalsBarFillTime.Value[1]
    DOTween.Sequence()
           :AppendInterval(waitTime)
           :SetUpdate(true) -- 不受游戏暂停影响
           :AppendCallback(function()
        -- 移除遮罩
        GameMainRoot.UI:SubUIMaskLayerRef()
        -- 安全执行回调
        if finishCallback then
            pcall(finishCallback)
        end
    end)
end

--- 启动游戏结束消息队列
function GameCoreManager:StartCheckGameOverMsgQueue()
    self:StopCheckGameOverMsgQueue()

    self._HandleGameOverMsgQueueCoroutine = HonorCoroutineHelper:StartAlone(function()
        self:CheckGameOverMsgQueueCoroutine()
    end)
end

--- 停止游戏结束消息队列
function GameCoreManager:StopCheckGameOverMsgQueue()
    if self._HandleGameOverMsgQueueCoroutine then
        HonorCoroutineHelper:StopAlone(self._HandleGameOverMsgQueueCoroutine)
        self._HandleGameOverMsgQueueCoroutine = nil
    end
    self._HandleGameOverMsgQueue:Clear()
end

--- 协程：处理游戏结束消息队列
function GameCoreManager:CheckGameOverMsgQueueCoroutine()
    local waitUtil1, waitUtil2
    local data

    while not self._IsGameOver do
        -- 等待队列有数据
        if self._HandleGameOverMsgQueue:Count() <= 0 then
            waitUtil1 = waitUtil1 or UnityWaitUntil(function()
                return self._HandleGameOverMsgQueue:Count() > 0
            end)
            coroutine.yield(waitUtil1)
        end

        data = self._HandleGameOverMsgQueue:Peek()
        if data then
            if not self._IsGameOver then
                -- 判断是否触发限制条件
                if self._GameCoreManagerData:IsTriggerLimit(data.gameOverType) then
                    self:FireTriggerGameOver(data.gameOverType, data.bounds)
                else
                    self:CheckGameResult(data.gameOverType, data.bounds)
                end

                -- 等待当前消息处理完成
                waitUtil2 = waitUtil2 or UnityWaitUntil(function()
                    return data.isHandle
                end)
                coroutine.yield(waitUtil2)
                data = nil
            end
        end
    end
end

--- 获取退出提示列表
function GameCoreManager:GetExitSureTip()
    return self._ExitSureTip
end

--- 是否存在退出提示
---@return boolean
function GameCoreManager:HasExitSureTip()
    return not TableIsEmpty(self._ExitSureTip)
end

--- 注册退出提示
---@param id any 提示ID
---@param strNum string 提示文本编号
function GameCoreManager:RegisterExitSureTip(id, strNum)
    local tip = self:FindExitSureTip(id)
    if tip then
        tip.strNum = strNum
    else
        table.insert(self._ExitSureTip, { id = id, strNum = strNum })
    end
end

--- 移除退出提示
---@param id any 提示ID
function GameCoreManager:UnRegisterExitSureTip(id)
    for i, t in ipairs(self._ExitSureTip) do
        if t.id == id then
            table.remove(self._ExitSureTip, i)
            break
        end
    end
end

--- 查找提示（内部方法）
---@param id any 提示ID
---@return table
function GameCoreManager:FindExitSureTip(id)
    for _, t in ipairs(self._ExitSureTip) do
        if t.id == id then
            return t
        end
    end
    return nil
end

--region 目标进度（主/副目标）

--- 获取主目标配置与进度
---@return number, number, number 目标ID, 目标总数, 当前进度
function GameCoreManager:GetMainGoalCfg()
    local mainId = self._GameCoreManagerData:CurRootGoalID()
    local mainNum = self._GameCoreManagerData:TotalRootGoalCount()
    local mainProgress = self._GameCoreManagerData:CurRootGoalCount()
    return mainId, mainNum, mainProgress
end

--- 根据ID获取副目标进度
---@param goalId string 副目标ID
---@return number, number 总数, 当前进度
function GameCoreManager:GetAdditionalGoalProgressById(goalId)
    if not self._GameCoreManagerData:IsEnableAdditionalGoal(goalId) then
        return nil, nil
    end

    local total = self._GameCoreManagerData:TotalAdditionalGoalCount(goalId)
    local progress = self._GameCoreManagerData:CurAdditionalGoalCount(goalId)
    return total, progress
end

--- 请求使用技能/道具
---@param boosterID string 道具ID
---@param num number 数量
function GameCoreManager:OnRequestUsingSkill(boosterID, num)
    if not boosterID then
        return
    end
    self._GameCoreManagerUseItem:OnUsingBooster(boosterID, num)
end

--- 是否允许使用道具
---@param boosterID string 道具ID
---@return boolean
function GameCoreManager:AllowUsingBooster(boosterID)
    if boosterID == EResourceIDs.Candy then
        return self._GameCoreManagerUseItem:AllowUsingCandy()
    elseif boosterID == EResourceIDs.WonderDish then
        return self._GameCoreManagerUseItem:AllowUsingWonderDish()
    elseif boosterID == EResourceIDs.Freeze then
        return self._GameCoreManagerUseItem:AllowUsingFreeze()
    elseif boosterID == EResourceIDs.SuperDelivery then
        return self._GameCoreManagerUseItem:AllowUsingSuperDelivery()
    end

    return false
end

--- 是否冻结道具生效中
---@return boolean
function GameCoreManager:IsFreezeInEffect()
    return ItemManager:InActive(EResourceIDs.Freeze)
end
