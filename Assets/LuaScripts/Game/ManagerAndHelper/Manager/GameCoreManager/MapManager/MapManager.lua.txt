---@class MapManager @ 地图管理器：管理地图加载、卸载、生命周期、场景机器
local MapManager = class("MapManager")

--region 构造 & 初始化
---=========================================================================
--- 构造函数
---@brief 初始化地图状态、配置、缓存容器
---=========================================================================
function MapManager:ctor()
    --- 准备加载的章节ID（预加载目标）
    self.ReadyLoadChapterId = SaveRootData.GameData.latestChapterId
    --- 当前加载中的章节ID
    self.CurLoadChapterId = nil
    --- 当前地图类型（默认主线）
    self.CurLoadMapType = EMapType.Main
    --- 当前加载的地图GameObject
    self.CurLoadMapGo = nil
    --- 地图数据配置
    self.CurMapData = nil
    --- 地图逻辑基类实例
    self.CurGameMapBase = nil
    --- 场景内所有机器实例字典
    self.CurMapMachines = {}
end

---=========================================================================
--- 创建管理器单例实例
---@return MapManager 地图管理器实例
---=========================================================================
function MapManager:Create()
    local instance = MapManager.new()
    PrintInfo("[MapManager] 初始化完成")
    return instance
end
--endregion

--region 地图卸载清理（前后两段）
---=========================================================================
--- 卸载地图【前】清理
---@brief 调用地图自身预清理逻辑（行为树、协程、监听）
---=========================================================================
function MapManager:UnloadMapBeforeClear()
    if self.CurGameMapBase then
        self.CurGameMapBase:UnloadMapBeforeClear()
    end
end

---=========================================================================
--- 卸载地图【后】清理
---@brief 清空所有引用、数据、状态、资源引用，防止内存泄漏
---=========================================================================
function MapManager:UnloadMapAfterClear()
    if self.CurGameMapBase then
        self.CurGameMapBase:UnloadMapAfterClear()
    end

    -- 清空所有地图数据
    self.CurLoadMapType = EMapType.Main
    self.CurLoadChapterId = nil
    self.CurLoadMapGo = nil
    self.CurMapData = nil
    self.CurGameMapBase = nil
    self.CurMapMachines = {}

    -- 清理主界面推送数据
    GameMainUIPushManager:ClearElementData()
end
--endregion

--=========================================================================
--region 地图加载/切换/卸载（核心流程）

---=========================================================================
--- 获取地图类型（目前固定为主线地图）
---@param chapterId string 章节ID
---@return EMapType 地图类型
---=========================================================================
function MapManager:GetMapType(chapterId)
    return EMapType.Main
end

---=========================================================================
--- 切换地图（外部统一入口）
--- 逻辑：相同章节→直接回调；不同→播放插屏广告→加载新地图
---@param chapterId string 目标章节ID
---@param finishCallback function 加载完成回调
---@param refuseAd boolean 是否跳过广告
---=========================================================================
function MapManager:OnSetSwitchMap(chapterId, finishCallback, refuseAd)
    -- 相同章节无需切换
    if self.CurLoadChapterId == chapterId then
        if finishCallback then finishCallback() end
        return
    end

    -- 尝试播放切章节广告
    local adShown = ADMgr:TryShowInterAd(EInterAdID.ChangeChapter, function()
        self:ReadyToLoadMap(chapterId, finishCallback)
    end)

    -- 不播放广告 或 广告播放失败 → 直接加载
    if refuseAd or not adShown then
        self:ReadyToLoadMap(chapterId, finishCallback)
    end
end

---=========================================================================
--- 准备加载地图：切换到加载状态
---@brief 进入LoadMapState，由状态机接管加载流程
---@param chapterId string 目标章节ID
---@param finishCallback function 完成回调
---=========================================================================
function MapManager:ReadyToLoadMap(chapterId, finishCallback)
    self.ReadyLoadChapterId = chapterId
    GameStateMachine:ChangeState(GameStateCmd.LoadMapState, finishCallback)
end

---=========================================================================
--- 加载预准备好的地图（状态机内部调用）
---=========================================================================
function MapManager:LoadReadyMap(luaParams)
    if self.ReadyLoadChapterId then
        self:LoadMapById(self.ReadyLoadChapterId, luaParams)
    end
end

---=========================================================================
--- 根据章节ID加载地图
---@param chapterId string 章节ID
---@param luaParams table 加载参数
---@return boolean 是否加载成功
---=========================================================================
function MapManager:LoadMapById(chapterId, luaParams)
    local mapName = self:GetLoadMapResName(chapterId)
    if not mapName then
        PrintError("[MapManager] 加载地图失败，资源名为空：" .. chapterId)
        return false
    end

    self.CurLoadChapterId = chapterId
    self.CurLoadMapType = self:GetMapType(chapterId)
    GamerDataManager:SetCurChapterId(chapterId)

    -- 同步加载地图Prefab
    self.CurLoadMapGo = self:LoadMapResByName(mapName, luaParams)
    return true
end

---=========================================================================
--- 同步加载地图Prefab到场景
---@param mapName string 地图资源名
---@param luaParams table 实例化参数
---@return UnityEngine.Object 地图GameObject
---=========================================================================
function MapManager:LoadMapResByName(mapName, luaParams)
    local parentTrans = GameMainRoot.Scene.SceneRootGO.transform
    local resPath = LoadRes["GameObject_" .. mapName].ABPath
    return GameMainRoot.Asset:LoadPrefabSync(resPath, mapName, parentTrans, luaParams)
end

---=========================================================================
--- 根据章节ID获取配置的地图资源名
---@param chapterId string 章节ID
---@return string 地图资源名
---=========================================================================
function MapManager:GetLoadMapResName(chapterId)
    local chapterData = ChapterDataManager:GetChapterDataByChapterType(chapterId)
    if chapterData then
        return chapterData.MapName
    end
    PrintError("[MapManager] 未找到章节配置：" .. chapterId)
    return nil
end

---=========================================================================
--- 卸载旧地图（完整流程）
---@brief 清理特效 → 预清理 → 销毁物体 → 后清理
---=========================================================================
function MapManager:UnloadOldMap()
    if string.isNullOrEmpty(self.CurLoadChapterId) or IsNull(self.CurLoadMapGo) then
        return
    end

    -- 清理全局特效
    GameEffectManager:ClearData()
    -- 地图预清理
    self:UnloadMapBeforeClear()
    -- 销毁Prefab
    DestroyPrefabWithGO(self.CurLoadMapGo)
    -- 地图后清理
    self:UnloadMapAfterClear()
end

--endregion

--=========================================================================
--region 游戏生命周期（统一转发给地图 + 所有机器）
---@brief 关卡生命周期统一调度：地图逻辑 + 所有机器行为树

---=========================================================================
--- 游戏初始化
---=========================================================================
function MapManager:OnInitGame()
    if not self.CurGameMapBase then return end

    for _, machine in pairs(self.CurMapMachines) do
        machine:OnInitGame()
    end
    self.CurGameMapBase:OnInitGame()
end

---=========================================================================
--- 进入游戏
---=========================================================================
function MapManager:OnEnterGame()
    if not self.CurGameMapBase then return end

    for _, machine in pairs(self.CurMapMachines) do
        machine:OnEnterGame()
    end
    self.CurGameMapBase:OnEnterGame()
end

---=========================================================================
--- 游戏开始
---=========================================================================
function MapManager:OnBeginGame()
    if not self.CurGameMapBase then return end

    for _, machine in pairs(self.CurMapMachines) do
        machine:OnBeginGame()
    end
    self.CurGameMapBase:OnBeginGame()
end

---=========================================================================
--- 游戏暂停
---=========================================================================
function MapManager:OnPauseGame()
    if not self.CurGameMapBase then return end

    for _, machine in pairs(self.CurMapMachines) do
        machine:OnPauseGame()
    end
    self.CurGameMapBase:OnPauseGame()
end

---=========================================================================
--- 游戏恢复
---=========================================================================
function MapManager:OnResumeGame()
    if not self.CurGameMapBase then return end

    for _, machine in pairs(self.CurMapMachines) do
        machine:OnResumeGame()
    end
    self.CurGameMapBase:OnResumeGame()
end

---=========================================================================
--- 游戏结束
---=========================================================================
function MapManager:OnGameOver()
    if not self.CurGameMapBase then return end

    for _, machine in pairs(self.CurMapMachines) do
        machine:OnGameOver()
    end
    self.CurGameMapBase:OnGameOver()
end

---=========================================================================
--- 退出游戏
---=========================================================================
function MapManager:OnExitGame()
    if not self.CurGameMapBase then return end

    for _, machine in pairs(self.CurMapMachines) do
        machine:OnExitGame()
    end
    self.CurGameMapBase:OnExitGame()
end

---=========================================================================
--- 重新开始游戏
---=========================================================================
function MapManager:OnRePlayGame()
    if not self.CurGameMapBase then return end

    for _, machine in pairs(self.CurMapMachines) do
        machine:OnInitGame()
        machine:OnRePlayGame()
    end
    self.CurGameMapBase:OnRePlayGame()
end

--endregion

return MapManager
