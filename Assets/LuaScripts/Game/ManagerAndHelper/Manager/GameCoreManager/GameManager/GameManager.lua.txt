---@class OpenCanvas @ 已打开完成的UI数据结构
---@field Canvas UnityEngine.Canvas  UI画布
---@field UIInfo Tables.UIs_Item    UI配置表数据

---@class GameManager : Singleton @ 游戏全局管理器
---@field _pauseGameRef number              游戏暂停引用计数
---@field _openOverCanvasArray OpenCanvas[] 已打开完成的UI列表
---@field _devicePerformanceLevel Honor.DevicePerformanceLevel 设备性能等级（缓存）
---@field _waitForFrames GameLib.WaitForFrames 公共等待帧对象

local GameManager = class("GameManager", import("Singleton"))

--region 构造 & 初始化
---=========================================================================
--- 构造函数
---@brief 初始化全局状态、引用计数、缓存对象
---=========================================================================
function GameManager:ctor(args)
    GameManager.super.ctor(self, args)
    self.CurrentMode=ELevelMode.NORMAL
    self._pauseGameRef = 0                          -- 暂停引用计数
    self._openOverCanvasArray = {}                  -- 已打开完成的UI数组
    self._devicePerformanceLevel = nil              -- 设备性能缓存
    self._waitForFrames = GameLib.WaitForFrames(6)  -- 公共等待6帧
    self.CharacterList={}    --可选选中物的集合
    self.CharacterChooseList={}    --选中物的集合
    self.NumberOfPropsUsed=0    --道具剩余次数
    self.LifeNumber=3    --生命次数
end

---=========================================================================
--- 创建单例实例
---@return GameManager 管理器实例
---=========================================================================
function GameManager:Create(args)
    local instance = GameManager.new(args)
    PrintInfo("[GameManager] 初始化完成")
    return instance
end

---=========================================================================
--- 初始化
---=========================================================================
function GameManager:Init()
    GameManager.super.Init(self)
    self:RegisterEvents()
end

---=========================================================================
--- 清理数据
---@brief 切号/重载时重置全局状态
---=========================================================================
function GameManager:ClearData()
    GameManager.super.ClearData(self)
    self:UnregisterEvents()
end

---@return boolean
---=========================================================================

    return GameMainRoot.Launcher.DevelopMode
end

---@param eLevelMode ELevelMode 关卡模式枚举
---=========================================================================
function GameManager:SetLevelMode(eLevelMode)
    self.CurrentMode = eLevelMode
end

---=========================================================================
--- 获取当前关卡运行模式
---@return ELevelMode 关卡模式枚举
---=========================================================================
function GameManager:GetLevelMode()
    return self.CurrentMode
end

---=========================================================================
--- 获取当前关卡数据信息
---@return ELevelMode 关卡模式枚举
---=========================================================================
function GameManager:GetMapRootData()
    local chapterId = ChapterListManager:GetRuntimeChapterId()
    if chapterId then
        local levelId = GamerDataManager:GetLevelLvByChapterId(chapterId)
        return PlayHelper:LoadLevelData(chapterId, levelId)
    end
    return nil
end

---@return number 角色总数量

    return GetTableCount(self.CharacterList)
end

---获取已选中角色数量
---@return number 选中角色数量

    return GetTableCount(self.CharacterChooseList)
end

---@return number 已使用道具数
function GameManager:GetNumberOfPropsUsed()
    return self.NumberOfPropsUsed
end

---获取剩余生命值
---@return number 生命值数量
function GameManager:GetLifeNumber()
    return self.LifeNumber
end

--- 注册全局事件监听
function GameManager:RegisterEvents()
    HonorEventHelper:AddListener(EventCmd.OnClickCharacter, handler(self, self.OnClickCharacter))
end

--- 取消全局事件监听
function GameManager:UnregisterEvents()
    HonorEventHelper:RemoveListener(EventCmd.OnClickCharacter, handler(self, self.OnClickCharacter))
end
---@type fun(character):void
---@param character CharacterBase @ 被点击的角色
function GameManager:OnClickCharacter(character)

    if  ContainsKey(self.CharacterChooseList,character.ID)==false then
        self.CharacterChooseList[character.ID]=character
    else
        
    end
end

---=========================================================================
--- 获取设备性能等级（缓存结果，避免重复调用）
---@return Honor.DevicePerformanceLevel
---=========================================================================
function GameManager:GetDevicePerformanceLevel()
    if self._devicePerformanceLevel ~= nil then
        return self._devicePerformanceLevel
    end

    self._devicePerformanceLevel = Honor.DevicePerformance.GetDevicePerformanceLevel()

    -- 失败降级
    if self._devicePerformanceLevel == nil then
        PrintError("[GameManager] 获取设备性能等级失败，使用默认低性能")
        self._devicePerformanceLevel = Honor.DevicePerformanceLevel.Low
    end

    return self._devicePerformanceLevel
end

---=========================================================================
--- 获取设备性能等级数字（0=低 1=中 2=高）
---@return number
---=========================================================================
function GameManager:GetDevicePerformanceLevelNumber()
    local level = self:GetDevicePerformanceLevel()

    if level == Honor.DevicePerformanceLevel.High then
        return 2
    elseif level == Honor.DevicePerformanceLevel.Mid then
        return 1
    else
        return 0
    end
end

--=========================================================================
--region 全屏UI性能优化：自动管理相机/Canvas渲染（核心优化）
--=========================================================================

---=========================================================================
--- UI打开完成回调（全屏UI关闭底层画布）
---@brief 全屏UI打开时自动禁用下层Canvas，节省渲染性能
---@param canvas UnityEngine.Canvas 画布对象
---@param uiName string UI名称
---=========================================================================
function GameManager:OnCanvasOpenOver(canvas, uiName)
    -- 清理空引用
    for i = #self._openOverCanvasArray, 1, -1 do
        local data = self._openOverCanvasArray[i]
        if IsNull(data.Canvas) then
            table.remove(self._openOverCanvasArray, i)
        end
    end

    if IsNull(canvas) then
        return
    end

    local uiCfg = UIs[uiName]
    if not uiCfg then
        return
    end

    -- 全屏UI：禁用所有已打开的Canvas
    if uiCfg.IsFullScreenUI then
        for _, data in ipairs(self._openOverCanvasArray) do
            data.Canvas.enabled = false
        end
    end

    -- 加入已打开列表
    table.insert(self._openOverCanvasArray, {
        Canvas = canvas,
        UIInfo = uiCfg
    })

    -- 检查是否存在全屏UI → 关闭场景相机
    local hasFullScreenUI = self:HasFullScreenUI()
    if hasFullScreenUI then
        local mainCamera = GameMainRoot.Scene.SceneCameras[0]
        if not IsNull(mainCamera) and mainCamera.activeSelf then
            mainCamera:SetActive(false)
        end
    end
end

---=========================================================================
--- UI立即关闭（直接清理）
---@param uiName string UI名称
---=========================================================================
function GameManager:OnCanvasCloseRightNow(uiName)
    for i = #self._openOverCanvasArray, 1, -1 do
        local data = self._openOverCanvasArray[i]
        if data.UIInfo.Name == uiName then
            table.remove(self._openOverCanvasArray, i)
        end
    end
end

---=========================================================================
--- UI关闭（内部逻辑）
---@brief 恢复最高层级Canvas显示，保证层级正确
---@param uiName string UI名称
---=========================================================================
function GameManager:_OnCanvasClose(uiName)
    -- 清理无效数据
    for i = #self._openOverCanvasArray, 1, -1 do
        local data = self._openOverCanvasArray[i]
        if data.UIInfo.Name == uiName or IsNull(data.Canvas) then
            table.remove(self._openOverCanvasArray, i)
        end
    end

    -- 启用最高层级全屏UI及之后的画布
    local count = #self._openOverCanvasArray
    if count <= 0 then
        return
    end

    local maxIndex = 1
    for i = 2, count do
        local curr = self._openOverCanvasArray[i]
        local max = self._openOverCanvasArray[maxIndex]

        if curr.UIInfo.IsFullScreenUI and curr.UIInfo.ZOrder >= max.UIInfo.ZOrder then
            maxIndex = i
        end
    end

    for i = maxIndex, count do
        self._openOverCanvasArray[i].Canvas.enabled = true
    end
end

---=========================================================================
--- UI关闭（外部调用，带异常保护）
---@brief 安全封装，关闭后自动恢复场景相机
---@param uiName string UI名称
---=========================================================================
function GameManager:OnCanvasClose(uiName)
    -- 安全执行
    local success = pcall(function()
        self:_OnCanvasClose(uiName)
    end)

    -- 无全屏UI → 恢复场景相机
    local hasFullScreenUI = success and self:HasFullScreenUI()
    if not hasFullScreenUI then
        local mainCamera = GameMainRoot.Scene.SceneCameras[0]
        if not IsNull(mainCamera) and not mainCamera.activeSelf then
            mainCamera:SetActive(true)
        end
    end
end

---=========================================================================
--- 所有UI关闭完成 → 强制恢复相机
---@brief 兜底保护，防止相机永久关闭
---=========================================================================
function GameManager:OnAfterCanvasCloseAll()
    ClearArray(self._openOverCanvasArray)

    local mainCamera = GameMainRoot.Scene.SceneCameras[0]
    if not IsNull(mainCamera) and not mainCamera.activeSelf then
        mainCamera:SetActive(true)
    end
end

---=========================================================================
--- 判断当前是否存在全屏UI（内部工具方法）
---@return boolean
---=========================================================================
function GameManager:HasFullScreenUI()
    for _, data in ipairs(self._openOverCanvasArray) do
        if data.UIInfo.IsFullScreenUI then
            return true
        end
    end
    return false
end
