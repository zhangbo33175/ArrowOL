--=====================================================================================================
-- @brief  UI 栈管理器
-- @desc   对 UI 进行分组栈管理（支持多栈），控制显示层级、返回逻辑、历史记录、弹窗序列
-- @struct 多栈隔离：不同业务栈互不干扰
-- @life   内置生命周期：OnStackPush / OnStackPop
-- @dependency Singleton, Stack, EUIStackName
--=====================================================================================================

---@class UIStackManager : Singleton @UI栈管理器（支持多分组栈隔离管理）
---@field _dict table<string, Stack> 多栈容器：key=栈名称(EUIStackName)，value=栈实例
local UIStackManager = class("UIStackManager", import("Singleton"))

--- 构造函数
---@param args table 自定义参数
function UIStackManager:ctor(args)
    UIStackManager.super.ctor(self, args)
    self._dict = {}
end

--- 创建单例
---@param args table 自定义参数
---@return UIStackManager
function UIStackManager:Create(args)
    return UIStackManager.new(args)
end

--- 清理数据（切号/重载/重启时调用）
function UIStackManager:ClearData()
    UIStackManager.super.ClearData(self)
    -- 清空所有栈并触发回调
    for name, _ in pairs(self._dict) do
        self:Clear(name)
    end
    self._dict = {}
end

---=========================================================================
--- 【压入 UI 到指定栈】
--- 自动创建不存在的栈
--- 触发 UI.OnStackPush()
---@param name string 栈名称（使用 EUIStackName 枚举）
---@param ui any UI实例
---=========================================================================
function UIStackManager:Push(name, ui)
    if string.isNullOrEmpty(name) or not ui then
        return
    end

    -- 自动创建栈
    local stack = self._dict[name]
    if not stack then
        stack = Stack:Create()
        self._dict[name] = stack
    end

    stack:Push(ui)

    -- 安全触发生命周期
    xpcall(function()
        if type(ui.OnStackPush) == "function" then
            ui:OnStackPush()
        end
    end, function(err)
        print_error("[UIStackManager] Push OnStackPush error: " .. tostring(err))
    end)
end

---=========================================================================
--- 【获取栈顶 UI（不弹出）】
---@param name string 栈名称
---@return any 栈顶UI / nil
---=========================================================================
function UIStackManager:Peek(name)
    if string.isNullOrEmpty(name) then return nil end

    local stack = self._dict[name]
    if not stack or stack:Count() <= 0 then return nil end

    return stack:Peek()
end

---=========================================================================
--- 【弹出栈顶 UI】
--- 触发 UI.OnStackPop()
---@param name string 栈名称
---@return any 弹出的UI
---=========================================================================
function UIStackManager:Pop(name)
    if string.isNullOrEmpty(name) then return nil end

    local stack = self._dict[name]
    if not stack or stack:Count() <= 0 then return nil end

    local ui = stack:Pop()

    -- 安全触发
    xpcall(function()
        if ui and type(ui.OnStackPop) == "function" then
            ui:OnStackPop()
        end
    end, function(err)
        print_error("[UIStackManager] Pop OnStackPop error: " .. tostring(err))
    end)

    return ui
end

---=========================================================================
--- 【清空整个栈】
--- 依次弹出所有 UI，逐个触发 OnStackPop
---@param name string 栈名称
---=========================================================================
function UIStackManager:Clear(name)
    if string.isNullOrEmpty(name) then return end

    local stack = self._dict[name]
    if not stack or stack:Count() == 0 then return end

    -- 安全清空
    while stack:Count() > 0 do
        local ui = stack:Pop()
        xpcall(function()
            if ui and type(ui.OnStackPop) == "function" then
                ui:OnStackPop()
            end
        end, function(err) end)
    end
end

return UIStackManager
