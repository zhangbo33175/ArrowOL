---@class BuffModifyConfig @ buff开始
---@field ID string@ buffID
---@field oldNum number @ 旧值
---@field newNum number @ 新值

---@class BuffManager:Singleton @限时道具管理
local BuffManager = class("BuffManager", import("Singleton"))

---构造函数
---@type fun(args:table):void
---@param args table @自定义参数
function BuffManager:ctor(args)
    BuffManager.super.ctor(self, args)
end

---创建函数
---@type fun(args:table):BuffMgr
---@param args table @自定义参数
---@return BuffMgr @限时道具管理
function BuffManager:Create(args)
    local obj = BuffManager.new(args)
    return obj
end

---是否是前置道具Buff
---@type fun(boosterID:string):boolean
---@param boosterID string @道具ID
---@return boolean @是否是前置道具Buff
function BuffManager:IsBuffPreBooster(boosterID)
    if boosterID ~= nil then
        local boosterData = ResourceDataManager:GetResourceData(boosterID)
        if boosterData and boosterData.Type == EResourceType.Buff and boosterData.SubType == EItemSubType.UnlimitedPreBooster then
            return true
        end
    end
    return false
end

---buff是否有效
---@type fun(id:string):boolean
---@param id string @ buff id (Resource表中id)
---@return boolean @ 是否有效
function BuffManager:IsBuffValid(id)
    return self:GetRemainTime(id) > 0
end

---添加buff
---@type fun(id:string,duration:number):number
---@param id string @ buff id (Resource表中id)
---@param duration number @ 持续时间
---@return number @ 剩余时间(单位秒)
function BuffManager:AddBuff(id, duration)
 
    local resData = ResourceDataManager:GetResourceData(id)
    if resData and resData.Type == EResourceType.Buff then
        local oldEndTimeStamp = PbRootData.GameData.resourceInfo.nums[id]
        local newUTC0 = GameTimeHelper:GetUTC0FutureTimeStamp()
        local newEndTimeStamp = nil

        if oldEndTimeStamp == nil or oldEndTimeStamp == 0 or oldEndTimeStamp < newUTC0 then
            newEndTimeStamp = newUTC0 + duration
        else
            newEndTimeStamp = oldEndTimeStamp + duration
        end
        local result = self:SetBuffTime(id, newEndTimeStamp, oldEndTimeStamp)
        return result
    else
        PrintError("BuffMgr:AddBuff error , 请检查是否是Buffer : id = {0} , duration = {1}", id, duration)
        return 0
    end
end

---设置buff的结束时间戳
---@type fun(id:string,endTimeStamp:number,oldEndTimeStamp:number):number
---@param id string @ buff id (Resource表中id)
---@param endTimeStamp number @ 结束时间戳
---@param oldEndTimeStamp number @ 旧的结束时间戳
---@return number @ 剩余时间(单位秒)
function BuffManager:SetBuffTime(id, endTimeStamp, oldEndTimeStamp)
    local nowUTC0 = GameTimeHelper:GetUTC0FutureTimeStamp()
    if endTimeStamp < 0 or endTimeStamp < nowUTC0 then
        PrintError(string.format("BuffManager:SetBuffTime error: endTimeStamp < 0 or endTimeStamp < 当前时间戳  id = {0} , endTimeStamp = {1}", id, endTimeStamp))
    end
    PbRootData.GameData.resourceInfo.nums[id] = endTimeStamp
    PbIO.GameData.Save_resourceInfo()
    HonorEventHelper:Fire(EventCmd.OnBufferTimeModifyEvent, { ID = id, oldNum = oldEndTimeStamp, newNum = endTimeStamp })  -- buff时间修改事件

    --启动倒计时监听Buff结束
    local buffEventKey = string.format("%s_%s", EventCmd.OnBufferTimeEndEvent, id)
    GameTimeHelper:UnRegisterTime(buffEventKey)
    self:_RegisterBuffEndTimeStamp(id, endTimeStamp)
    self:SetBuffTimeAfter(id)
    return self:GetRemainTime(id)
end

---@type fun(id:string,endTimeStamp:number):void
---@param id string @ buffID
---@param endTimeStamp number @ buff结束
function BuffManager:_RegisterBuffEndTimeStamp(id, endTimeStamp)
    local buffEventKey = string.format("%s_%s", EventCmd.OnBufferTimeEndEvent, id)
    GameTimeHelper:RegisterTime(buffEventKey, endTimeStamp, function()
        PrintInfo(string.format("BuffMgr:BuffEnd : %s", id))
        HonorEventHelper:FireNow(EventCmd.OnBufferTimeEndEvent, id)    -- buff结束事件
    end)
end

---更新Buff时间后需要处理的逻辑
---@type fun(id:string):void
---@param id string @ buff id (Resource表中id)
function BuffManager:SetBuffTimeAfter(id)
    local resData = ResourceDataManager:GetResourceData(id)
    if resData and resData.SubType == EItemSubType.UnlimitedLife then
        ---体力回满                                                             
        ItemMgr:AddLife(ItemMgr:GetMaxLifeNum())
    end
end

---获取buff剩余时间(单位秒)                                                                
---@type fun(id:string):number
---@param id string @ buff id (Resource表中id)
---@return number @ 剩余时间(单位秒)
function BuffManager:GetRemainTime(id)
    local endTimeStamp = PbRootData.GameData.resourceInfo.nums[id]
    if endTimeStamp == nil or endTimeStamp == 0 then
        return 0
    end
    local nowUTC0 = GameTimeHelper:GetUTC0FutureTimeStamp()
    if endTimeStamp < nowUTC0 then
        return 0
    end
    return endTimeStamp - nowUTC0
end

---初始化buff结束计时
---@type fun():void
---@return void
function BuffManager:Init()
    BuffManager.super.Init(self)
    local resTable = ResourceDataManager:GetResourceIDs(EResourceType.Buff)
    ---@param resId ResourceID
    for _, resId in ipairs(resTable) do
        if self:IsBuffValid(resId.ID) then
            local endTimeStamp = PbRootData.GameData.resourceInfo.nums[resId.ID]
            ---注入一个倒计时
            self:_RegisterBuffEndTimeStamp(resId.ID, endTimeStamp)
        end
    end
    HonorEventHelper:AddListener(EventCmd.OnBufferDeltaTimeTickEvent, handler(self, self.OnBufferDeltaTimeTickEvent))
    HonorTimerHelper:RegisterDeltaTime(EventCmd.OnBufferDeltaTimeTickEvent, 1, "ProcedurePlaying", true)
end

---清理数据
---@type fun():void
---@return void
function BuffManager:ClearData()
    BuffManager.super.ClearData(self)
    HonorTimerHelper:UnregisterDeltaTime(true, EventCmd.OnBufferDeltaTimeTickEvent, 1)
    HonorEventHelper:RemoveListener(EventCmd.OnBufferDeltaTimeTickEvent, handler(self, self.OnBufferDeltaTimeTickEvent))
end

---Buff的Tick事件
function BuffManager:OnBufferDeltaTimeTickEvent()
    local resTable = ResourceDataManager:GetResourceIDs(EResourceType.Buff)
    ---@param resId ResourceID
    for _, resId in ipairs(resTable) do
        local remainTime = self:GetRemainTime(resId.ID) -- 剩余时间
        if remainTime > 0 then
            HonorEventHelper:FireNow(EventCmd.OnBufferTimeTickEvent, resId.ID, remainTime)
        end
    end
end

return BuffManager
