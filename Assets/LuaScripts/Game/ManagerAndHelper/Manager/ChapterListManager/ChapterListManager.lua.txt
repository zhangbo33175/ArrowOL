---@class ChapterListManager : Singleton @章节&区域全局管理器，负责章节解锁、进度存档、运行时章节状态管理
local ChapterListManager = class("ChapterListManager", import("Singleton"))

--region 构造 & 初始化

---=========================================================================
--- @brief 构造函数
--- @param args table 初始化参数
--- @desc 初始化运行时成员变量，不做业务逻辑，业务初始化交由Init执行
---=========================================================================
function ChapterListManager:ctor(args)
    ChapterListManager.super.ctor(self, args)
    self._runtimeChapterID = "1"        ---@type string 当前运行时正在浏览/游玩的章节ID
    self._changeChapterFunc = nil       ---@type function 切换章节外部回调函数

    self.LevelData = nil                ---@type table 关卡临时数据对象
end

---=========================================================================
--- @brief 创建管理器单例实例
--- @param args table 初始化参数
--- @return ChapterListManager 章节管理器实例对象
---=========================================================================
function ChapterListManager:Create(args)
    return ChapterListManager.new(args)
end

---=========================================================================
--- @brief 清理运行时数据
--- @desc 切换场景、游戏重置时调用，仅清空内存运行时状态，不修改持久化存档
---=========================================================================
function ChapterListManager:ClearData()
    ChapterListManager.super.ClearData(self)
    self._runtimeChapterID = "1"
    self._changeChapterFunc = nil
end

---=========================================================================
--- @brief 管理器初始化
--- @desc 注册全局事件、设置默认运行章节、自动检测并解锁满足条件的后续章节、同步配置表最大章节ID至存档
---=========================================================================
function ChapterListManager:Init()
    ChapterListManager.super.Init(self)
    -- 监听关卡通关成功事件
    HonorEventHelper:AddListener(EventCmd.OnLevelSuccess, handler(self, self.OnLevelSuccess))

    --SaveRootData.GameData.latestChapterId = "1"
    --SaveIO.GameData.Save_latestChapterId()
    --SaveRootData.GameData.chapterInfos["1"].lv = 1
    -- 将运行时章节默认设置为存档内最新已解锁章节
    self:SetRuntimeChapter(self:GetLatestChapterId())
    -- 启动阶段检测下一章节是否满足解锁条件，满足则自动解锁，无UI弹窗提示
    local checkNextChapterId = tostring(tonumber(self:GetLatestChapterId()) + 1)
    if ChapterDataManager:GetChaptersDataByChapterId(checkNextChapterId) ~= nil then
        local meetCondition = self:MeetConditionForUnlock(checkNextChapterId)
        if meetCondition then
            self:UnlockChapter(checkNextChapterId)
        end
    end
    -- 同步配置表里最大章节ID到玩家存档，用于边界校验
    self:SetupMaxChapterID()
end

---=========================================================================
--- @brief 关卡通关成功回调
--- @param lvInfo ChapterLevel 通关关卡信息对象
--- @desc 关卡胜利后触发，校验当前章节进度，判断是否可以解锁下一章节，预留解锁UI调用点
---=========================================================================
function ChapterListManager:OnLevelSuccess(lvInfo)
    local latestChapterID = self:GetLatestChapterId()
    -- 非最新解锁章节通关，直接返回，不处理解锁逻辑
    if latestChapterID ~= tostring(lvInfo.Chapter) or latestChapterID == "1" then
        return
    end

    local next = tostring(lvInfo.Chapter + 1)
    local nextChapterId = tonumber(next)
    local lastChapterData = ChapterDataManager:GetChaptersDataByChapterId(tostring(nextChapterId - 1))
    if not lastChapterData then
        return
    end
    local lastChapterLv = self:GetRuntimeChapterLvByChapterId(lastChapterData.ID) - 1
    local lastChapterMaxLv = lastChapterData.UnlockNextLevel - 1

    -- 当前章节进度达到解锁下一章节要求
    if lastChapterLv == lastChapterMaxLv then
        local maxChapterId = ChapterDataManager:GetConfigMaxChapterID()
        if nextChapterId <= maxChapterId then
            --self:OpenChapterUnlockUI(next) --预留章节解锁弹窗打开接口
        end
    end
end

---=========================================================================
--- @brief 同步配置最大章节ID到存档
--- @desc 将配置表内最大章节ID写入存档，做解锁边界判断，发生变更时持久化保存
---=========================================================================
function ChapterListManager:SetupMaxChapterID()
    local cfgMax = ChapterDataManager:GetConfigMaxChapterID()
    if SaveRootData.GameData.maxChapterId ~= cfgMax then
        SaveRootData.GameData.maxChapterId = cfgMax
        SaveIO.GameData.Save_maxChapterId()
    end
end

--endregion

--region 公共方法：解锁、完成、条件判断

---=========================================================================
--- @brief 判断章节是否满足解锁条件
--- @param chapterId string 目标待解锁章节ID
--- @return boolean true:满足解锁条件 false:不满足解锁条件
--- @desc 依赖上一章节进度，上一章等级达到配置UnlockNextLevel才可解锁
---=========================================================================
function ChapterListManager:MeetConditionForUnlock(chapterId)
    local prevChapterId = tostring(tonumber(chapterId) - 1)
    local lastChapterData = ChapterDataManager:GetChaptersDataByChapterId(prevChapterId)
    if not lastChapterData then
        return false
    end
    local lastChapterLv = self:GetRuntimeChapterLvByChapterId(lastChapterData.ID)
    local unlockNextLv = lastChapterData.UnlockNextLevel or 1
    return lastChapterLv >= unlockNextLv
end

---=========================================================================
--- @brief 设置存档最新解锁章节ID
--- @param chapterId string 章节ID
--- @desc 更新存档latestChapterId并执行单字段持久化保存
---=========================================================================
function ChapterListManager:SetLatestChapterId(chapterId)
    SaveRootData.GameData.latestChapterId = chapterId
    SaveIO.GameData.Save_latestChapterId()
end

---=========================================================================
--- @brief 解锁指定章节
--- @param chapterId string 需要解锁的章节ID
--- @desc 更新存档，清理大于当前章节的旧存档数据，初始化章节存档结构，发送章节解锁全局事件
--- @note GM回退章节时会清除后续章节存档；自动初始化历史章节默认存档数据
---=========================================================================
function ChapterListManager:UnlockChapter(chapterId)
    -- 仅当待解锁章节ID大于存档最新章节，更新最新解锁章节标记
    if tonumber(chapterId) > tonumber(self:GetLatestChapterId()) then
        self:SetLatestChapterId(chapterId)
    end

    SaveRootData.GameData.latestChapterId = chapterId

    for id, ___ in pairs(SaveRootData.GameData.chapterInfos) do
        if tonumber(id) > tonumber(chapterId) then
            SaveRootData.GameData.chapterInfos[id] = nil
        end
    end

    -- 收集全部配置章节，初始化从1到目标chapterId所有章节的存档
    local chapterIDsForUnlock = {}
    local chapterIDs = Tables.Chapters
    table.insert(chapterIDsForUnlock, chapterIDs)
    for _, chapterIDForUnlock in pairs(chapterIDsForUnlock[1]) do
        local chapterIdCfg = chapterIDForUnlock.ID
        local chapterData = ChapterDataManager:GetChaptersDataByChapterId(chapterIdCfg)
        if not chapterData then
            goto continue
        end
        -- 如果该章节还没有存档数据，则初始化默认存档
        if SaveRootData.GameData.chapterInfos[chapterIdCfg] == nil then
            ---@type SaveMsgDef.SaveGameData.MachineInfo @关卡进度默认存档结构体
            local saveMachineInfo = {
                levelGroupId = 1,    -- 当前关卡组ID
                levelStamina = 3,    -- 关卡剩余体力
                levelData = {},      -- 关卡运行时数据
            }
            ---@type SaveMsgDef.SaveGameData.ChapterInfo @章节默认存档结构体
            local saveChapterInfo = {
                lv = 1,                              -- 章节玩家等级
                levelProgress = saveMachineInfo,     -- 关卡进度信息
                CareerDataStatistics = {},           -- 章节业务统计指标数据
                currentChapterData = 1               -- 当前章节索引标记
            }
            SaveRootData.GameData.chapterInfos[chapterIdCfg] = saveChapterInfo
        end
        ::continue::
    end

    -- 将新解锁章节等级强制初始化为1
    SaveRootData.GameData.chapterInfos[chapterId].lv = 1

    -- 全部游戏数据持久化落地
    SaveIO.GameData.SaveAll()

    -- 派发新章节解锁全局事件，UI/业务模块监听做响应
    HonorEventHelper:Fire(EventCmd.OnNewChapterUnlockEvent)
end

---=========================================================================
--- @brief 创建章节初始存档数据结构
--- @param chapterId string 章节ID
--- @return SaveMsgDef.SaveGameData.ChapterInfo 章节默认存档table
--- @desc 返回全新的章节默认存档对象，用于外部手动新建章节存档
---=========================================================================
function ChapterListManager:CreateChapterInitData(chapterId)
    ---@type SaveMsgDef.SaveGameData.MachineInfo @关卡进度默认存档结构体
    local saveLevelProgress = {
        levelGroupId = 1,
        levelStamina = 2,
        levelData = {},
    }
    ---@type SaveMsgDef.SaveGameData.ChapterInfo @章节默认存档结构体
    local saveChapterInfo = {
        lv = 1,
        levelProgress = saveLevelProgress,
        CareerDataStatistics = {},
        currentChapterData = 1
    }
    return saveChapterInfo
end

---=========================================================================
--- @brief 判断章节是否已经解锁
--- @param chapterId string 待检测章节ID
--- @return boolean true=已解锁 false=未解锁
---=========================================================================
function ChapterListManager:IsUnlockedChapter(chapterId)
    return tonumber(chapterId) <= tonumber(self:GetLatestChapterId())
end

---=========================================================================
--- @brief 判断该章节是否已经完成通关
--- @param chapterId string 章节ID
--- @return boolean true=章节已完成 false=未完成
--- @desc 通过章节等级与配置的UnlockNextLevel对比判断
---=========================================================================
function ChapterListManager:IsChapterCompleted(chapterId)
    local curLv = self:GetRuntimeChapterLvByChapterId(chapterId)
    local chapterData = ChapterDataManager:GetChapterDataByChapterType(chapterId)
    if not chapterData then
        return false
    end

    local unlockNextLv = chapterData.UnlockNextLevel or 1
    return curLv >= unlockNextLv
end

--endregion

--region 运行时：当前正在显示/游玩的章节/区域

---=========================================================================
--- @brief 设置运行时章节，内部校验配置有效性
--- @desc 自动绑定对应区域数据，外部调用入口
--- @param chapterId string 目标章节ID
---=========================================================================
function ChapterListManager:SetRuntimeChapter(chapterId)
    local areaData = ChapterDataManager:GetChaptersDataByChapterId(chapterId)
    if areaData then
        self:SetRuntimeChapterId(chapterId)
    end
end

---=========================================================================
--- @brief 直接设置内存运行时章节ID（无配置校验）
--- @param chapterId string 章节ID，传nil会回退默认值"1"
---=========================================================================
function ChapterListManager:SetRuntimeChapterId(chapterId)
    self._runtimeChapterID = chapterId or "1"
end

---=========================================================================
--- @brief 获取当前运行时章节ID
--- @return string 章节ID
--- @desc 如果内存运行时ID为空，则返回存档最新解锁章节ID作为兜底
---=========================================================================
function ChapterListManager:GetRuntimeChapterId()
    return self._runtimeChapterID or self:GetLatestChapterId()
end

---=========================================================================
--- @brief 根据章节ID读取存档中的章节等级
--- @param chapterId string 章节ID
--- @return number 章节等级；无存档返回0
---=========================================================================
function ChapterListManager:GetRuntimeChapterLvByChapterId(chapterId)
    local info = SaveRootData.GameData.chapterInfos[chapterId]
    return info and info.lv or 0
end

--endregion

--region 存档相关：最新解锁章节

---=========================================================================
--- @brief 获取存档内最新解锁章节ID
--- @return string 章节ID
--- @desc 读取持久化存档，空值兜底为"1"，并回写存档防止空脏数据
---=========================================================================
function ChapterListManager:GetLatestChapterId()
    local chapterId = SaveRootData.GameData.latestChapterId
    if string.isNullOrEmpty(chapterId) then
        chapterId = "1"
        SaveRootData.GameData.latestChapterId = chapterId
        SaveIO.GameData.Save_latestChapterId()
    end
    return chapterId
end

--endregion

--region 章节等级比较工具

---=========================================================================
--- @brief 比较目标章节等级是否大于最新章节等级
--- @param targetChapterLv table 目标章节等级数据 {Chapter:number, Level:number}
--- @param latestChapterLv table|nil 对比基准等级，不传则读取GamerDataManager
--- @return boolean true:目标进度更大 false:目标进度小于等于基准进度
---=========================================================================
function ChapterListManager:LargeThanLatestChapterLv(targetChapterLv, latestChapterLv)
    latestChapterLv = latestChapterLv or GamerDataManager:GetLatestChapterLevel()
    local lessOrEqual = targetChapterLv.Chapter < latestChapterLv.Chapter
            or (targetChapterLv.Chapter == latestChapterLv.Chapter and targetChapterLv.Level <= latestChapterLv.Level)
    return not lessOrEqual
end

--endregion

return ChapterListManager
