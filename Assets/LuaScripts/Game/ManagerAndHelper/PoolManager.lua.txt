-- @desc:     管理所有 GameObject 的缓存、复用、销毁，大幅减少 Instantiate/Destroy 开销
-- @feat:     自动创建池、预热、上限控制、安全回收、内存优化
-- @dependency: Singleton、GlobalUtility、GlobalDefine

---@class PoolManager : Singleton @游戏内的对象加载/缓存管理器
---@field pools table<string, Pool> @对象池字典：key=资源名，value=Pool实例
---@field maxItemCountOfOnePool number @单个池最大缓存数量（超过直接销毁）
local PoolManager = class("PoolManager", import("Singleton"))

--region 生命周期 & 基础函数
---=========================================================================
--- 构造函数
---@brief 初始化对象池容器与默认配置
---=========================================================================
function PoolManager:ctor()
    PoolManager.super.ctor(self)
    self.pools = {}
    self.maxItemCountOfOnePool = 30  -- 单个对象池最大缓存数量
end

---=========================================================================
--- 创建单例实例
---@return PoolManager 单例对象
---=========================================================================
function PoolManager:Create()
    return PoolManager.new()
end

---=========================================================================
--- 初始化（重写Singleton）
---@brief 单例初始化入口
---=========================================================================
function PoolManager:Init()
    PoolManager.super.Init(self)
end

---=========================================================================
--- 清空所有数据（销毁所有池）
---@brief 全局清理接口，清空所有对象池与资源
---=========================================================================
function PoolManager:ClearData()
    PoolManager.super.ClearData(self)
    self:DestroyAllPools()
end
--endregion

--region 预热功能
---=========================================================================
--- 【预热接口】预加载指定数量对象到池，避免运行时卡顿
---@param resName string 资源名（必须在RES表中配置）
---@param count number 预热数量
---=========================================================================
function PoolManager:Prewarm(resName, count)
    if string.isNullOrEmpty(resName) or count <= 0 then
        return
    end

    -- 池已存在，直接预热
    if self.pools[resName] then
        local p = self:GetPool(resName)
        p:Prewarm(count)
        return
    end

    -- 检查资源配置
    local resDefItem = LoadRes[resName]
    if not resDefItem then
        PrintError("【对象池】资源未在RES表配置: " .. resName)
        return
    end

    -- 创建新池并预热
    local newPool = self:CreatePool(resName, resDefItem, CreateNewPoolRoot().transform, function() end)
    if newPool then
        newPool:Prewarm(count)
    end
end
--endregion

--region 获取对象
---=========================================================================
--- 【获取对象】从池获取，无池则自动创建（最简接口）
---@param resName string 资源名
---@param newParent Transform 父物体
---@return GameObject 复用/新建的对象
---=========================================================================
function PoolManager:Get(resName, newParent)
    return self:GetWithHandle(resName, newParent, function() end)
end

---=========================================================================
--- 【获取对象（带重置回调）】
---@param resName string 资源名
---@param newParent Transform 父物体
---@param resetHandler function 物体激活时的重置逻辑
---@return GameObject 获取到的游戏对象
---=========================================================================
function PoolManager:GetWithHandle(resName, newParent, resetHandler)
    if string.isNullOrEmpty(resName) then
        PrintError("【对象池】GetWithHandle 资源名不能为空")
        return nil
    end

    -- 池已存在 → 直接取
    if self.pools[resName] then
        local p = self:GetPool(resName)
        p.resetHandler = resetHandler
        return p:Get(newParent)
    end

    -- 池不存在 → 创建新池
    local resDefItem = LoadRes[resName]
    if not resDefItem then
        PrintError("【对象池】资源未在RES表配置: " .. resName)
        return nil
    end

    local newPool = self:CreatePool(resName, resDefItem, CreateNewPoolRoot().transform, resetHandler)
    if newPool then
        return newPool:Get(newParent)
    end

    return nil
end
--endregion

--region 回收对象
---=========================================================================
--- 【回收对象】将GameObject放回池（超过上限则直接销毁）
---@param go GameObject 需要回收的游戏对象
---=========================================================================
function PoolManager:Put(go)
    if IsNull(go) or not UnityApplication.isPlaying then
        return
    end

    -- 池名规则：GameObject_资源名
    local poolName = "GameObject_" .. go.name
    local pool = self.pools[poolName]

    if not pool then
        -- 无对应池，直接销毁
        DestroyPrefabWithGO(go)
        return
    end

    -- 未超上限 → 入池缓存
    if pool.unusedQueue:Count() < self.maxItemCountOfOnePool then
        pool:Put(go)
    else
        -- 超上限 → 直接销毁
        if pool.usedList:Contains(go) then
            pool.usedList:Remove(go)
        end
        DestroyPrefabWithGO(go)
    end
end
--endregion

--region 池管理（私有）
---=========================================================================
--- 【私有】创建一个新对象池
---@param name string 池名
---@param resDefItem table 资源配置
---@param preParent Transform 默认父节点
---@param resetHandler function 重置回调
---@return Pool 创建完成的对象池
---@private
---=========================================================================
function PoolManager:CreatePool(name, resDefItem, preParent, resetHandler)
    if string.isNullOrEmpty(name) or not resDefItem then
        return nil
    end

    -- 只支持GameObject类型
    if resDefItem.TypeName ~= "GameObject" then
        PrintError("【对象池】创建失败：仅支持GameObject类型 → " .. name)
        return nil
    end

    -- 已存在则不重复创建
    if self.pools[name] then
        return self.pools[name]
    end

    -- 创建Pool实例
    local ok, pool = xpcall(function()
        return require("HonorPool"):Create(name, resDefItem, preParent, resetHandler)
    end, function(err)
        PrintError("【对象池】创建池失败: " .. tostring(err))
    end)

    if ok and pool then
        self.pools[name] = pool
    end

    return self.pools[name]
end
--endregion

--region 池查询 & 统计
---=========================================================================
--- 【获取指定对象池】
---@param name string 池名
---@return Pool 对应资源的对象池
---=========================================================================
function PoolManager:GetPool(name)
    return self.pools[name]
end

---=========================================================================
--- 【获取当前池数量】
---@return number 当前已创建的对象池总数
---=========================================================================
function PoolManager:Count()
    local count = 0
    for _, _ in pairs(self.pools) do
        count = count + 1
    end
    return count
end
--endregion

--region 销毁功能
---=========================================================================
--- 【销毁指定池】
---@param name string 池名
---@return boolean 是否成功销毁
---=========================================================================
function PoolManager:DestroyPool(name)
    if self.pools[name] then
        xpcall(function()
            self.pools[name]:Destroy()
        end, function(err) end)
        self.pools[name] = nil
        return true
    end
    return false
end

---=========================================================================
--- 【销毁所有池 + 清理根节点】
---@brief 完全释放所有对象池资源
---=========================================================================
function PoolManager:DestroyAllPools()
    -- 安全销毁所有池
    for name, pool in pairs(self.pools) do
        xpcall(function()
            pool:Destroy()
        end, function(err)
            PrintError("【对象池】销毁池失败: " .. name .. " err: " .. tostring(err))
        end)
    end
    self.pools = {}

    -- 销毁池根节点
    local poolRoot = GetPoolRoot()
    if not IsNull(poolRoot) then
        Unity.GameObject.Destroy(poolRoot)
    end
end

---=========================================================================
--- 【只销毁未使用的缓存对象】（保留池结构，优化内存）
---@brief 场景切换/低内存时调用
---=========================================================================
function PoolManager:DestroyAllPoolsUnused()
    for _, pool in pairs(self.pools) do
        xpcall(function()
            pool:DestroyUnused()
        end, function(err) end)
    end
end
--endregion

return PoolManager
