--=====================================================================================================
-- @brief  游戏系统解锁管理器
-- @desc   根据玩家当前章节/关卡，判断各个游戏系统（升级、装修、任务、收藏）是否解锁
-- @trigger 关卡胜利 / 新章节解锁 时自动刷新
-- @event  OnGameSysUnlock - 系统解锁时抛出
--=====================================================================================================

---@class SystemGameUnlock : Singleton @游戏系统解锁查询
local SystemGameUnlock = class("SystemGameUnlock", import("Singleton"))

--- 构造函数
---@param args table 自定义参数
function SystemGameUnlock:ctor(args)
    SystemGameUnlock.super.ctor(self, args)

    --- 系统解锁状态表 key: 系统名(ESystemName) value: 是否已解锁
    self.map = {}

    --- 系统解锁配置表 key: 系统名 value: {Chapter, Level} 解锁所需条件
    self.unlockLv = {}
end

--- 创建单例
---@param args table 自定义参数
---@return SystemGameUnlock
function SystemGameUnlock:Create(args)
    return SystemGameUnlock.new(args)
end

---=========================================================================
--- 【对外接口】查询某个系统是否已解锁
---@param sys string 系统名称（来自 ESystemName 枚举）
---@return boolean 是否已解锁
---=========================================================================
function SystemGameUnlock:SystemIsUnlock(sys)
    if not sys or not self.map then
        return false
    end
    return self.map[sys] == true
end

---=========================================================================
--- 【监听】关卡胜利 → 刷新所有系统解锁状态
---@param chapterLv ChapterLevel 最新章节关卡
---=========================================================================
function SystemGameUnlock:_ListenLevelSuccess(chapterLv)
    if not chapterLv or not chapterLv.Chapter or not chapterLv.Level then
        return
    end

    local isAllUnlock = true

    -- 遍历所有需要解锁的系统
    for sys, unlockCfg in pairs(self.unlockLv) do
        if not self.map[sys] then
            -- 安全校验配置
            if not unlockCfg or not unlockCfg.Chapter or not unlockCfg.Level then
                self.map[sys] = true
                goto continue
            end

            -- 解锁判断：章节更小 或 同章节关卡达标
            local isUnlock = unlockCfg.Chapter < chapterLv.Chapter
                    or (unlockCfg.Chapter == chapterLv.Chapter and unlockCfg.Level <= chapterLv.Level)

            self.map[sys] = isUnlock

            -- 解锁成功 → 抛出事件
            if isUnlock then
                HonorEventHelper:Fire(EventCmd.OnGameSysUnlock, sys)
            end
        end

        ::continue::
        -- 只要有一个没解锁，整体就不是全解锁
        isAllUnlock = isAllUnlock and self.map[sys]
    end

    -- 全部解锁后，移除监听，避免无效开销
    if isAllUnlock then
        HonorEventHelper:RemoveListener(EventCmd.OnLevelSuccess, handler(self, self._ListenLevelSuccess))
        HonorEventHelper:RemoveListener(EventCmd.OnNewChapterUnlockEvent, handler(self, self._ListenChapterUnlock))
    end
end

---=========================================================================
--- 【监听】新章节解锁 → 刷新系统解锁
---=========================================================================
function SystemGameUnlock:_ListenChapterUnlock()
    local latestChapterLv = GamerDataManager:GetLatestChapterLevel()
    self:_ListenLevelSuccess(latestChapterLv)
end

---=========================================================================
--- 单例优先级（越高越早初始化）
---@return number
---=========================================================================
function SystemGameUnlock:Priority()
    return ESingletonPriority.GAME_SYS_UNLOCK
end
