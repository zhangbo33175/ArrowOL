--=====================================================================================================
-- 【全局气泡管理器】
-- 功能：管理游戏内所有提示气泡（跟随目标、自动消失、互斥显示、对象池复用）
-- 特点：支持UGUI/场景物体跟随、屏幕约束、互斥显示、自动回收
--=====================================================================================================

---@class BubbleGenerateConfig : table @ 气泡生成配置
---@field id number @ 气泡唯一ID
---@field key string @ 气泡唯一Key
---@field bubbleHost UnityEngine.Transform @ 跟随目标
---@field hostInUGUI boolean @ 宿主是否为UGUI
---@field pivot UnityEngine.Vector2 @ 中心点
---@field bubbleTextKey string @ 文本Key
---@field duration string @ 持续时间
---@field followPosition boolean @ 是否跟随目标移动
---@field bubbleAnchorGetter function @ 动态获取锚点
---@field useScreenSize boolean @ 是否受屏幕边界约束
---@field onlyOneTag string @ 互斥Tag（同Tag只显示一个）
---@field nearCenter boolean @ 是否靠近屏幕中心

---@class BubbleManager : Singleton @气泡生成与关闭
---@field _activeBubble table<string, UIBubble> @ 当前活跃气泡字典
---@field _autoGrowID number @ 自增ID生成器
local BubbleManager = class("BubbleManager", import("Singleton"))

--- 构造函数
---@param args table 自定义参数
function BubbleManager:ctor(args)
    BubbleManager.super.ctor(self, args)
    self._activeBubble = {}   --- 活跃气泡
    self._autoGrowID = 10086 --- 自增ID初始值
end

--- 创建单例
---@param args table
---@return BubbleManager
function BubbleManager:Create(args)
    local obj = BubbleManager.new(args)
    return obj
end

--- 清理数据（切号/重载）
function BubbleManager:ClearData()
    BubbleManager.super.ClearData(self)
    ClearDic(self._activeBubble)
    self._autoGrowID = 10086
end

---=========================================================================
--- 【全局接口】关闭所有活跃气泡
---=========================================================================
function BubbleManager:CloseAllBubble()
    if self._activeBubble and not TableIsEmpty(self._activeBubble) then
        for _, bubble in pairs(self._activeBubble) do
            bubble:OnPutToPool(true)
        end
    end
    ClearDic(self._activeBubble)
end

--- 释放（同CloseAll）
function BubbleManager:Dispose()
    self:CloseAllBubble()
end

---=========================================================================
--- 【核心接口】生成一个气泡
--- 逻辑：
--- 1. 检查互斥Tag → 关闭同Tag旧气泡
--- 2. 检查Key是否已存在 → 存在则更新文本
--- 3. 不存在则从对象池创建 → 加入管理
---@param bubbleGenerateConfig BubbleGenerateConfig 气泡配置
---@return UIBubble
---=========================================================================
function BubbleManager:GenerateBubble(bubbleGenerateConfig)
    -- 互斥逻辑：同 onlyOneTag 只允许显示一个
    if bubbleGenerateConfig.onlyOneTag and not TableIsEmpty(self._activeBubble) then
        for _, bubble in pairs(self._activeBubble) do
            local cfg = bubble:GetBubbleCfg()
            if cfg and cfg.onlyOneTag == bubbleGenerateConfig.onlyOneTag then
                if not bubble:IsClosing() then
                    xpcall(function()
                        bubble:OnPutToPool()
                    end, function(msg)
                        PrintError(debug.traceback(msg, 5))
                    end)
                end
            end
        end
    end

    -- 如果已存在相同Key → 直接更新文本
    local existBubble = self._activeBubble[bubbleGenerateConfig.key]
    if existBubble then
        existBubble:UpdateTextContent(bubbleGenerateConfig.bubbleTextKey, bubbleGenerateConfig.duration)
        return existBubble
    end

    -- 分配新ID
    bubbleGenerateConfig.id = self:GetNewBubbleID()

    -- 从对象池创建气泡
    local go = PoolManager:Get(RES.GameObject_UIBubble.Name, GameMainRoot.UI.ScreenUICanvas.transform)
    local luaBehaviour = go:GetComponent("LuaBehaviour")
    ---@type UIBubble
    local bubble = luaBehaviour.ValidLuaClass

    -- 初始化气泡
    bubble:OnGetFromPool(bubbleGenerateConfig)
    self._activeBubble[bubbleGenerateConfig.key] = bubble

    return bubble
end

---=========================================================================
--- 强制关闭指定Key的气泡
---@param bubbleKey string 气泡Key
---=========================================================================
function BubbleManager:ForceCloseBubble(bubbleKey)
    local bubble = self._activeBubble[bubbleKey]
    self._activeBubble[bubbleKey] = nil

    if bubble then
        bubble:OnPutToPool()
    end
end

--- 关闭气泡（外部调用）
---@param bubble UIBubble
function BubbleManager:CloseBubble(bubble)
    bubble:OnPutToPool()
end

---=========================================================================
--- 【气泡内部回调】气泡真正关闭并回池
---@param bubbleKey string
---=========================================================================
function BubbleManager:OnBubbleClose(bubbleKey)
    if self._activeBubble[bubbleKey] then
        -- 回收GameObject
        PoolManager:Put(self._activeBubble[bubbleKey].cs.gameObject)
        self._activeBubble[bubbleKey] = nil
    end
end

--- 生成新气泡ID（自增）
---@return number
function BubbleManager:GetNewBubbleID()
    self._autoGrowID = self._autoGrowID + 1
    return self._autoGrowID
end

return BubbleManager
