--region 设置开关获取 & 设置
---=========================================================================
---@brief  获取存档中记录的设置开关状态
---@type fun(settingType:ESettingType):boolean
---@param settingType ESettingType 设置类型（音乐/音效/推送等）
---@return boolean 开关状态：true=开启，false=关闭
---=========================================================================
function SettingDataManager:GetSettingState(settingType)
    -- 安全校验
    if not settingType or not SaveRootData or not SaveRootData.GameData or not SaveRootData.GameData.settingInfo then
        PrintError("[SettingDataManager] 获取设置失败：存档结构异常")
        return true -- 默认开启
    end

    local state = SaveRootData.GameData.settingInfo[settingType]
    -- 未初始化默认开启
    return state ~= nil and state or true
end

---=========================================================================
---@brief  修改设置开关状态并保存存档
---@type fun(settingType:ESettingType,newState:boolean):void
---@param settingType ESettingType 设置类型
---@param newState boolean 新状态：true=开启，false=关闭
---=========================================================================
function SettingDataManager:SetSettingNewState(settingType, newState)
    -- 安全校验
    if not settingType or type(newState) ~= "boolean" then
        PrintError("[SettingDataManager] 设置失败：参数无效")
        return
    end
    if not SaveRootData or not SaveRootData.GameData or not SaveRootData.GameData.settingInfo then
        PrintError("[SettingDataManager] 设置失败：settingInfo 不存在")
        return
    end

    -- 写入并保存
    SaveRootData.GameData.settingInfo[settingType] = newState

    -- 安全保存存档
    xpcall(function()
        if SaveIO and SaveIO.GameData and SaveIO.GameData.Save_settingInfo then
            SaveIO.GameData.Save_settingInfo()
        end
    end, function(err)
        PrintError("[SettingDataManager] 保存设置失败：" .. tostring(err))
    end)

    -- 音乐开关同步
    if settingType == ESettingType.MUSIC then
        xpcall(function()
            SoundMgr:OnBGMCloseStateChanged(settingType, newState)
        end, function() end)
    end

    -- 音效开关同步
    if settingType == ESettingType.SOUND then
        xpcall(function()
            SoundMgr:OnSoundCloseStateChanged(settingType, newState)
        end, function() end)
    end
end
