---@class LanguageSettingDataManager : Singleton @多语言配置管理中心
local LanguageSettingDataManager = class("LanguageSettingDataManager", import("Singleton"))

--region 构造 & 实例化
---@brief 构造函数
---@type fun():void
function LanguageSettingDataManager:ctor()
    LanguageSettingDataManager.super.ctor(self)
    self._useLangs = nil -- 启用语言缓存
end

---@brief 创建单例
---@type fun():LanguageSettingDataManager
---@return LanguageSettingDataManager
function LanguageSettingDataManager:Create()
    return LanguageSettingDataManager.new()
end
--endregion

--region 初始化 & 生命周期
---@brief 初始化
---@type fun():void
function LanguageSettingDataManager:Init()
    LanguageSettingDataManager.super.Init(self)
    self._useLangs = nil -- 重置缓存
end

---@brief 清理数据（切号/重载）
---@type fun():void
function LanguageSettingDataManager:ClearData()
    LanguageSettingDataManager.super.ClearData(self)
    self._useLangs = nil
end
--endregion

--region 多语言配置获取
---=========================================================================
---@brief  获取游戏中已启用的语言类型列表（自动缓存）
---@type fun():table
---@return table 启用的语言配置数组
---=========================================================================
function LanguageSettingDataManager:GetInUseLangs()
    -- 已有缓存直接返回
    if self._useLangs ~= nil then
        return self._useLangs
    end

    self._useLangs = {}

    -- 安全校验配置表
    if not Tables or not Tables.LocalizationSetting then
        PrintError("[LanguageSettingDataManager] 多语言配置表 LocalizationSetting 不存在！")
        return self._useLangs
    end

    -- 遍历并收集启用的语言
    for i = 1, #Tables.LocalizationSetting do
        local cfg = Tables.LocalizationSetting[i]
        if cfg and cfg.InUse then
            table.insert(self._useLangs, cfg)
        end
    end

    return self._useLangs
end
--endregion

return LanguageSettingDataManager
