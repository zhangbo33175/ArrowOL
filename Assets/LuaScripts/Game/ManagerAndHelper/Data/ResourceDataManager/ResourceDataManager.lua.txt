--=====================================================================================================
-- @brief  资源物品表格数据管理中心
-- @desc   读取物品配置表、按类型/子类型获取道具列表、获取资源类型、缓存分类道具数据
-- @feat   全局单例、自动缓存、安全查询、分类排序、配置容错
--=====================================================================================================

---@class ResourceID
---@field ID string 道具ID

---@class ResourceDataManager : Singleton 资源物品表格数据管理中心
---@field _boosterIDs table<EItemSubType, ResourceID[]> 按子类型缓存的道具ID列表
---@field _resources table<EResourceType, ResourceID[]> 按资源类型缓存的道具ID列表
local ResourceDataManager = class("ResourceDataManager", import("Singleton"))

--- 构造函数
function ResourceDataManager:ctor()
    ResourceDataManager.super.ctor(self)
    self:Init()
end

--- 创建单例
---@return ResourceDataManager
function ResourceDataManager:Create()
    return ResourceDataManager.new()
end

--- 初始化（清空缓存）
function ResourceDataManager:Init()
    self._boosterIDs = nil
    self._resources = nil
end

--- 清理数据（切号/重载）
function ResourceDataManager:ClearData()
    ResourceDataManager.super.ClearData(self)
    self:Init()
end

---=========================================================================
--- 获取物品资源总表
---@return Tables.Resource_Item[]
---=========================================================================
function ResourceDataManager:GetResourceDatas()
    if not Tables or not Tables.Items then
        PrintError("[ResourceDataManager] Tables.Items 配置表不存在！")
        return {}
    end
    return Tables.Items
end

---=========================================================================
--- 根据资源ID获取物品配置
---@param resID string 资源ID
---@return Tables.Resource_Item|nil
---=========================================================================
function ResourceDataManager:GetResourceData(resID)
    if not resID then return nil end
    local items = self:GetResourceDatas()
    return items[tostring(resID)]
end

---=========================================================================
--- 根据资源ID获取资源类型
---@param resID string 资源ID
---@return EResourceType|nil
---=========================================================================
function ResourceDataManager:GetResourceTypeById(resID)
    local data = self:GetResourceData(resID)
    return data and data.Type or nil
end

---=========================================================================
--- 获取指定子类型的所有道具ID（已排序）
---@param boosterSubType EItemSubType
---@return ResourceID[]
---=========================================================================
function ResourceDataManager:GetBoosterIDs(boosterSubType)
    if not boosterSubType then return {} end

    self._boosterIDs = self._boosterIDs or {}
    if not self._boosterIDs[boosterSubType] then
        local list = {}
        local items = self:GetResourceDatas()

        for k, v in pairs(items) do
            if v.Type == EResourceType.Item and v.SubType == boosterSubType then
                table.insert(list, { ID = k })
            end
        end

        -- 稳定数字排序
        table.sort(list, function(a, b)
            local idA = tonumber(a.ID) or 99999
            local idB = tonumber(b.ID) or 99999
            return idA < idB
        end)

        self._boosterIDs[boosterSubType] = list
    end

    return self._boosterIDs[boosterSubType] or {}
end

---=========================================================================
--- 获取指定资源类型的所有道具ID（已排序）
---@param resType EResourceType
---@return ResourceID[]
---=========================================================================
function ResourceDataManager:GetResourceIDs(resType)
    if not resType then return {} end

    self._resources = self._resources or {}
    if not self._resources[resType] then
        local list = {}
        local items = self:GetResourceDatas()

        for k, v in pairs(items) do
            if v.Type == resType then
                table.insert(list, { ID = k })
            end
        end

        -- 稳定数字排序
        table.sort(list, function(a, b)
            local idA = tonumber(a.ID) or 99999
            local idB = tonumber(b.ID) or 99999
            return idA < idB
        end)

        self._resources[resType] = list
    end

    return self._resources[resType] or {}
end

return ResourceDataManager
