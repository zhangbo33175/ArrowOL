---@class CharacterGenerationDataManager @角色数据管理器
---@field _cacheByType table<ECharacterType, List<Tables.Characters_Item>> @按角色类型缓存
local CharacterGenerationDataManager = class("CharacterGenerationDataManager")

--region 【构造与创建】

---=========================================================================
---@brief  构造函数
---@type fun(args:table):void
---@param args table @自定义参数
---=========================================================================
function CharacterGenerationDataManager:ctor(args)
    self._cacheByType = {}
end

---=========================================================================
---@brief  创建实例
---@type fun(args:table):CharacterGenerationDataManager
---@param args table @自定义参数
---@return CharacterGenerationDataManager @角色数据管理器实例
---=========================================================================
function CharacterGenerationDataManager:Create(args)
    return CharacterGenerationDataManager.new(args)
end

--endregion

--region 【数据获取接口】

---=========================================================================
---@brief  根据角色ID获取角色配置
---@type fun(characterId:string):Tables.Characters_Item
---@param characterId string @角色ID
---@return Tables.Characters_Item @角色配置
---=========================================================================
function CharacterGenerationDataManager:GetDataById(characterId)
    if table.kIn(Tables.Characters, characterId) then
        return Tables.Characters[characterId]
    else
        PrintError(string.format("角色ID不存在：%s", characterId))
        return nil
    end
end

--endregion

return CharacterGenerationDataManager
