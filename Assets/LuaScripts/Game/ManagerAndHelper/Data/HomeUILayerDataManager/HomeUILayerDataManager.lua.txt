---@class HomeUILayerDataManager : Singleton @主界面UI层级数据管理器
local HomeUILayerDataManager = class("HomeUILayerDataManager", import("Singleton"))

--region 构造 & 实例化
---@brief 构造函数
---@type fun():void
function HomeUILayerDataManager:ctor()
    HomeUILayerDataManager.super.ctor(self)
end

---@brief 创建单例
---@type fun():HomeUILayerDataManager
---@return HomeUILayerDataManager
function HomeUILayerDataManager:Create()
    return HomeUILayerDataManager.new()
end
--endregion

--region 初始化 & 生命周期
---@brief 初始化
---@type fun():void
function HomeUILayerDataManager:Init()
    HomeUILayerDataManager.super.Init(self)
end

---@brief 清理数据
---@type fun():void
function HomeUILayerDataManager:ClearData()
    HomeUILayerDataManager.super.ClearData(self)
end
--endregion

--region UI层级配置获取
---=========================================================================
---@brief  根据层级ID获取UI层级配置
---@type fun(id:EGameMainUIPushSortOrder):Tables.HomeUILayers_Item
---@param id EGameMainUIPushSortOrder UI层级ID
---@return Tables.HomeUILayers_Item UI层级配置
---=========================================================================
function HomeUILayerDataManager:GetDataById(id)
    -- 配置表不存在防护
    if not Tables or not Tables.HomeUILayers then
        PrintError("[HomeUILayerDataManager] Tables.HomeUILayers 配置表不存在！")
        return nil
    end

    -- 空ID → 返回默认NormalUI
    if id == nil then
        PrintError("[HomeUILayerDataManager] GetDataById id is nil → 使用默认 NormalUI")
        return Tables.HomeUILayers[EGameMainUIPushSortOrder.NormalUI]
    end

    local cfg = Tables.HomeUILayers[id]
    -- 配置不存在 → 降级默认
    if not cfg then
        PrintDebug("[HomeUILayerDataManager] 未找到ID: " .. tostring(id) .. " → 使用默认 NormalUI")
        return Tables.HomeUILayers[EGameMainUIPushSortOrder.NormalUI]
    end

    return cfg
end

---=========================================================================
---@brief  根据层级ID获取UI显示顺序
---@type fun(id:EGameMainUIPushSortOrder):number
---@param id EGameMainUIPushSortOrder UI层级ID
---@return number 排序值（数值越小越先显示）
---=========================================================================
function HomeUILayerDataManager:GetOrderById(id)
    local data = self:GetDataById(id)
    if not data then
        PrintError("[HomeUILayerDataManager] GetOrderById data is nil → return 0")
        return 0
    end
    return data.Order or 0
end
--endregion

return HomeUILayerDataManager
