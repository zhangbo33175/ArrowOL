---@class PlayerInfoDataManager : Singleton @玩家信息存档管理器
local PlayerInfoDataManager = class("PlayerInfoDataManager", import("Singleton"))

--region 构造 & 实例化
---@brief 构造函数
---@type fun():void
function PlayerInfoDataManager:ctor()
    PlayerInfoDataManager.super.ctor(self)
end

---@brief 创建单例
---@type fun():PlayerInfoDataManager
---@return PlayerInfoDataManager
function PlayerInfoDataManager:Create()
    return PlayerInfoDataManager.new()
end
--endregion

--region 初始化 & 生命周期
---@brief 初始化
---@type fun():void
function PlayerInfoDataManager:Init()
    PlayerInfoDataManager.super.Init(self)
end

---@brief 清理数据
---@type fun():void
function PlayerInfoDataManager:ClearData()
    PlayerInfoDataManager.super.ClearData(self)
end
--endregion

--region 用户名管理
---=========================================================================
---@brief  获取玩家当前显示的用户名（自动判断默认名/自定义名）
---@type fun():string
---@return string 用户名
---=========================================================================
function PlayerInfoDataManager:GetPlayerInfoUserName()
    -- 安全校验存档
    if not SaveRootData or not SaveRootData.GameData or not SaveRootData.GameData.playerInfo then
        return Lang["PlayerDefaultNameError"] or "Player_Error"
    end

    local playerInfo = SaveRootData.GameData.playerInfo

    -- 未自定义过用户名
    if playerInfo.userInput == false then
        -- UID 异常
        if not SaveRootData.UserData or SaveRootData.UserData.uid <= 0 then
            return Lang["PlayerDefaultNameError"] or "Player_Error"
        else
            -- 返回默认用户名：玩家_UID
            return mstr(Lang["PlayerDefaultName"], { str1 = SaveRootData.UserData.uid })
        end
    else
        -- 返回自定义名字（空值防护）
        return playerInfo.name or Lang["PlayerDefaultNameError"]
    end
end

---=========================================================================
---@brief  设置玩家自定义用户名，并标记为已输入
---@type fun(newName:string):void
---@param newName string 用户新输入的用户名
---=========================================================================
function PlayerInfoDataManager:SetPlayerInfoUserName(newName)
    if not newName or type(newName) ~= "string" or newName == "" then
        PrintError("[PlayerInfoDataManager] 设置用户名失败：名字为空")
        return
    end

    -- 安全校验存档结构
    if not SaveRootData or not SaveRootData.GameData or not SaveRootData.GameData.playerInfo then
        PrintError("[PlayerInfoDataManager] 设置用户名失败：playerInfo 存档不存在")
        return
    end

    -- 写入数据
    SaveRootData.GameData.playerInfo.name = newName
    SaveRootData.GameData.playerInfo.userInput = true

    -- 保存存档
    if SaveIO and SaveIO.GameData and SaveIO.GameData.Save_playerInfo then
        SaveIO.GameData.Save_playerInfo()
    else
        PrintError("[PlayerInfoDataManager] 保存失败：存档接口不存在")
    end
end
--endregion

return PlayerInfoDataManager
