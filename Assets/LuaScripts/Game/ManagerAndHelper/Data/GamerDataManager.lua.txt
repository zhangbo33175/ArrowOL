--=====================================================================================================
-- @brief  游戏存档管理器
-- @desc   管理：当前章节、关卡进度、挑战关卡、玩家存档数据、区域数据、关卡次数统计
-- @feat   全局单例、存档安全、自动初始化、空值防护、错误捕获
--=====================================================================================================

---@class GamerDataManager : Singleton 游戏存档管理器
local GamerDataManager = class("GamerDataManager", import("Singleton"))

--- 构造函数
function GamerDataManager:ctor()
    GamerDataManager.super.ctor(self)
end

--- 创建单例
---@return GamerDataManager 存档管理器实例
function GamerDataManager:Create()
    return GamerDataManager.new()
end

--- 初始化
function GamerDataManager:Init()
    GamerDataManager.super.Init(self)
end

--- 清理数据
function GamerDataManager:ClearData()
    GamerDataManager.super.ClearData(self)
end

---=========================================================================
--- 获取最新解锁的章节ID（存档记录）
---@return string 章节Id
---=========================================================================
function GamerDataManager:GetLatestChapterId()
    if not SaveRootData or not SaveRootData.GameData then
        return "1"
    end

    local chapterId = SaveRootData.GameData.latestChapterId
    if string.isNullOrEmpty(chapterId) then
        SaveRootData.GameData.latestChapterId = "1"
        xpcall(function()
            SaveIO.GameData.Save_latestChapterId()
        end, function() end)
        return "1"
    end
    return chapterId
end

---=========================================================================
--- 获取当前选中/正在游玩的章节ID
---@return string 章节Id
---=========================================================================
function GamerDataManager:GetCurChapterId()
    if not SaveRootData or not SaveRootData.GameData then
        return "1"
    end

    local chapterId = SaveRootData.GameData.curChapterId
    if string.isNullOrEmpty(chapterId) then
        SaveRootData.GameData.curChapterId = "1"
        xpcall(function()
            SaveIO.GameData.Save_curChapterId()
        end, function() end)
        return "1"
    end
    return chapterId
end

---=========================================================================
--- 判断当前所在章节与关卡是否等于玩家最大进度
---@return boolean true-是最新进度
---=========================================================================
function GamerDataManager:CurrentChapterLevelIsLatest()
    local latestChapter = self:GetLatestChapterId()
    local currChapter = self:GetCurChapterId()

    if latestChapter ~= currChapter then
        return false
    end

    local latestLevel = self:GetLevelLvByChapterId(latestChapter)
    local currLevel = self:GetLevelLvByChapterId(currChapter)
    return latestLevel == currLevel
end

---=========================================================================
--- 设置当前游玩的章节ID
---@param chapterId string 目标章节ID
---=========================================================================
function GamerDataManager:SetCurChapterId(chapterId)
    if not chapterId or not SaveRootData or not SaveRootData.GameData then
        return
    end

    SaveRootData.GameData.curChapterId = chapterId
    xpcall(function()
        SaveIO.GameData.Save_curChapterId()
    end, function() end)
end

---=========================================================================
--- 根据章节ID获取当前关卡等级
---@param chapterId string 章节ID
---@param checkChallengeLevel boolean 是否检查挑战关卡
---@return number 关卡等级
---=========================================================================
function GamerDataManager:GetLevelLvByChapterId(chapterId, checkChallengeLevel)
    if not chapterId then return 1 end
    checkChallengeLevel = checkChallengeLevel or false

    -- 优先检查挑战关卡
    if checkChallengeLevel and ChallengeLevelHelper then
        local challengeLevelId = ChallengeLevelHelper:GetCurChallengeLevel(chapterId)
        if challengeLevelId > 0 then
            return challengeLevelId
        end
    end

    local chapterInfos = SaveRootData.GameData.chapterInfos
    if not chapterInfos or not table.kIn(chapterInfos, chapterId) then
        PrintError("[GamerDataManager] 未找到章节数据：" .. chapterId)
        return 1
    end

    local chapterInfo = chapterInfos[chapterId]
    return chapterInfo.lv or 1
end

---=========================================================================
--- 获取玩家最大进度的章节&关卡
---@return ChapterLevel 章节等级结构{Chapter, Level}
---=========================================================================
function GamerDataManager:GetLatestChapterLevel()
    local chapterId = self:GetLatestChapterId()
    local level = self:GetLevelLvByChapterId(chapterId)
    return { Chapter = tonumber(chapterId) or 1, Level = level }
end

---=========================================================================
--- 获取指定章节的当前关卡
---@param chapterId string 章节ID
---@return ChapterLevel 章节等级结构{Chapter, Level}
---=========================================================================
function GamerDataManager:GetTargetChapterLevel(chapterId)
    local level = self:GetLevelLvByChapterId(chapterId)
    return { Chapter = tonumber(chapterId) or 1, Level = level }
end

---=========================================================================
--- 获取当前章节&关卡
---@param checkChallengeLevel boolean 是否检查挑战关卡
---@return ChapterLevel 章节等级结构{Chapter, Level}
---=========================================================================
function GamerDataManager:GetChapterLevel(checkChallengeLevel)
    checkChallengeLevel = checkChallengeLevel or false
    local chapterId = self:GetCurChapterId()

    if checkChallengeLevel and ChallengeLevelHelper then
        local challengeLevelId = ChallengeLevelHelper:GetCurChallengeLevel(chapterId)
        if challengeLevelId > 0 then
            return { Chapter = tonumber(chapterId) or 1, Level = challengeLevelId }
        end
    end

    local level = self:GetLevelLvByChapterId(chapterId)
    return { Chapter = tonumber(chapterId) or 1, Level = level }
end

---=========================================================================
--- 获取玩家已解锁的最大章节ID
---@return string 最大章节ID
---=========================================================================
function GamerDataManager:GetCurMaxChapterId()
    local chapterInfos = SaveRootData.GameData.chapterInfos
    if not chapterInfos then return "1" end

    local keys = table.keys(chapterInfos)
    if #keys == 0 then return "1" end

    table.sort(keys, function(a, b)
        return tonumber(a) > tonumber(b)
    end)
    return keys[1]
end


---=========================================================================
--- 增加章节关卡等级
---@param chapterId string 章节ID
---@return boolean 是否增加成功
---=========================================================================
function GamerDataManager:AddLevelLvByChapterId(chapterId)
    return self:AddLevelLvOfMainChapter(chapterId)
end

---=========================================================================
--- 增加主线章节的关卡等级
---@param chapterId string 章节ID
---@return boolean 是否增加成功
---=========================================================================
function GamerDataManager:AddLevelLvOfMainChapter(chapterId)
    return false
end

---=========================================================================
--- 直接设置当前章节的关卡等级
---@param lv number 目标等级
---@return boolean 设置结果
---=========================================================================
function GamerDataManager:SetLevelLv(lv)
    if not lv or lv <= 0 then
        PrintError("[GamerDataManager] 关卡等级不能 <= 0")
        return false
    end

    local chapterId = self:GetCurChapterId()
    local chapterInfos = SaveRootData.GameData.chapterInfos

    if not chapterInfos or not table.kIn(chapterInfos, chapterId) then
        PrintError("[GamerDataManager] 未找到章节，无法设置等级：" .. chapterId)
        return false
    end

    local chapterInfo = chapterInfos[chapterId]
    chapterInfo.lv = lv

    xpcall(function()
        SaveIO.GameData.Save_chapterInfos()
    end, function() end)
    return true
end

---=========================================================================
--- 获取玩家首次注册时间
---@return number 注册时间戳
---=========================================================================
function GamerDataManager:GetFirstRegUserTime()
    local info = SaveRootData.GameData.playerInfo
    return info and info.regUserTime or 0
end

return GamerDataManager
