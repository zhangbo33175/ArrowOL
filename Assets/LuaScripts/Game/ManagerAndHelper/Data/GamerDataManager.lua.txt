--=====================================================================================================
-- @brief  游戏存档管理器
-- @desc   管理：当前章节、关卡进度、挑战关卡、玩家存档数据、区域数据、关卡次数统计
-- @feat   全局单例、存档安全、自动初始化、空值防护、错误捕获
--=====================================================================================================

---@class GamerDataManager : Singleton 游戏存档管理器
local GamerDataManager = class("GamerDataManager", import("Singleton"))

--- 构造函数
function GamerDataManager:ctor()
    GamerDataManager.super.ctor(self)
end

--- 创建单例
---@return GamerDataManager
function GamerDataManager:Create()
    return GamerDataManager.new()
end

--- 初始化
function GamerDataManager:Init()
    GamerDataManager.super.Init(self)
end

--- 清理数据
function GamerDataManager:ClearData()
    GamerDataManager.super.ClearData(self)
end

---=========================================================================
--- 获取最新解锁的章节ID（存档记录）
---@return string 章节Id
---=========================================================================
function GamerDataManager:GetLatestChapterId()
    if not PbRootData or not PbRootData.GameData then
        return "1"
    end

    local chapterId = PbRootData.GameData.latestChapterId
    if string.isNullOrEmpty(chapterId) then
        PbRootData.GameData.latestChapterId = "1"
        xpcall(function()
            PbIO.GameData.Save_latestChapterId()
        end, function() end)
        return "1"
    end
    return chapterId
end

---=========================================================================
--- 获取当前选中/正在游玩的章节ID
---@return string 章节Id
---=========================================================================
function GamerDataManager:GetCurChapterId()
    if not PbRootData or not PbRootData.GameData then
        return "1"
    end

    local chapterId = PbRootData.GameData.curChapterId
    if string.isNullOrEmpty(chapterId) then
        PbRootData.GameData.curChapterId = "1"
        xpcall(function()
            PbIO.GameData.Save_curChapterId()
        end, function() end)
        return "1"
    end
    return chapterId
end

---=========================================================================
--- 判断当前所在章节与关卡是否等于玩家最大进度
---@return boolean
---=========================================================================
function GamerDataManager:CurrentChapterLevelIsLatest()
    local latestChapter = self:GetLatestChapterId()
    local currChapter = self:GetCurChapterId()

    if latestChapter ~= currChapter then
        return false
    end

    local latestLevel = self:GetLevelLvByChapterId(latestChapter)
    local currLevel = self:GetLevelLvByChapterId(currChapter)
    return latestLevel == currLevel
end

---=========================================================================
--- 设置当前游玩的章节ID
---@param chapterId string
---=========================================================================
function GamerDataManager:SetCurChapterId(chapterId)
    if not chapterId or not PbRootData or not PbRootData.GameData then
        return
    end

    PbRootData.GameData.curChapterId = chapterId
    xpcall(function()
        PbIO.GameData.Save_curChapterId()
    end, function() end)
end

---=========================================================================
--- 根据章节ID获取当前关卡等级
---@param chapterId string
---@param checkChallengeLevel boolean
---@return number
---=========================================================================
function GamerDataManager:GetLevelLvByChapterId(chapterId, checkChallengeLevel)
    if not chapterId then return 1 end
    checkChallengeLevel = checkChallengeLevel or false

    -- 优先检查挑战关卡
    if checkChallengeLevel and ChallengeLevelHelper then
        local challengeLevelId = ChallengeLevelHelper:GetCurChallengeLevel(chapterId)
        if challengeLevelId > 0 then
            return challengeLevelId
        end
    end

    local chapterInfos = PbRootData.GameData.chapterInfos
    if not chapterInfos or not table.kIn(chapterInfos, chapterId) then
        PrintError("[GamerDataManager] 未找到章节数据：" .. chapterId)
        return 1
    end

    local chapterInfo = chapterInfos[chapterId]
    return chapterInfo.lv or 1
end

---=========================================================================
--- 获取玩家最大进度的章节&关卡
---@return ChapterLevel
---=========================================================================
function GamerDataManager:GetLatestChapterLevel()
    local chapterId = self:GetLatestChapterId()
    local level = self:GetLevelLvByChapterId(chapterId)
    return { Chapter = tonumber(chapterId) or 1, Level = level }
end

---=========================================================================
--- 获取指定章节的当前关卡
---@param chapterId string
---@return ChapterLevel
---=========================================================================
function GamerDataManager:GetTargetChapterLevel(chapterId)
    local level = self:GetLevelLvByChapterId(chapterId)
    return { Chapter = tonumber(chapterId) or 1, Level = level }
end

---=========================================================================
--- 获取当前章节&关卡
---@param checkChallengeLevel boolean
---@return ChapterLevel
---=========================================================================
function GamerDataManager:GetChapterLevel(checkChallengeLevel)
    checkChallengeLevel = checkChallengeLevel or false
    local chapterId = self:GetCurChapterId()

    if checkChallengeLevel and ChallengeLevelHelper then
        local challengeLevelId = ChallengeLevelHelper:GetCurChallengeLevel(chapterId)
        if challengeLevelId > 0 then
            return { Chapter = tonumber(chapterId) or 1, Level = challengeLevelId }
        end
    end

    local level = self:GetLevelLvByChapterId(chapterId)
    return { Chapter = tonumber(chapterId) or 1, Level = level }
end

---=========================================================================
--- 获取玩家已解锁的最大章节ID
---@return string
---=========================================================================
function GamerDataManager:GetCurMaxChapterId()
    local chapterInfos = PbRootData.GameData.chapterInfos
    if not chapterInfos then return "1" end

    local keys = table.keys(chapterInfos)
    if #keys == 0 then return "1" end

    table.sort(keys, function(a, b)
        return tonumber(a) > tonumber(b)
    end)
    return keys[1]
end

---=========================================================================
--- 判断当前关卡是否为产出星星的正常关卡
---@param chapterId number
---@param levelId number
---@return boolean
---=========================================================================
function GamerDataManager:IsOutputStarLevel(chapterId, levelId)
    if ChallengeLevelHelper then
        local challengeLevelId = ChallengeLevelHelper:GetCurChallengeLevel(chapterId)
        if challengeLevelId > 0 then
            return false
        end
    end

    local chapterData = ChapterDataMgr:GetChapterDataByChapterId(tostring(chapterId))
    if not chapterData or not chapterData.UnlockNextLevel then
        return false
    end
    return chapterData.UnlockNextLevel > levelId
end

function GamerDataManager:AddLevelLvOfMainChapter(chapterId)
    return false

---=========================================================================
--- 直接设置当前章节的关卡等级
---@param lv number
---@return boolean
---=========================================================================
function GamerDataManager:SetLevelLv(lv)
    if not lv or lv <= 0 then
        PrintError("[GamerDataManager] 关卡等级不能 <= 0")
        return false
    end

    local chapterId = self:GetCurChapterId()
    local chapterInfos = PbRootData.GameData.chapterInfos

    if not chapterInfos or not table.kIn(chapterInfos, chapterId) then
        PrintError("[GamerDataManager] 未找到章节，无法设置等级：" .. chapterId)
        return false
    end

    local chapterInfo = chapterInfos[chapterId]
    chapterInfo.lv = lv

    xpcall(function()
        PbIO.GameData.Save_chapterInfos()
    end, function() end)
    return true
end

---=========================================================================
--- 获取指定关卡的开始/重试次数
---@param chapterId string
---@param levelId number
---@return number
---=========================================================================
function GamerDataManager:GetLevelStartTimes(chapterId, levelId)
    local playerInfo = PbRootData.GameData.playerInfo
    if not playerInfo.lastInLevel then
        playerInfo.lastInLevel = {}
    end

    local key = chapterId .. "_" .. tostring(levelId)
    return playerInfo.lastInLevel[key] or 0
end

---=========================================================================
--- 增加关卡开始/重试次数，并清理同章节旧记录
---@param chapterId string
---@param levelId number
---@return number
---=========================================================================
function GamerDataManager:AddLevelStartTimes(chapterId, levelId)
    local playerInfo = PbRootData.GameData.playerInfo
    if not playerInfo.lastInLevel then
        playerInfo.lastInLevel = {}
    end

    local key = chapterId .. "_" .. tostring(levelId)
    local lastInLevel = playerInfo.lastInLevel

    -- 次数+1
    lastInLevel[key] = (lastInLevel[key] or 0) + 1

    -- 清理同章节旧关卡
    local prefix = chapterId .. "_"
    for k, _ in pairs(lastInLevel) do
        if string.startsWith(k, prefix) and k ~= key then
            lastInLevel[k] = nil
        end
    end

    xpcall(function()
        PbIO.GameData.Save_playerInfo_lastInLevel()
    end, function() end)

    return lastInLevel[key]
end

---=========================================================================
--- 获取玩家指定区域的存档数据
---@param areaId string
---@return SavePBMsgDef.SaveGameData.PlayerAreaInfo
---=========================================================================
function GamerDataManager:GetPlayerAreaInfo(areaId)
    if string.isNullOrEmpty(areaId) then
        areaId = ChapterListManager and ChapterListManager:GetLatestAreaId() or "1"
    end

    local playerAreaInfos = PbRootData.GameData.playerInfo.playerAreaInfos
    if not playerAreaInfos then
        PrintError("[GamerDataManager] playerAreaInfos 不存在")
        return nil
    end

    if not playerAreaInfos[areaId] then
        playerAreaInfos[areaId] = {
            areaId = areaId,
            areaTotalCoinSpend = 0,
            areaTotalUpgradeCoinSpend = 0
        }
        xpcall(function()
            PbIO.GameData.Save_playerInfo_playerAreaInfos()
        end, function() end)
    end

    return playerAreaInfos[areaId]
end

---=========================================================================
--- 获取玩家首次注册时间
---@return number
---=========================================================================
function GamerDataManager:GetFirstRegUserTime()
    local info = PbRootData.GameData.playerInfo
    return info and info.regUserTime or 0
end

return GamerDataManager
