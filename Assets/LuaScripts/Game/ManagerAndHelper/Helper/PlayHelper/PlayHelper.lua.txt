---@class PlayHelper @开始游戏帮助类：统一管理关卡启动、UI打开、特效播放、奖励展示等游戏核心逻辑
---@field _loopLevelHelper LoopLevelHelper @ 循环关卡辅助类（预留）
local PlayHelper = class("PlayHelper")

--region 【构造与初始化】
---@type fun():void
---@brief 构造函数：初始化常量配置、颜色配置、格式字符串
function PlayHelper:ctor()
    self._mainMinChapterId = 1                                          ---@type number                                 ---主线最小章节ID
    self._mainMaxChapterId = 1000                                       ---@type number                                 ---主线最大章节ID
    self._goldenTicketPlayNameTextColorConfigData = nil                 ---@type TextColorConfigData                    ---金票玩家名字颜色配置
    self._goldenTicketHighPlayNameTextColorConfigData = nil             ---@type TextColorConfigData                    ---高级金票玩家名字颜色配置
    self._OutLineName = "Material_RGame_SDF_Static___Outline_%s"        ---@type string                                 ---文字描边材质名称格式化串
end

---@type fun():PlayHelper
---@brief 创建PlayHelper单例对象
---@return PlayHelper 帮助类实例
function PlayHelper:Create()
    local obj = PlayHelper.new()
    return obj
end
--endregion

--region 【关卡状态判断】
---@type fun(chapterId:string):boolean
---@brief 判断是否为主线章节
---@param chapterId string 章节ID
---@return boolean 是否主线章节
function PlayHelper:IsMainChapter(chapterId)
    if string.isNullOrEmpty(chapterId) then
        return false
    end
    local chapterIdNum = tonumber(chapterId)
    return chapterIdNum >= self._mainMinChapterId and chapterIdNum <= self._mainMaxChapterId
end

---@type fun():boolean
---@brief 判断当前是否处于游戏关卡状态
---@return boolean 是否在游戏中
function PlayHelper:IsInLevelState()
    if GameStateMachine:CurrentStateType() == GameStateCmd.PlayingState then
        return true
    end
    return false
end

---@type fun():boolean
---@brief 判断是否满足游玩第一关条件
---@return boolean 是否满足
function PlayHelper:IsPlayFirstLevel()
    local chapterId = ChapterListManager:GetLatestChapterId()
    if chapterId == "1" and GamerDataManager:GetLevelLvByChapterId(chapterId) == 1 then
        return true
    end
    return false
end

---@type fun(chapterId:string,levelId:number):boolean
---@brief 判断是否需要显示准备关卡界面
---@param chapterId string 章节ID
---@param levelId number 关卡ID
---@return boolean 是否显示
function PlayHelper:IsPlayShowPrepareLevel(chapterId, levelId)
    return false
end

---@type fun():boolean
---@brief 是否正在游玩最新章节
---@return boolean
function PlayHelper:IsPlayingNewChapter()
    return SaveRootData.GameData.latestChapterId == SaveRootData.GameData.curChapterId
end
--endregion

--region 【关卡数据与难度奖励】
---@type fun(chapterId:string,levelId:number):Tables.LevelData_Item
---@brief 加载并获取关卡配置数据
---@param chapterId string 章节ID
---@param levelId number 关卡ID
---@return Tables.LevelData_Item 关卡数据
function PlayHelper:LoadLevelData(chapterId, levelId)
    ---@type Tables.Chapters_Item
    local chapterConfig = Tables.MainLevels[tostring(chapterId)]
    if chapterConfig and chapterConfig.MainLevels and not ChallengeLevelHelper:IsChallengeLevel(levelId) and levelId > chapterConfig.MainLevels[2] then
        return nil
    end
    local levelName = table.concat({ "LevelData_", chapterId, "_", levelId })
    local levelData = Tables[levelName]
    return levelData
end

--endregion

--region 【游戏开始/结束/退出逻辑】
---@type fun(levelData:Tables.LevelData_Item):boolean
---@brief 开始游戏（显示心心）
---@param levelData Tables.LevelData_Item 关卡数据
---@return boolean 是否成功开始
function PlayHelper:PlayLevel(levelData)
    if levelData == nil then
        return false
    end
    if GameCoreManager:IsLifeEnough() and self:PlayLevelImmediate(levelData) then
        return true
    end
    return false
end

---@type fun(levelData:Tables.LevelData_Item):boolean
---@brief 立即开始游戏（不检查体力）
---@param levelData Tables.LevelData_Item 关卡数据
---@return boolean 是否成功
function PlayHelper:PlayLevelImmediate(levelData)
    if levelData then
        --GameCoreManager:ClearData()
        GameCoreManager._CurLevelData = levelData
        return true
    end
    PrintError("PlayLevelImmediate is not implement,LevelData is nil")
    return false
end

---@type fun(isSuccess:boolean):void
---@brief 游戏结束
---@param isSuccess boolean 是否胜利
function PlayHelper:GameOver(isSuccess)
    PrintInfo("GameOver,IsSuccess:" .. tostring(isSuccess))
    GameCoreManager._IsGameOver = true
    GameCoreManager._IsSuccessFinishGame = isSuccess
    GameCoreManager:SetPlayingLevelTag('')
end

---@type fun():void
---@brief 主动点击退出游戏
function PlayHelper:ClickGameExit()
    PrintInfo("ClickGameExit")
    GameCoreManager:DoGameFailed(EGameOverType.PlayerGiveUp)
    self:GameExitPlayHud()
end

---@type fun(id:string,num:string):void
---@brief 注册退出二次确认提示
---@param id string 提示ID
---@param strNum string 提示数字字符串
function PlayHelper:RegisterExitSureTip(id, strNum)

end

---@type fun():void
---@brief 退出关卡，返回主界面
function PlayHelper:GameExitPlayHud()
    GameStateMachine:ChangeState(GameStateCmd.MainState, nil)
end

---@type fun():void
---@brief 游戏胜利后离开关卡（结算专用）
function PlayHelper:OnLevelWinExitPlayGameFun()
    if GuideContext:IsTutorialExecute() or not ADMgr:TryShowInterAd(EInterAdID.LevelWin, function()
        PlayHelper:GameExitPlayHud()
    end, nil) then
        PlayHelper:GameExitPlayHud()
    end
end

---@type fun():void
---@brief 等级礼包领奖关闭动画回调
function PlayHelper:OnLevelWinRewardBeginCloseFun()
    CloseUI(UIs.UILevelSuccessBg)
end
--endregion

--region 【通用游戏入口】
---@type fun(lvData:Tables.LevelData_Item,view:GameViewSuper,boostersLua:UIBoosterButton[]):boolean
---@brief 通用游戏开始入口（处理道具、打点、状态切换）
---@param lvData Tables.LevelData_Item 关卡数据
---@param view GameViewSuper 界面（可为空）
---@param boostersLua UIBoosterButton[] 前置道具（可为空）
---@return boolean 是否成功
function PlayHelper:CommonOnPlayGame(lvData, view, boostersLua)
    local preboosterUse = {}
    if boostersLua then
        for i = 1, #boostersLua do
            if boostersLua[i]:GetSelectedState() then
                table.insert(preboosterUse, boostersLua[i]:GetPropID())
            end
        end
    end

    if self:PlayLevel(lvData) then
        if boostersLua then
            for i = 1, #boostersLua do
                boostersLua[i]:OnConfirmBooster()
            end
        end
        if self:IsInLevelState() then
            PlayHudCycle:OnAction(EPlayHudCycleAction.RePlayGame)
        else
            GameStateMachine:ChangeState(GameStateCmd.PlayingState, nil)
        end
        if view then
            view:Close()
        end
        return true
    else
        if not GameCoreManager:IsLifeEnough() then
            OpenUISync(UIs.UILife)
        else
            if view then
                view:Close()
            end
        end
        return false
    end
end
--endregion

--region 【音频管理】
---@type fun():void
---@brief 播放关卡内背景音乐
function PlayHelper:PlayInGameBGM()
    local levelData = GameCoreManager._CurLevelData
    if levelData then
        if MapManager.CurLoadMapType == EMapType.Main then
            local peakId = levelData.PeakId
            local peakLevelData = PeakLevelsDataMgr:GetDataById(tostring(peakId))
            if peakLevelData and peakId ~= ELevelType.Normal and not string.isNullOrEmpty(peakLevelData.LevelBGM) then
                SoundManager:PlayBGM(peakLevelData.LevelBGM)
            else
                self:PlayCommonInGameBGM()
            end
            return
        end
        if not string.isNullOrEmpty(MapManager.CurLoadChapterId) then
            PrintError("PlayHelper:PlayInGameBGM() 未知的地图类型 chapterId" .. MapManager.CurLoadChapterId)
        end
    end
end

---@type fun():void
---@brief 播放普通关卡背景音乐
function PlayHelper:PlayCommonInGameBGM()
    --if GameCoreManager:IsInChallengeLevel() then
    --    SoundManager:PlayBGM(Tables.Sounds.BGM_Challenge.Name)
    --    return
    --end

    local chapterData = ChapterDataManager:GetChapterDataByChapterType(MapManager.CurLoadChapterId)
    if chapterData and chapterData.MapMusic and chapterData.MapMusic.ingame then
        SoundManager:PlayBGM(chapterData.MapMusic.ingame)
    end
end
--endregion

--region 【资源/地图加载】
---@type fun(chapterId:number, callback:fun(), refuseAd:boolean):void
---@brief 加载主线地图
---@param chapterId number 章节ID
---@param callback fun() 加载完成回调
---@param refuseAd boolean 不播放广告
function PlayHelper:LoadMainMap(chapterId, callback, refuseAd)
    local areaData = ChapterDataManager:GetAreaDataByChapterId(chapterId)
    if not areaData then
        PrintError("没有找到章节所在区域数据:" .. chapterId)
        return
    end
    MapManager:OnSetSwitchMap(chapterId, callback, refuseAd)
end
--endregion

--region 【UI 资源刷新（图片/材质）】
---@type fun(ui:LuaBehaviourSuper, uiText:TMPro.TextMeshProUGUI,outlineMatName:string):void
---@brief 刷新文本描边材质
function PlayHelper:RefreshTextOutLineMaterial(ui, uiText, outlineMatName)
    if ui and uiText then
        ---@type RES_DEF_ITEM
        local info = LoadRes[outlineMatName]
        if info and (not uiText.fontMaterial or uiText.fontMaterial.name ~= info.AssetName) then
            uiText.fontMaterial = ui:LoadAsset(info)
        end
    end
end

---@type fun(ui:LuaBehaviourSuper, uiImage:UnityEngine.UI.Image,spriteName:string):void
---@brief 刷新图片精灵
function PlayHelper:RefreshImageSprite(ui, uiImage, spriteName)
    if ui and uiImage then
        ---@type RES_DEF_ITEM
        local info = LoadRes["Sprite_"..spriteName]
        if info and (not uiImage.sprite or uiImage.sprite.name ~= info.AssetName) then
            uiImage.sprite = ui:LoadAsset(info)
            uiImage:SetNativeSize()
        end
    end
end
--endregion

return PlayHelper
