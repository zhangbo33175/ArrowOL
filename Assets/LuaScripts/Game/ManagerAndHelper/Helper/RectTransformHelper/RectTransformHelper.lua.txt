---@class RectTransformHelper RectTransform 工具类：封装 UGUI 常用操作
local RectTransformHelper = class("RectTransformHelper")

--region 构造 & 实例化
---=========================================================================
--- 构造函数
---@param args table 预留扩展参数
---=========================================================================
function RectTransformHelper:ctor(args)
end

---=========================================================================
--- 创建工具类实例
---@param args table 预留扩展参数
---@return RectTransformHelper
---=========================================================================
function RectTransformHelper:Create(args)
    return RectTransformHelper.new(args)
end
--endregion

--region 锚点位置操作
---=========================================================================
--- 设置锚点位置 X 坐标
---@param rectTransform UnityEngine.RectTransform 目标组件
---@param posX number 目标 X 值
---=========================================================================
function RectTransformHelper:SetAnchoredPositionX(rectTransform, posX)
    local anchoredPos = rectTransform.anchoredPosition
    anchoredPos.x = posX
    rectTransform.anchoredPosition = anchoredPos
end

---=========================================================================
--- 设置锚点位置 Y 坐标
---@param rectTransform UnityEngine.RectTransform 目标组件
---@param posY number 目标 Y 值
---=========================================================================
function RectTransformHelper:SetAnchoredPositionY(rectTransform, posY)
    local anchoredPos = rectTransform.anchoredPosition
    anchoredPos.y = posY
    rectTransform.anchoredPosition = anchoredPos
end
--endregion

return RectTransformHelper
