--=====================================================================================================
-- @brief:     可暂停协程工具类（关卡专用）
-- @desc:      支持按信号量分组暂停/恢复，解决关卡弹窗、游戏暂停时冻结逻辑的问题
--=====================================================================================================

--region 内部元表与暂停信号定义
--- 全局暂停信号数组
--- 1 = 通用全局暂停(Any)
--- 2 = 关卡逻辑暂停(Level)
local _pauseSignal = { false, false }

--- 协程结束标记
local _coroutineEnd = {}

--- 可暂停协程迭代器元表
local _pauseableCoroutineMetaTable = {
    __index = {
        --- 协程每帧迭代
        MoveNext = function(cor)
            -- 暂停中：卡住协程
            if _pauseSignal[cor.pauseSignal] then
                return true
            end

            cor.Current = cor.coFunc()
            if cor.Current == _coroutineEnd then
                cor.Current = nil
                return false
            else
                return true
            end
        end,

        --- 重置协程
        Reset = function(cor)
            cor.coFunc = coroutine.wrap(cor.originFunc)
        end
    }
}
--endregion

---@class PauseableCoroutineHelper 可暂停协程工具类（关卡专用）
---@field Any number 通用全局暂停信号
---@field Level number 关卡逻辑暂停信号
local PauseableCoroutineHelper = class("PauseableCoroutineHelper")

---@brief 构造函数
function PauseableCoroutineHelper:ctor()
    self.Any = 1       --- 通用全局暂停
    self.Level = 2     --- 关卡逻辑暂停
end

---@brief 创建实例
---@return PauseableCoroutineHelper
function PauseableCoroutineHelper:Create()
    return PauseableCoroutineHelper.new()
end

---@brief 内部：创建可暂停协程迭代器
---@param func function 协程函数
---@param pauseSignal number 暂停信号
---@param from any 调用来源（调试）
---@return table 协程迭代器
function PauseableCoroutineHelper:_createPauseableEnumerator(func, pauseSignal, from)
    local enumerator = setmetatable({
        originFunc = function()
            func()
            return _coroutineEnd
        end,
        pauseSignal = pauseSignal,
        from = from
    }, _pauseableCoroutineMetaTable)

    enumerator:Reset()
    return enumerator
end

---@brief 启动可暂停协程
---@param handler handler 方法包装器
---@param pauseSignal number 暂停信号类型
---@param from any 调试来源
---@return UnityEngine.Coroutine
function PauseableCoroutineHelper:StartPauseableCoroutine(handler, pauseSignal, from)
    pauseSignal = pauseSignal or self.Any

    if GameMainRoot.Launcher then
        local enumerator = self:_createPauseableEnumerator(handler.func, pauseSignal, from)
        return GameMainRoot.Launcher:StartCoroutine(enumerator)
    end

    return nil
end

---@brief 停止可暂停协程
---@param cor UnityEngine.Coroutine
function PauseableCoroutineHelper:StopPauseableCoroutine(cor)
    if not cor then return end
    if GameMainRoot.Launcher then
        GameMainRoot.Launcher:StopCoroutine(cor)
    end
end

---@brief 暂停一组协程
---@param pauseSignal number 信号类型
function PauseableCoroutineHelper:PauseCoroutineGroup(pauseSignal)
    pauseSignal = pauseSignal or self.Any
    _pauseSignal[pauseSignal] = true
end

---@brief 恢复一组协程
---@param pauseSignal number 信号类型
function PauseableCoroutineHelper:ResumeCoroutineGroup(pauseSignal)
    pauseSignal = pauseSignal or self.Any
    _pauseSignal[pauseSignal] = false
end

return PauseableCoroutineHelper
