---@class AnimatorHelper 动画控制器通用工具类（纯工具方法）
local AnimatorHelper = class("AnimatorHelper")

--region 构造 & 实例化
---@brief 构造函数
---@type fun(args:table):void
---@param args table 构造参数
function AnimatorHelper:ctor(args)
end

---@brief 创建工具类实例
---@type fun(args:table):AnimatorHelper
---@param args table 构造参数
---@return AnimatorHelper
function AnimatorHelper:Create(args)
    return AnimatorHelper.new(args)
end
--endregion

--region 动画时长获取
---@brief 获取单个动画片段时长
---@type fun(animator:UnityEngine.Animator,clipName:string):number
---@param animator UnityEngine.Animator 动画控制器
---@param clipName string 动画片段名称
---@return number 时长（秒），未找到返回 0
function AnimatorHelper:GetAnimationLength(animator, clipName)
    if not animator or string.isNullOrEmpty(clipName) then
        return 0
    end

    local controller = animator.runtimeAnimatorController
    if not controller then return 0 end

    local clips = controller.animationClips
    for i = 0, clips.Length - 1 do
        if clips[i].name == clipName then
            return clips[i].length
        end
    end
    return 0
end

---@brief 获取所有动画总时长
---@type fun(animator:UnityEngine.Animator):number
---@param animator UnityEngine.Animator 动画控制器
---@return number 总时长（秒）
function AnimatorHelper:GetTotalAnimationLength(animator)
    if not animator or not animator.runtimeAnimatorController then
        return 0
    end

    local clips = animator.runtimeAnimatorController.animationClips
    local totalTime = 0
    for i = 0, clips.Length - 1 do
        totalTime = totalTime + clips[i].length
    end
    return totalTime
end
--endregion

--region 动画播放控制
---@brief 从头播放指定动画
---@type fun(animator:UnityEngine.Animator,stateName:string):void
---@param animator UnityEngine.Animator 动画控制器
---@param stateName string 状态机名称
function AnimatorHelper:PlayAnimation(animator, stateName)
    if not animator or string.isNullOrEmpty(stateName) then
        return
    end

    GameLib.AnimatorUtil.PlayByNormalizedTime(animator, stateName, 0, 0)
    animator:Update(0)
end

---@brief 采样动画（直接跳转到指定时间点）
---@type fun(animator:UnityEngine.Animator,stateName:string,time:number):void
---@param animator UnityEngine.Animator 动画控制器
---@param stateName string 状态机名称
---@param time number 目标时间（秒）
function AnimatorHelper:SampleAnimation(animator, stateName, time)
    if not animator or string.isNullOrEmpty(stateName) then
        return
    end

    GameLib.AnimatorUtil.PlayByNormalizedTime(animator, stateName, 0, 0)
    animator:Update(0)
    animator:Update(time)
end

---@brief 安全播放动画（避免重复播放）
---@type fun(animator:UnityEngine.Animator,stateName:string):void
---@param animator UnityEngine.Animator 动画控制器
---@param stateName string 状态机名称
function AnimatorHelper:SafePlayAnimation(animator, stateName)
    if not animator or string.isNullOrEmpty(stateName) then
        return
    end

    local stateInfo = animator:GetCurrentAnimatorStateInfo(0)
    if stateInfo then
        -- 非目标动画 或 动画已播放完毕 → 重新播放
        if not stateInfo:IsName(stateName) or stateInfo.normalizedTime >= 1 then
            self:PlayAnimation(animator, stateName)
        end
    else
        -- 获取状态失败，直接播放
        self:PlayAnimation(animator, stateName)
    end
end
--endregion

return AnimatorHelper
