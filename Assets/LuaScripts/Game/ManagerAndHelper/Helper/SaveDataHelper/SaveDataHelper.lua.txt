--=====================================================================================================
-- @brief  存档同步助手（全局单例）
-- @desc   负责登录、心跳、存档上传/下载、第三方账号绑定、数据覆盖、错误码处理
--         核心作用：本地存档 ↔ 云端存档 双向同步，账号体系入口
-- @规范   Unity Lua 商业项目编码标准 / 注释完整 / 健壮性 / 性能优化
--=====================================================================================================

---@class SaveDataHelper 存档&登录&云端同步核心单例
local SaveDataHelper = class("SaveDataHelper")

--region 构造 & 实例化
---=========================================================================
--- 构造函数
---=========================================================================
function SaveDataHelper:ctor(args)
    ---@type string 登录鉴权 Token
    self._loginToken = ""
end

---=========================================================================
--- 创建单例实例
---@return SaveDataHelper
---=========================================================================
function SaveDataHelper:Create(args)
    return SaveDataHelper.new(args)
end
--endregion

--=========================================================================
--region 登录信息本地保存
--=========================================================================
---=========================================================================
--- 保存登录返回的用户信息（UID、注册时间、登录时间、绑定时间）
---@param returnData table 登录返回数据
---=========================================================================
function SaveDataHelper:SaveSignInInfos(returnData)
    if not returnData then return end

    -- 保存 UID
    if returnData.user_id and returnData.user_id > 0 then
        SaveRootData.UserData.uid = returnData.user_id
        SaveIO.UserData.Save_uid()
    end

    -- 保存注册时间
    if returnData.create_time and returnData.create_time > 0 then
        local time = returnData.create_time
        SaveRootData.UserData.regTime.utc0 = time
        SaveRootData.UserData.regTime.utc0String = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", time)
        SaveRootData.UserData.regTime.locality = HonorTimerHelper:GetLocalTimeStamp(time)
        SaveRootData.UserData.regTime.localString = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", SaveRootData.UserData.regTime.locality)
        SaveIO.UserData.Save_regTime()
    end

    -- 保存登录时间（上次 & 当前）
    if returnData.login_time and returnData.login_time > 0 then
        local lastLogin = SaveRootData.UserData.loginTime.current
        SaveRootData.UserData.loginTime.last.utc0 = lastLogin.utc0
        SaveRootData.UserData.loginTime.last.utc0String = lastLogin.utc0String
        SaveRootData.UserData.loginTime.last.locality = lastLogin.locality
        SaveRootData.UserData.loginTime.last.localString = lastLogin.localString

        local currLoginTime = returnData.login_time
        SaveRootData.UserData.loginTime.current.utc0 = currLoginTime
        SaveRootData.UserData.loginTime.current.utc0String = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", currLoginTime)
        SaveRootData.UserData.loginTime.current.locality = HonorTimerHelper:GetLocalTimeStamp(currLoginTime)
        SaveRootData.UserData.loginTime.current.localString = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", SaveRootData.UserData.loginTime.current.locality)
        SaveIO.UserData.Save_loginTime()
    end

    -- 保存国家地区
    if returnData.country then
        UserTypeMgr:TrySaveUserCountry(returnData.country)
    end
end

---=========================================================================
--- 仅保存登录 Token
---@param returnData table 登录数据
---=========================================================================
function SaveDataHelper:SaveToken(returnData)
    if returnData and returnData.token then
        self._loginToken = returnData.token
    end
end

---=========================================================================
--- 保存云端第三方账号信息
---@param returnData table 服务器数据
---=========================================================================
function SaveDataHelper:SaveSignInAid(returnData)
    if returnData and returnData.ServerUserInfo then
        SaveRootData.UserData.aid = returnData.ServerUserInfo.server_account
        SaveIO.UserData.Save_aid()
    end
end
--endregion

--=========================================================================
--region 云端存档 → 本地覆盖
--=========================================================================
---=========================================================================
--- 动态重载存档结构并替换数据
---@param otherGameData table 外部数据
---@return table 新的游戏数据
---=========================================================================
function SaveDataHelper:DynamicReplacementGameData(otherGameData)
    package.loaded["SaveGameData"] = nil
    require('SaveGameData')

    local result = SaveRootData.GameData
    self:CopyGameDataTable(result, otherGameData)
    return result
end

---=========================================================================
--- 递归复制存档表（云端 → 本地）
---@param dest table 目标表
---@param source table 源表
---=========================================================================
function SaveDataHelper:CopyGameDataTable(dest, source)
    for key, value in pairs(source) do
        if type(value) ~= "table" then
            dest[key] = value
        else
            dest[key] = dest[key] or {}
            self:CopyGameDataTable(dest[key], value)
        end
    end
end

---=========================================================================
--- 登录成功后：保存信息 + 自动全量同步
---@param returnData table 登录数据
---@param completeCallback function 完成回调
---=========================================================================
function SaveDataHelper:SaveLocalDataAndAutoSyncData(returnData, completeCallback)
    self:SaveSignInInfos(returnData)
    if completeCallback then
        completeCallback()
    end
end
--endregion

--=========================================================================

return SaveDataHelper
