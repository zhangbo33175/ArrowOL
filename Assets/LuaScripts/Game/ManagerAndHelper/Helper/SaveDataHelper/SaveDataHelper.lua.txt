--region 登录信息本地保存
--=========================================================================

---@brief 保存登录返回的用户信息（UID、注册时间、登录时间、绑定时间）
---@param returnData table 登录返回数据
function SaveDataHelper:SaveSignInInfos(returnData)
    if not returnData then return end

    -- 保存 UID
    if returnData.user_id and returnData.user_id > 0 then
        PbRootData.UserData.uid = returnData.user_id
        PbIO.UserData.Save_uid()
    end

    -- 保存注册时间
    if returnData.create_time and returnData.create_time > 0 then
        local time = returnData.create_time
        PbRootData.UserData.regTime.utc0 = time
        PbRootData.UserData.regTime.utc0String = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", time)
        PbRootData.UserData.regTime.locality = HonorTimerHelper:GetLocalTimeStamp(time)
        PbRootData.UserData.regTime.localString = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", PbRootData.UserData.regTime.locality)
        PbIO.UserData.Save_regTime()
    end

    -- 保存登录时间（上次 & 当前）
    if returnData.login_time and returnData.login_time > 0 then
        local lastLogin = PbRootData.UserData.loginTime.current
        PbRootData.UserData.loginTime.last.utc0 = lastLogin.utc0
        PbRootData.UserData.loginTime.last.utc0String = lastLogin.utc0String
        PbRootData.UserData.loginTime.last.locality = lastLogin.locality
        PbRootData.UserData.loginTime.last.localString = lastLogin.localString

        local currLoginTime = returnData.login_time
        PbRootData.UserData.loginTime.current.utc0 = currLoginTime
        PbRootData.UserData.loginTime.current.utc0String = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", currLoginTime)
        PbRootData.UserData.loginTime.current.locality = HonorTimerHelper:GetLocalTimeStamp(currLoginTime)
        PbRootData.UserData.loginTime.current.localString = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", PbRootData.UserData.loginTime.current.locality)
        PbIO.UserData.Save_loginTime()
    end

    -- 保存账号绑定时间
    if returnData.binding_time and returnData.binding_time > 0 then
        local bindTime = returnData.binding_time
        PbRootData.UserData.aidBindTime.utc0 = bindTime
        PbRootData.UserData.aidBindTime.utc0String = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", bindTime)
        PbRootData.UserData.aidBindTime.locality = HonorTimerHelper:GetLocalTimeStamp(bindTime)
        PbRootData.UserData.aidBindTime.localString = HonorTimerHelper:GetTimeStringByTimeStamp("yyyy-MM-dd HH:mm:ss", PbRootData.UserData.aidBindTime.locality)
        PbIO.UserData.Save_aidBindTime()
    end

    -- 保存登录 Token
    if returnData.token then
        self._loginToken = returnData.token
    end

    -- 保存国家地区
    if returnData.country then
        UserTypeMgr:TrySaveUserCountry(returnData.country)
    end
end

---@brief 仅保存登录 Token
---@param returnData table 登录数据
function SaveDataHelper:SaveToken(returnData)
    if returnData and returnData.token then
        self._loginToken = returnData.token
    end
end

---@brief 保存云端第三方账号信息
---@param returnData table 服务器数据
function SaveDataHelper:SaveSignInAid(returnData)
    if returnData and returnData.ServerUserInfo then
        PbRootData.UserData.aid = returnData.ServerUserInfo.server_account
        PbIO.UserData.Save_aid()
    end
end

--=========================================================================
--region 云端存档 → 本地覆盖
--=========================================================================

---@brief 将服务器存档覆盖到本地
---@param serverGameData table 云端游戏数据
---@param uid string 用户ID
---@param aid string 第三方账号
---@param platform number 账号平台
function SaveDataHelper:SaveServerUserInfoToLocalByGameData(serverGameData, uid, aid, platform)
    -- 覆盖游戏存档
    if serverGameData then
        PbRootData.GameData = self:DynamicReplacementGameData(serverGameData)
        PbIO.GameData.SaveAll()
    end

    -- 覆盖 UID
    PbRootData.UserData.uid = uid
    PbIO.UserData.Save_uid()

    -- 覆盖绑定账号信息
    PbRootData.GameData.aidData.aid = aid
    PbRootData.GameData.aidData.platform = platform
    PbIO.GameData.Save_aidData()
end

---@brief 动态重载存档结构并替换数据
---@param otherGameData table 外部数据
---@return table 新的游戏数据
function SaveDataHelper:DynamicReplacementGameData(otherGameData)
    package.loaded["SaveGameData"] = nil
    require('SaveGameData')

    local result = PbRootData.GameData
    self:CopyGameDataTable(result, otherGameData)
    return result
end

---@brief 递归复制存档表（云端 → 本地）
---@param dest table 目标表
---@param source table 源表
function SaveDataHelper:CopyGameDataTable(dest, source)
    for key, value in pairs(source) do
        if type(value) ~= "table" then
            dest[key] = value
        else
            dest[key] = dest[key] or {}
            self:CopyGameDataTable(dest[key], value)
        end
    end
end

---@brief 登录成功后：保存信息 + 自动全量同步
---@param returnData table 登录数据
---@param completeCallback function 完成回调
function SaveDataHelper:SaveLocalDataAndAutoSyncData(returnData, completeCallback)
    self:SaveSignInInfos(returnData)
    if completeCallback then
        completeCallback()
    end
end
