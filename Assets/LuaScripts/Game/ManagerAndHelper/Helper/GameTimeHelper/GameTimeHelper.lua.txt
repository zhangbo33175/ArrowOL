--region 时间格式枚举定义
---@class ETimeCustomFormat @自定义时间格式枚举
---@field Time_0m00s number @0m 00s 格式
---@field Time_0_00 number @0:00 格式
ETimeCustomFormat = {
    Time_0m00s = 1, --- 0m 00s 格式
    Time_0_00  = 2  --- 0:00 格式
}

---@class EFormatTimeString @时间格式化字符串常量
---@field Time_0d00h string @ 0d 00h 格式化字符串
---@field Time_0h00m string @ 0h 00m 格式化字符串
---@field Time_0m00s string @0m 00s 格式化字符串
---@field Time_0_00 string @0:00 格式化字符串
EFormatTimeString = {
    Time_0d00h = "%sd %02dh", --- 天 小时
    Time_0h00m = "%sh %02dm", --- 小时 分钟
    Time_0m00s = "%sm %02ds", --- 分钟 秒
    Time_0_00  = "%s:%02d",    --- 分:秒
}
--endregion

--region 数据结构定义
---@class TimestampRecorder @定时任务记录结构
---@field timestamp number @ UTC时间戳
---@field callback function @ 到达时间回调
---@field key string @ 任务唯一标识（用于取消）
--endregion

--region 类定义
---@class GameTimeHelper : Singleton @游戏时间管理单例
---@field timestampRecords table @已排序定时任务列表（二分插入）
---@field waitingExecute table @待执行回调缓存
---@field eventCmdProcSecond string @每秒触发事件Key
---@field timestampRecordsCache table @任务缓存队列
---@field futureTimeStamp number @基准未来时间戳（防改时间）
---@field dayStr string @多语言：天
---@field hourStr string @多语言：小时
---@field minStr string @多语言：分钟
---@field secStr string @多语言：秒
local GameTimeHelper = class("GameTimeHelper", import("Singleton"))
--endregion

--region 构造 & 实例化
---@brief 构造函数
---@type fun(args:table):void
---@param args table @自定义参数
function GameTimeHelper:ctor(args)
    GameTimeHelper.super.ctor(self, args)

    self.timestampRecords      = {}     --- 定时任务有序列表（按时间戳升序）
    self.waitingExecute        = {}     --- 待执行任务缓存
    self.timestampRecordsCache = {}     --- 任务注册缓存队列
    self.eventCmdProcSecond    = "EventCmd_ProcSecond"
    self.futureTimeStamp       = 0      --- 防时间作弊基准时间戳

    self.dayStr   = nil
    self.hourStr  = nil
    self.minStr   = nil
    self.secStr   = nil

    --- 初始化多语言时间单位
    self:_InitLanguageText()
end

---@brief 创建单例实例
---@type fun(args:table):GameTimeHelper
---@param args table @自定义参数
---@return GameTimeHelper @时间工具实例
function GameTimeHelper:Create(args)
    local obj = GameTimeHelper.new(args)
    return obj
end
--endregion

--region 初始化 & 生命周期
---@brief 初始化（注册事件、启动每秒定时器）
---@type fun():void
function GameTimeHelper:Init()
    GameTimeHelper.super.Init(self)

    --- 语言切换时刷新时间单位文本
    HonorEventHelper:AddListener(EventCmd.OnLanguageChange, handler(self, self._OnLanguageChange))
    --- 注册每秒执行事件
    HonorEventHelper:AddListener(self.eventCmdProcSecond, handler(self, self.ProcSecond))
    --- 启动每秒定时器
    HonorTimerHelper:RegisterDeltaTime(self.eventCmdProcSecond, 1, "ProcedurePlaying", true)
end

---@brief 获取单例优先级
---@type fun():number
---@return number @优先级
function GameTimeHelper:Priority()
    return ESingletonPriority.GAME_TIME_HELPER
end

---@brief 清理数据（切换账号/场景时调用）
---@type fun():void
function GameTimeHelper:ClearData()
    GameTimeHelper.super.ClearData(self)

    --- 清空所有任务队列
    self.timestampRecords      = {}
    self.waitingExecute        = {}
    self.timestampRecordsCache = {}

    --- 重置防作弊基准时间
    self:SetupFutureTimeStamp()
end
--endregion

--region 多语言时间单位
---@brief 多语言切换刷新事件
---@type fun():void
function GameTimeHelper:_OnLanguageChange()
    self:_InitLanguageText()
end

---@brief 初始化多语言时间单位文本
---@type fun():void
function GameTimeHelper:_InitLanguageText()
    self.dayStr   = GetLocalization('TimeShowDay')
    self.hourStr  = GetLocalization('TimeShowHour')
    self.minStr   = GetLocalization('TimeShowMin')
    self.secStr   = GetLocalization('TimeShowSec')
end
--endregion

--region 定时任务：注册 / 取消 / 插入
---@brief 二分法插入有序任务列表（保证按时间戳升序）
---@type fun(record :TimestampRecorder) :void
---@param record TimestampRecorder @ 定时任务
function GameTimeHelper:_Insert(record)
    local count = #self.timestampRecords
    local min = 1
    local max = count

    --- 二分查找合适的插入位置
    while count > 0 and max - min > 0 do
        local mid = min + math.floor((max - min) / 2)
        if self.timestampRecords[mid].timestamp > record.timestamp then
            max = math.clamp(mid - 1, min, max)
        elseif self.timestampRecords[mid].timestamp < record.timestamp then
            min = math.clamp(mid + 1, min, max)
        else
            min = mid
            break
        end
    end

    --- 插入任务到有序列表
    if self.timestampRecords[min] == nil or self.timestampRecords[min].timestamp >= record.timestamp then
        table.insert(self.timestampRecords, min, record)
    else
        table.insert(self.timestampRecords, min + 1, record)
    end
end

---@brief 注册UTC时间戳定时任务
---@type fun(key : string ,timestamp:number,func :function):void
---@param key string @ 取消任务用唯一Key
---@param timestamp number @UTC0时间戳
---@param func function @到达时间回调
function GameTimeHelper:RegisterTime(key, timestamp, func)
    table.insert(self.timestampRecordsCache, { key = key, timestamp = timestamp, callback = func })
end

---@brief 根据Key取消指定定时任务
---@type fun(key : string):void
---@param key string @ 任务唯一Key
function GameTimeHelper:UnRegisterTime(key)
    --- 从缓存队列移除
    for i = 1, #self.timestampRecordsCache do
        local task = self.timestampRecordsCache[i]
        if task.key and task.key == key then
            table.remove(self.timestampRecordsCache, i)
            break
        end
    end

    --- 从有序执行队列移除
    for i = 1, #self.timestampRecords do
        local task = self.timestampRecords[i]
        if task.key and task.key == key then
            table.remove(self.timestampRecords, i)
            break
        end
    end
end
--endregion

--region 每秒任务执行
---@brief 每秒执行：检查并执行所有到期任务
---@type fun():void
function GameTimeHelper:ProcSecond()
    --- 将缓存队列任务插入有序列表
    local count = #self.timestampRecordsCache
    for i = 1, count do
        self:_Insert(self.timestampRecordsCache[i])
        self.timestampRecordsCache[i] = nil
    end

    local total = #self.timestampRecords
    if total == 0 then return end

    --- 查找所有已到期任务
    local idx = 0
    for i = 1, total do
        if self.timestampRecords[i].timestamp > self:GetUTC0FutureTimeStamp() then
            break
        end
        idx = i
    end

    if idx == 0 then return end

    --- 收集待执行回调
    ClearArray(self.waitingExecute)
    for i = 1, idx do
        table.insert(self.waitingExecute, self.timestampRecords[i])
    end

    --- 移除已执行任务
    for i = 1, total do
        if i <= total - idx then
            self.timestampRecords[i] = self.timestampRecords[i + idx]
        else
            self.timestampRecords[i] = nil
        end
    end

    --- 执行到期回调
    for i = 1, #self.waitingExecute do
        self.waitingExecute[i].callback()
    end

    ClearArray(self.waitingExecute)
end
--endregion

--region 时间格式化
---@brief 固定格式时间格式化
---@type fun(time:number,format:ETimeCustomFormat):string
---@param time number @总秒数
---@param format ETimeCustomFormat @目标格式
---@return string @格式化后的时间字符串
function GameTimeHelper:FormatTime(time, format)
    if format == ETimeCustomFormat.Time_0m00s then
        local minute = math.floor(time / 60)
        local second = math.floor(time % 60)
        return string.format(EFormatTimeString.Time_0m00s, minute, second)
    elseif format == ETimeCustomFormat.Time_0_00 then
        local minute = math.floor(time / 60)
        local second = math.floor(time % 60)
        return string.format(EFormatTimeString.Time_0_00, minute, second)
    end
    return ""
end

---@brief 通用智能时间格式化（自动适配天/时/分/秒，支持多语言）
---@type fun(time:number,inTick:boolean):string
---@param time number @总秒数
---@param inTick boolean @ 是否为倒计时状态（倒计时强制显示秒数）
---@return string @最终显示文本
function GameTimeHelper:FormatTimeInCommonRule(time, inTick)
    if time >= 86400 then
        local day  = math.floor(time / 86400)
        local hour = math.floor((time - day * 86400) / 3600)
        if hour == 0 then
            return string.format("%s%s", day, self.dayStr)
        else
            return string.format("%s%s %s%s", day, self.dayStr, hour, self.hourStr)
        end
    elseif time >= 3600 then
        local hour   = math.floor(time / 3600)
        local minute = math.floor((time - hour * 3600) / 60)
        if minute == 0 and not inTick then
            return string.format("%s%s", hour, self.hourStr)
        else
            return string.format("%s%s %s%s", hour, self.hourStr, minute, self.minStr)
        end
    elseif time >= 60 then
        local minute = math.floor(time / 60)
        local second = math.floor(time % 60)
        if second == 0 and not inTick then
            return string.format("%s%s", minute, self.minStr)
        else
            return string.format("%s%s %02d%s", minute, self.minStr, second, self.secStr)
        end
    else
        local second = math.floor(time % 60)
        return string.format("%02d%s", second, self.secStr)
    end
end
--endregion

--region 防篡改时间戳（防止玩家修改本地时间作弊）
---@brief 登录时初始化基准未来时间戳
---@type fun():void
function GameTimeHelper:SetupFutureTimeStamp()
    if not SaveRootData.GameData.timestamps[ETimestampID.FUTURE_TIME_STAMP] then
        SaveRootData.GameData.timestamps[ETimestampID.FUTURE_TIME_STAMP] = SaveRootData.UserData.launchTime.current.utc0
    end

    local loginTime  = SaveRootData.UserData.launchTime.current.utc0
    local futureTime = SaveRootData.GameData.timestamps[ETimestampID.FUTURE_TIME_STAMP]

    if loginTime == nil then
        return
    end

    --- 使用真实启动时间矫正基准，防止回退时间
    if futureTime < loginTime then
        SaveRootData.GameData.timestamps[ETimestampID.FUTURE_TIME_STAMP] = loginTime - Unity.Time.realtimeSinceStartup
        SaveIO.GameData.Save_timestamps()
    end

    self.futureTimeStamp = SaveRootData.GameData.timestamps[ETimestampID.FUTURE_TIME_STAMP]
end

---@brief 切后台/退出游戏时刷新并保存基准时间
---@type fun():void
function GameTimeHelper:RefreshFutureTimeStamp()
    if not self.futureTimeStamp then
        return
    end

    SaveRootData.GameData.timestamps[ETimestampID.FUTURE_TIME_STAMP] = self:GetUTC0FutureTimeStamp()
    SaveIO.GameData.Save_timestamps()
end

---@brief 获取当前可靠UTC0时间戳（防篡改）
---@type fun():number
---@return number @ 安全可靠的服务器基准时间戳
function GameTimeHelper:GetUTC0FutureTimeStamp()
    if not self.futureTimeStamp then
        PrintError("GameTimeHelper => SetupFutureTimeStamp must be call...")
        self.futureTimeStamp = SaveRootData.UserData.launchTime.current.utc0
    end

    return self.futureTimeStamp + Unity.Time.realtimeSinceStartup
end
--endregion

return GameTimeHelper
