--=====================================================================================================
-- 【UI文本帮助类】
-- 功能：通用数字滚动动画（支持 Text / TextMeshPro / TextMeshProUGUI）
-- 用途：金币增加、分数跳动、数值变化、结算动画
--=====================================================================================================

---@class UITextHelper @UI文本通用工具类
local UITextHelper = class("UITextHelper")

--- 构造函数
function UITextHelper:ctor(args)
end

--- 创建实例
---@return UITextHelper
function UITextHelper:Create(args)
    return UITextHelper.new(args)
end

--- 播放数字动画（推荐：绑定UI，关闭UI自动停止协程，安全）
---@param luaBehaviourSuper LuaBehaviourSuper 脚本宿主
---@param text UnityEngine.UI.MaskableGraphic 文本组件
---@param originCount number 起始数字
---@param targetCount number 目标数字
---@param duration number 动画时长
---@param prefix string 前缀
---@param suffix string 后缀
---@param onChangeAction function 变化回调
---@param onEndAction function 结束回调
---@param animDelay number 延迟播放
---@param isRealtime boolean 是否真实时间（不受暂停影响）
---@return UnityEngine.Coroutine
function UITextHelper:PlayNumberAnim(luaBehaviourSuper, text, originCount, targetCount, duration, prefix, suffix, onChangeAction, onEndAction, animDelay, isRealtime)
    return HonorCoroutineHelper:Start(handler(luaBehaviourSuper, function()
        self:CoPlayNumberAnim(text, originCount, targetCount, duration, prefix, suffix, onChangeAction, onEndAction, animDelay, isRealtime)
    end))
end

--- 播放数字动画（独立协程：不绑定UI，需手动管理，容易泄漏）
---@param text UnityEngine.UI.MaskableGraphic
---@param originCount number
---@param targetCount number
---@param duration number
---@param prefix string
---@param suffix string
---@param onChangeAction function
---@param onEndAction function
---@param animDelay number
---@return UnityEngine.Coroutine
function UITextHelper:PlayNumberAnimAlone(text, originCount, targetCount, duration, prefix, suffix, onChangeAction, onEndAction, animDelay)
    return HonorCoroutineHelper:StartAlone(function()
        self:CoPlayNumberAnim(text, originCount, targetCount, duration, prefix, suffix, onChangeAction, onEndAction, animDelay)
    end)
end

--- 协程数字动画主逻辑（支持所有文本组件）
---@param text UnityEngine.UI.MaskableGraphic
---@param originCount number
---@param targetCount number
---@param duration number
---@param prefix string
---@param suffix string
---@param onChangeAction function
---@param onEndAction function
---@param animDelay number
---@param isRealtime boolean
function UITextHelper:CoPlayNumberAnim(text, originCount, targetCount, duration, prefix, suffix, onChangeAction, onEndAction, animDelay, isRealtime)
    if not text then return end

    -- 默认参数
    isRealtime = isRealtime or false
    originCount = originCount or 0
    targetCount = targetCount or 0
    duration = duration or 0.1
    prefix = prefix or ""
    suffix = suffix or ""

    -- 延迟播放
    if animDelay and animDelay > 0 then
        if isRealtime then
            HonorCoroutineHelper:WaitForSecondsRealtime(animDelay)
        else
            HonorCoroutineHelper:WaitForSeconds(animDelay)
        end
    end

    -- 获取三种文本组件
    local tmpUGUI = text.gameObject:GetComponent("TextMeshProUGUI")
    local lab = text.gameObject:GetComponent("Text")
    local tmp = text.gameObject:GetComponent("TextMeshPro")

    -- 无需动画直接赋值
    if originCount < 0 or originCount == targetCount then
        local str = prefix .. targetCount .. suffix
        if tmpUGUI then tmpUGUI.text = str
        elseif lab then lab.text = str
        elseif tmp then tmp.text = str end

        if onEndAction then xpcall(onEndAction, __G__TRACK_BACK__, targetCount) end
        return
    end

    -- 计算步进
    local deltaTime = isRealtime and UnityTime.unscaledDeltaTime or UnityTime.deltaTime
    local waitFrameCount = duration / deltaTime
    local baseStep = math.floor(math.abs(targetCount - originCount) / waitFrameCount) + 1
    local step = originCount > targetCount and baseStep or -baseStep

    local delay = -1
    if waitFrameCount > math.abs(targetCount - originCount) then
        delay = duration / math.abs(targetCount - originCount)
        step = originCount > targetCount and 1 or -1
    end

    -- 滚动动画
    while originCount ~= targetCount do
        originCount = originCount - step
        if step > 0 and originCount <= targetCount then
            originCount = targetCount
        elseif step <= 0 and originCount >= targetCount then
            originCount = targetCount
        end

        -- 赋值显示
        local str = prefix .. originCount .. suffix
        if tmpUGUI then tmpUGUI.text = str
        elseif lab then lab.text = str
        elseif tmp then tmp.text = str end

        -- 变化回调
        if onChangeAction then
            xpcall(onChangeAction, __G__TRACK_BACK__, originCount)
        end

        -- 等待
        if delay > 0 then
            if isRealtime then
                HonorCoroutineHelper:WaitForSecondsRealtime(delay)
            else
                HonorCoroutineHelper:WaitForSeconds(delay)
            end
        else
            coroutine.yield()
        end
    end

    -- 最终赋值
    local finalStr = prefix .. targetCount .. suffix
    if tmpUGUI then tmpUGUI.text = finalStr
    elseif lab then lab.text = finalStr
    elseif tmp then tmp.text = finalStr end

    -- 结束回调
    if onEndAction then
        xpcall(onEndAction, __G__TRACK_BACK__, targetCount)
    end
end

return UITextHelper
