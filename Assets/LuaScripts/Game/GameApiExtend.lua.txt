--region Table 扩展工具

--=====================================================================================================
-- Table 扩展工具
--=====================================================================================================

---@brief 判断键是否存在于表中
---@param tbl table 目标表
---@param key any 键
---@return boolean 是否存在
function table.kIn(tbl, key)
    if type(tbl) ~= "table" then
        return false
    end
    return tbl[key] ~= nil
end

---@brief 判断值是否存在于表中
---@param tbl table 目标表
---@param value any 值
---@return boolean 是否存在
function table.vIn(tbl, value)
    if type(tbl) ~= "table" then
        return false
    end
    for _, v in pairs(tbl) do
        if v == value then
            return true
        end
    end
    return false
end

--endregion

--region Math 扩展工具

--=====================================================================================================
-- Math 扩展
--=====================================================================================================

---@brief 数值限制在 [min, max]
---@param value number 输入值
---@param min number 最小值
---@param max number 最大值
---@return number clamped 结果
function math.clamp(value, min, max)
    if value < min then return min end
    if value > max then return max end
    return value
end

---@brief 浮点数近似相等判断
---@param a number 数值A
---@param b number 数值B
---@return boolean 是否近似相等
function math.Approximately(a, b)
    return math.abs(a - b) < 1e-6
end

---@brief 限制值在 0~1 之间
---@param value number 输入值
---@return number 0~1 之间的值
function math.Clamp01(value)
    return math.clamp(value, 0, 1)
end

---@brief 平滑插值 SmoothStep（Unity 标准平滑曲线）
---@param from number 起始值
---@param to number 目标值
---@param t number 0~1 插值系数
---@return number 平滑插值结果
function math.SmoothStep(from, to, t)
    t = math.Clamp01(t)
    t = t * t * (3 - 2 * t)
    return from + (to - from) * t
end

--endregion

--region String 扩展工具

--=====================================================================================================
-- String 扩展
--=====================================================================================================

---@brief 判断字符串是否为空或 nil
---@param str string 目标字符串
---@return boolean 是否为空
function string.isNullOrEmpty(str)
    return str == nil or str == ""
end

--endregion
