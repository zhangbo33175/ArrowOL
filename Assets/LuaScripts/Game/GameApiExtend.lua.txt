--- 判断键是否存在于表中
---@param tbl table 目标表
---@param key any 键
---@return boolean 是否存在
function table.kIn(tbl, key)
    if type(tbl) ~= "table" then
        return false
    end
    return tbl[key] ~= nil
end

--- 判断值是否存在于表中
---@param tbl table 目标表
---@param value any 值
---@return boolean 是否存在
function table.vIn(tbl, value)
    if type(tbl) ~= "table" then
        return false
    end
    for _, v in pairs(tbl) do
        if v == value then
            return true
        end
    end
    return false
end

--- 判断表是否包含子表所有值
---@param tbl table 主表
---@param valueTable table 子表
---@return boolean 是否全部包含
function table.tableIn(tbl, valueTable)
    if type(tbl) ~= "table" or type(valueTable) ~= "table" then
        return false
    end
    for _, v in pairs(valueTable) do
        if not table.vIn(tbl, v) then
            return false
        end
    end
    return true
end

--- 判断是否为纯数组（连续数字索引）
---@param t table 目标表
---@return boolean 是否为纯数组
function table.isArrayTable(t)
    if type(t) ~= "table" then
        return false
    end
    local length = #t
    for k, _ in pairs(t) do
        if type(k) ~= "number" or k < 1 or k > length then
            return false
        end
    end
    return true
end

--- 递归打印表（调试专用）
---@param t table 目标表
---@return string 格式化字符串
function table.PrintDicTable(t)
    if type(t) ~= "table" then
        return tostring(t)
    end
    local result = {}
    for k, v in pairs(t) do
        local keyStr = tostring(k)
        local valStr = type(v) == "table" and table.PrintDicTable(v) or tostring(v)
        table.insert(result, keyStr .. " = " .. valStr)
    end
    return table.concat(result, "\n")
end

--- 无序比较两个数组是否相等
---@param t1 table 数组1
---@param t2 table 数组2
---@return boolean 是否相等
function table.compareTablesWithDisorder(t1, t2)
    if type(t1) ~= "table" or type(t2) ~= "table" then
        return false
    end
    if #t1 ~= #t2 then
        return false
    end
    for _, v in pairs(t1) do
        if not table.vIn(t2, v) then
            return false
        end
    end
    return true
end

--- 数组安全过滤删除（遍历过程中安全删除元素）
---@param t table 数组
---@param predicate function 判定函数(key, value):boolean 满足条件保留，不满足删除
function table.filterRemove(t, predicate)
    if type(t) ~= "table" or type(predicate) ~= "function" then
        return
    end
    local index = 1
    while index <= #t do
        if not predicate(index, t[index]) then
            table.remove(t, index)
        else
            index = index + 1
        end
    end
end

--- 判断 t1 是否包含 t2 所有元素
---@param t1 table 主表
---@param t2 table 子表
---@return boolean 是否全部包含
function table.isContains(t1, t2)
    if type(t1) ~= "table" or type(t2) ~= "table" then
        return false
    end
    for _, v in pairs(t2) do
        if not table.vIn(t1, v) then
            return false
        end
    end
    return true
end

--- Fisher-Yates 打乱数组（原地修改）
---@param t table 数组
function table.shuffle(t)
    if type(t) ~= "table" or #t <= 1 then
        return
    end
    local i = #t
    while i > 1 do
        local j = math.random(i)
        t[i], t[j] = t[j], t[i]
        i = i - 1
    end
end

--- 求两个数组的交集
---@param arr1 table 数组1
---@param arr2 table 数组2
---@return table 交集结果数组
function table.Intersection(arr1, arr2)
    if type(arr1) ~= "table" or type(arr2) ~= "table" then
        return {}
    end
    local lookup = {}
    local result = {}
    for _, v in ipairs(arr2) do
        lookup[v] = true
    end
    for _, v in ipairs(arr1) do
        if lookup[v] then
            table.insert(result, v)
        end
    end
    return result
end

--- 数值限制在 [min, max]
---@param value number 输入值
---@param min number 最小值
---@param max number 最大值
---@return number clamped 结果
function math.clamp(value, min, max)
    if value < min then return min end
    if value > max then return max end
    return value
end

--- 四舍五入
---@param value number 输入值
---@return number 四舍五入结果
function math.roundToNearest(value)
    return value >= 0 and math.floor(value + 0.5) or math.ceil(value - 0.5)
end

--- 保留 N 位小数（截断）
---@param value number 原数值
---@param decimals number 保留位数，默认0
---@return number 截断后数值
function math.GetPreciseDecimal(value, decimals)
    if type(value) ~= "number" then
        return value
    end
    decimals = math.max(math.floor(decimals or 0), 0)
    local multiplier = 10 ^ decimals
    return math.floor(value * multiplier) / multiplier
end

--- 浮点数近似相等判断
---@param a number 数值A
---@param b number 数值B
---@return boolean 是否近似相等
function math.Approximately(a, b)
    return math.abs(a - b) < 1e-6
end

--- 限制值在 0~1 之间
---@param value number 输入值
---@return number 0~1 之间的值
function math.Clamp01(value)
    return math.clamp(value, 0, 1)
end

--- 平滑插值 SmoothStep（Unity 标准平滑曲线）
---@param from number 起始值
---@param to number 目标值
---@param t number 0~1 插值系数
---@return number 平滑插值结果
function math.SmoothStep(from, to, t)
    t = math.Clamp01(t)
    t = t * t * (3 - 2 * t)
    return from + (to - from) * t
end

--- 查找子串第一次出现的索引，找不到返回-1
---@param str string 原字符串
---@param substr string 子串
---@param startIndex number 起始位置，默认1
---@return number 索引位置
function string.indexOf(str, substr, startIndex)
    if type(str) ~= "string" or type(substr) ~= "string" then
        return -1
    end
    startIndex = startIndex or 1
    local pos = string.find(str, substr, startIndex, true)
    return pos or -1
end

--- 判断字符串是否为空或 nil
---@param str string 目标字符串
---@return boolean 是否为空
function string.isNullOrEmpty(str)
    return str == nil or str == ""
end
