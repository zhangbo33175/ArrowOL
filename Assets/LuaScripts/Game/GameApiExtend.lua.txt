---查看某值是否为表tbl中的key值
---@param tbl table 目标表
---@param key any 要查找的键
---@return boolean 是否存在
function table.kIn(tbl, key)
    if tbl == nil then
        return false
    end
    for k, _ in pairs(tbl) do
        if k == key then
            return true
        end
    end
    return false
end

---查看某值是否为表tbl中的value值
---@param tbl table 目标表
---@param value any 要查找的值
---@return boolean 是否存在
function table.vIn(tbl, value)
    if tbl == nil then
        return false
    end

    for _, v in pairs(tbl) do
        if v == value then
            return true
        end
    end
    return false
end

---查看表tbl中是否包含 valueTable 里的所有值
---@param tbl table 主表
---@param valueTable table 子值表
---@return boolean 是否全部包含
function table.tableIn(tbl, valueTable)
    if tbl == nil then
        return false
    end

    for _, v in pairs(valueTable) do
        if not table.vIn(tbl, v) then
            return false
        end
    end
    return true
end

---检查table是否为纯数组（只有连续数字索引）
---@param t table
---@return boolean
function table.isArrayTable(t)
    if type(t) ~= "table" then
        return false
    end
    local n = #t
    for i, _ in pairs(t) do
        if type(i) ~= "number" then
            return false
        end
        if i > n then
            return false
        end
    end
    return true
end

---递归打印字典表（方便调试）
---@param t table
---@return string
function table.PrintDicTable(t)
    local str = ""
    for k, v in pairs(t) do
        if type(v) == "table" then
            str = str .. k .. " = " .. table.PrintDicTable(v) .. "\n"
        else
            str = str .. k .. " = " .. tostring(v) .. "\n"
        end
    end
    return str
end

---无序比较两个表的值是否相等
---@param t1 table
---@param t2 table
---@return boolean
function table.compareTablesWithDisorder(t1, t2)
    if #t1 ~= #t2 then
        return false
    end
    for k, v in pairs(t1) do
        if not table.vIn(t2, v) then
            return false
        end
    end
    return true
end

---数组过滤删除（安全删除，不会跳索引）
---@param t table 数组
---@param fn function 回调 function(key,value):boolean
function table.filterRemove(t, fn)
    local key = 1
    while key <= #t do
        if not fn(key, t[key]) then
            table.remove(t, key)
        else
            key = key + 1
        end
    end
end

---判断 t1 是否包含 t2 所有元素
---@param t1 table
---@param t2 table
---@return boolean
function table.isContains(t1, t2)
    for _, v in pairs(t2) do
        if not table.vIn(t1, v) then
            return false
        end
    end
    return true
end

---随机打乱数组（直接修改原表）
---@param t table
function table.shuffle(t)
    if type(t) ~= "table" then
        return
    end
    local num = #t
    while num > 1 do
        local next = math.random(1, num - 1)
        t[next], t[num] = t[num], t[next]
        num = num - 1
    end
end

---求两个数组的交集
---@param table1 table 数组1
---@param table2 table 数组2
---@return table 交集数组
function table.Intersection(table1, table2)
    local intersection = {}
    local set = {}

    for _, value in ipairs(table2) do
        set[value] = true
    end

    for _, value in ipairs(table1) do
        if set[value] then
            table.insert(intersection, value)
        end
    end

    return intersection
end

---数值限制在 [min, max] 之间
---@param v number
---@param minValue number
---@param maxValue number
---@return number
function math.clamp(v, minValue, maxValue)
    if v < minValue then
        return minValue
    end
    if v > maxValue then
        return maxValue
    end
    return v
end

---四舍五入
---@param num number
---@return number
function math.roundToNearest(num)
    if num >= 0 then
        return math.floor(num + 0.5)
    else
        return math.ceil(num - 0.5)
    end
end

---保留 N 位小数（截断，不四舍五入）
---@param nNum number 原数
---@param n number 保留位数
---@return number
function math.GetPreciseDecimal(nNum, n)
    if type(nNum) ~= "number" then
        return nNum
    end
    n = n or 0
    n = math.floor(n)
    if n < 0 then n = 0 end

    local decimal = 10 ^ n
    return math.floor(nNum * decimal) / decimal
end

---浮点数相等判断（防精度误差）
---@param a number
---@param b number
---@return boolean
function math.Approximately(a, b)
    return math.abs(a - b) < 0.000001
end

---限制值在 0~1
---@param value number
---@return number
function math.Clamp01(value)
    if value < 0 then
        return 0
    else
        return value > 1 and 1 or value
    end
end

---平滑插值（SmoothStep）
---@param from number 起始值
---@param to number 目标值
---@param t number 0~1
---@return number
function math.SmoothStep(from, to, t)
    t = math.Clamp01(t)
    t = -2 * t * t * t + 3 * t * t
    return to * t + from * (1 - t)
end

------------------------------
-- String 扩展
------------------------------

---查找字符第一次出现的索引（找不到返回-1）
---@param s string 原字符串
---@param pattern string 查找内容
---@param init number 起始位置
---@return number 索引
function string.indexOf(s, pattern, init)
    init = init or 1
    local index = string.find(s, pattern, init, true)
    return index or -1
end

---判断字符串是否为空或nil
---@param s string
---@return boolean
function string.isNullOrEmpty(s)
    return s == nil or s == ""
end
