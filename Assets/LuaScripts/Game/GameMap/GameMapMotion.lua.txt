---@class MapMotionRange
---@field min number X 轴最小边界
---@field max number X 轴最大边界

---@class GameMapMotion 地图相机控制主类
---@field FollowPosition UnityEngine.Vector3 相机目标跟随位置
---@field MapMotionLayer GameLib.MapMotionLayer 地图运动控制层
---@field ScreenCamera UnityEngine.Camera 主相机引用
---@field SwipeBegin function 滑动开始回调
---@field Swipe function 滑动中回调
---@field SwipeEnd function 滑动结束回调
---@field Vector3Num UnityEngine.Vector3 临时偏移向量
---@field InertialSlidingVector UnityEngine.Vector3 惯性滑动向量
---@field PositionWhenPaused UnityEngine.Vector3 暂停时保存的位置
---@field StepAddFactor number 平滑插值增长系数
---@field IsMapWork boolean 地图交互开关
---@field MapHalfWidth number 地图半宽
---@field MapHalfHeigh number 地图半高
---@field MapCenterX number 地图中心点 X
---@field MapCenterY number 地图中心点 Y
---@field MapBounds UnityEngine.Bounds 地图总包围盒
---@field SwipeSmoothStepNum number 滑动平滑系数
---@field InertialSlidingFactor number 惯性衰减系数
---@field ScreenCameraAspect number 相机宽高比
---@field ScreenCameraOrthographicSize number 相机正交大小
---@field VectorContainer1 table 屏幕转世界临时向量
---@field VectorContainer2 table 屏幕转世界临时向量
---@field PointStartFromUI boolean 是否从UI区域开始触摸
---@field IsMouseDown boolean 鼠标左键是否按下
---@field MouseLastScreenPos UnityEngine.Vector2 上一帧鼠标屏幕坐标
local GameMapMotion = {
    StepAddFactor = 2,
    SwipeSmoothStepNum = 0,
    InertialSlidingFactor = 0,
    VectorContainer1 = { x = 0, y = 0, z = 0 },
    VectorContainer2 = { x = 0, y = 0, z = 0 },
    PointStartFromUI = false,
    lastMousePos = nil,
    IsMouseDown = false,
    MouseLastScreenPos = { x = 0, y = 0 },

    -- 相机缩放限制参数
    MinOrthoSize = 200,
    MaxOrthoSize = 1024,
    ZoomSpeed = 1.1,
    LastPinchDistance = 0,
    IsPinching = false,
}

--region 【核心更新逻辑】
---=========================================================================
---@brief 帧更新：驱动相机平滑跟随、视差滚动、惯性、缩放、鼠标拖动
---@type fun():void
---=========================================================================
function GameMapMotion:OnUpdate()
    if GameManager:IsGamePaused() or not self.IsMapWork then
        return
    end

    -- 限制相机缩放范围
    self.ScreenCamera.orthographicSize = math.clamp(
            self.ScreenCamera.orthographicSize,
            self.MinOrthoSize,
            self.MaxOrthoSize
    )

    -- 相机 X 轴平滑跟随
    local currCameraX = self.ScreenCamera.transform:GetPositionX()
    if not math.Approximately(currCameraX, self.FollowPosition.x) then
        local smoothX = math.SmoothStep(currCameraX, self.FollowPosition.x, self.SwipeSmoothStepNum)
        self.ScreenCamera.transform:SetPositionX(smoothX)
    end

    -- 相机 Y 轴平滑跟随
    local currCameraY = self.ScreenCamera.transform:GetPositionY()
    if not math.Approximately(currCameraY, self.FollowPosition.y) then
        local smoothY = math.SmoothStep(currCameraY, self.FollowPosition.y, self.SwipeSmoothStepNum)
        self.ScreenCamera.transform:SetPositionY(smoothY)
    end

    -- 惯性滑动
    if self.InertialSlidingFactor > 0 then
        self:OnInertialSliding(self.InertialSlidingFactor)
    end

    -- 双指缩放
    self:CheckPinchZoom()

    -- 鼠标滚轮缩放
    self:UpdateMouseScrollZoom()

    -- 鼠标左键拖动
    self:UpdateMouseDrag()
end
--endregion

--region 【PC鼠标拖动】
---=========================================================================
---@brief PC 鼠标左键拖动控制逻辑
---@type fun():void
---=========================================================================
function GameMapMotion:UpdateMouseDrag()
    -- 鼠标按下
    if CS.UnityEngine.Input.GetMouseButtonDown(0) then
        self.PointStartFromUI = self:IsPointerOverGameObject()
        if self.PointStartFromUI then
            self.IsMouseDown = false
            return
        end

        self.IsMouseDown = true
        self.MouseLastScreenPos.x = CS.UnityEngine.Input.mousePosition.x
        self.MouseLastScreenPos.y = CS.UnityEngine.Input.mousePosition.y

        self.InertialSlidingFactor = 0
        self.SwipeSmoothStepNum = 0
        self.Vector3Num.x = 0
        self.Vector3Num.y = 0
    end

    -- 鼠标拖动中
    if CS.UnityEngine.Input.GetMouseButton(0) and self.IsMouseDown and not self.PointStartFromUI then
        local currMouseX = CS.UnityEngine.Input.mousePosition.x
        local currMouseY = CS.UnityEngine.Input.mousePosition.y

        self.Vector3Num.x = currMouseX
        self.Vector3Num.y = currMouseY
        GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer1)

        self.Vector3Num.x = self.MouseLastScreenPos.x
        self.Vector3Num.y = self.MouseLastScreenPos.y
        GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer2)

        local deltaWorldX = self.VectorContainer1.x - self.VectorContainer2.x
        local deltaWorldY = self.VectorContainer1.y - self.VectorContainer2.y

        -- 方向修正
        deltaWorldX = deltaWorldX * -1
        deltaWorldY = deltaWorldY * -1

        self.Vector3Num.x = deltaWorldX
        self.Vector3Num.y = deltaWorldY
        self:MovingFollowPosition(self.Vector3Num)

        self.SwipeSmoothStepNum = math.clamp(
                self.SwipeSmoothStepNum + CS.UnityEngine.Time.deltaTime * self.StepAddFactor,
                0.3, 1
        )

        self.MouseLastScreenPos.x = currMouseX
        self.MouseLastScreenPos.y = currMouseY
    end

    -- 鼠标抬起
    if CS.UnityEngine.Input.GetMouseButtonUp(0) and self.IsMouseDown then
        self.IsMouseDown = false
        self.PointStartFromUI = false
        self.InertialSlidingFactor = 1 - self.SwipeSmoothStepNum
        self.SwipeSmoothStepNum = 0.3
    end
end
--endregion

--region 【触摸检测】
---=========================================================================
---@brief 判断当前触摸是否在UI之上
---@type fun():boolean
---@return boolean 在UI上返回true
---=========================================================================
function GameMapMotion:IsPointerOverGameObject()
    return GameLib.RSetupLuaTableValue.IsPointerOverUIObject()
end
--endregion

--region 【移动端滑动】
---=========================================================================
---@brief 触摸滑动开始回调
---@type fun(gesture:any):void
---@param gesture any 手势对象
---=========================================================================
function GameMapMotion:OnSwipeBegin(gesture)
    if GameManager:IsGamePaused() then
        return
    end
    self.PointStartFromUI = self:IsPointerOverGameObject()
    if self.PointStartFromUI then
        return
    end

    self.InertialSlidingFactor = 0
    self.SwipeSmoothStepNum = 0
    self.Vector3Num.x = 0
    self.Vector3Num.y = 0
end

---=========================================================================
---@brief 触摸滑动中回调
---@type fun(gesture:any):void
---@param gesture any 手势对象
---=========================================================================
function GameMapMotion:OnSwipe(gesture)
    if GameManager:IsGamePaused() or self.PointStartFromUI or not self.IsMapWork then
        return
    end

    if gesture.touchCount >= 2 then
        return
    end

    if gesture.touchCount == 1 then
        self.Vector3Num.x = gesture:GetPositionX()
        self.Vector3Num.y = gesture:GetPositionY()
        GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer1)

        self.Vector3Num.x = gesture:GetPositionX() - gesture:GetDeltaPositionX()
        self.Vector3Num.y = gesture:GetPositionY() - gesture:GetDeltaPositionY()
        GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer2)

        local deltaWorldX = self.VectorContainer1.x - self.VectorContainer2.x
        local deltaWorldY = self.VectorContainer1.y - self.VectorContainer2.y
        deltaWorldX = deltaWorldX * -1
        deltaWorldY = deltaWorldY * -1

        self.Vector3Num.x = deltaWorldX
        self.Vector3Num.y = deltaWorldY
        self:MovingFollowPosition(self.Vector3Num)

        self.SwipeSmoothStepNum = math.clamp(
                self.SwipeSmoothStepNum + CS.UnityEngine.Time.deltaTime * self.StepAddFactor,
                0.3, 1
        )
    end
end

---=========================================================================
---@brief 触摸滑动结束回调
---@type fun(gesture:any):void
---@param gesture any 手势对象
---=========================================================================
function GameMapMotion:OnSwipeEnd(gesture)
    if GameManager:IsGamePaused() then
        return
    end
    if self.PointStartFromUI then
        self.InertialSlidingFactor = 0
        self.SwipeSmoothStepNum = 0
        self.Vector3Num.x = 0
        self.Vector3Num.y = 0
    end

    self.PointStartFromUI = false
    self.IsPinching = false
    self.InertialSlidingFactor = 1 - self.SwipeSmoothStepNum
    self.SwipeSmoothStepNum = 0.3
end
--endregion

--region 【惯性滑动】
---=========================================================================
---@brief 惯性滑动逻辑
---@type fun(factor:number):void
---@param factor number 惯性系数
---=========================================================================
function GameMapMotion:OnInertialSliding(factor)
    self.InertialSlidingVector.x = self.Vector3Num.x * self.InertialSlidingFactor
    self:MovingFollowPosition(self.InertialSlidingVector)
    self.InertialSlidingFactor = math.clamp(self.InertialSlidingFactor - 0.025, 0, 1)

    self.SwipeSmoothStepNum = math.clamp(
            self.SwipeSmoothStepNum + GameLib.RSetupLuaTableValue.GetTimeDeltaTime(),
            0.75, 1
    )
end
--endregion

--region 【相机移动与边界限制】
---=========================================================================
---@brief 执行相机移动并限制边界，同步更新视差层
---@type fun(worldDelta:table):void
---@param worldDelta table 世界坐标系偏移量
---=========================================================================
function GameMapMotion:MovingFollowPosition(worldDelta)
    local aspect = self.ScreenCamera.aspect
    local orthoSize = self.ScreenCamera.orthographicSize
    local halfW = aspect * orthoSize
    local halfH = orthoSize

    -- 计算无黑边安全边界
    local minX = self.MapCenterX - self.MapHalfWidth + halfW
    local maxX = self.MapCenterX + self.MapHalfWidth - halfW
    local minY = self.MapCenterY - self.MapHalfHeigh + halfH
    local maxY = self.MapCenterY + self.MapHalfHeigh - halfH

    local oldX = self.FollowPosition.x
    self.FollowPosition.x = self.FollowPosition.x + worldDelta.x
    self.FollowPosition.y = self.FollowPosition.y + worldDelta.y

    -- 边界钳位
    self.FollowPosition.x = math.clamp(self.FollowPosition.x, minX, maxX)
    self.FollowPosition.y = math.clamp(self.FollowPosition.y, minY, maxY)

    -- 坐标精度优化
    self.FollowPosition.x = math.floor(self.FollowPosition.x * 1000) / 1000
    self.FollowPosition.y = math.floor(self.FollowPosition.y * 1000) / 1000
end
--endregion

--region 【缩放控制】
---=========================================================================
---@brief 移动端双指缩放逻辑
---@type fun():void
---=========================================================================
function GameMapMotion:CheckPinchZoom()
    if not self.ScreenCamera or not self.IsMapWork or self.PointStartFromUI then
        return
    end

    local touchCount = CS.UnityEngine.Input.touchCount
    if touchCount == 2 then
        local t1 = CS.UnityEngine.Input.GetTouch(0)
        local t2 = CS.UnityEngine.Input.GetTouch(1)

        if t1.phase == CS.UnityEngine.TouchPhase.Began or t2.phase == CS.UnityEngine.TouchPhase.Began then
            self.LastPinchDistance = self:GetTwoTouchDistance(t1, t2)
            self.IsPinching = true
            self.InertialSlidingFactor = 0
            return
        end

        if t1.phase == CS.UnityEngine.TouchPhase.Moved or t2.phase == CS.UnityEngine.TouchPhase.Moved then
            local currDistance = self:GetTwoTouchDistance(t1, t2)
            local diff = currDistance - self.LastPinchDistance

            if math.abs(diff) > 1 then
                local ortho = self.ScreenCamera.orthographicSize
                ortho = diff > 0 and (ortho / self.ZoomSpeed) or (ortho * self.ZoomSpeed)
                ortho = math.clamp(ortho, self.MinOrthoSize, self.MaxOrthoSize)

                self.ScreenCamera.orthographicSize = ortho
                self.LastPinchDistance = currDistance

                self:ClampCameraAfterZoom()
            end
        end
    else
        self.IsPinching = false
    end
end

---=========================================================================
---@brief 计算两个触摸点之间的距离
---@type fun(t1:any, t2:any):number
---@param t1 any 触摸点1
---@param t2 any 触摸点2
---@return number 两点间距离
---=========================================================================
function GameMapMotion:GetTwoTouchDistance(t1, t2)
    local dx = t1.position.x - t2.position.x
    local dy = t1.position.y - t2.position.y
    return math.sqrt(dx * dx + dy * dy)
end

---=========================================================================
---@brief PC 鼠标滚轮缩放
---@type fun():void
---=========================================================================
function GameMapMotion:UpdateMouseScrollZoom()
    if not self.ScreenCamera or not self.IsMapWork then
        return
    end

    local scroll = CS.UnityEngine.Input.GetAxis("Mouse ScrollWheel")
    if math.abs(scroll) < 0.001 then
        return
    end

    local ortho = self.ScreenCamera.orthographicSize
    ortho = scroll > 0 and (ortho / self.ZoomSpeed) or (ortho * self.ZoomSpeed)
    ortho = math.clamp(ortho, self.MinOrthoSize, self.MaxOrthoSize)

    self.ScreenCamera.orthographicSize = ortho
    self:ClampCameraAfterZoom()
end

---=========================================================================
---@brief 缩放后自动修正相机位置，防止出现黑边
---@type fun():void
---=========================================================================
function GameMapMotion:ClampCameraAfterZoom()
    local aspect = self.ScreenCamera.aspect
    local orthoSize = self.ScreenCamera.orthographicSize
    local halfW = aspect * orthoSize
    local halfH = orthoSize

    local minX = self.MapCenterX - self.MapHalfWidth + halfW
    local maxX = self.MapCenterX + self.MapHalfWidth - halfW
    local minY = self.MapCenterY - self.MapHalfHeigh + halfH
    local maxY = self.MapCenterY + self.MapHalfHeigh - halfH

    self.FollowPosition.x = math.clamp(self.FollowPosition.x, minX, maxX)
    self.FollowPosition.y = math.clamp(self.FollowPosition.y, minY, maxY)
end
--endregion

--region 【触摸事件注册】
---=========================================================================
---@brief 注册或注销触摸事件
---@type fun(action:string):void
---@param action string 注册/注销标记
---=========================================================================
function GameMapMotion:RegisterTouchEvents(action)
    if action == "+" then
        self.SwipeBegin = self.SwipeBegin or function(g)
            self:OnSwipeBegin(g)
        end
        self.Swipe = self.Swipe or function(g)
            self:OnSwipe(g)
        end
        self.SwipeEnd = self.SwipeEnd or function(g)
            self:OnSwipeEnd(g)
        end

        CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragStart("+", self.SwipeBegin)
        CS.HedgehogTeam.EasyTouch.EasyTouch.On_Drag("+", self.Swipe)
        CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragEnd("+", self.SwipeEnd)
    else
        if self.SwipeBegin then
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragStart("-", self.SwipeBegin)
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_Drag("-", self.Swipe)
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragEnd("-", self.SwipeEnd)
        end
    end
end
--endregion

--region 【地图控制开关】
---=========================================================================
---@brief 恢复地图交互功能
---@type fun():void
---=========================================================================
function GameMapMotion:ResumeMapMotion()
    if self.IsMapWork then
        return
    end

    self.SwipeSmoothStepNum = 0
    self.FollowPosition.x = self.ScreenCamera.transform:GetPositionX()
    self.FollowPosition.y = self.ScreenCamera.transform:GetPositionY()
    self.InertialSlidingFactor = 0
    self.Vector3Num.x = 0
    self.Vector3Num.y = 0
    self.IsMapWork = true
    self.IsMouseDown = false
    self.PointStartFromUI = false
end

---=========================================================================
---@brief 暂停地图交互功能
---@type fun():void
---=========================================================================
function GameMapMotion:PauseMapMotion()
    self.IsMapWork = false
    self.IsMouseDown = false
    self.PositionWhenPaused = {
        self.ScreenCamera.transform:GetPositionX(),
        self.ScreenCamera.transform:GetPositionY(),
        self.ScreenCamera.transform:GetPositionZ()
    }
end
--endregion

--region 【初始化与销毁】
---=========================================================================
---@brief 地图控制器初始化
---@type fun(mapTrans:UnityEngine.Transform):void
---@param mapTrans UnityEngine.Transform 地图根节点Transform
---=========================================================================
function GameMapMotion:OnMapStart(mapTrans)
    if not mapTrans then
        return
    end
    self.MapMotionLayer = mapTrans:GetComponent("GameLib.MapMotionLayer")
    if not self.MapMotionLayer then
        return
    end

    self.ScreenCamera = GameLib.MapCamHelper.MapCamera
    self.IsMapWork = true
    self.SwipeSmoothStepNum = 0
    self.IsMouseDown = false

    -- 初始化相机缩放
    self.ScreenCamera.orthographicSize = (self.MinOrthoSize + self.MaxOrthoSize) / 2
    self.InitialOrthoSize = self.ScreenCamera.orthographicSize

    -- 【关键修复：强制使用Bounds的extents和center】
    self.MapHalfWidth = self.MapMotionLayer.m_AreaBounds.extents.x
    self.MapHalfHeigh = self.MapMotionLayer.m_AreaBounds.extents.y
    self.MapCenterX = self.MapMotionLayer.m_AreaBounds.center.x
    self.MapCenterY = self.MapMotionLayer.m_AreaBounds.center.y
    self.MapBounds = self.MapMotionLayer.m_AreaBounds

    -- 初始化相机位置（放在地图中心）
    self.FollowPosition = {
        x = self.MapCenterX,
        y = self.MapCenterY,
        z = self.ScreenCamera.transform:GetPositionZ()
    }
    self.PositionWhenPaused = { x = self.FollowPosition.x, y = self.FollowPosition.y, z = self.FollowPosition.z }

    self.InertialSlidingVector = { x = 0, y = 0, z = 0 }
    self.Vector3Num = { x = 0, y = 0, z = 0 }

    self.ScreenCameraAspect = self.ScreenCamera.aspect
    self.ScreenCameraOrthographicSize = self.ScreenCamera.orthographicSize
    self:RegisterTouchEvents("+")
end

---=========================================================================
---@brief 限制并返回合法的X轴位置
---@type fun(posX:number, aspect:number, orthographicSize:number):number
---@param posX number 目标X
---@param aspect number 相机宽高比
---@param orthographicSize number 相机大小
---@return number 合法位置
---=========================================================================
function GameMapMotion:ResetPositionX(posX, aspect, orthographicSize)
    local halfWidth = aspect * orthographicSize
    local min = self.MapCenterX - self.MapHalfWidth + halfWidth
    local max = self.MapCenterX + self.MapHalfWidth - halfWidth
    return math.clamp(posX, min, max)
end

---=========================================================================
---@brief 限制并返回合法的Y轴位置
---@type fun(posY:number, orthographicSize:number):number
---@param posY number 目标Y
---@param orthographicSize number 相机大小
---@return number 合法位置
---=========================================================================
function GameMapMotion:ResetPositionY(posY, orthographicSize)
    local halfHeigh = orthographicSize
    local min = self.MapCenterY - self.MapHalfHeigh + halfHeigh
    local max = self.MapCenterY + self.MapHalfHeigh - halfHeigh
    return math.clamp(posY, min, max)
end

---=========================================================================
---@brief 获取当前地图边界
---@type fun():UnityEngine.Bounds
---@return UnityEngine.Bounds 地图Bounds
---=========================================================================
function GameMapMotion:GetCurrMapRect()
    return self.MapBounds
end

---=========================================================================
---@brief 销毁并清理控制器
---@type fun():void
---=========================================================================
function GameMapMotion:OnMapEnd()
    self:RegisterTouchEvents("-")
    self:PauseMapMotion()
end
--endregion

return GameMapMotion
