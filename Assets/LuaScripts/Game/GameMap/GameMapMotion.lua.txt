---@class MapMotionRange 地图边界范围结构
---@field min number X 轴最小边界
---@field max number X 轴最大边界

---@class GameMapMotion 地图相机控制主类 | 拖动+单击分离+缩放+惯性+边界限制
local GameMapMotion = {
    StepAddFactor = 2,                          -- 平滑步进系数
    SwipeSmoothStepNum = 0,                     -- 滑动平滑值
    InertialSlidingFactor = 0,                  -- 惯性滑动系数

    VectorContainer1 = { x = 0, y = 0, z = 0 },  -- 坐标临时容器1
    VectorContainer2 = { x = 0, y = 0, z = 0 },  -- 坐标临时容器2
    Vector3Num = { x = 0, y = 0, z = 0 },       -- 通用临时向量
    InertialSlidingVector = { x=0,y=0,z=0 },     -- 惯性方向向量

    PointStartFromUI = false,                   -- 是否从UI开始按压
    lastMousePos = nil,                         -- 上一帧鼠标位置
    IsMouseDown = false,                        -- 鼠标是否按下
    MouseLastScreenPos = { x = 0, y = 0 },      -- 鼠标最后屏幕坐标

    -- 相机缩放配置
    MinOrthoSize = 200,                         -- 最小正交尺寸
    MaxOrthoSize = 1024,                        -- 最大正交尺寸
    ZoomSpeed = 1.1,                            -- 缩放速度
    LastPinchDistance = 0,                      -- 上一次双指距离
    IsPinching = false,                         -- 是否双指缩放

    -- ===================== 【严格单击 / 拖动区分】 =====================
    ClickEnable = true,                         -- 单击开关
    ClickStartX = 0,                            -- 单击起始X
    ClickStartY = 0,                            -- 单击起始Y
    IsPressing = false,                         -- 是否正在按压
    DragThreshold = 12,                         -- 拖动阈值（像素）
    OnMapClick = nil,                           -- 地图单击回调

    -- 运行时数据
    IsMapWork = false,                          -- 地图是否运行
    ScreenCamera = nil,                         -- 地图相机
    MapMotionLayer = nil,                       -- 地图层组件
    FollowPosition = nil,                       -- 相机跟随目标
    MapBounds = nil,                            -- 地图包围盒
    MapCenterX = 0, MapCenterY = 0,             -- 地图中心点
    MapHalfWidth = 0, MapHalfHeigh = 0,         -- 地图半宽高
    ScreenCameraAspect = 1,                     -- 相机宽高比
    ScreenCameraOrthographicSize = 0,           -- 相机正交尺寸
    InitialOrthoSize = 0,                       -- 初始正交尺寸
    PositionWhenPaused = nil                    -- 暂停时位置
}

--region 【核心更新逻辑】每帧驱动
---@brief 地图控制主更新函数
function GameMapMotion:OnUpdate()
    if  not self.IsMapWork then
        return
    end

    -- 安全矫正相机尺寸
    if self.ScreenCamera.orthographicSize == 0 then
        self.ScreenCamera.orthographicSize = (self.MinOrthoSize + self.MaxOrthoSize) / 2
    else
        self.ScreenCamera.orthographicSize = math.clamp(
                self.ScreenCamera.orthographicSize,
                self.MinOrthoSize,
                self.MaxOrthoSize
        )
    end

    -- X轴平滑跟随
    local currCameraX = self.ScreenCamera.transform:GetPositionX()
    if not math.Approximately(currCameraX, self.FollowPosition.x) then
        local smoothX = math.SmoothStep(currCameraX, self.FollowPosition.x, self.SwipeSmoothStepNum)
        self.ScreenCamera.transform:SetPositionX(smoothX)
    end

    -- Y轴平滑跟随
    local currCameraY = self.ScreenCamera.transform:GetPositionY()
    if not math.Approximately(currCameraY, self.FollowPosition.y) then
        local smoothY = math.SmoothStep(currCameraY, self.FollowPosition.y, self.SwipeSmoothStepNum)
        self.ScreenCamera.transform:SetPositionY(smoothY)
    end

    -- 惯性滑动
    if self.InertialSlidingFactor > 0 then
        self:OnInertialSliding(self.InertialSlidingFactor)
    end

    -- 输入控制
    self:CheckPinchZoom()
    self:UpdateMouseScrollZoom()
    self:UpdateMouseDrag()
    self:UpdateClickAndDrag()
end
--endregion

--region 【PC端鼠标拖动】
---@brief PC鼠标拖拽移动地图
function GameMapMotion:UpdateMouseDrag()
    -- 按下
    if CS.UnityEngine.Input.GetMouseButtonDown(0) then
        self.PointStartFromUI = self:IsPointerOverGameObject()
        local mp = CS.UnityEngine.Input.mousePosition

        -- UI区域 / 屏幕外 不响应
        if self.PointStartFromUI or mp.x < 0 or mp.y < 0 or mp.x > CS.UnityEngine.Screen.width or mp.y > CS.UnityEngine.Screen.height then
            self.IsMouseDown = false
            return
        end

        self.IsMouseDown = true
        self.MouseLastScreenPos.x = mp.x
        self.MouseLastScreenPos.y = mp.y

        -- 重置状态
        self.InertialSlidingFactor = 0
        self.SwipeSmoothStepNum = 0
        self.Vector3Num.x = 0
        self.Vector3Num.y = 0
    end

    -- 拖拽中
    if CS.UnityEngine.Input.GetMouseButton(0) and self.IsMouseDown and not self.PointStartFromUI then
        local curr = CS.UnityEngine.Input.mousePosition
        if curr.x < 0 or curr.y < 0 or curr.x > CS.UnityEngine.Screen.width or curr.y > CS.UnityEngine.Screen.height then
            return
        end

        -- 坐标转换：屏幕 → 世界
        self.Vector3Num.x = curr.x
        self.Vector3Num.y = curr.y
        GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer1)

        self.Vector3Num.x = self.MouseLastScreenPos.x
        self.Vector3Num.y = self.MouseLastScreenPos.y
        GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer2)

        -- 计算偏移
        local dx = self.VectorContainer1.x - self.VectorContainer2.x
        local dy = self.VectorContainer1.y - self.VectorContainer2.y
        dx = dx * -1
        dy = dy * -1

        self.Vector3Num.x = dx
        self.Vector3Num.y = dy
        self:MovingFollowPosition(self.Vector3Num)

        -- 平滑系数递增
        self.SwipeSmoothStepNum = math.clamp(self.SwipeSmoothStepNum + CS.UnityEngine.Time.deltaTime * self.StepAddFactor, 0.3, 1)
        self.MouseLastScreenPos.x = curr.x
        self.MouseLastScreenPos.y = curr.y
    end

    -- 抬起
    if CS.UnityEngine.Input.GetMouseButtonUp(0) and self.IsMouseDown then
        self.IsMouseDown = false
        self.PointStartFromUI = false
        self.InertialSlidingFactor = 1 - self.SwipeSmoothStepNum
        self.SwipeSmoothStepNum = 0.3
    end
end
--endregion

--region 【UI触摸检测】
---@brief 判断当前是否触摸在UI上
---@return boolean 是否在UI上
function GameMapMotion:IsPointerOverGameObject()
    return GameLib.RSetupLuaTableValue.IsPointerOverUIObject()
end
--endregion

--region 【核心：严格区分 单击 / 拖动】
---@brief 单击/拖动分离逻辑，超过阈值=拖动，不触发单击
function GameMapMotion:UpdateClickAndDrag()
    if not self.IsMapWork or not self.ClickEnable or self.PointStartFromUI then
        return
    end

    -- 鼠标按压开始
    if CS.UnityEngine.Input.GetMouseButtonDown(0) then
        local mp = CS.UnityEngine.Input.mousePosition
        self.IsPressing = true
        self.ClickStartX = mp.x
        self.ClickStartY = mp.y
    end

    -- 鼠标按压结束
    if CS.UnityEngine.Input.GetMouseButtonUp(0) then
        if self.IsPressing then
            self.IsPressing = false
            local mp = CS.UnityEngine.Input.mousePosition
            local dx = mp.x - self.ClickStartX
            local dy = mp.y - self.ClickStartY
            local dist = math.sqrt(dx * dx + dy * dy)

            -- 小于阈值 = 单击
            if dist < self.DragThreshold then
                self:TriggerClick(mp)
            end
        end
    end

    -- 触摸单击判断
    local tc = CS.UnityEngine.Input.touchCount
    if tc == 1 then
        local touch = CS.UnityEngine.Input.GetTouch(0)
        if touch.phase == CS.UnityEngine.TouchPhase.Began then
            self.IsPressing = true
            self.ClickStartX = touch.position.x
            self.ClickStartY = touch.position.y
        elseif touch.phase == CS.UnityEngine.TouchPhase.Ended or touch.phase == CS.UnityEngine.TouchPhase.Canceled then
            if self.IsPressing then
                self.IsPressing = false
                local dx = touch.position.x - self.ClickStartX
                local dy = touch.position.y - self.ClickStartY
                local dist = math.sqrt(dx * dx + dy * dy)
                if dist < self.DragThreshold then
                    self:TriggerClick(touch.position)
                end
            end
        end
    end
end

---@brief 触发单击回调（仅真正单击时调用）
---@param screenPos Vector2 点击屏幕坐标
function GameMapMotion:TriggerClick(screenPos)
    if not self.ScreenCamera then return end
    if screenPos.x < 0 or screenPos.y < 0 or screenPos.x > CS.UnityEngine.Screen.width or screenPos.y > CS.UnityEngine.Screen.height then
        return
    end

    local isHit = false
    local hitTarget = nil

    -- 遍历Character层碰撞体进行点击检测（无射线、全平台兼容）
    local allColliders = CS.UnityEngine.Object.FindObjectsOfType(typeof(CS.UnityEngine.BoxCollider2D))
    if allColliders ~= nil then
        for i = 0, allColliders.Length - 1 do
            local collider = allColliders[i]
            if collider ~= nil and collider.gameObject ~= nil then
                local go = collider.gameObject
                -- 只检测Character层
                if go.layer == CS.UnityEngine.LayerMask.NameToLayer("Character") then
                    -- 世界坐标转屏幕
                    local screenPoint = self.ScreenCamera:WorldToScreenPoint(collider.transform.position)
                    local bounds = collider.bounds
                    -- 构建屏幕Rect
                    local screenRect = CS.UnityEngine.Rect(
                            screenPoint.x - bounds.extents.x,
                            screenPoint.y - bounds.extents.y,
                            bounds.size.x,
                            bounds.size.y
                    )
                    -- 判断点击
                    if screenRect:Contains(CS.UnityEngine.Vector2(screenPos.x, screenPos.y)) then
                        isHit = true
                        hitTarget = go
                        break
                    end
                end
            end
        end
    end

    -- 日志输出
    if isHit and hitTarget then
        print("✅ 单击命中物体：", hitTarget.name)
    else
        print("✅ 单击空地")
        HonorEventHelper:Fire(EventCmd.OnClickCharacter, nil)
    end

    -- 外部回调
    if self.OnMapClick then
        self.OnMapClick(nil, isHit, hitTarget)
    end
end
--endregion

--region 【移动端EasyTouch滑动】
---@brief 滑动开始
function GameMapMotion:OnSwipeBegin(gesture)
    if GameManager:IsGamePaused() then return end
    self.PointStartFromUI = self:IsPointerOverGameObject()
    self.IsPressing = false
    if self.PointStartFromUI then return end

    self.InertialSlidingFactor = 0
    self.SwipeSmoothStepNum = 0
    self.Vector3Num.x = 0
    self.Vector3Num.y = 0
end

---@brief 滑动中
function GameMapMotion:OnSwipe(gesture)
    if  self.PointStartFromUI or not self.IsMapWork then return end
    if gesture.touchCount >= 2 then return end

    if gesture.touchCount == 1 then
        local tx = gesture:GetPositionX()
        local ty = gesture:GetPositionY()
        if tx < 0 or ty < 0 or tx > CS.UnityEngine.Screen.width or ty > CS.UnityEngine.Screen.height then return end

        -- 坐标转换
        self.Vector3Num.x = tx
        self.Vector3Num.y = ty
        GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer1)

        self.Vector3Num.x = tx - gesture.GetDeltaPositionX()
        self.Vector3Num.y = ty - gesture.GetDeltaPositionY()
        GameLib.RSetupLuaTableValue.RScreenToWorldPoint(self.ScreenCamera, self.Vector3Num, self.VectorContainer2)

        -- 偏移计算
        local dx = self.VectorContainer1.x - self.VectorContainer2.x
        local dy = self.VectorContainer1.y - self.VectorContainer2.y
        dx = dx * -1
        dy = dy * -1

        self.Vector3Num.x = dx
        self.Vector3Num.y = dy
        self:MovingFollowPosition(self.Vector3Num)

        self.SwipeSmoothStepNum = math.clamp(self.SwipeSmoothStepNum + CS.UnityEngine.Time.deltaTime * self.StepAddFactor, 0.3, 1)
    end
end

---@brief 滑动结束
function GameMapMotion:OnSwipeEnd(gesture)
    if GameManager:IsGamePaused() then return end
    if self.PointStartFromUI then
        self.InertialSlidingFactor = 0
        self.SwipeSmoothStepNum = 0
        self.Vector3Num.x = 0
        self.Vector3Num.y = 0
    end
    self.PointStartFromUI = false
    self.IsPinching = false
    self.InertialSlidingFactor = 1 - self.SwipeSmoothStepNum
    self.SwipeSmoothStepNum = 0.3
end
--endregion

--region 【惯性滑动】
---@brief 拖拽结束后的惯性移动
---@param factor number 惯性系数
function GameMapMotion:OnInertialSliding(factor)
    self.InertialSlidingVector.x = self.Vector3Num.x * self.InertialSlidingFactor
    self:MovingFollowPosition(self.InertialSlidingVector)
    self.InertialSlidingFactor = math.clamp(self.InertialSlidingFactor - 0.025, 0, 1)
    self.SwipeSmoothStepNum = math.clamp(self.SwipeSmoothStepNum + GameLib.RSetupLuaTableValue.GetTimeDeltaTime(), 0.75, 1)
end
--endregion

--region 【地图边界限制】
---@brief 移动相机并限制在地图范围内
---@param worldDelta table 世界偏移量
function GameMapMotion:MovingFollowPosition(worldDelta)
    local aspect = self.ScreenCamera.aspect
    local ortho = self.ScreenCamera.orthographicSize
    local hw = aspect * ortho
    local hh = ortho

    -- 计算边界
    local minX = self.MapCenterX - self.MapHalfWidth + hw
    local maxX = self.MapCenterX + self.MapHalfWidth - hw
    local minY = self.MapCenterY - self.MapHalfHeigh + hh
    local maxY = self.MapCenterY + self.MapHalfHeigh - hh

    -- 更新位置
    self.FollowPosition.x = self.FollowPosition.x + worldDelta.x
    self.FollowPosition.y = self.FollowPosition.y + worldDelta.y

    -- 边界限制
    self.FollowPosition.x = math.clamp(self.FollowPosition.x, minX, maxX)
    self.FollowPosition.y = math.clamp(self.FollowPosition.y, minY, maxY)
    self.FollowPosition.x = math.floor(self.FollowPosition.x * 1000) / 1000
    self.FollowPosition.y = math.floor(self.FollowPosition.y * 1000) / 1000
end
--endregion

--region 【双指缩放 + 鼠标滚轮缩放】
---@brief 移动端双指缩放检测
function GameMapMotion:CheckPinchZoom()
    if not self.ScreenCamera or not self.IsMapWork or self.PointStartFromUI then return end
    local tc = CS.UnityEngine.Input.touchCount
    if tc == 2 then
        local t1 = CS.UnityEngine.Input.GetTouch(0)
        local t2 = CS.UnityEngine.Input.GetTouch(1)
        if t1.phase == CS.UnityEngine.TouchPhase.Began or t2.phase == CS.UnityEngine.TouchPhase.Began then
            self.LastPinchDistance = self:GetTwoTouchDistance(t1, t2)
            self.IsPinching = true
            self.InertialSlidingFactor = 0
            return
        end
        if t1.phase == CS.UnityEngine.TouchPhase.Moved or t2.phase == CS.UnityEngine.TouchPhase.Moved then
            local curr = self:GetTwoTouchDistance(t1, t2)
            local diff = curr - self.LastPinchDistance
            if math.abs(diff) > 1 then
                local ortho = self.ScreenCamera.orthographicSize
                ortho = diff > 0 and (ortho / self.ZoomSpeed) or (ortho * self.ZoomSpeed)
                ortho = math.clamp(ortho, self.MinOrthoSize, self.MaxOrthoSize)
                self.ScreenCamera.orthographicSize = ortho
                self.LastPinchDistance = curr
                self:ClampCameraAfterZoom()
            end
        end
    else
        self.IsPinching = false
    end
end

---@brief 获取两个触摸点距离
function GameMapMotion:GetTwoTouchDistance(t1, t2)
    local dx = t1.position.x - t2.position.x
    local dy = t1.position.y - t2.position.y
    return math.sqrt(dx * dx + dy * dy)
end

---@brief 鼠标滚轮缩放
function GameMapMotion:UpdateMouseScrollZoom()
    if not self.ScreenCamera or not self.IsMapWork then return end
    local scroll = CS.UnityEngine.Input.GetAxis("Mouse ScrollWheel")
    if math.abs(scroll) < 0.001 then return end

    local ortho = self.ScreenCamera.orthographicSize
    ortho = scroll > 0 and (ortho / self.ZoomSpeed) or (ortho * self.ZoomSpeed)
    ortho = math.clamp(ortho, self.MinOrthoSize, self.MaxOrthoSize)
    if ortho == 0 then ortho = (self.MinOrthoSize + self.MaxOrthoSize) / 2 end

    self.ScreenCamera.orthographicSize = ortho
    self:ClampCameraAfterZoom()
end

---@brief 缩放后重新限制相机边界
function GameMapMotion:ClampCameraAfterZoom()
    local aspect = self.ScreenCamera.aspect
    local ortho = self.ScreenCamera.orthographicSize
    local hw = aspect * ortho
    local hh = ortho

    local minX = self.MapCenterX - self.MapHalfWidth + hw
    local maxX = self.MapCenterX + self.MapHalfWidth - hw
    local minY = self.MapCenterY - self.MapHalfHeigh + hh
    local maxY = self.MapCenterY + self.MapHalfHeigh - hh

    self.FollowPosition.x = math.clamp(self.FollowPosition.x, minX, maxX)
    self.FollowPosition.y = math.clamp(self.FollowPosition.y, minY, maxY)
end
--endregion

--region 【EasyTouch 触摸注册/反注册】
---@brief 注册/反注册触摸事件
function GameMapMotion:RegisterTouchEvents(action)
    if action == "+" then
        self.SwipeBegin = self.SwipeBegin or function(g) self:OnSwipeBegin(g) end
        self.Swipe = self.Swipe or function(g) self:OnSwipe(g) end
        self.SwipeEnd = self.SwipeEnd or function(g) self:OnSwipeEnd(g) end

        CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragStart("+", self.SwipeBegin)
        CS.HedgehogTeam.EasyTouch.EasyTouch.On_Drag("+", self.Swipe)
        CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragEnd("+", self.SwipeEnd)
    else
        if self.SwipeBegin then
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragStart("-", self.SwipeBegin)
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_Drag("-", self.Swipe)
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_DragEnd("-", self.SwipeEnd)
        end
    end
end
--endregion

--region 【开关控制：暂停 / 恢复】
---@brief 恢复地图控制
function GameMapMotion:ResumeMapMotion()
    if self.IsMapWork then return end

    self.SwipeSmoothStepNum = 0
    self.FollowPosition.x = self.ScreenCamera.transform:GetPositionX()
    self.FollowPosition.y = self.ScreenCamera.transform:GetPositionY()
    self.InertialSlidingFactor = 0
    self.Vector3Num.x = 0
    self.Vector3Num.y = 0
    self.IsMapWork = true
    self.IsMouseDown = false
    self.PointStartFromUI = false
    self.IsPressing = false
end

---@brief 暂停地图控制
function GameMapMotion:PauseMapMotion()
    self.IsMapWork = false
    self.IsMouseDown = false
    self.IsPressing = false

    -- 记录暂停位置
    self.PositionWhenPaused = {
        self.ScreenCamera.transform:GetPositionX(),
        self.ScreenCamera.transform:GetPositionY(),
        self.ScreenCamera.transform:GetPositionZ()
    }
end
--endregion

--region 【初始化与销毁】
---@brief 地图启动初始化
function GameMapMotion:OnMapStart(mapTrans)
    if not mapTrans then return end
    self.MapMotionLayer = mapTrans:GetComponent("GameLib.MapMotionLayer")
    if not self.MapMotionLayer then return end

    -- 基础配置
    self.ScreenCamera = GameLib.MapCamHelper.MapCamera
    self.IsMapWork = true
    self.SwipeSmoothStepNum = 0
    self.IsMouseDown = false
    self.IsPressing = false

    -- 相机初始尺寸
    self.ScreenCamera.orthographicSize = (self.MinOrthoSize + self.MaxOrthoSize) / 2
    self.InitialOrthoSize = self.ScreenCamera.orthographicSize

    -- 地图边界数据
    self.MapHalfWidth = self.MapMotionLayer.m_AreaBounds.extents.x
    self.MapHalfHeigh = self.MapMotionLayer.m_AreaBounds.extents.y
    self.MapCenterX = self.MapMotionLayer.m_AreaBounds.center.x
    self.MapCenterY = self.MapMotionLayer.m_AreaBounds.center.y
    self.MapBounds = self.MapMotionLayer.m_AreaBounds

    -- 初始位置
    self.FollowPosition = {
        x = self.MapCenterX,
        y = self.MapCenterY,
        z = self.ScreenCamera.transform:GetPositionZ()
    }
    self.PositionWhenPaused = { x = self.FollowPosition.x, y = self.FollowPosition.y, z = self.FollowPosition.z }

    -- 临时向量
    self.InertialSlidingVector = { x = 0, y = 0, z = 0 }
    self.Vector3Num = { x = 0, y = 0, z = 0 }

    -- 相机信息
    self.ScreenCameraAspect = self.ScreenCamera.aspect
    self.ScreenCameraOrthographicSize = self.ScreenCamera.orthographicSize

    -- 注册触摸
    self:RegisterTouchEvents("+")
    self:ClampCameraAfterZoom()
end

---@brief 重置X坐标（带边界）
function GameMapMotion:ResetPositionX(posX, aspect, ortho)
    local hw = aspect * ortho
    local min = self.MapCenterX - self.MapHalfWidth + hw
    local max = self.MapCenterX + self.MapHalfWidth - hw
    return math.clamp(posX, min, max)
end

---@brief 重置Y坐标（带边界）
function GameMapMotion:ResetPositionY(posY, ortho)
    local hh = ortho
    local min = self.MapCenterY - self.MapHalfHeigh + hh
    local max = self.MapCenterY + self.MapHalfHeigh - hh
    return math.clamp(posY, min, max)
end

---@brief 获取当前地图包围盒
function GameMapMotion:GetCurrMapRect()
    return self.MapBounds
end

---@brief 地图结束，清理
function GameMapMotion:OnMapEnd()
    self:RegisterTouchEvents("-")
    self:PauseMapMotion()
end
--endregion

return GameMapMotion
