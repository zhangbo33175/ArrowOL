---@class GameMapBase : LuaBehaviourSuper 地图基类
local GameMapBase = class('GameMapBase', import('LuaBehaviourSuper'))

--region 【构造 & 实例化】
---@brief 构造函数
---@type fun(args:table):void
---@param args table 自定义参数
function GameMapBase:ctor(args)
    GameMapBase.super.ctor(self, args)

    self.Swipe = nil                          ---@type function(HedgehogTeam.EasyTouch.Gesture) 滑动手势回调
    self.SwipeEnd = nil                       ---@type function(HedgehogTeam.EasyTouch.Gesture) 滑动结束回调
    self.SimpleTap = nil                      ---@type function(HedgehogTeam.EasyTouch.Gesture) 点击回调
    self.SwipePos = Unity.Vector3(0, 0, 0)    ---@type UnityEngine.Vector3 滑动位置
    self.ScreenCamera = self:GetSafeScreenCamera()  ---@type UnityEngine.Camera 地图相机

    self._IsPlaySwipeSound = false            ---@type boolean 是否正在播放滑动音效
   
    self.GameMapRoot=self.cs.transform:Find("RawImage/Root")
    --- 全局当前地图实例
    MapManager.CurGameMapBase = self
    --- 获取当前地图配置组件
    MapManager.CurMapData = self.cs:GetComponent("MapData")
    
    self:LoadCharacterByMapData()
end


---@brief 安全获取地图相机，自动判空
---@return UnityEngine.Camera|nil
function GameMapBase:GetSafeScreenCamera()
    local cam = GameLib.MapCamHelper.MapCamera
    if cam == nil then
        print("[GameMapBase] 警告：MapCamera 为空")
        return nil
    end
    return cam
end

---@brief 创建实例
---@type fun(args:table):GameMapBase
---@param args table 自定义参数
---@return GameMapBase 地图基类实例
function GameMapBase:Create(args)
    return GameMapBase.new(args)
end
--endregion

--region 【事件 & 初始化】
---@brief 注册事件监听（父类自动注销）
---@type fun():void
function GameMapBase:OnAddListeners()
    GameMapBase.super.OnAddListeners(self)
end

---@brief 初始化
---@type fun():void
function GameMapBase:Init()
    GameMapBase.super.Init(self)
end

---@brief 唤醒回调
---@type fun():void
function GameMapBase:Awake()
    GameMapBase.super.Awake(self)

    --- 初始化地图相机参数
    GameLib.MapCamHelper.InitMapCamera(MapManager.CurMapData.m_NormalCamData)
    --- 初始化机器到游戏初始状态
    MapManager:OnInitGame()
end

---@brief 开始
---@type fun():void
function GameMapBase:Start()
    GameMapBase.super.Start(self)
    GameMapMotion:OnMapStart(self.cs.transform)
end

---@brief 销毁回调
---@type fun():void
function GameMapBase:OnDestroy()
    GameMapBase.super.OnDestroy(self)
    --- 清理触摸手势数据
    self:ClearSwipeData()
end
--endregion

--region 【游戏生命周期】
---@brief 游戏初始化
---@type fun():void
function GameMapBase:OnInitGame()
end

---@brief 进入游戏
---@type fun():void
function GameMapBase:OnEnterGame()
end

---@brief 游戏开始
---@type fun():void
function GameMapBase:OnBeginGame()
    --- 注册触摸事件
    self:RegOrUnRegEvent('+')
end

---@brief 游戏暂停
---@type fun():void
function GameMapBase:OnPauseGame()
end

---@brief 游戏恢复
---@type fun():void
function GameMapBase:OnResumeGame()
end

---@brief 游戏结束
---@type fun():void
function GameMapBase:OnGameOver()
    --- 注销触摸事件
    self:RegOrUnRegEvent('-')

    --- 胜利/失败表现
    if GameCoreManager._IsSuccessFinishGame == true then
        self:GameOverSuccessAni()
    else
        self:GameOverFailedAni()
    end
end

---@brief 游戏退出
---@type fun():void
function GameMapBase:OnExitGame()
end

---@brief 游戏重玩
---@type fun():void
function GameMapBase:OnRePlayGame()
    self:RegOrUnRegEvent('+')
end

---@brief 地图卸载前清理
---@type fun():void
function GameMapBase:UnloadMapBeforeClear()
end

---@brief 地图卸载后清理
---@type fun():void
function GameMapBase:UnloadMapAfterClear()
end
--endregion

--region 【胜利/失败动画与音效】
---@brief 游戏胜利表现（音效+粒子）
---@type fun():void
function GameMapBase:GameOverSuccessAni()
    SoundManager:PlaySound(Tables.Sounds.Level_Success.Name)
    --- 播放胜利彩带粒子
    GameEffectManager:AddEffect(RES.GameObject_TX_coloredribbon.Name, GameMainRoot.Effect.efRoot, ELayerName.UI, Unity.Vector3(0, 0, 0), EPos.World, 3.5)
end

---@brief 游戏失败表现（音效）
---@type fun():void
function GameMapBase:GameOverFailedAni()
    --- 玩家主动放弃不播放失败音效
    if GameCoreManager._eGameOverType ~= EGameOverType.PlayerGiveUp then
        SoundManager:PlaySound(Tables.Sounds.Level_Failed.Name)
    end
end
--endregion

--region 【触摸滑动与点击】
---@brief 清除滑动/点击数据
---@type fun():void
function GameMapBase:ClearSwipeData()
    self.Swipe = nil
    self.SimpleTap = nil
    self.SwipeEnd = nil
    self._IsPlaySwipeSound = false
end

---@brief 检测点击/滑动与机器的交互
---@type fun(gesture:HedgehogTeam.EasyTouch.Gesture, isSwipe:boolean):void
---@param gesture 手势对象
---@param isSwipe boolean 是否为滑动
function GameMapBase:CheckOrderMachines(gesture, isSwipe)
    if GameManager:IsGamePaused() then
        return
    end

    if gesture.touchCount == 1 then
        self.SwipePos.x = gesture.position.x
        self.SwipePos.y = gesture.position.y
        self.SwipePos.z = 0

        local worldPosition = self.ScreenCamera:ScreenToWorldPoint(self.SwipePos)
        worldPosition.z = -1

        if MapManager.CurMapData and MapManager.CurMapData.tableOfViewBounds then
            if MapManager.CurMapData.tableOfViewBounds.Count > 0 then
                local leftBounds = MapManager.CurMapData.tableOfViewBounds[0]
                local leftList = MapManager:GetOrderMachineListOfLeft()
            end

            if MapManager.CurMapData.tableOfViewBounds.Count > 1 then
                local rightBounds = MapManager.CurMapData.tableOfViewBounds[1]
                local rightList = MapManager:GetOrderMachineListOfRight()
            end
        end
    end
end

---@brief 地图收集金币消息的注册和反注册
---@type fun(action:string):void
---@param action string + or -
function GameMapBase:RegOrUnRegEvent(action)
    if action == '+' then
        self.Swipe = self.Swipe or function(gesture)
            self:OnSwipe(gesture)
        end
        self.SwipeEnd = self.SwipeEnd or function(gesture)
            self:OnSwipeEnd(gesture)
        end
        self.SimpleTap = self.SimpleTap or function(gesture)
            self:OnSimpleTap(gesture)
        end

        CS.HedgehogTeam.EasyTouch.EasyTouch.On_Swipe('+', self.Swipe)
        CS.HedgehogTeam.EasyTouch.EasyTouch.On_UIElementTouchUp('+', self.SwipeEnd)
        CS.HedgehogTeam.EasyTouch.EasyTouch.On_OverUIElement('+', self.SimpleTap)
    elseif action == '-' then
        if self.Swipe and self.SimpleTap then
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_Swipe('-', self.Swipe)
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_UIElementTouchUp('-', self.SwipeEnd)
            CS.HedgehogTeam.EasyTouch.EasyTouch.On_OverUIElement('-', self.SimpleTap)
        end
    end
end

---@brief 滑动
---@type fun(gesture:HedgehogTeam.EasyTouch.Gesture):void
---@param gesture HedgehogTeam.EasyTouch.Gesture 滑动手势
function GameMapBase:OnSwipe(gesture)
    self:CheckOrderMachines(gesture, true)
end

---@brief 点击
---@type fun(gesture:HedgehogTeam.EasyTouch.Gesture):void
---@param gesture HedgehogTeam.EasyTouch.Gesture 点击手势
function GameMapBase:OnSimpleTap(gesture)
    self:CheckOrderMachines(gesture, false)
end
--endregion

--region 加载地图寻找物信息
---根据地图配置数据生成角色实体并初始化显示
function GameMapBase:LoadCharacterByMapData()
    -- 获取地图根配置数据
    local _mapData = GameManager.GetMapRootData()

    -- 遍历所有地图角色数据
    for _, v in ipairs(_mapData.MapData) do
        -- 生成角色实体
        local CharacterBase = CharacterGenerationManager:CharacterGeneration(v.Id, self.GameMapRoot)

        -- 设置角色位置（字符串转Vector3）
        CharacterBase._GameObject.transform.localPosition = StringToVector3(v.Position)

        -- 获取角色图标UI对象
        local CharacterIcon = CharacterBase._GameObject.transform:Find("CharacterIcon")
        local uiImage = CharacterIcon.transform:GetComponent(typeof(CS.UnityEngine.UI.Image))

        -- 获取角色配置数据并刷新图标
        local catData = CharacterGenerationDataManager:GetDataById(v.Id)
        PlayHelper:RefreshImageSprite(CharacterBase, uiImage, catData.Icon)

        -- 将角色存入全局管理器列表
        GameManager.CharacterList[v.Id] = CharacterBase
    end
end
--endregion

return GameMapBase
