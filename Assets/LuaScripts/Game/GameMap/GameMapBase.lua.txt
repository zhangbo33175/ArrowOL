function GameMapBase:OnGameOver()
    -- 注销触摸事件
    self:RegOrUnRegEvent('-')

    -- 胜利/失败表现
    if GameCoreManager._IsSuccessFinishGame == true then
        self:GameOverSuccessAni()
    else
        self:GameOverFailedAni()
    end
end

--========================================================================
-- 胜利/失败动画与音效
--========================================================================
---游戏胜利表现（音效+粒子）
function GameMapBase:GameOverSuccessAni()
    SoundMgr:PlaySound(Tables.Sounds.Level_Success.Name)
    -- 播放胜利彩带粒子
    GameEffectManager:AddEffect(RES.GameObject_TX_coloredribbon.Name, GameMainRoot.Effect.efRoot, ELayerName.UI, Unity.Vector3(0, 0, 0), EPos.World, 3.5)
end

---游戏失败表现（音效）
function GameMapBase:GameOverFailedAni()
    -- 玩家主动放弃不播放失败音效
    if GameCoreManager._eGameOverType ~= EGameOverType.PlayerGiveUp then
        SoundMgr:PlaySound(Tables.Sounds.Level_Failed.Name)
    end
end

--========================================================================

---清除滑动/点击数据
---@type fun():void
function GameMapBase:ClearSwipeData()
    self.Swipe = nil
    self.SimpleTap = nil
    self.SwipeEnd = nil
    self._IsPlaySwipeSound = false
end
