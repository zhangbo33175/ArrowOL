---@class HonorLogin
local HonorLogin         = {}

--region 依赖引入
local HonorSDKMap        = require("Game.GameSDK.HonorSDK.HonorSDKMap")
local HonorLoginCallback = require("Game.GameSDK.HonorSDK.HonorLoginCallback")
local SGCloudSDKProtocol = require("Game.GameSDK.HonorSDK.SGCloudSDKProtocol")
--endregion

--region 登录接口

---@brief HonorSDK 登录
local function CSGHonorSDKLoginLogin()
    local loginData = HonorSDKMap.CSSGLoginEx
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.P_SGLoginLogin, data, HonorLoginCallback.OnHonorOnLoginCallback)
    end
end

---@brief 获取当前账号信息
---@param callback function 获取账号信息回调
local function CSGHonorSDKGetUserInfo(callback)
    local loginData = HonorSDKMap.CSGHonorSDKGetUserInfo
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.p_SGGetUserInfo, data, callback)
    end
end

---@brief 判断当前是否处于登录状态
---@param callback function 查询回调
local function CSGHonorSDKIsLogin(callback)
    local loginData = HonorSDKMap.CSGHonorSDKIsLogin
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.p_IsLogin, data, callback)
    end
end

---@brief 切换账号
local function CSGHonorSDKSwitchUser()
    local loginData      = {}
    loginData.useLaunchUser = true
    local data           = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.P_SGSwitchUser, data, HonorLoginCallback.OnHonorOnLoginCallback)
    end
end

---@brief 登出
local function CSGHonorSDKLogout()
    local loginData = HonorSDKMap.CSGHonorSDKLogout
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.P_SGLogout, data, HonorLoginCallback.OnHonorOnLoginCallback)
    end
end

---@brief 退出游戏
local function CSGHonorSDKExit()
    local loginData = HonorSDKMap.CSGHonorSDKExit
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.P_SGExit, data, HonorLoginCallback.OnHonorOnLoginCallback)
    end
end

---@brief 判断 SDK 是否有退出框
local function CSGHonorSDKHasExitDialog()
    local loginData = HonorSDKMap.CSGHonorSDKHasExitDialog
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.p_SGHasExitDialog, data, HonorLoginCallback.OnHonorOnLoginCallback)
    end
end

---@brief 查询区服信息
---@param call function 区服信息回调
local function CSGHonorQueryServerList(call)
    local loginData = HonorSDKMap.SCGHonorQueryServerList
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.p_SGHasQueryServerList, data, function(cmd, content)
            if cmd == SGCloudSDKProtocol.CloudSDKProtocol.p_SGHasQueryServerList then
                call(json.decode(content))
            else
                call()
            end
        end)
    end
end
--endregion

--region 推送绑定

---@brief 绑定登录相关回调
local function CSGHonorSDKLoginBind()
    if GameCfg.IsSDK then
        SGHonorSDKImpl.BindPushMethod(SGCloudSDKProtocol.CloudSDKProtocol.P_SGSwitchUser, HonorLoginCallback.OnHonorOnLoginCallback)
        SGHonorSDKImpl.BindPushMethod(SGCloudSDKProtocol.CloudSDKProtocol.P_SGExit, HonorLoginCallback.OnHonorOnLoginCallback)
        SGHonorSDKImpl.BindPushMethod(SGCloudSDKProtocol.CloudSDKProtocol.P_SGLogout, HonorLoginCallback.OnHonorOnLoginCallback)
        SGHonorSDKImpl.BindPushMethod(SGCloudSDKProtocol.CloudSDKProtocol.P_SGExitError, HonorLoginCallback.OnHonorOnLoginCallback)
    end
end

---@brief 解绑登录相关回调
local function CSGHonorSDKLoginRemove()
    if GameCfg.IsSDK then
        SGHonorSDKImpl.RemovePushMethod(SGCloudSDKProtocol.CloudSDKProtocol.P_SGSwitchUser)
        SGHonorSDKImpl.RemovePushMethod(SGCloudSDKProtocol.CloudSDKProtocol.P_SGExit)
        SGHonorSDKImpl.RemovePushMethod(SGCloudSDKProtocol.CloudSDKProtocol.P_SGLogout)
    end
end
--endregion

--region 用户中心与悬浮窗

---@brief 打开用户中心
local function CSGHonorOpenUserCenter()
    local loginData = HonorSDKMap.CSGHonorSDKHasExitDialog
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.P_SGOpenUserCenter, data, HonorLoginCallback.OnHonorOnLoginCallback)
    end
end

---@brief 隐藏悬浮窗
local function CSGHonorHideFloatView()
    local loginData = HonorSDKMap.CSGHonorSDKHasExitDialog
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.P_SGHideFloatView, data, HonorLoginCallback.OnHonorOnLoginCallback)
    end
end

---@brief 显示悬浮窗
local function CSGHonorShowFloatView()
    local loginData = HonorSDKMap.CSGHonorSDKHasExitDialog
    local data      = json.encode(loginData)
    if GameCfg.IsSDK then
        SGHonorSDKImpl.Request(SGCloudSDKProtocol.CloudSDKProtocol.P_SGShowFloatView, data, HonorLoginCallback.OnHonorOnLoginCallback)
    end
end
--endregion

---隐藏悬浮窗

---显示悬浮窗
HonorLogin.CSGHonorShowFloatView = CSGHonorShowFloatView
---HonorSDK 绑定
HonorLogin.CSGHonorSDKLoginBind = CSGHonorSDKLoginBind
---HonorSDK 解绑
HonorLogin.CSGHonorSDKLoginRemove = CSGHonorSDKLoginRemove
---HonorSDK 登录
HonorLogin.CSGHonorSDKLoginLogin = CSGHonorSDKLoginLogin
---获取当前账号信息
HonorLogin.CSGHonorSDKGetUserInfo = CSGHonorSDKGetUserInfo
---判断当前是否处于登录状态

---切换账号

--endregion

return HonorLogin
