---@class HonorLoginCallback
local HonorLoginCallback = {}

--region 依赖加载
local SGCloudSDKProtocol = require("Game.GameSDK.HonorSDK.SGCloudSDKProtocol")
--endregion

--region 回调函数声明
local OnSGLoginLogin        --登录
local OnSGLoginFail         --登录失败
local OnSGLogout            --登出
local OnSGSwitchUser        --切换游戏账号请求
local OnSGExit              --退出回调
local OnSGGetUserInfo       --获取当前账号信息
local OnSGHasExitDialog     --唤判断 SDK 是否有退出框
local OnSGExitError         --失败信息
local OnOpenUserCenter      --用户信息

-- 返回方法list
local _OnLoginCallbackList
--endregion

--region 登录回调事件

---@brief 登录回调统一入口
---@param cmd number 协议号
---@param content string 消息体
function HonorLoginCallback.OnHonorOnLoginCallback(cmd, content)
    _OnLoginCallbackList[cmd](content)
end

--登录 回调
OnSGLoginLogin = function(content)
    local data = json.decode(content)
    -- 扩展信息
    LoginManager.OnGetTencentServerList(data)
end

--登录失败 回调
OnSGLoginFail = function(content)
    local data = json.decode(content)
    LoginManager.OnSGLoginFail(data)
end

--登出 回调
OnSGLogout = function(content)
    local data = json.decode(content)
    LoginManager.OnSGLogout(data)
end

--切换游戏账号回调
OnSGSwitchUser = function(content)
    local data = json.decode(content)
    LoginManager.OnSGSwitchUser(data)
end

--退出回调
OnSGExit = function(content)
    local data = json.decode(content)
    LoginManager.OnSGExit(data)
end

--获取当前账号信息 回调
OnSGGetUserInfo = function(content)
    local data = json.decode(content)
    LoginManager.OnSGGetUserInfo(data)
end

--唤判断 SDK 是否有退出框 回调
OnSGHasExitDialog = function(content)
    local data = json.decode(content)
    LoginManager.OnSGHasExitDialog(data)
end

--回调失败
OnSGExitError = function(content)
    local data = json.decode(content)
    LoginManager.OnSGLoginFail(data)
end

--回调失败
OnOpenUserCenter = function(content)
    local data = json.decode(content)
end
--endregion

--region 回调分发表
-- 方法list
_OnLoginCallbackList = {
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGExitError]      = OnSGExitError,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGOpenUserCenter] = OnOpenUserCenter,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGLoginLogin]     = OnSGLoginLogin,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGLoginFail]      = OnSGLoginFail,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGLogout]         = OnSGLogout,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGSwitchUser]     = OnSGSwitchUser,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGExit]           = OnSGExit,
    [SGCloudSDKProtocol.CloudSDKProtocol.p_SGGetUserInfo]     = OnSGGetUserInfo,
    [SGCloudSDKProtocol.CloudSDKProtocol.p_SGHasExitDialog]  = OnSGHasExitDialog,
}
--endregion

return HonorLoginCallback
