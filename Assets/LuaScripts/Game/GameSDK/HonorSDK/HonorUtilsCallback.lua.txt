---@class HonorUtilsCallback
local HonorUtilsCallback = {}

--region 依赖加载
local SGCloudSDKProtocol = require("Game.GameSDK.HonorSDK.SGCloudSDKProtocol")
--endregion

--region 回调函数声明
--请求权限
local OnSGHonorSDKRequestPermission = nil
--获取权限状态
local OnSGHonorSDKGetPermissionState = nil
--获取网络信息
local OnSGHonorSDKGetNetworkInfo = nil
--获取磁盘信息
local OnSGHonorSDKGetDiskInfo = nil
--获取应用信息
local OnSGHonorSDKGetAppInfo = nil
--获取内存信息
local OnSGHonorSDKGetMemory = nil
--获取电池信息
local OnSGHonorSDKGetBattery = nil
--获取硬件信息
local OnSGHonorSDKGetHardwareInfo = nil
--获取设备刘海屏信息
local OnSGHonorSDKGetScreenNotchInfo = nil

-- 返回方法list
local _UtilsCallbackList
--endregion

--region 工具类回调事件

---@brief 工具类回调统一入口
---@param cmd number 协议号
---@param content string 消息体
function HonorUtilsCallback.OnHonorUtilsCallback(cmd, content)
    _UtilsCallbackList[cmd](content)
end

--请求权限
OnSGHonorSDKRequestPermission = function(content)
    local data = json.decode(content)
    ChatGVoice.SetRequestHonorSDKPermission(data)
end

--获取权限状态
OnSGHonorSDKGetPermissionState = function(content)
    local data = json.decode(content)
    ChatGVoice.SetGVoicePermissionState(data)
end

--获取网络信息
OnSGHonorSDKGetNetworkInfo = function(content)
    local data = json.decode(content)
    SDKData.NetworkType = data.NetworkType
    SDKData.IsConnect = data.IsConnect
end

--获取磁盘信息
OnSGHonorSDKGetDiskInfo = function(content)
    local data = json.decode(content)
    SDKData.AvailSize = data.AvailSize
    SDKData.TotalSize = data.TotalSize
    SDKManager.HonorUtils.GetHonorSDKAppInfo()
end

--获取应用信息
OnSGHonorSDKGetAppInfo = function(content)
    local data = json.decode(content)
    SDKData.AppId = data.AppId
    SDKData.DeviceId = data.DeviceId
    SDKData.Version = data.Version
    SDKData.Platform = data.Platform
    SDKData.AppName = data.AppName
    SDKData.Release = data.Release
    SDKData.Language = data.Language
    SDKData.Country = data.Country
    SDKManager.HonorUtils.GetHonorSDKMemory()
end

--获取内存信息
OnSGHonorSDKGetMemory = function(content)
    local data = json.decode(content)
    SDKData.AvailMem = data.AvailMem
    SDKData.TotalMem = data.TotalMem
    SDKData.PssMem = data.PssMem
    SDKManager.HonorUtils.GetHonorSDKBattery()
end

--获取电池信息
OnSGHonorSDKGetBattery = function(content)
    local data = json.decode(content)
    SDKData.Level = data.Level
    SDKData.Scale = data.Scale
    SDKData.Temperature = data.Temperature
    SDKData.IsCharging = data.IsCharging
    SDKManager.HonorUtils.GetHonorSDKHardwareInfo()
end

--获取硬件信息
OnSGHonorSDKGetHardwareInfo = function(content)
    local data = json.decode(content)
    if data.CpuInfo ~= nil then
        SDKData.CpuModel = data.CpuInfo.CpuModel
        SDKData.CpuNum = data.CpuInfo.CpuNum
        SDKData.CpuMaxFreq = data.CpuInfo.CpuMaxFreq
    end
    if data.GpuInfo ~= nil then
        SDKData.GpuModel = data.GpuInfo.GpuModel
        SDKData.GlVersion = data.GpuInfo.GlVersion
    end
    if data.Temperature ~= nil then
        SDKData.AvailMem = data.Temperature.AvailMem
        SDKData.TotalMem = data.Temperature.TotalMem
        SDKData.PssMem = data.Temperature.PssMem
    end
    SDKData.DeviceModel = data.DeviceModel
    SDKData.SystemHardware = data.DeviceModel
    SDKData.SystemSoftware = data.OsVersion
    SDKData.TelecomOper = data.CarrierType
    SDKData.Ipv4 = data.Ipv4
    SDKData.Ipv6 = data.Ipv6
end

--获取设备刘海屏信息
OnSGHonorSDKGetScreenNotchInfo = function(content)
    --local data = json.decode(content)
end
--endregion

--region 回调分发表
-- 方法list
_UtilsCallbackList = {
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGHonorSDKRequestPermission] = OnSGHonorSDKRequestPermission,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGHonorSDKGetPermissionState] = OnSGHonorSDKGetPermissionState,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGHonorSDKGetNetworkInfo] = OnSGHonorSDKGetNetworkInfo,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGHonorSDKGetDiskInfo] = OnSGHonorSDKGetDiskInfo,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGHonorSDKGetAppInfo] = OnSGHonorSDKGetAppInfo,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGHonorSDKGetMemory] = OnSGHonorSDKGetMemory,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGHonorSDKGetBattery] = OnSGHonorSDKGetBattery,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGHonorSDKGetHardwareInfo] = OnSGHonorSDKGetHardwareInfo,
    [SGCloudSDKProtocol.CloudSDKProtocol.P_SGHonorSDKGetScreenNotchInfo] = OnSGHonorSDKGetScreenNotchInfo,
}
--endregion

return HonorUtilsCallback
