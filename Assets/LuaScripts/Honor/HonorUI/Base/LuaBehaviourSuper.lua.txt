--=====================================================================================================
-- 生命周期：构造函数
--=====================================================================================================

---@brief 构造函数
---@desc 绑定环境表变量并初始化全部字段（参数、引用、动画时长、资源缓存、行为树节点缓存等）
---@param args table 构造参数（包含 env 环境表）
function LuaBehaviourSuper:ctor(args)
    -- 绑定环境表变量
    if args ~= nil and args.env ~= nil then
        for __, v in pairs(args.env) do
            self[__] = v
        end
    end

    self._args = args                                -- 传入参数
    self._params = self.cs.LuaParams                 -- C#配置的自定义参数
    self._name = self.__cname                        -- 类名
    self._isClosing = false                          -- 是否正在关闭

    -- 行为树节点缓存
    self._btActions = {}                             -- 行为树Action节点
    self._btComposites = {}                          -- 行为树Composite节点
    self._btConditionals = {}                        -- 行为树Conditional节点
    self._btDecorators = {}                          -- 行为树Decorator节点

    -- UI 通用元素引用
    self._bgImage = nil                              -- 背景图
    self._bottomMaskLayerImage = nil                 -- 底部遮罩
    self._topMaskLayerImage = nil                    -- 顶部遮罩
    self._bottomCloseLayerButton = nil               -- 底部点击关闭按钮

    -- 动画时长配置
    self._bottomMaskLayerOpenAnimationDelay = 0.5
    self._bottomMaskLayerCloseAnimationDelay = 0.5
    self._alphaOpenAnimationDelay = 0.5
    self._alphaCloseAnimationDelay = 0.5
    self._rootOpenAnimationDelay = 0.5
    self._rootCloseAnimationDelay = 0.5

    self._loadedAssets = {}                           -- 动态加载的资源（自动管理释放）
    self._oriPosForUIStack = Unity.Vector3.zero       -- UI栈原始位置

    -- 统一生命周期管理（对齐 HonorUI 自动注销风格）
    self._globalEventListenerList = {}                -- 全局事件监听记录表（OnDestroy 统一注销）
    self._timerList = {}                              -- 协程定时器句柄列表（OnDestroy 统一停止）
    --TODO: [新增] ctor 初始化全局事件记录表与定时器句柄表
end

end

---@brief 销毁（自动释放所有资源）
---@desc 释放引用、销毁全部行为树节点、卸载资源并移除事件与协程
function LuaBehaviourSuper:OnDestroy()
    self._args = nil

    -- 销毁所有行为树节点
    for _, btAction in pairs(self._btActions) do
        btAction:OnDestroy()
    end
    self._btActions = {}

    for _, btComposite in pairs(self._btComposites) do
        btComposite:OnDestroy()
    end
    self._btComposites = {}

    for _, btConditional in pairs(self._btConditionals) do
        btConditional:OnDestroy()
    end
    self._btConditionals = {}

    for _, btDecorator in pairs(self._btDecorators) do
        btDecorator:OnDestroy()
    end
    self._btDecorators = {}

    -- 释放资源、事件、协程
    self:UnloadAllAssets()
    HonorEventHelper:RemoveListenerByTarget(self)
    HonorCoroutineHelper:StopByGO(self.cs.gameObject)

    --TODO: [新增] 统一停止本对象注册的循环定时器协程（StopByGO 已按 GO 兜底，此处按记录表显式收口）
    if self._timerList ~= nil then
        for _, co in ipairs(self._timerList) do
            HonorCoroutineHelper:Stop(co)
        end
        self._timerList = {}
    end

    --TODO: [新增] 清空全局事件监听记录表（实际移除已由 RemoveListenerByTarget(self) 完成，此处仅释放记录引用）
    self._globalEventListenerList = {}

    if self._uiListenerList ~= nil then
        for _, listener in ipairs(self._uiListenerList) do
            OnRemoveListener(listener[1], listener[2], listener[3])
        end
        self._uiListenerList = {}
    end
end

--=====================================================================================================
-- UI 监听注册（自动注销）
--=====================================================================================================

---@brief 注册UI组件交互监听（随界面销毁自动注销）
---@desc 封装 AddUIListenerFunction：注册按钮/Toggle/输入框等交互事件，并记录到 self._uiListenerList，
---      界面销毁时由 OnDestroy 统一 OnRemoveListener 自动移除，无需手动配对注销
---@param obj userdata C#注入的UI对象（Button/Toggle/Input等）
---@param cmd string 事件类型：Clicked/Changed/End/Chosen 或 Evt_ 开头事件
---@param handler table 回调处理器（handler 包装）
function LuaBehaviourSuper:AddUIListener(obj, cmd, handler)
    if obj == nil then return end
    if self._uiListenerList == nil then
        self._uiListenerList = {}
    end
    table.insert(self._uiListenerList, {obj, cmd, handler})
    AddUIListenerFunction(obj, cmd, handler)
end

--=====================================================================================================
-- 全局事件与定时器（自动注销，对齐 HonorUI UIBaseComponent 风格）
--=====================================================================================================

---@brief 注册全局事件监听（随界面销毁自动注销）
---@desc 用 handler(self, callback) 包装回调并通过 HonorEventHelper 注册，回调句柄记录到 _globalEventListenerList；
---      OnDestroy 时由 HonorEventHelper:RemoveListenerByTarget(self) 统一移除，无需手动配对注销
---@param key string 事件名（FwEventCmd / EventCmd）
---@param callback function 事件回调（function(...) end）
--TODO: [新增] 提供 self:AddGlobalEvent 封装，全局事件随对象销毁自动注销（对齐 HonorUI UIBaseComponent:AddGlobalEvent）
function LuaBehaviourSuper:AddGlobalEvent(key, callback)
    if callback == nil then return end
    if self._globalEventListenerList == nil then
        self._globalEventListenerList = {}
    end
    -- handler(self, callback).target == self，便于 RemoveListenerByTarget(self) 统一回收
    local h = handler(self, callback)
    HonorEventHelper:AddListener(key, h)
    table.insert(self._globalEventListenerList, { key, h })
end

---@brief 注册循环定时器（随界面销毁自动停止）
---@desc 启动一个循环协程：每隔 delay 秒执行一次 func，记录协程句柄到 _timerList；
---      OnDestroy 时遍历 HonorCoroutineHelper:Stop(co) 统一停止，无需手动配对关闭。
---      说明：HonorTimerHelper:RegisterDeltaTime 需 procedure 且按事件名派发，与本"实例级循环回调"语义不符，故采用协程方案
---@param delay number 间隔秒数
---@param func function 每次间隔到点的回调
---@param isRealTime boolean 是否使用真实时间（不受时间缩放影响），默认 false
---@return UnityEngine.Coroutine 协程句柄（可用于提前手动停止）
--TODO: [新增] 提供 self:AddUpdateTimer 封装，实例级循环定时随对象销毁自动停止（对齐 HonorUI UIBaseComponent:AddUpdateTimer）
function LuaBehaviourSuper:AddUpdateTimer(delay, func, isRealTime)
    if func == nil then return nil end
    if delay == nil or delay <= 0 then delay = 0 end
    if self._timerList == nil then
        self._timerList = {}
    end
    -- handler(self, ...).target.cs.gameObject 已绑定本对象生命周期，StopByGO 亦可兜底
    local co = HonorCoroutineHelper:Start(handler(self, function()
        while true do
            if isRealTime then
                HonorCoroutineHelper:WaitForSecondsRealtime(delay)
            else
                HonorCoroutineHelper:WaitForSeconds(delay)
            end
            func()
        end
    end))
    table.insert(self._timerList, co)
    return co
end

-- 资源管理
--=====================================================================================================

---@brief 加载资源（自动管理）
---@desc 按资源配置同步加载资源并缓存，重复加载直接返回缓存
---@param assetInfo LOAD_RES_DEF_ITEM 资源配置
---@return UnityEngine.Object 加载的资源对象
function LuaBehaviourSuper:LoadAsset(assetInfo)
    if assetInfo ~= nil then
        local keyName = assetInfo.Name
        if self._loadedAssets[keyName] == nil then
            -- 统一经 AssetUtil 加载（含异常防护与 JsonAsset 解析）
            self._loadedAssets[keyName] = AssetUtil.LoadAssetSync(assetInfo)
        end
        return self._loadedAssets[keyName]
    end
    return nil
end

---@brief 卸载单个资源
---@desc 按资源配置卸载对应缓存资源，并移除缓存记录
---@param assetInfo LOAD_RES_DEF_ITEM 资源配置
---@return boolean 是否卸载成功
function LuaBehaviourSuper:UnloadAsset(assetInfo)
    if assetInfo ~= nil then
        local keyName = assetInfo.Name
        local asset = self._loadedAssets[keyName]
        if asset ~= nil then
            -- 统一经 AssetUtil 卸载（内部含 C# 空守卫）
            AssetUtil.RemoveAsset(asset)
            self._loadedAssets[keyName] = nil
            return true
        end
    end
    return false
end

---@brief 按对象卸载资源
---@desc 遍历缓存按资源对象匹配并卸载，同时移除缓存记录
---@param asset UnityEngine.Object 资源对象
---@return boolean 是否卸载成功
function LuaBehaviourSuper:UnloadAssetByAsset(asset)
    if asset ~= nil then
        for keyName, vAsset in pairs(self._loadedAssets) do
            if vAsset == asset then
                AssetUtil.RemoveAsset(asset)
                self._loadedAssets[keyName] = nil
                return true
            end
        end
    end
    return false
end

---@brief 卸载所有资源
---@desc 遍历卸载全部缓存资源并清空缓存表
function LuaBehaviourSuper:UnloadAllAssets()
    for _, vAsset in pairs(self._loadedAssets) do
        if vAsset ~= nil then
            AssetUtil.RemoveAsset(vAsset)
        end
    end
    self._loadedAssets = {}
end
