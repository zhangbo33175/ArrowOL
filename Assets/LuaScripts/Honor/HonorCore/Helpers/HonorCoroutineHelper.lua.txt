-- @brief     协程助手（全局协程管理器 + GC 优化）
-- @desc      启动/停止/管理 Lua 协程，支持绑定 GameObject 在物体销毁时自动停协程，
--            缓存 WaitForSeconds 等待句柄以大幅减少 GC 分配，支持帧等待与真实时间等待
-- @core      xlua SystemUtil 协程生成、target 绑定追踪、等待句柄缓存池、逆序匹配停止
-- @scope     全游戏协程统一调度入口；界面销毁经 StopByGO 释放绑定协程
-- @dependency xlua SystemUtil, GameMainRoot.HonorUIManager, UnityEngine.WaitForSeconds
-- @note      依赖 xlua SystemUtil；等待对象缓存复用，避免频繁 new 等待句柄
--=====================================================================================================

--region 协程管理助手
---@class CoroutineWaitHelper @ 等待对象缓存池（减少GC）
---@field waitForSeconds table @ 普通等待缓存（按毫秒索引的 WaitForSeconds 复用队列）
---@field WaitForSecondsRealtime table @ 真实时间等待缓存（按毫秒索引的 WaitForSecondsRealtime 复用队列）
local CoroutineWaitHelper = class("CoroutineWaitHelper")

---@class HonorCoroutineHelper @ 协程管理助手
---@field _util table @ xlua SystemUtil 协程工具
---@field _coroutines table[] @ 协程列表：{target, coroutine}
---@field waitForEndOfFrame userdata @ 等待帧末句柄
---@field waitForSeconds CoroutineWaitHelper @ 等待时间缓存池
local HonorCoroutineHelper = class("HonorCoroutineHelper")

---@brief 构造函数
---@desc 加载 xlua SystemUtil，初始化协程列表与等待句柄缓存池
---@param args table 自定义参数（预留）
function HonorCoroutineHelper:ctor(args)
    self._util = require("SystemUtil")         --- xlua 协程工具
    self._coroutines = {}                      --- 协程列表：{target, coroutine}
    self.waitForEndOfFrame = UnityEngine.WaitForEndOfFrame  --- 等待帧末
    self.waitForSeconds = CoroutineWaitHelper          --- 等待时间缓存池
    self.waitForSeconds.waitForSeconds = {}            --- 普通等待缓存
    self.waitForSeconds.WaitForSecondsRealtime = {}    --- 真实时间等待缓存
end

---@brief 创建实例
---@param args table 初始化参数（可选）
---@return HonorCoroutineHelper 协程管理实例
function HonorCoroutineHelper:Create(args)
    local obj = HonorCoroutineHelper.new(args)
    return obj
end

---@brief 启动协程（自动绑定 target，方便后续销毁）
---@desc 经 Launcher 启动 cs_generator 协程，并记录 target 以便 Stop/StopByGO 释放
---@param handler table 协程处理器（含 func 与 target）
---@return UnityEngine.Coroutine 协程句柄
function HonorCoroutineHelper:Start(handler)
    local co = nil
    if GameMainRoot.HonorUIManager.Launcher then
        co = GameMainRoot.HonorUIManager:StartCoroutine(self._util.cs_generator(handler.func))
        -- 绑定 target 便于销毁
        local target = handler.target.cs ~= nil and handler.target.cs.gameObject or handler.target
        table.insert(self._coroutines, { target = target, coroutine = co })
    end
    return co
end

---@brief 独立启动协程（不绑定 target）
---@desc 不记录 target，需调用方自行持有句柄并 StopAlone
---@param func function 协程执行函数
---@return UnityEngine.Coroutine 协程句柄
function HonorCoroutineHelper:StartAlone(func)
    if GameMainRoot.HonorUIManager.Launcher then
        return GameMainRoot.HonorUIManager:StartCoroutine(self._util.cs_generator(func))
    end
    return nil
end

---@brief 停止单个独立协程
---@desc 停止 StartAlone 启动的未绑定协程
---@param co UnityEngine.Coroutine 协程句柄
function HonorCoroutineHelper:StopAlone(co)
    if not co then return end
    if GameMainRoot.HonorUIManager.Launcher then
        GameMainRoot.HonorUIManager:StopCoroutine(co)
    end
end

---@brief 停止所有协程
---@desc 调用 Launcher 停全部协程并清空追踪列表
function HonorCoroutineHelper:StopAll()
    if GameMainRoot.HonorUIManager.Launcher then
        GameMainRoot.HonorUIManager:StopAllCoroutines()
    end
    self._coroutines = {}
end

---@brief 停止指定协程
---@desc 逆序遍历追踪列表，按句柄匹配停止对应协程
---@param co UnityEngine.Coroutine 协程句柄
function HonorCoroutineHelper:Stop(co)
    if not co then return end
    self:_StopMatchingCoroutines(function(entry) return entry.coroutine == co end)
end

---@brief 根据 GameObject 停止其所有协程（界面销毁必备）
---@desc 界面销毁时按 target GameObject 释放其绑定的全部协程
---@param go UnityEngine.GameObject 目标物体
function HonorCoroutineHelper:StopByGO(go)
    if not go then return end
    self:_StopMatchingCoroutines(function(entry) return entry.target == go end)
end

---@brief 内部：逆序遍历协程列表，停止并移除所有命中匹配的协程（私有，供 Stop/StopByGO 复用）
---@param match fun(entry:table):boolean 匹配谓词，返回 true 即停止该协程
function HonorCoroutineHelper:_StopMatchingCoroutines(match)
    for i = #self._coroutines, 1, -1 do
        local entry = self._coroutines[i]
        if match(entry) then
            GameMainRoot.HonorUIManager:StopCoroutine(entry.coroutine)
            table.remove(self._coroutines, i)
        end
    end
end

---@brief 等待指定帧数
---@desc 复位 waitter 后协程 yield 该帧等待对象
---@param waitter any 帧等待对象
function HonorCoroutineHelper:WaitForFrame(waitter)
    waitter:Reset()
    coroutine.yield(waitter)
end

---@brief 等待秒数（带缓存，无GC）
---@desc 按毫秒精度取缓存池 WaitForSeconds，取出等待后回收，全程无 GC 分配
---@param waitTime number 等待秒数
function HonorCoroutineHelper:WaitForSeconds(waitTime)
    waitTime = math.floor(waitTime * 1000) / 1000  -- 精度处理
    local cache = self.waitForSeconds.waitForSeconds[waitTime]
    -- 无缓存则新建
    if not cache or #cache == 0 then
        cache = {}
        table.insert(cache, UnityEngine.WaitForSeconds(waitTime))
        self.waitForSeconds.waitForSeconds[waitTime] = cache
    end
    -- 取出、等待、回收（完全无GC）
    local waitter = table.remove(cache)
    coroutine.yield(waitter)
    table.insert(cache, waitter)
end

---@brief 等待真实秒数（不受时间缩放影响，带缓存）
---@desc 按毫秒精度取缓存池 WaitForSecondsRealtime，取出等待后回收
---@param waitTime number 等待真实秒数
function HonorCoroutineHelper:WaitForSecondsRealtime(waitTime)
    waitTime = math.floor(waitTime * 1000) / 1000
    local cache = self.waitForSeconds.WaitForSecondsRealtime[waitTime]
    if not cache or #cache == 0 then
        cache = {}
        table.insert(cache, UnityEngine.WaitForSecondsRealtime(waitTime))
        self.waitForSeconds.WaitForSecondsRealtime[waitTime] = cache
    end
    local waitter = table.remove(cache)
    waitter:Reset()
    coroutine.yield(waitter)
    table.insert(cache, waitter)
end
--endregion

return HonorCoroutineHelper
