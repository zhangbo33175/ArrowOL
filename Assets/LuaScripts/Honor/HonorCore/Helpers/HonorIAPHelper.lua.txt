--region 内购（IAP）管理助手 | 游戏支付统一入口，自动分发官方/第三方支付
---@class HonorIAPHelper @ 内购（IAP）管理助手，游戏支付统一入口，自动分发官方/第三方支付
---@field _buyID number @ 当前正在购买的商品配置表ID
---@field _customGoodsListUI table @ 第三方支付商品列表UI实例
---@field _customPayUI table @ 第三方支付网页弹窗UI实例
---@field _debugIP string @ 测试环境专用IP，用于内网调试支付
local HonorIAPHelper = class("HonorIAPHelper")

---@brief 构造函数
--- 初始化内购相关临时变量，存储当前支付上下文
---@param args table 自定义参数（预留）
--TODO: [优化] 补齐 ctor 的 @param 注解（签名不变）
function HonorIAPHelper:ctor(args)
    self._buyID = nil                  ---@type number 当前正在购买的商品配置表ID
    self._customGoodsListUI = nil     ---@type table 第三方支付商品列表UI实例
    self._customPayUI = nil           ---@type table 第三方支付网页弹窗UI实例
    self._debugIP = nil               ---@type string 测试环境专用IP，用于内网调试支付
end

---@brief 创建内购助手实例
---@param args table 可选初始化参数
---@return HonorIAPHelper 内购管理实例
function HonorIAPHelper:Create(args)
    local obj = HonorIAPHelper.new(args)
    return obj
end

---@brief C# 回调：IAP SDK 初始化完成
---@desc 由 C# IAP SDK 反向调用；初始化结果全局派发 IAPInitFinish 事件
---@param success boolean 是否初始化成功
---@param failReason string 失败原因（失败时有效）
function HonorIAPHelper:OnIAPInitFinish(success, failReason)
    local eventArgs = {Success = success, FailReason = failReason}
    HonorEventHelper:Fire(FwEventCmd.IAPInitFinish, eventArgs)
end

---@brief 获取【第三方支付】全局开关状态
---@desc 读取 GM 后台 THIRD_IAP_OPEN 配置，决定走官方商店还是第三方网页支付
---@return boolean true=开启第三方支付，false=关闭（仅走官方商店支付）
function HonorIAPHelper:GetThirdPayOpenState()
    return HonorConfigHelper:GetSystemConfig("THIRD_IAP_OPEN")
end

---@brief 开始购买商品（自动选择：官方支付 / 第三方支付）
---@desc 第三方开关开启且非 Amazon 平台 → 走第三方网页支付；否则走官方商店支付
---@param tableID number 商品配置表ID（Purchases表）
---@param customTable table 自定义透传参数（服务器/订单扩展字段）
---@param customGoodsListUI table 第三方支付商品列表UI
---@param customPayUI table 第三方支付网页UI
---@param debugIP string 调试用IP（测试环境）
---@return boolean true=使用第三方支付，false=使用官方Google/iOS支付
function HonorIAPHelper:BuyThirdPayGood(tableID, customTable, customGoodsListUI, customPayUI, debugIP)
    self._buyID = tableID
    self._customGoodsListUI = customGoodsListUI
    self._customPayUI = customPayUI
    self._debugIP = debugIP

    -- 第三方支付开启 && 不是Amazon平台 → 使用第三方网页支付
    if HonorConfigHelper:GetSystemConfig("THIRD_IAP_OPEN") == true and HonorRoot.Launcher.IsAmazonStore == false then
        GameMainRoot.SDK.IAPHelper:PayThird(tableID, customTable ~= nil and TableEncode(customTable) or nil, self._debugIP)
        return true
    else
        -- 走官方应用商店支付（Google / iOS / Amazon）
        GameMainRoot.SDK.IAPHelper:PayNormal(tableID, customTable, nil)
        return false
    end
end

---@brief 第三方支付：主动创建新订单
---@desc 透传商品/渠道/类型参数至 C# SDK 创建第三方订单
---@param tableID number 商品配置表ID
---@param channelID number 支付渠道ID（微信/支付宝等）
---@param typeInfoID number 商品类型ID
function HonorIAPHelper:GotoThirdCreateNewTransaction(tableID, channelID, typeInfoID)
    HonorRoot.SDK.IAPHelper:CreateThirdTransaction(tableID, channelID, typeInfoID, self._debugIP)
end

---@brief 第三方支付：从WebView查询订单状态
---@desc 由 WebView 回调触发，转发至 C# SDK 查询网络侧订单状态
---@param tableID number 商品配置表ID
---@param orderID string 订单号
---@param orderStatus number 订单状态（自定义枚举）
function HonorIAPHelper:GetThirdPayStatusFromWebView(tableID, orderID, orderStatus)
    HonorRoot.SDK.IAPHelper:GetTransactionStatusFromNetView(tableID)
end

---@brief 订单支付【成功】→ 全局派发成功事件，业务层发放奖励
---@desc C# SDK 反向调用；服务器校验通过后派发 IAPPaySuccess，业务层据此发奖
---@param tableID number 商品配置表ID
---@param isThird boolean 是否为第三方支付
---@param canDeliver boolean 是否可发放道具（服务器校验通过）
---@param customTable table 自定义透传参数
---@param SUBData table 订阅商品信息（非订阅为nil）
function HonorIAPHelper:OnTransaction_Success(tableID, isThird, canDeliver, customTable, SUBData)
    local returnData = {
        tableID = tableID,
        isThird = isThird,
        customTable = customTable,
        canDeliver = canDeliver,
        SUBData = SUBData
    }
    HonorEventHelper:FireNow(FwEventCmd.IAPPaySuccess, returnData)
end

---@brief 订单支付【失败】→ 全局派发失败事件，业务层提示用户
---@desc C# SDK 反向调用；派发 IAPPayFail，业务层提示取消/网络错误/支付失败
---@param tableID number 商品配置表ID
---@param isThird boolean 是否为第三方支付
---@param customTable table 自定义透传参数
---@param failReason string 失败原因（取消/网络错误/支付失败）
function HonorIAPHelper:OnTransaction_Fail(tableID, isThird, customTable, failReason)
    local returnData = {
        tableID = tableID,
        isThird = isThird,
        customTable = customTable,
        failReason = failReason
    }
    HonorEventHelper:FireNow(FwEventCmd.IAPPayFail, returnData)
end

---@brief 订阅恢复完成（换设备/重装游戏恢复会员/订阅商品）
---@desc C# SDK 反向调用；换设备/重装后恢复订阅，派发 IAPRestoreFinish
---@param success boolean 恢复是否成功
---@param gameUIDStr string 玩家UID
---@param SUBInfo table 订阅详细信息
---@param isForceRestore boolean 是否强制恢复
function HonorIAPHelper:OnSubscriptionRestoreFinish(success, gameUIDStr, SUBInfo, isForceRestore)

        success = success,
        uid = gameUIDStr,
        SUBInfo = SUBInfo,
        isForceRestore = isForceRestore
    }
    HonorEventHelper:FireNow(FwEventCmd.IAPRestoreFinish, restoreData)
end

---@brief 获取商品类型（普通消耗品 / 非消耗品 / 订阅）
---@desc 查 Tables.Purchases 配置表 ProductType，经 IAPWrapper 枚举转换
---@param tableID number 商品配置表ID
---@return IAPProductType 商品类型枚举（Consumable/NonConsumable/Subscription）
function HonorIAPHelper:GetProductType(tableID)
    if Tables.Purchases[tableID] ~= nil then
        return IAPWrapper.ProductType.__CastFrom(tonumber(Tables.Purchases[tableID].ProductType))
    end
    return nil
end
