-- 2. 码值由 UniqueEventCode() 生成全局唯一，杜绝硬编码与冲突
-- 3. C# 层通过 EventCodes.X 读取码值做消息匹配
-- 架构：全局表 EventCodes，依赖 SerialProducer 的 UniqueEventCode 生成器
-- @note  须在 SerialProducer 之后加载；本文件 return EventCodes 供 require 引用
--=====================================================================================================

-- 缓存全局唯一码生成器为局部引用（避免事件定义阶段频繁全局查找）
local UniqueEventCode = UniqueEventCode

--region 全局通用事件 C# <-> Lua
---@class EventCodes 全局事件定义（C# <-> Lua 互通）
EventCodes = {
    --- 服务器时间校正事件
    --- 参数：IsTimeCorrectionByServer(boolean)
    --- true  = 服务器校正
    --- false = 客户端校正
    S_OnTimeCorrectionByServer = UniqueEventCode(),

    --- 全局通知消息回调
    --- 用途：飘字、提示、系统通知
    NotificationMessage = UniqueEventCode(),
}
--endregion

return EventCodes
