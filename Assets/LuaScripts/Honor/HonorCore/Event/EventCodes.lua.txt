    --- 参数：IsTimeCorrectionByServer(boolean)
    --- true  = 服务器校正

    S_OnTimeCorrectionByServer = UniqueEventCode(),

    --- 全局通知消息回调
    --- 用途：飘字、提示、系统通知
    NotificationMessage = UniqueEventCode(),
}
--endregion
