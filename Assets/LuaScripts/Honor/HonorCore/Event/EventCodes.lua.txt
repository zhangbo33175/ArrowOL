-- @desc      提供跨语言通用事件码（如服务器时间校正、全局通知回调）；码值由 UniqueEventCode() 生成全局唯一，杜绝硬编码与冲突；C# 层通过 EventCodes.X 读取码值做消息匹配
-- @core      全局表 EventCodes，每项经 UniqueEventCode() 自增赋唯一数字码
-- @scope     C# <-> Lua 跨语言事件码统一定义；由 C# 侧读取做消息匹配
-- @dependency SerialProducer 的 UniqueEventCode（全局函数）
-- @note  须在 SerialProducer 之后加载；本文件 return EventCodes 供 require 引用
--=====================================================================================================

-- 缓存全局唯一码生成器为局部引用（避免事件定义阶段频繁全局查找）
local UniqueEventCode = UniqueEventCode

--region 全局通用事件 C# <-> Lua
---@class EventCodes 全局事件定义（C# <-> Lua 互通）
EventCodes = {
    --- 服务器时间校正事件
    --- 参数：IsTimeCorrectionByServer(boolean)
    --- true  = 服务器校正
    --- false = 客户端校正
    S_OnTimeCorrectionByServer = UniqueEventCode(),

    --- 全局通知消息回调
    --- 用途：飘字、提示、系统通知
    NotificationMessage = UniqueEventCode(),
}
--endregion

return EventCodes
