-- @core      模块替换、upvalue/数据保留、OnReload 回调、单例/类热更

--region Lua 热重载核心（运行时代码热更）
---@brief 主入口：Lua 脚本热重载
--- 功能：重新加载指定 Lua 文件，替换旧代码，保留运行时数据，实现无重启更新
---@param filename string 需要热重载的 Lua 文件名（模块名）
function __RELOAD_LUA_HOTFIX__(filename)
    PrintDebug("开始热重载Lua脚本: {0}", filename)

    -- 1. 取出旧模块（如果已经加载过）
    local oldModule
    if package.loaded[filename] then
        oldModule = package.loaded[filename]  -- 保存旧模块
        package.loaded[filename] = nil        -- 清空缓存，让 require 重新加载
    else
        PrintDebug("该Lua脚本从未加载过: {0}",filename)
        return
    end

    -- 2. 尝试重新加载新模块
    local ok, err = pcall(require, filename)
    if not ok then
        -- 加载失败：恢复旧模块，避免崩溃
        package.loaded[filename] = oldModule
        PrintDebug("热重载Lua脚本失败，错误信息：", err)
        return
    end

    -- 3. 获取刚加载的新模块
    local newModule = package.loaded[filename]

    -- 4. 深度递归更新：把新模块的函数/table 同步到旧模块
    --    保留旧模块的所有运行时数据，只替换逻辑
    ---TODO 规范命名：updated_tables/old_upvalue_map/old_value/type_value/old_meta/new_meta/new_func/old_func/new_table/old_table 等蛇形局部与参数改为驼峰（全部文件内私有，逻辑等价）
    local updatedTables = {}
    __RELOAD_LUA_HOTFIX_UPDATE_TABLE__(newModule, oldModule, updatedTables)

    -- 5. 如果旧模块有 OnReload 方法，执行重载回调
    if "table" == type(oldModule) and oldModule.OnReload ~= nil then
        oldModule:OnReload()
    end

    -- 6. 恢复旧模块到 package.loaded（保证外部引用不变）
    PrintDebug("Lua脚本替换成功！")
    package.loaded[filename] = oldModule
end

---@brief 更新 Lua 函数（热重载核心）
--- 功能：替换函数体，**保留旧函数的 upvalue（外部变量）**，保证数据不丢失
---@param newFunc function 新代码的函数
---@param oldFunc function 旧的运行中函数
function __RELOAD_LUA_HOTFIX_UPDATE_FUNC__(newFunc, oldFunc)
    if "function" == type(newFunc) and "function" == type(oldFunc) then
        -- 第一步：收集旧函数的所有 upvalue（外部引用变量）
        local oldUpvalueMap = {}
        for i = 1, math.huge do
            local name, value = debug.getupvalue(oldFunc, i)
            if not name then break end
            oldUpvalueMap[name] = value
        end

        -- 第二步：把旧的 upvalue 赋值给新函数
        -- 作用：新代码可以继续使用旧的运行数据，不会重置
        for i = 1, math.huge do
            local name, value = debug.getupvalue(newFunc, i)
            if not name then break end
            PrintDebug("设置upvalue:{0}", name)
            local oldValue = oldUpvalueMap[name]
            if oldValue then
                debug.setupvalue(newFunc, i, oldValue)
            end
        end
    end
end

---@brief 递归更新 Lua table（模块、类、子 table）
--- 功能：遍历新旧 table，只替换 function，保留旧数据，递归处理嵌套 table
---@param newTable table 新代码的 table
---@param oldTable table 旧的运行中 table
---@param updatedTables table 防止循环引用，记录已更新的 table
function __RELOAD_LUA_HOTFIX_UPDATE_TABLE__(newTable, oldTable, updatedTables)
    if "table" == type(newTable) and "table" == type(oldTable) then
        -- 遍历新 table 的所有 key，同步到旧 table
        for key, value in pairs(newTable) do
            local oldValue = oldTable[key]
            local typeValue = type(value)

            if typeValue == "function" then
                -- 函数：替换并保留 upvalue
                __RELOAD_LUA_HOTFIX_UPDATE_FUNC__(value, oldValue)
                oldTable[key] = value

            elseif typeValue == "table" then
                -- 子表：递归更新，避免循环引用
                if updatedTables[value] == nil then
                    updatedTables[value] = true
                    __RELOAD_LUA_HOTFIX_UPDATE_TABLE__(value, oldValue, updatedTables)
                end
            end
        end

        -- 最后更新元表（保证类的 __index / 方法也被热更）
        local oldMeta = debug.getmetatable(oldTable)
        local newMeta = debug.getmetatable(newTable)
        if type(oldMeta) == "table" and type(newMeta) == "table" then
            __RELOAD_LUA_HOTFIX_UPDATE_TABLE__(newMeta, oldMeta, updatedTables)
        end
    end
end
