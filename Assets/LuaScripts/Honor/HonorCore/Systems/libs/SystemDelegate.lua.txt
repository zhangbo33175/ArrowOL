-- 2. 支持 once 一次性监听与带参广播
-- 架构：基于 SystemSet 维护回调集合
-- @note  依赖 SystemSet，需先加载无序集合模块
--=====================================================================================================

--region 多播委托（C# 风格 Delegate）
local SystemSet = require("SystemSet")

---@class SystemDelegate 多播委托（C# 风格 Delegate）
local SystemDelegate = {}
SystemDelegate.__index = SystemDelegate

--TODO: [优化] 修正 __call 文档形参名为 t（与 function(t, ...) 一致），运行签名不变
---@brief 委托实例直接调用：delegate(...) 触发所有监听
---@param t Delegate 委托实例
---@param ... 事件参数
SystemDelegate.__call = function(t, ...)
    -- 遍历所有监听器并执行
    for k,_ in pairs(t._listeners._data) do
        k(...)
    end
end

---@brief 创建一个新的代理器实例
---@return Delegate 代理器实例
function SystemDelegate.new()
    ---@class Delegate 代理器实例
    ---@field _listeners SystemSet 监听器集合（保证唯一、快速增删）
    local tb = {}
    tb._listeners = SystemSet.new()
    return setmetatable(tb, SystemDelegate)
end

---@brief 添加监听函数（自动去重，重复添加会报错）
---@param self Delegate 代理器实例
---@param f function 要注册的回调函数
function SystemDelegate.add(self, f)
    -- 插入失败说明已存在，assert 强制抛出警告
    assert(SystemSet.insert(self._listeners, f))
end

---@brief 移除监听函数
---@param self Delegate 代理器实例
---@param f function 要取消注册的回调函数
---@return boolean 成功返回true，失败返回false
function SystemDelegate.remove(self, f)
    return SystemSet.remove(self._listeners, f)
end

---@brief 判断是否包含某个监听函数
---@param self Delegate 代理器实例
---@param f function 查询的回调函数
---@return boolean 包含返回true，否则false
function SystemDelegate.has(self, f)
    return self._listeners:has(f)
end
--endregion

return SystemDelegate
