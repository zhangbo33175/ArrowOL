-- @brief     序列号生成器（全局唯一码 / 唯一ID / 唯一名生成函数）
-- @desc      UniqueEventCode / UniqueID 自增生成全局唯一数字码；UniqueName 生成「前缀 + 自增序号」唯一名；FunctionID / FunctionNegativeID / OrnamentID 自减生成负向功能标识
-- @core      文件级闭包自增 / 自减计数器，全局函数暴露为工程唯一来源
-- @scope     全局函数集合（全局注入，不 return 模块表属设计）；由 StartGame require 时生效，供 EventCodes / UIEventCode 等引用
-- @dependency 无外部依赖（纯自增 / 自减计数器）
-- @note  全局函数无同名冲突；须先于 EventCodes / UIEventCode 加载
--=====================================================================================================

--region 全局唯一事件码
---TODO 规范命名：文件级计数器 unique_eventCode 改为驼峰 uniqueEventCode（私有局部，逻辑等价）
local uniqueEventCode = 1
---@brief 生成全局唯一事件码（自增，从 2 开始）
---@type fun():number
---@return number 唯一事件码
function UniqueEventCode()
    uniqueEventCode = uniqueEventCode + 1
    return uniqueEventCode
end
--endregion

--region 全局唯一ID
---TODO 规范命名：文件级计数器 unique_id 改为驼峰 uniqueId（私有局部，逻辑等价）
local uniqueId = 1
---@brief 生成全局唯一ID（自增，从 2 开始）
---@type fun():number
---@return number 唯一ID
function UniqueID()
    uniqueId = uniqueId + 1
    return uniqueId
end
--endregion

--region 全局唯一名称
---TODO 规范命名：文件级计数器 unique_name_id 改为驼峰 uniqueNameId（私有局部，逻辑等价）
local uniqueNameId = 1
---@brief 生成全局唯一名称（名称前缀 + 自增序号）
---@type fun(name:string):string
---@param name string 名称前缀
---@return string 唯一名称
function UniqueName(name)
    uniqueNameId = uniqueNameId + 1
    return string.format('%s%s', tostring(name), tostring(uniqueNameId))
end
--endregion

--region 功能负ID（自减，负数）
---TODO 规范命名：文件级计数器 function_negative_id 改为驼峰 functionNegativeId（私有局部，逻辑等价）
local functionNegativeId = 0
---@brief 生成功能负ID（每次自减 1，用于负向标识）
---@type fun():number
---@return number 负ID
function FunctionNegativeID()
    functionNegativeId = functionNegativeId - 1
    return functionNegativeId
end
--endregion

--region 功能ID（自减，负数）
---TODO 规范命名：文件级计数器 function_id 改为驼峰 functionId（私有局部，逻辑等价）
local functionId = 0
---@brief 生成功能ID（每次自减 1，用于功能唯一标识）
---@type fun():number
---@return number 功能ID
function FunctionID()
    functionId = functionId - 1
    return functionId
end
--endregion

--region 装饰物体唯一ID（自减，从 -2 开始）

---@brief 生成装饰物体唯一ID（每次自减 1，从 -2 开始）
---@type fun():number
---@return number 装饰物体ID
function OrnamentID()
    ornamentId = ornamentId - 1
    return ornamentId
end
--endregion
