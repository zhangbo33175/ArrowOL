-- descrip:   单种 Prefab 的 GameObject 对象池（复用游戏对象，减少实例化/销毁与 GC 开销）
-- 功能：
-- 1. Prewarm 预热预创建、Get 取用（无空闲则新建）、Put 回收入空闲队列
-- 2. 空闲队列(HonorQueue)+使用中列表(HonorList)双结构管理，取回时执行 resetHandler 重置
-- 3. RecycleAll 一键回收活跃对象、DestroyUnused 仅销毁空闲、Destroy 整池释放
-- 架构：class("HonorPool") + HonorQueue/HonorList 组合容器 + UnityUtil 空判与 C# 资源创建桥接
-- @note  由 HonorPoolHelper/HonorAssetManager 统一创建管理；父节点为空时拒绝回收，取出对象丢失会容错重建。
--=====================================================================================================

--region GameObject 对象池（性能优化核心）
---@class HonorPool 单个GameObject对象池（管理一种Prefab的复用）
---@field name string 池名称（唯一标识）
---@field unusedQueue HonorQueue 空闲对象队列（等待复用，隐藏状态）
---@field usedList HonorList 正在使用的对象列表（活跃状态）
---@field resDefItem LOAD_RES_DEF_ITEM 资源配置表项（路径、类型、预加载数量）
---@field preParent UnityEngine.Transform 对象池父节点（统一管理空闲对象）
---@field resetHandler function 对象取出时的重置函数（重置状态、位置、动画等）
local HonorPool = class("HonorPool")

--region 依赖引入
-- 框架内自洽：空判断走 HonorCore/Utils 同域工具，避免依赖业务层 GameFunctions 全局
local UnityUtil = require('UnityUtil')
--endregion

--region 内部工具方法
---@brief 倒序取出并销毁全部空闲对象（Destroy / DestroyUnused 逐字相同块，文件内私有去重）
---@param self HonorPool 对象池实例
local function DestroyAllUnused(self)
    for _ = self.unusedQueue:Count(), 1, -1 do
        local go = self.unusedQueue:Dequeue()
        DestroyPrefabWithGO(go)
    end
end

---@brief 更新池父物体名称（实时显示空闲数量，Put/Get/DestroyUnused 三处复用）
---@param self HonorPool 对象池实例
local function UpdatePoolDisplayName(self)
    self.preParent.gameObject.name = "HonorPool:" .. self.name .. "(" .. self.unusedQueue:Count() .. ")"
end
--endregion

---@brief 构造函数
---@param name string 对象池唯一名称
---@param resDefItem LOAD_RES_DEF_ITEM 资源配置数据
---@param preParent UnityEngine.Transform 空闲对象父节点
---@param resetHandler function 从对象池取出对象时的重置逻辑
function HonorPool:ctor(name, resDefItem, preParent, resetHandler)
    self.name = name
    self.unusedQueue = require("HonorQueue"):Create()
    self.usedList = require("HonorList"):Create()
    self.resDefItem = resDefItem
    self.preParent = preParent
    self.resetHandler = resetHandler
    -- 更新父物体名称，实时显示空闲数量
    UpdatePoolDisplayName(self)
end

---@brief 创建对象池实例（工厂方法）
---@return HonorPool 对象池实例
function HonorPool:Create(name, resDefItem, preParent, resetHandler)
    local obj = HonorPool.new(name, resDefItem, preParent, resetHandler)
    return obj
end

---@brief 预热对象池（预创建指定数量对象，避免游戏运行中卡顿）
---@param count number 预创建的对象数量
function HonorPool:Prewarm(count)
    for _ = 1, count do
        local go = CreatePrefabGOSync(self.resDefItem, self.preParent)
        self.usedList:Add(go)
        -- 预创建后直接放入空闲队列
        self:Put(go)
    end
end

---@brief 回收对象（将使用中的对象放回对象池）
---@param go UnityEngine.GameObject 要回收的游戏对象
function HonorPool:Put(go)
    if UnityUtil.IsNull(self.preParent) then
        PrintWarning("对象池【" .. self.name .. "】的父节点已为空，无法回收对象！")
        return
    end
    if not UnityUtil.IsNull(go) then
        -- 仅回收当前对象池管理的对象
        if self.usedList:Contains(go) then
            self.usedList:Remove(go)
            self.unusedQueue:Enqueue(go)
            -- 放回池父节点并隐藏
            go.transform:SetParent(self.preParent)
            go:SetActive(false)
            -- 更新名称显示空闲数量
            UpdatePoolDisplayName(self)
        else
            PrintWarning("待回收对象不属于对象池【" .. self.name .. "】，请检查业务逻辑")
        end
    end
end

---@brief 获取对象（无空闲则创建，有空闲则复用）
---@param newParent UnityEngine.Transform 新父节点（不传则使用池默认父节点）
---@return UnityEngine.GameObject 可用的游戏对象
function HonorPool:Get(newParent)
    if UnityUtil.IsNull(newParent) then
        newParent = self.preParent
    end

    local go = nil
    -- 无空闲对象，创建新对象
    if self.unusedQueue:Count() == 0 then
        go = CreatePrefabGOSync(self.resDefItem, newParent)
    else
        -- 从空闲队列取出复用
        go = self.unusedQueue:Dequeue()
        -- 容错：对象丢失则重新创建
        if UnityUtil.IsNull(go) then
            PrintError("对象池【" .. self.name .. "】获取对象为空，执行容错创建！")
            go = CreatePrefabGOSync(self.resDefItem, newParent)
        end
        go.transform:SetParent(newParent)
    end

    if UnityUtil.IsNull(go) then
        PrintError("对象池【" .. self.name .. "】获取对象彻底失败，请检查资源配置！")
        return nil
    end

    -- 加入使用中列表并激活
    self.usedList:Add(go)
    go:SetActive(true)
    -- 执行对象重置逻辑
    if self.resetHandler then
        self.resetHandler(go)
    end
    -- 更新空闲数量显示
    UpdatePoolDisplayName(self)
    return go
end

---@brief 回收所有正在使用的对象（一键回收）
function HonorPool:RecycleAll()
    local count = self.usedList:Count()
    -- 倒序遍历安全回收
    for i = count, 1, -1 do
        local item = self.usedList:Get(i)
        self:Put(item)
    end
    self.usedList:Clear()
end

---@brief 销毁整个对象池（释放全部内存，场景切换时调用）
function HonorPool:Destroy()
    -- 先回收所有活跃对象
    self:RecycleAll()
    -- 销毁所有空闲对象
    DestroyAllUnused(self)
    -- 清空数据
    self.unusedQueue:Clear()
    self.usedList:Clear()
    self.resDefItem = nil
    -- 销毁池根节点
    if not UnityUtil.IsNull(self.preParent) then
        Unity.GameObject.Destroy(self.preParent.gameObject)
        self.preParent = nil
    end
end

---@brief 仅销毁空闲对象（释放闲置内存，不影响正在使用的对象）
function HonorPool:DestroyUnused()
    DestroyAllUnused(self)
    self.unusedQueue:Clear()
    -- 更新数量显示
    UpdatePoolDisplayName(self)
end
--endregion

return HonorPool
