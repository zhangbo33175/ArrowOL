-- @brief     通用队列容器（FIFO 先进先出，尾部入队、头部出队）
-- @desc      Enqueue 队尾入队、Dequeue 队首出队、Peek 仅查看队首；Count/IsEmpty 状态查询、Clear 清空；空队列安全返回 nil
-- @core      class 封装内部 items 原生数组；table.remove 队首出队
-- @scope     通用任务 / 消息 / 网络请求队列，及对象池空闲队列（HonorPool.unusedQueue）复用
-- @dependency 无外部依赖（仅内部 items 原生数组）
-- @note  出队采用 table.remove 队首，元素量大时性能一般，按需选用。
--=====================================================================================================

--region 通用队列容器（FIFO 先进先出）
---@class HonorQueue 通用队列容器（FIFO 先进先出）
---@field items table 内部存储元素的数组
local HonorQueue = class("HonorQueue")

---@brief 构造函数
function HonorQueue:ctor()
    ---@type table 队列元素数组
    self.items = {}
end

---@brief 创建队列实例
---@return HonorQueue 新队列对象
function HonorQueue:Create()
    local queue = HonorQueue.new()
    return queue
end

---@brief 获取队首元素（仅查看，不移除）
---@return any 队首元素，空队列返回 nil
function HonorQueue:Peek()
    return self.items[1]
end

---@brief 元素入队（添加到队尾）
---@param item any 任意类型要入队的元素
function HonorQueue:Enqueue(item)
    table.insert(self.items, item)
end

---@brief 元素出队（从队首移除并返回）
---@return any 出队的元素，空队列返回 nil
function HonorQueue:Dequeue()
    if self:Count() <= 0 then
        return nil
    end
    local item = self:Peek()
    table.remove(self.items, 1)
    return item
end

---@brief 获取队列元素数量
---@return number 队列元素个数
function HonorQueue:Count()
    return #self.items
end

---@brief 判断队列是否为空
---@return boolean true=空, false=非空
function HonorQueue:IsEmpty()
    return self:Count() == 0
end

---@brief 清空整个队列
function HonorQueue:Clear()
    self.items = {}
end
--endregion

return HonorQueue
