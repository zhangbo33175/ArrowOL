-- @brief     通用动态列表容器（仿 C# List<T>，有序存储一组元素）
-- @desc      Add/Get/Set/Remove/RemoveAt 有序增删改查（索引从 1 开始）；Count/IsEmpty/Contains 状态查询；Clear 清空、ToArray 导出原生数组、ForEach 安全遍历
-- @core      class 封装内部 items 原生数组；Contains 复用 TableUtil
-- @scope     框架 / 业务通用顺序容器；被 SingletonManager、对象池（HonorPool.usedList）等模块组合复用
-- @dependency TableUtil
-- @note  遍历过程中不建议删除元素；仅作顺序容器，不保证跨帧安全。
--=====================================================================================================

--region 通用动态列表容器（仿 C# List）
local TableUtil = require("TableUtil")

---@class HonorList 通用动态列表容器（仿 C# List）
---@field items table 内部存储元素的原生数组
local HonorList = class("HonorList")

---@brief 构造函数
function HonorList:ctor()
    ---@type table 内部元素数组
    self.items = {}
end

---@brief 创建列表实例（静态工厂方法）
---@return HonorList 新的列表实例
function HonorList:Create()
    local list = HonorList.new()
    return list
end

---@brief 添加元素到列表末尾
---@param item any 要添加的任意类型元素
function HonorList:Add(item)
    table.insert(self.items, item)
end

---@brief 获取列表中元素的总数量
---@return number 元素个数
function HonorList:Count()
    return #self.items
end

---@brief 判断列表是否为空
---@return boolean true=空, false=非空
function HonorList:IsEmpty()
    return self:Count() == 0
end

---@brief 判断列表是否包含指定元素
---@param item any 要查找的元素
---@return boolean 是否存在
function HonorList:Contains(item)
    return TableUtil.ContainsValue(self.items, item)
end

---@brief 根据索引获取元素
---@param index number 索引（从 1 开始）
---@return any 对应位置的元素，索引越界返回 nil
function HonorList:Get(index)
    return self.items[index]
end

---@brief 根据索引设置元素
---@param index number 索引（从 1 开始）
---@param item any 要设置的元素
function HonorList:Set(index, item)
    if index >= 1 and index <= self:Count() then
        self.items[index] = item
    end
end

---@brief 移除指定的元素（第一个匹配项）
---@param item any 要移除的元素
---@return boolean 是否移除成功
function HonorList:Remove(item)
    for i = 1, self:Count() do
        if self.items[i] == item then
            table.remove(self.items, i)
            return true
        end
    end
    return false
end

---@brief 移除指定索引位置的元素
---@param index number 索引（从 1 开始）
---@return boolean 是否移除成功
function HonorList:RemoveAt(index)
    if index < 1 or index > self:Count() then
        return false
    end
    table.remove(self.items, index)
    return true
end

---@brief 清空列表所有元素
function HonorList:Clear()
    self.items = {}
end

---@brief 转换为 Lua 原生数组
---@return table 原生数组
function HonorList:ToArray()
    local array = {}
    for i = 1, self:Count() do
        array[i] = self.items[i]
    end
    return array
end

---@brief 遍历列表（安全迭代，不建议在遍历中删除）
---@param action fun(index:number, item:any) 遍历回调
function HonorList:ForEach(action)
    if type(action) ~= "function" then return end
    for i = 1, self:Count() do
        action(i, self.items[i])
    end
end
--endregion

return HonorList
