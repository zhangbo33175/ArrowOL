-- descrip:   通用栈容器（LIFO 后进先出，压栈/弹栈/栈顶窥视）
-- 功能：
-- 1. Push 压栈顶、Pop 弹出栈顶、Peek 仅查看栈顶不弹出
-- 2. Count/IsEmpty/Contains 状态查询（Contains 复用 TableUtil）
-- 3. Clear 清空，适用于 UI 面板栈/返回栈/状态回溯与撤销
-- 架构：class("HonorStack") 继承 + 内部 items 原生数组封装（依赖 TableUtil）
-- @note  空栈 Pop/Peek 安全返回 nil；不支持弱引用，仅作后进先出容器。
--=====================================================================================================

--region 通用栈容器（LIFO 后进先出）
local TableUtil = require("TableUtil")

---@class HonorStack 通用栈容器（LIFO 后进先出）
---@field items table 内部存储元素的数组
local HonorStack = class("HonorStack")

---@brief 构造函数
function HonorStack:ctor()
    ---@type table 栈元素数组
    self.items = {}
end

---@brief 创建栈实例
---@return HonorStack 新的栈对象
function HonorStack:Create()
    local obj = HonorStack.new()
    return obj
end

---@brief 获取栈顶元素（仅查看，不弹出）
---@return any 栈顶元素，空栈返回 nil
function HonorStack:Peek()
    if self:IsEmpty() then
        return nil
    end
    return self.items[#self.items]
end

---@brief 元素入栈（压入栈顶）
---@param item any 任意类型要压入的元素
function HonorStack:Push(item)
    table.insert(self.items, item)
end

---@brief 元素出栈（弹出栈顶并返回）
---@return any 出栈元素，空栈返回 nil
function HonorStack:Pop()
    if self:IsEmpty() then
        return nil
    end
    local item = table.remove(self.items)
    return item
end

---@brief 获取栈内元素数量
---@return number 元素个数
function HonorStack:Count()
    return #self.items
end

---@brief 判断栈是否为空
---@return boolean true=空, false=非空
function HonorStack:IsEmpty()
    return self:Count() == 0
end

---@brief 判断栈是否包含指定元素
---@param item any 要查找的元素
---@return boolean 是否包含
function HonorStack:Contains(item)
    return TableUtil.ContainsValue(self.items, item)
end

---@brief 清空整个栈
function HonorStack:Clear()
    self.items = {}
end
--endregion

return HonorStack
