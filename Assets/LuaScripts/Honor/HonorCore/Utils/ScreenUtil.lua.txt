-- descrip:   屏幕边界工具（相机视口世界坐标四向边界计算）
-- 功能：
-- 1. GetScreenBorderLeft/RightWorldX 取左右世界 X 边界
-- 2. GetScreenBorderTop/BottomWorldY 取上下世界 Y 边界
-- 架构：纯工具表，依赖 UnityEngine/GameMainRoot.Scene 与同域 UnityUtil
-- @note  依据场景相机视口推算世界坐标边界
--=====================================================================================================

local ScreenUtil = {}

--region 依赖引入
-- 框架内自洽：空判断走同域工具，避免依赖业务层 GameFunctions 全局
local UnityUtil = require('UnityUtil')
--endregion

--region 屏幕边界计算

--=====================================================================================================
-- 屏幕边界计算：获取相机视口世界坐标
--=====================================================================================================

---@brief 获取屏幕左边界世界坐标 X
---@param sceneCamera UnityEngine.Camera 场景相机
---@return number 左边界世界X坐标
function ScreenUtil.GetScreenBorderLeftWorldX(sceneCamera)
    if UnityUtil.IsNull(sceneCamera) then
        sceneCamera = GameMainRoot.Scene.SceneCameras[0]
    end
    return sceneCamera:ViewportToWorldPoint(Vector3(0, 0, 0)).x
end

---@brief 获取屏幕右边界世界坐标 X
---@param sceneCamera UnityEngine.Camera 场景相机
---@return number 右边界世界X坐标
function ScreenUtil.GetScreenBorderRightWorldX(sceneCamera)
    if UnityUtil.IsNull(sceneCamera) then
        sceneCamera = GameMainRoot.Scene.SceneCameras[0]
    end
    return sceneCamera:ViewportToWorldPoint(Vector3(1, 0, 0)).x
end

---@brief 获取屏幕上边界世界坐标 Y
---@param sceneCamera UnityEngine.Camera 场景相机
---@return number 上边界世界Y坐标
function ScreenUtil.GetScreenBorderTopWorldY(sceneCamera)
    if UnityUtil.IsNull(sceneCamera) then
        sceneCamera = GameMainRoot.Scene.SceneCameras[0]
    end
    return sceneCamera:ViewportToWorldPoint(Vector3(0, 1, 0)).y
end

---@brief 获取屏幕下边界世界坐标 Y
---@param sceneCamera UnityEngine.Camera 场景相机
---@return number 下边界世界Y坐标
function ScreenUtil.GetScreenBorderBottomWorldY(sceneCamera)
    if UnityUtil.IsNull(sceneCamera) then
        sceneCamera = GameMainRoot.Scene.SceneCameras[0]
    end
    return sceneCamera:ViewportToWorldPoint(Vector3(0, 0, 0)).y
end

--endregion

return ScreenUtil
