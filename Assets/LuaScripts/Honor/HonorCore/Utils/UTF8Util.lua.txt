---@class UTF8Util UTF8 工具类
local UTF8Util = {}

---@brief 获取utf8编码字符串正确长度（支持中文）
---@param str string 字符串
---@return number 长度
function UTF8Util.GetUTF8StrLength(str)
    local len = #str
    local left = len
    local cnt = 0
    local arr = { 0, 0xc0, 0xe0, 0xf0, 0xf8, 0xfc }
    while left ~= 0 do
        local tmp = string.byte(str, -left)
        local i = #arr
        while arr[i] do
            if tmp >= arr[i] then
                left = left - i;
                break ;
            end
            i = i - 1
        end
        cnt = cnt + 1
    end
    return cnt
end

return UTF8Util
