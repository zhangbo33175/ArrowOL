local RewardUtil = {}

--region 道具 / 奖励发放

--=====================================================================================================
-- 道具奖励工具：发放、显示格式化、批量处理
--=====================================================================================================

---@brief 根据资源类型获取预设名
---@param resType EResourceType 资源类型
---@return string 预设名称
function RewardUtil.GetItemPrefabNameByType(resType)
    if resType == EResourceType.CURRENCY then
        return "GameObject_CurrencyProto"
    elseif resType == EResourceType.ITEM then
        return "GameObject_BoosterProto"
    elseif resType == EResourceType.BUFF then
        return "GameObject_BuffProto"
    end
    return "GameObject_ItemProto"
end

---@brief 直接发放道具
---@param itemId string 道具ID
---@param count number 数量
---@param areaId string 区域ID
function RewardUtil.GrantItem(itemId, count, areaId)
    if string.isNullOrEmpty(itemId) or not count or count <= 0 then
        PrintError("GrantItem failed: invalid params")
        return
    end

    local data = ResourceDataManager and ResourceDataManager:GetResourceData(itemId)
    if not data then return end

    if data.Type == EResourceType.Currency then
        if CurrencyMgr then CurrencyMgr:AddNum(itemId, count, areaId) end
    elseif data.Type == EResourceType.Item or data.Type == EResourceType.Profile then
        if ItemManager then ItemManager:AddNum(itemId, count) end
    elseif data.Type == EResourceType.Buff then
        if BuffManager then BuffManager:AddBuff(itemId, count) end
    elseif data.Type == EResourceType.Card then
        if CollectionMgr then CollectionMgr:AddPack(itemId, count) end
    else
        if ItemManager then ItemManager:AddNum(itemId, count) end
    end
end

---@brief 批量发放奖励
---@param rewardList table 奖励列表
---@param areaId string 区域ID
function RewardUtil.GrantRewardList(rewardList, areaId)
    if not rewardList or not rewardList.items then return end
    for _, v in ipairs(rewardList.items) do
        RewardUtil.GrantItem(v.Id, v.Num, areaId)
    end
end

---@brief 格式化道具数量显示
---@param itemId string 道具ID
---@param count number 数量
---@param showPrefix boolean 是否显示前缀x
---@return string 格式化后字符串
function RewardUtil.FormatItemCountDisplay(itemId, count, showPrefix)
    showPrefix = showPrefix ~= nil and showPrefix or true
    local format = showPrefix and "x%d" or "%d"

    if string.isNullOrEmpty(itemId) then return string.format(format, count) end
    local data = ResourceDataManager:GetResourceData(itemId)
    if not data then return string.format(format, count) end

    if data.Type == EResourceType.Buff then
        return GameTimeHelper and GameTimeHelper:FormatTimeInCommonRule(count, false) or tostring(count)
    else
        return string.format(format, count)
    end
end

---@brief 从RewardData格式化数量
---@param reward table 奖励数据
---@return string 格式化数量字符串
function RewardUtil.FormatRewardCount(reward)
    return reward and RewardUtil.FormatItemCountDisplay(reward.Id, reward.Num) or ""
end

--endregion

return RewardUtil
