---@class SingletonManager 单例管理器（全局唯一）
---@field _singletonList HonorList 单例列表 List<Singleton>
local SingletonManager = class("SingletonManager")

--=====================================================================================================
-- 生命周期：构造函数
--=====================================================================================================

---@brief 构造函数
---@desc 创建单例管理器实例，并初始化内部单例管理列表
---@param args table 自定义参数
function SingletonManager:ctor(args)
    --- 单例列表 List<Singleton>
    ---@type HonorList
    self._singletonList = HonorList:Create()
end

--=====================================================================================================
-- 工厂方法：创建实例
--=====================================================================================================

---@brief 创建单例管理器实例
---@desc 通过类方法创建并返回一个新的单例管理器对象，作为全局单例的创建入口
---@param args table 自定义参数
---@return SingletonManager 创建的单例管理器实例
function SingletonManager:Create(args)
    local obj = SingletonManager.new(args)
    return obj
end

--=====================================================================================================
-- 单例注册与注销
--=====================================================================================================

---@brief 添加单例到管理列表
---@desc 将指定单例对象注册到管理列表，交由管理器统一执行初始化与数据清理
---@param singleton Singleton 待添加的单例对象
function SingletonManager:AddSingleton(singleton)
    if singleton then
        self._singletonList:Add(singleton)
    end
end

---@brief 从管理列表移除单例
---@desc 将指定单例对象从管理列表中移除，取消其统一生命周期管理
---@param singleton Singleton 待移除的单例对象
function SingletonManager:RemoveSingleton(singleton)
    if singleton then
        self._singletonList:Remove(singleton)
    end
end

--=====================================================================================================
-- 数据清理：清空所有单例数据
--=====================================================================================================

---@brief 清理所有单例数据（切换账号/重启游戏时调用）
---@desc 停止全部动画、协程、定时器、通知与事件处理，再按优先级【降序】依次清理所有单例数据
function SingletonManager:ClearAllSingleData()
    Try(function()
        -- 清理所有动画
        DOTween.KillAll(false)

        ----------------------------------------
        -- 框架级单例数据清理
        ----------------------------------------
        HonorCoroutineHelper:StopAll()
        HonorTimerHelper:UnregisterAll(true)
        HonorNotificationHelper:ClearNotificationData()
        HonorEventHelper:Cleanup()
        -- 重启异步事件处理协程
        HonorEventHelper:StartProcessEventAsyncCoroutine()
        ----------------------------------------

        -- 按优先级【降序】清理（先业务后框架）
        self:OrderSingleton(true)

        -- 遍历所有单例并执行 ClearData
        ---@param v Singleton
        for _, v in ipairs(self._singletonList.items) do
            v:ClearData()
        end
    end)
end

--=====================================================================================================
-- 初始化：分步预加载初始化所有单例
--=====================================================================================================

---@brief 初始化所有单例（预加载流程分步执行）
---@desc 按优先级【升序】对全部单例排序，并将每个单例的 Init 作为独立步骤加入预加载流程
---@param preload ProcedurePreload 预加载流程对象
function SingletonManager:InitAllSingleForStep(preload)
    -- 按优先级【升序】初始化（先框架后业务）
    self:OrderSingleton(false)

    -- 遍历所有单例并依次加入预加载步骤
    ---@param v Singleton
    for _, v in ipairs(self._singletonList.items) do
        -- 加入预加载步骤
        preload.cs:AddStep(function()
            v:Init()
        end)
    end
end

--=====================================================================================================
-- 排序：按优先级重排单例列表
--=====================================================================================================

---@brief 单例按优先级排序
---@desc 依据各单例返回的 Priority() 值对管理列表进行排序，决定初始化/清理的执行顺序
---@param isDescending boolean 是否降序（true=降序 false=升序）
function SingletonManager:OrderSingleton(isDescending)
    if isDescending == nil then
        isDescending = false
    end

    -- 排序依据：单例返回的 Priority() 优先级
    table.sort(self._singletonList.items, function(a, b)
        if isDescending then
            return a:Priority() < b:Priority()
        else
            return a:Priority() > b:Priority()
        end
    end)
end

return SingletonManager
