--region 全局事件管理助手（发布-订阅模式）
---@class HonorEventHelper 全局事件管理助手（发布-订阅模式）
local HonorEventHelper = class("HonorEventHelper")

--- 构造函数
--- 初始化事件存储容器、异步事件队列、启动异步处理协程
function HonorEventHelper:ctor(args)
    self._targetHandlers = {}        ---@type table<string, table> 事件回调字典：key=事件名，value=回调函数列表
    self._eventsCallNextFrame = {}   ---@type table<function> 下一帧执行的异步事件队列
    self._csCoroutine = nil          ---@type userdata 异步事件协程句柄
    self:StartProcessEventAsyncCoroutine()
end

--- 创建单例/实例对象
---@param args table 初始化参数（可选）
---@return HonorEventHelper 事件助手实例
function HonorEventHelper:Create(args)
    local obj = HonorEventHelper.new(args)
    return obj
end

--- 启动异步事件处理协程
--- 停止旧协程并创建新协程，避免协程重复启动
function HonorEventHelper:StartProcessEventAsyncCoroutine()
    HonorCoroutineHelper:Stop(self._csCoroutine)
    self._csCoroutine = HonorCoroutineHelper:Start(handler(self, self.ProcessEventAsync))
end

--- 异步执行事件协程逻辑
--- 等待事件队列非空 → 等待一帧 → 批量执行所有异步事件
function HonorEventHelper:ProcessEventAsync()
    while true do
        -- 等待有事件需要执行
        coroutine.yield(Unity.WaitUntil(function() return #self._eventsCallNextFrame > 0 end))
        -- 等待一帧，实现下一帧执行效果
        coroutine.yield(nil)
        -- 安全执行所有延迟事件，执行后清空队列
        if #self._eventsCallNextFrame > 0 then
            for _, callFunc in pairs(self._eventsCallNextFrame) do
                callFunc()
            end
            self._eventsCallNextFrame = {}
        end
    end
end

--- 注册事件监听
---@param evtCmd string 事件名称（唯一标识）
---@param targetHandler table 回调处理器（包含target/method/func）
---@remark 自带防重复注册、参数安全校验
function HonorEventHelper:AddListener(evtCmd, targetHandler)
    -- 安全校验
    if type(evtCmd) ~= "string" then
        PrintError("[Event] evtCmd 必须是 string 类型")
        return
    end
    if not targetHandler then return end

    -- 初始化事件列表
    if not self._targetHandlers[evtCmd] then
        self._targetHandlers[evtCmd] = {}
    end
    -- 防重复注册校验
    for _, info in pairs(self._targetHandlers[evtCmd]) do
        if info.target == targetHandler.target and info.method == targetHandler.method then
            PrintWarning("[Event] 重复注册事件: " .. evtCmd)
            return
        end
    end
    table.insert(self._targetHandlers[evtCmd], targetHandler)
end

--- 移除指定事件监听
---@param evtCmd string 事件名称
---@param targetHandler table 需要移除的回调处理器
function HonorEventHelper:RemoveListener(evtCmd, targetHandler)
    if type(evtCmd) ~= "string" or not targetHandler then return end
    if self._targetHandlers[evtCmd] then
        for i, info in pairs(self._targetHandlers[evtCmd]) do
            if info.target == targetHandler.target and info.method == targetHandler.method then
                self._targetHandlers[evtCmd][i] = nil
                break
            end
        end
    end
end

--- 根据事件名移除该事件的所有监听
---@param evtCmd string 事件名称
function HonorEventHelper:RemoveListenerByEventID(evtCmd)
    if self._targetHandlers[evtCmd] then
        self._targetHandlers[evtCmd] = {}
    end
end

--- 根据对象移除该对象绑定的所有事件监听
---@param target table 监听对象（常用：界面销毁时调用）
function HonorEventHelper:RemoveListenerByTarget(target)
    if not target then return end
    for _, handlers in pairs(self._targetHandlers) do
        for i, info in pairs(handlers) do
            if info.target == target then
                handlers[i] = nil
            end
        end
    end
end

--- 根据回调函数移除监听
---@param method function 回调函数
function HonorEventHelper:RemoveListenerByFunc(method)
    if not method then return end
    for _, handlers in pairs(self._targetHandlers) do
        for i, info in pairs(handlers) do
            if info.method == method then
                handlers[i] = nil
            end
        end
    end
end

--- 【立即同步派发】事件
---@param evtCmd string 事件名称
---@param ... any 传递的事件参数
---@remark 立即执行所有注册的回调，适合同步逻辑
function HonorEventHelper:FireNow(evtCmd, ...)
    if type(evtCmd) ~= "string" then return end
    local handlers = self._targetHandlers[evtCmd]
    if not handlers then return end

    -- 遍历执行所有回调
    for _, info in pairs(handlers) do
        if info.func then
            info.func(...)
        end
    end
end

--- 【下一帧异步派发】事件
---@param evtCmd string 事件名称
---@param ... any 传递的事件参数
---@remark 不卡当前帧，自带xpcall异常捕获，单个回调报错不影响其他
function HonorEventHelper:Fire(evtCmd, ...)
    if type(evtCmd) ~= "string" then return end
    local params = {...}
    -- 加入下一帧执行队列
    table.insert(self._eventsCallNextFrame, function()
        -- 异常捕获保护：防止单个回调崩溃影响整体
        xpcall(function()
            self:FireNow(evtCmd, table.unpack(params))
        end, __G__TRACK_BACK__)
    end)
end

--- 清空所有事件数据
--- 重置事件容器、异步队列，释放内存
function HonorEventHelper:Cleanup()
    self._targetHandlers = {}
    self._eventsCallNextFrame = {}
end
--endregion

return HonorEventHelper
