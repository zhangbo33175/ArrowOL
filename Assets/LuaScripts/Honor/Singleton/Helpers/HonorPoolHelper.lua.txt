--=====================================================================================================
-- descrip:   对象池助手（全局对象池管理器）
--              作用：统一创建、管理、销毁所有 GameObject 对象池
--              定位：对象池系统的顶层入口，方便全局调用与管理
--=====================================================================================================

---@class HonorPoolHelper @全局对象池管理助手
---@field pools table<string, HonorPool> @所有对象池字典（key=池名称, value=池对象）
local HonorPoolHelper = class("HonorPoolHelper")

--- 构造函数
---@type fun(args:table):void
function HonorPoolHelper:ctor(args)
    self.pools = {}  -- 初始化空的对象池集合
end

--- 创建实例
---@type fun(args:table):HonorPoolHelper
---@return HonorPoolHelper
function HonorPoolHelper:Create(args)
    local obj = HonorPoolHelper.new(args)
    return obj
end

--- 创建一个新的对象池
---@type fun(name:string, resDefItem:RES_DEF_ITEM, preParent:UnityEngine.Transform, resetHandler:handler):HonorPool
---@param name 对象池名称（唯一标识）
---@param resDefItem 资源配置
---@param preParent 对象池父节点
---@param resetHandler 对象复用前的重置函数
---@return HonorPool
function HonorPoolHelper:CreatePool(name, resDefItem, preParent, resetHandler)
    -- 只支持 GameObject 类型
    if resDefItem.TypeName == "GameObject" then
        if self.pools[name] == nil then
            -- 创建并缓存对象池
            self.pools[name] = require("HonorPool"):Create(name, resDefItem, preParent, resetHandler)
        else
            PrintError("对象池" .. name .. "已存在，无需再次创建。")
        end
        return self.pools[name]
    else
        PrintError("对象池" .. name .. "创建失败：资源类型不是 GameObject。")
        return nil
    end
end

--- 根据名称获取对象池
---@type fun(name:string):HonorPool
function HonorPoolHelper:GetPool(name)
    return self.pools[name]
end

--- 获取当前总对象池数量
---@type fun():number
function HonorPoolHelper:Count()
    local count = 0
    for _, __ in pairs(self.pools) do
        count = count + 1
    end
    return count
end

--- 销毁指定名称的对象池
---@type fun(name:string):boolean
function HonorPoolHelper:DestroyPool(name)
    if self.pools[name] ~= nil then
        self.pools[name]:Destroy()
        self.pools[name] = nil
        return true
    end
    return false
end

--- 销毁所有对象池（场景切换/游戏退出时调用）
---@type fun():void
function HonorPoolHelper:DestroyAllPools()
    for _, pool in pairs(self.pools) do
        pool:Destroy()
    end
    self.pools = {}
end

--- 销毁所有池中的未使用对象（释放内存）
---@type fun():void
function HonorPoolHelper:DestroyPoolUnused()
    for _, pool in pairs(self.pools) do
        pool:DestroyUnused()
    end
end

return HonorPoolHelper
