--=====================================================================================================
-- descrip:   推送通知助手（本地/远程推送管理）
--              功能：
--              1. 本地定时推送通知（离线消息、活动提醒）
--              2. 跨平台推送开关（iOS / Android）
--              3. 定时轮询拉取通知
--              4. 领奖通知、后台运营推送协议交互
--              5. 推送数据存档与清除
--=====================================================================================================

---@class HonorNotificationHelper @推送通知管理助手
local HonorNotificationHelper = class("HonorNotificationHelper")

--- 构造函数
---@type fun(args:table):void
function HonorNotificationHelper:ctor(args)
    -- 发送通知达到 X 条后，强制刷新一次系统推送
    self._nativeNotificationMessageMaxCount = 5
    -- 每 X 秒轮询检测一次通知消息
    self._nativeNotificationMessageTimeDelta = 30
    -- 当前已发送通知计数
    self._nativeNotificationMessageCurCount = 0
    -- 初始化监听与定时
    self:Init()
end

--- 创建实例
---@type fun(args:table):HonorNotificationHelper
---@return HonorNotificationHelper
function HonorNotificationHelper:Create(args)
    local obj = HonorNotificationHelper.new(args)
    return obj
end

--- 初始化（注册事件 + 定时轮询）
---@type fun():void
function HonorNotificationHelper:Init(args)
    -- 监听通知消息事件
    HonorEventHelper:AddListener(EventCmd.NotificationMessage, handler(self, self.RequestNotificationMessage))
    -- 定时轮询拉取通知（每30秒执行一次，锁定不删除）
    HonorTimerHelper:RegisterDeltaTime(
            EventCmd.NotificationMessage,
            self._nativeNotificationMessageTimeDelta,
            "ProcedurePlaying",
            true
    )
end

--- 检查推送功能是否开启（受系统配置表控制）
---@type fun():boolean
---@return boolean iOS/Android 推送总开关
function HonorNotificationHelper:CheckNotifyFuncIsOpen()
    if Unity.Application.platform == Unity.RuntimePlatform.Android then
        return SystemConfigHelper:GetSystemConfig("ANDROID_NOTIFY_FUNCTION_OPEN")
    elseif Unity.Application.platform == Unity.RuntimePlatform.IPhonePlayer then
        return SystemConfigHelper:GetSystemConfig("IOS_NOTIFY_FUNCTION_OPEN")
    else
        return SystemConfigHelper:GetSystemConfig("ANDROID_NOTIFY_FUNCTION_OPEN")
    end
end

--- 设置/取消 本地定时通知
---@type fun(messageType:number, isCancel:boolean, titleIDS:string,UTC0TimeStamp:number)
---@param messageType 消息类型 >0
---@param isCancel 是否取消该通知
---@param titleIDS 后台配置ID组，如 "1001,2001"
---@param UTC0TimeStamp 通知触发时间（UTC0），取消时可不传
function HonorNotificationHelper:SetNotificationMessage(messageType, isCancel, titleIDS, UTC0TimeStamp)
end

--- 请求拉取通知消息（主动刷新）
---@type fun():void
function HonorNotificationHelper:RequestNotificationMessage()
end

--- 发送【领奖类通知】协议（每日奖励、限时奖励等）
--- 作用：告诉后台“已领取”，避免重复推送
---@type fun(flagType:number):void
function HonorNotificationHelper:SendRewardNotificationMessage(flagType)
end

--- 获取通知相关存档数据（记录已读/已发/已领）
---@type fun():table
function HonorNotificationHelper:GetNotificationSaveData()
end

--- 清空所有通知存档数据
---@type fun():void
function HonorNotificationHelper:ClearNotificationData()
end

--- 检测是否需要主动拉取通知
---@type fun():void
function HonorNotificationHelper:IsNeedRequestNotificationMessage()
end

return HonorNotificationHelper
