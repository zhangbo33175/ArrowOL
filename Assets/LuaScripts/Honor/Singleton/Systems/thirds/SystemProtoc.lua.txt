-- 元表创建工具：快速创建带 __name / __index 的类
local function meta(name, t)
   t = t or {}
   t.__name  = name
   t.__index = t
   return t
end

-- 默认值工具：如果表中无 key，则创建并返回空表
local function default(t, k, def)
   local v = t[k]
   if not v then
      v = def or {}
      t[k] = v
   end
   return v
end

--=====================================================================================================
-- Lexer：词法分析器
-- 负责：读取 token、跳过注释/空格、解析数字/字符串/标识符
--=====================================================================================================
local Lexer = meta "Lexer" do

   local escape = {
      a = "\a", b = "\b", f = "\f", n = "\n",
      r = "\r", t = "\t", v = "\v"
   }

   local function tohex(x) return string.byte(tonumber(x, 16)) end
   local function todec(x) return string.byte(tonumber(x, 10)) end
   local function toesc(x) return escape[x] or x end

   -- 创建 Lexer 实例
   function Lexer.new(name, src)
      local self = {
         name = name,
         src = src,
         pos = 1
      }
      return setmetatable(self, Lexer)
   end

   -- 模式匹配
   function Lexer:__call(patt, pos)
      return self.src:match(patt, pos or self.pos)
   end

   -- 测试并消耗模式
   function Lexer:test(patt)
      self:whitespace()
      local pos = self('^'..patt..'%s*()')
      if not pos then return false end
      self.pos = pos
      return true
   end

   -- 必须匹配，否则报错
   function Lexer:expected(patt, name)
      if not self:test(patt) then
         return self:error((name or "'"..patt.."'").." expected")
      end
      return self
   end

   -- 位置转换为行号+列号
   function Lexer:pos2loc(pos)
      local linenr = 1
      pos = pos or self.pos
      for start, stop in self.src:gmatch "()[^\n]*()\n?" do
         if start <= pos and pos <= stop then
            return linenr, pos - start + 1
         end
         linenr = linenr + 1
      end
   end

   -- 抛出带位置信息的错误
   function Lexer:error(fmt, ...)
      local ln, co = self:pos2loc()
      return error(("%s:%d:%d: "..fmt):format(self.name, ln, co, ...))
   end

   -- 可选报错
   function Lexer:opterror(opt, msg)
      if not opt then return self:error(msg) end
      return nil
   end

   -- 跳过空白字符 + 注释
   function Lexer:whitespace()
      local pos, c = self "^%s*()(%/?)"
      self.pos = pos
      if c == '' then return self end
      return self:comment()
   end

   -- 跳过 // 和 /* */ 注释
   function Lexer:comment()
      local pos = self "^%/%/[^\n]*\n?()"
      if not pos then
         if self "^%/%*" then
            pos = self "^%/%*.-%*%/()"
            if not pos then
               self:error "unfinished comment"
            end
         end
      end
      if not pos then return self end
      self.pos = pos
      return self:whitespace()
   end

   -- 行结束（支持 ;）
   function Lexer:line_end(opt)
      self:whitespace()
      local pos = self '^[%s;]*%s*()'
      if not pos then
         return self:opterror(opt, "';' expected")
      end
      self.pos = pos
      return pos
   end

   -- 是否到达文件末尾
   function Lexer:eof()
      self:whitespace()
      return self.pos > #self.src
   end

   -- 匹配关键字
   function Lexer:keyword(kw, opt)
      self:whitespace()
      local ident, pos = self "^([%a_][%w_]*)%s*()"
      if not ident or ident ~= kw then
         return self:opterror(opt, "''"..kw..'" expected')
      end
      self.pos = pos
      return kw
   end

   -- 匹配普通标识符
   function Lexer:ident(name, opt)
      self:whitespace()
      local b, ident, pos = self "^()([%a_][%w_]*)%s*()"
      if not ident then
         return self:opterror(opt, (name or 'name')..' expected')
      end
      self.pos = pos
      return ident, b
   end

   -- 匹配带点标识符（a.b.c）
   function Lexer:full_ident(name, opt)
      self:whitespace()
      local b, ident, pos = self "^()([%a_][%w_.]*)%s*()"
      if not ident or ident:match "%.%.+" then
         return self:opterror(opt, (name or 'name')..' expected')
      end
      self.pos = pos
      return ident, b
   end

   -- 解析整数（十进制/八进制/十六进制）
   function Lexer:integer(opt)
      self:whitespace()
      local ns, oct, hex, s, pos =
      self "^([+-]?)(0?)([xX]?)([0-9a-fA-F]+)%s*()"
      local n
      if oct == '0' and hex == '' then
         n = tonumber(s, 8)
      elseif oct == '' and hex == '' then
         n = tonumber(s, 10)
      elseif oct == '0' and hex ~= '' then
         n = tonumber(s, 16)
      end
      if not n then
         return self:opterror(opt, 'integer expected')
      end
      self.pos = pos
      return ns == '-' and -n or n
   end

   -- 解析浮点数（支持 nan / inf）
   function Lexer:number(opt)
      self:whitespace()
      if self:test "nan%f[%A]" then
         return 0.0/0.0
      elseif self:test "inf%f[%A]" then
         return 1.0/0.0
      end
      local ns, d1, s, d2, s2, pos = self "^([+-]?)(%.?)([0-9]+)(%.?)([0-9]*)()"
      if not ns then
         return self:opterror(opt, 'floating-point number expected')
      end
      local es, pos2 = self("(^[eE][+-]?[0-9]+)%s*()", pos)
      if d1 == "." and d2 == "." then
         return self:error "malformed floating-point number"
      end
      self.pos = pos2 or pos
      local n = tonumber(d1..s..d2..s2..(es or ""))
      return ns == '-' and -n or n
   end

   -- 解析字符串（支持转义）
   function Lexer:quote(opt)
      self:whitespace()
      local q, start = self '^(["\'])()'
      if not start then
         return self:opterror(opt, 'string expected')
      end
      self.pos = start
      local patt = '()(\\?'..q..')%s*()'
      while true do
         local stop, s, pos = self(patt)
         if not stop then
            self.pos = start-1
            return self:error "unfinished string"
         end
         self.pos = pos
         if s == q then
            return self.src:sub(start, stop-1)
                       :gsub("\\x(%x+)", tohex)
                       :gsub("\\(%d+)", todec)
                       :gsub("\\(.)", toesc)
         end
      end
   end

   -- 解析常量（标识符 / 数字 / 字符串）
   function Lexer:constant(opt)
      local c = self:full_ident('constant', 'opt') or
              self:number('opt') or
              self:quote('opt')
      if not c and not opt then
         return self:error "constant expected"
      end
      return c
   end

   -- 解析选项名
   function Lexer:option_name()
      local ident
      if self:test "%(" then
         ident = self:full_ident "option name"
         self:expected "%)"
      else
         ident = self:ident "option name"
      end
      while self:test "%." do
         ident = ident .. "." .. self:ident()
      end
      return ident
   end

   -- 解析类型名
   function Lexer:type_name()
      if self:test "%." then
         local id, pos = self:full_ident "type name"
         return "."..id, pos
      else
         return self:full_ident "type name"
      end
   end

end -- Lexer

--=====================================================================================================
-- Parser：语法分析器（构建 FileDescriptorProto）
--=====================================================================================================
local Parser = meta "Parser" do
   Parser.typemap = {}
   Parser.loaded  = {}
   Parser.paths   = { "." }

   function Parser.new()
      local self = {}
      self.typemap = {}
      self.loaded  = {}
      self.paths   = { "." }
      return setmetatable(self, Parser)
   end

   function Parser:error(msg)
      return self.lex:error(msg)
   end

   function Parser:addpath(path)
      self.paths[#self.paths+1] = path
   end

   -- 创建解析上下文
   local function make_context(self, lex)
      local ctx = {
         syntax  = "proto2";
         locmap  = {};
         prefix  = ".";
         lex     = lex;
         parser  = self;
      }
      ctx.loaded  = self.loaded
      ctx.typemap = self.typemap
      ctx.paths   = self.paths

      function ctx.import_fallback(import_name)
         if self.unknown_import == true then
            return true
         elseif type(self.unknown_import) == 'string' then
            return import_name:match(self.unknown_import) and true or nil
         elseif self.unknown_import then
            return self:unknown_import(import_name)
         end
      end

      function ctx.type_fallback(type_name)
         if self.unknown_type == true then
            return true
         elseif type(self.unknown_type) == 'string' then
            return type_name:match(self.unknown_type) and true
         elseif self.unknown_type then
            return self:unknown_type(type_name)
         end
      end

      function ctx.on_import(info)
         if self.on_import then
            return self.on_import(info)
         end
      end

      return setmetatable(ctx, Parser)
   end

   -- 解析文件（带缓存）
   function Parser:parsefile(name)
      local info = self.loaded[name]
      if info then return info end
      local errors = {}
      local proto = require(name)
      info = self:parse(proto, name)
      return info
   end

   -- Protobuf 标准 wire type 定义
   local labels = { optional = 1; required = 2; repeated = 3 }
   local key_types = {
      int32    = 5;  int64    = 3;  uint32   = 13;
      uint64   = 4;  sint32   = 17; sint64   = 18;
      fixed32  = 7;  fixed64  = 6;  sfixed32 = 15;
      sfixed64 = 16; bool     = 8;  string   = 9;
   }
   local com_types = {
      group    = 10; message  = 11; enum     = 14;
   }
   local types = {
      double   = 1;  float    = 2;  int32    = 5;
      int64    = 3;  uint32   = 13; uint64   = 4;
      sint32   = 17; sint64   = 18; fixed32  = 7;
      fixed64  = 6;  sfixed32 = 15; sfixed64 = 16;
      bool     = 8;  string   = 9;  bytes    = 12;
      group    = 10; message  = 11; enum     = 14;
   }

   -- 注册类型
   local function register_type(self, lex, tname, type)
      if not tname:match "%."then
         tname = self.prefix..tname
      end
      if self.typemap[tname] then
         return lex:error("type %s already defined", tname)
      end
      self.typemap[tname] = type
   end

   -- 获取类型信息
   local function type_info(lex, tname)
      local tenum = types[tname]
      if com_types[tname] then
         return lex:error("invalid type name: "..tname)
      elseif tenum then
         tname = nil
      end
      return tenum, tname
   end

   -- 解析 map<key, value>
   local function map_info(lex)
      local keyt = lex:ident "key type"
      if not key_types[keyt] then
         return lex:error("invalid key type: "..keyt)
      end
      local valt = lex:expected "," :type_name()
      local name = lex:expected ">" :ident()
      local ident = name:gsub("^%a", string.upper)
                        :gsub("_(%a)", string.upper).."Entry"
      local kt, ktn = type_info(lex, keyt)
      local vt, vtn = type_info(lex, valt)
      return name, types.message, ident, {
         name = ident,
         field = {
            { name = "key", number = 1, label = 1, type = kt, type_name = ktn },
            { name = "value", number = 2, label = 1, type = vt, type_name = vtn },
         },
         options = { map_entry = true }
      }
   end

   -- 解析 inline option [key=value]
   local function inline_option(lex, info)
      if lex:test "%[" then
         info = info or {}
         while true do
            local name  = lex:option_name()
            local value = lex:expected '=' :constant()
            info[name] = value
            if lex:test "%]" then
               return info
            end
            lex:expected ','
         end
      end
   end

   -- 解析字段
   local function field(self, lex, ident)
      local name, type, type_name, map_entry
      if ident == "map" and lex:test "%<" then
         name, type, type_name, map_entry = map_info(lex)
         self.locmap[map_entry.field[1]] = lex.pos
         self.locmap[map_entry.field[2]] = lex.pos
         register_type(self, lex, type_name, types.message)
      else
         type, type_name = type_info(lex, ident)
         name = lex:ident()
      end
      local info = {
         name      = name,
         number    = lex:expected "=":integer(),
         label     = ident == "map" and 3 or 1,
         type      = type,
         type_name = type_name
      }
      local options = inline_option(lex)
      if options then
         info.default_value = tostring(options.default)
         options.default = nil
         info.json_name = options.json_name
         options.json_name = nil
         if options.packed and options.packed == "false" then
            options.packed = false
         end
      end
      info.options = options
      if info.number <= 0 then
         lex:error("invalid tag number: "..info.number)
      end
      return info, map_entry
   end

   -- 解析带 label 的字段（required / optional / repeated）
   local function label_field(self, lex, ident)
      local label = labels[ident]
      local info, map_entry
      if not label then
         if self.syntax == "proto2" and ident ~= "map" then
            return lex:error("proto2 disallow missing label")
         end
         return field(self, lex, ident)
      end
      if label == 1 and self.syntax == "proto3" then
         return lex:error("proto3 disallow 'optional' label")
      end
      info, map_entry = field(self, lex, lex:type_name())
      info.label = label
      return info, map_entry
   end

   -- 顶层语法解析
   local toplevel = {} do

      function toplevel:package(lex, info)
         local package = lex:full_ident 'package name'
         lex:line_end()
         info.package = package
         self.prefix = "."..package.."."
         return self
      end

      function toplevel:import(lex, info)
         local mode = lex:ident('"weak" or "public"', 'opt') or "public"
         local name = lex:quote()
         lex:line_end()
         local result = self:parsefile(name)
         if self.on_import then self.on_import(result) end
         local dep = default(info, 'dependency')
         local index = #dep
         dep[index+1] = name
      end

      -- message 内部解析
      local msg_body = {} do
         function msg_body:message(lex, info)
            local nested_type = default(info, 'nested_type')
            nested_type[#nested_type+1] = toplevel.message(self, lex)
         end
         function msg_body:enum(lex, info)
            local nested_type = default(info, 'enum_type')
            nested_type[#nested_type+1] = toplevel.enum(self, lex)
         end
         function msg_body:extend(lex, info)
            local extension = default(info, 'extension')
            local nested_type = default(info, 'nested_type')
            local ft, mt = toplevel.extend(self, lex, {})
            for _,v in ipairs(ft) do extension[#extension+1]=v end
            for _,v in ipairs(mt) do nested_type[#nested_type+1]=v end
         end
         function msg_body:extensions(lex, info)
            local rt = default(info, 'extension_range')
            repeat
               local start = lex:integer()
               local stop = 2^29
               lex:keyword 'to'
               if not lex:keyword('max','opt') then stop=lex:integer() end
               rt[#rt+1] = {start=start, ['end']=stop}
            until not lex:test ','
            lex:line_end()
         end
         function msg_body:reserved(lex, info)
            if lex:test '%a' then
               local rt = default(info,'reserved_name')
               repeat rt[#rt+1]=lex:ident() until not lex:test ','
            else
               local rt=default(info,'reserved_range')
               repeat
                  local s=lex:integer()
                  if lex:keyword('to','opt') then
                     rt[#rt+1]={start=s,['end']=lex:integer()}
                  else
                     rt[#rt+1]={start=s,['end']=s}
                  end
               until not lex:test ','
            end
            lex:line_end()
         end
         function msg_body:oneof(lex, info)
            local fs=default(info,"field")
            local ts=default(info,"nested_type")
            local ot=default(info,"oneof_decl")
            local idx=#ot+1
            local o={name=lex:ident()}
            lex:expected"{"
            while not lex:test"}" do
               local id=lex:type_name()
               if id=="option" then
                  toplevel.option(self,lex,o)
               else
                  local f,t=field(self,lex,id)
                  f.oneof_index=idx-1
                  fs[#fs+1]=f
                  if t then ts[#ts+1]=t end
               end
               lex:line_end'opt'
            end
            ot[idx]=o
         end
      end -- msg_body

      -- 解析 message
      function toplevel:message(lex, info)
         local name=lex:ident'message name'
         local t={name=name}
         register_type(self,lex,name,types.message)
         local pre=self.prefix
         self.prefix=pre..name.."."
         lex:expected"{"
         while not lex:test"}" do
            local id=lex:type_name()
            if msg_body[id] then
               msg_body[id](self,lex,t)
            else
               local fs=default(t,'field')
               local f,m=label_field(self,lex,id)
               fs[#fs+1]=f
               if m then default(t,'nested_type')[#t.nested_type+1]=m end
            end
            lex:line_end'opt'
         end
         lex:line_end'opt'
         if info then default(info,'message_type')[#info.message_type+1]=t end
         self.prefix=pre
         return t
      end

      -- 解析 enum
      function toplevel:enum(lex, info)
         local name=lex:ident'enum name'
         local e={name=name}
         register_type(self,lex,name,types.enum)
         lex:expected"{"
         while not lex:test"}" do
            local id=lex:ident()
            if id=='option' then
               toplevel.option(self,lex,default(e,'options'))
            else
               local v=default(e,'value')
               v[#v+1]={name=id,number=lex:expected'=':integer(),options=inline_option(lex)}
               lex:line_end()
            end
            lex:line_end'opt'
         end
         lex:line_end'opt'
         if info then default(info,'enum_type')[#info.enum_type+1]=e end
         return e
      end

      -- 解析 option
      function toplevel:option(lex, info)
         local k=lex:option_name()
         lex:expected"="
         local v=lex:constant()
         lex:line_end()
         local o=info and default(info,'options') or {}
         o[k]=v
         return o
      end

      -- 解析 extend
      function toplevel:extend(lex, info)
         local tname=lex:type_name()
         local ft=info and default(info,'extension') or {}
         local mt=info and default(info,'message_type') or {}
         lex:expected"{"
         while not lex:test"}" do
            local id,p=lex:type_name()
            local f,m=label_field(self,lex,id)
            f.extendee=tname
            ft[#ft+1]=f
            if m then mt[#mt+1]=m end
            lex:line_end'opt'
         end
         return ft,mt
      end

      -- service 内部解析
      local svr_body = {} do
         function svr_body:rpc(lex, info)
            local name,p=lex:ident"rpc name"
            local r={name=name}
            self.locmap[r]=p
            lex:expected"%("
            r.client_stream=lex:keyword("stream","opt")
            local _,tn=type_info(lex,lex:type_name())
            r.input_type=tn
            lex:expected"%)":expected"returns":expected"%("
            r.server_stream=lex:keyword("stream","opt")
            _,tn=type_info(lex,lex:type_name())
            r.output_type=tn
            lex:expected"%)"
            if lex:test"{" then
               while not lex:test"}" do
                  lex:line_end"opt"
                  lex:keyword"option"
                  toplevel.option(self,lex,default(r,'options'))
               end
            end
            lex:line_end"opt"
            default(info,"method")[#info.method+1]=r
         end
      end -- svr_body

      -- 解析 service
      function toplevel:service(lex, info)
         local name=lex:ident'service name'
         local s={name=name}
         lex:expected"{"
         while not lex:test"}" do
            local id=lex:type_name()
            if svr_body[id] then
               svr_body[id](self,lex,s)
            else
               lex:error"expected rpc/option"
            end
            lex:line_end'opt'
         end
         lex:line_end'opt'
         if info then default(info,'service')[#info.service+1]=s end
         return s
      end

   end -- toplevel

   -- 主解析入口
   function Parser:parse(src, name)
      local loaded = self.loaded[name]
      if loaded then
         if loaded == true then error("loop loaded: "..name) end
         return loaded
      end
      name = name or "<input>"
      local lex = Lexer.new(name, src)
      local info = { name = lex.name }
      if name then self.loaded[name] = true end
      local ctx = make_context(self, lex)

      if lex:keyword('syntax','opt') then
         info.syntax = lex:expected'=':quote()
         ctx.syntax  = info.syntax
         lex:line_end()
      end

      while not lex:eof() do
         local id = lex:ident()
         if toplevel[id] then
            toplevel[id](ctx, lex, info)
         else
            lex:error("unknown keyword '"..id.."'")
         end
         lex:line_end "opt"
      end

      if name then self.loaded[name] = info end
      return ctx:resolve(lex, info)
   end

   --=====================================================================================================
   -- Resolver：类型解析 + 校验
   -- 检查：重复定义、类型合法性、作用域、字段编号
   --=====================================================================================================
   local function empty() end
   local function iter(t,k) local v=t[k] return v and ipairs(v) or empty end

   local function check_dup(self,lex,ty,map,k,v)
      local o=map[v[k]]
      if o then
         local ln,co=lex:pos2loc(self.locmap[o])
         lex:error("%s '%s' exists at %d:%d",ty,v[k],ln,co)
      end
      map[v[k]]=v
   end

   local function check_type(self,lex,tn)
      if tn:match"^%." then
         local t=self.typemap[tn]
         if not t then lex:error("unknown type '%s'",tn) end
         return t,tn
      end
      local p=self.prefix
      for i=#p+1,1,-1 do
         local op=p[i]
         p[i]=tn
         local n=table.concat(p,".",1,i)
         p[i]=op
         local t=self.typemap[n]
         if t then return t,n end
      end
      if self.type_fallback then
         local n,t=self.type_fallback(tn)
         if n then
            t=types[t or"message"]
            if n==true then n="."..tn end
            return t,n
         end
      end
      lex:error("unknown type '%s'",tn)
   end

   local function check_field(self,lex,f)
      if f.extendee then
         local t,tn=check_type(self,lex,f.extendee)
         if t~=types.message then lex:error"expected message" end
         f.extendee=tn
      end
      if f.type_name then
         local t,tn=check_type(self,lex,f.type_name)
         f.type=t;f.type_name=tn
      end
   end

   local function check_enum(self,lex,e)
      local n={}
      local v={}
      for _,vv in iter(e,'value') do
         lex.pos=self.locmap[vv]
         check_dup(self,lex,'enum name',n,'name',vv)
         if not (e.options and e.options.allow_alias) then
            check_dup(self,lex,'enum number',v,'number',vv)
         end
      end
   end

   local function check_message(self,lex,m)
      self.prefix[#self.prefix+1]=m.name
      local n={}
      local v={}
      for _,vv in iter(m,'field') do
         lex.pos=self.locmap[vv]
         check_dup(self,lex,'field name',n,'name',vv)
         check_dup(self,lex,'field number',v,'number',vv)
         check_field(self,lex,vv)
      end
      for _,vv in iter(m,'nested_type') do check_message(self,lex,vv) end
      for _,vv in iter(m,'extension') do
         lex.pos=self.locmap[vv];check_field(self,lex,vv)
      end
      self.prefix[#self.prefix]=nil
   end

   local function check_service(self,lex,s)
      local n={}
      for _,m in iter(s,'method') do
         lex.pos=self.locmap[m]
         check_dup(self,lex,'rpc',n,'name',m)
         local t,tn=check_type(self,lex,m.input_type)
         m.input_type=tn
         if t~=types.message then lex:error"expected message" end
         t,tn=check_type(self,lex,m.output_type)
         m.output_type=tn
         if t~=types.message then lex:error"expected message" end
      end
   end

   function Parser:resolve(lex,info)
      self.prefix={"",info.package}
      for _,m in iter(info,'message_type') do check_message(self,lex,m) end
      for _,e in iter(info,'enum_type') do check_enum(self,lex,e) end
      for _,s in iter(info,'service') do check_service(self,lex,s) end
      for _,e in iter(info,'extension') do
         lex.pos=self.locmap[e];check_field(self,lex,e)
      end
      self.prefix=nil
      return info
   end

end -- Parser

--=====================================================================================================
-- PB 集成：自动编译为 descriptor 并加载到 pb 库
--=====================================================================================================
local has_pb,pb=pcall(require,"pb") do
   if has_pb then
      -- 内置官方 descriptor.proto 二进制数据
      local descriptor_pb = "\n\google\protobuf\descriptor.proto"

      function Parser.reload()
         assert(pb.load(descriptor_pb))
      end

      local function do_compile(self,f,...)
         if self.include_imports then
            local old=self.on_import
            local r={}
            function self.on_import(i) r[#r+1]=i end
            local t=f(...)
            r[#r+1]=t
            self.on_import=old
            return {file=r}
         end
         return {file={f(...)}}
      end

      function Parser:compile(s,name)
         local d=do_compile(self,self.parse,self,s,name)
         return pb.encode('.google.protobuf.FileDescriptorSet',d)
      end

      function Parser:compilefile(fn)
         local d=do_compile(self,self.parsefile,self,fn)
         return pb.encode('.google.protobuf.FileDescriptorSet',d)
      end

      function Parser:load(s,name)
         return assert(pb.load(self:compile(s,name)))
      end

      function Parser:loadfile(fn)
         return assert(pb.load(self:compilefile(fn)))
      end

      Parser.reload()

   end
end

return Parser
