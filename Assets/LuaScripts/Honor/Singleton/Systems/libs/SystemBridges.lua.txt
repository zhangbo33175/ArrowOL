--- EmmyLua 断点调试器初始化（仅编辑器环境生效）
--- 功能：根据配置文件自动选择调试模式，连接 Rider/IDE 进行 Lua 断点调试
---@type fun():void
function __DEBUGGER__()
    if IsEditor == true then
        local debugType = Honor.GameDefinitions.DebugMode.None
        -- 读取调试模式配置文件
        if io.exists(Honor.GamePathUtils.IDEDebugger.GetLuaDebugModeLibraryConfigFileFullPath()) == true then
            local content = io.readfile(Honor.GamePathUtils.IDEDebugger.GetLuaDebugModeLibraryConfigFileFullPath())
            debugType = Honor.GameDefinitions.DebugMode.__CastFrom(tonumber(content))
            -- 加载 EmmyLua 调试插件
            package.cpath = package.cpath .. ';C:/Users/Administrator/AppData/Roaming/JetBrains/Rider2026.1/plugins/IntelliJ-EmmyLua/debugger/emmy/windows/x64/?.dll'

            -- 模式1：IDE主动连接 Unity
            if debugType == Honor.GameDefinitions.DebugMode.IDEFirst then
                local dbg = require('emmy_core')
                dbg.tcpConnect('localhost', 9966)
                print("<color=green>[EmmyLua] 调试器启动成功！</color>")

                -- 模式2：Unity 等待 IDE 连接
            elseif debugType == Honor.GameDefinitions.DebugMode.UnityFirst then
                local dbg = require('emmy_core')
                dbg.tcpListen('localhost', 9966)
                dbg.waitIDE()
            end
        end
    end

    -- 重复代码段（保留原逻辑，未做修改）
    if IsEditor == true then
        local debugType = Honor.GameDefinitions.DebugMode.None
        if io.exists(Honor.GamePathUtils.IDEDebugger.GetLuaDebugModeLibraryConfigFileFullPath()) == true then
            local content = io.readfile(Honor.GamePathUtils.IDEDebugger.GetLuaDebugModeLibraryConfigFileFullPath())
            debugType = Honor.GameDefinitions.DebugMode.__CastFrom(tonumber(content))
            package.cpath = package.cpath .. ';C:/Users/Administrator/AppData/Roaming/JetBrains/Rider2026.1/plugins/IntelliJ-EmmyLua/debugger/emmy/windows/x64/?.dll'
            if debugType == Honor.GameDefinitions.DebugMode.IDEFirst then
                local dbg = require('emmy_core')
                dbg.tcpConnect('localhost', 9966)
            elseif debugType == Honor.GameDefinitions.DebugMode.UnityFirst then
                local dbg = require('emmy_core')
                dbg.tcpListen('localhost', 9966)
                dbg.waitIDE()
            end
        end
    end
end

--- 从 C# 层创建 Lua 业务类（绑定生命周期）
--- 功能：C# 的 LuaBehaviour 调用此函数创建 Lua 类，并绑定 Awake/Start/Update/OnDestroy 等周期
---@type fun(_env:table, _luaScriptName:string):table
---@param _env table C#传递的环境表，用于绑定生命周期函数
---@param _luaScriptName string Lua脚本名称（require用）
---@return table 创建好的Lua类实例
function CreateLuaClassFromCS(_env, _luaScriptName)
    -- 创建Lua类实例
    local _createdClass     = require(_luaScriptName):Create({ env = _env })

    -- 绑定 C# 生命周期 -> Lua 函数
    _env.Awake              = function() _createdClass:Awake() end
    _env.OnEnable           = function() _createdClass:OnEnable() end
    _env.Start              = function() _createdClass:Start() end
    _env.Proc               = function() _createdClass:Proc() end
    _env.OnDisable          = function() _createdClass:OnDisable() end
    _env.OnDestroy          = function() _createdClass:OnDestroy() end

    -- 2D碰撞
    _env.OnCollisionEnter2D = function(collision) _createdClass:OnCollisionEnter2D(collision) end
    _env.OnCollisionStay2D  = function(collision) _createdClass:OnCollisionStay2D(collision) end
    _env.OnCollisionExit2D  = function(collision) _createdClass:OnCollisionExit2D(collision) end
    -- 3D碰撞
    _env.OnCollisionEnter3D = function(collision) _createdClass:OnCollisionEnter3D(collision) end
    _env.OnCollisionStay3D  = function(collision) _createdClass:OnCollisionStay3D(collision) end
    _env.OnCollisionExit3D  = function(collision) _createdClass:OnCollisionExit3D(collision) end

    -- 2D触发
    _env.OnTriggerEnter2D   = function(collision) _createdClass:OnTriggerEnter2D(collision) end
    _env.OnTriggerStay2D    = function(collision) _createdClass:OnTriggerStay2D(collision) end
    _env.OnTriggerExit2D    = function(collision) _createdClass:OnTriggerExit2D(collision) end
    -- 3D触发
    _env.OnTriggerEnter3D   = function(collision) _createdClass:OnTriggerEnter3D(collision) end
    _env.OnTriggerStay3D    = function(collision) _createdClass:OnTriggerStay3D(collision) end
    _env.OnTriggerExit3D    = function(collision) _createdClass:OnTriggerExit3D(collision) end

    return _createdClass
end

--- 从 C# 层创建流程(Procedure)Lua类
--- 功能：专门用于流程（场景/状态）的创建，绑定流程生命周期
---@type fun(_env:table, _luaScriptName:string):table
---@param _env table 环境表
---@param _luaScriptName string 流程类名
---@return table 流程实例
function CreatePocedureLuaClassFromCS(_env, _luaScriptName)
    local _createdClass = require(_luaScriptName):Create({ env = _env })

    -- 流程生命周期
    _env.OnEnter        = function(...) _createdClass:OnEnter(...) end    -- 进入流程
    _env.OnUpdate       = function(...) _createdClass:OnUpdate(...) end  -- 流程更新
    _env.OnLeave        = function(...) _createdClass:OnLeave(...) end    -- 离开流程

    return _createdClass
end

--- 为UI注入对象注册事件监听（按钮、Toggle、输入框、列表等自动绑定）
--- 功能：框架核心函数，C#可视化配置的事件自动绑定到Lua函数
---@type fun(_injectionObj:userdata, _injectionCmd:string, _handler:handler):void
---@param _injectionObj userdata C#注入的UI对象（Button/Toggle/Input等）
---@param _injectionCmd string 事件类型：Clicked/Changed/End/Chosen 或 Evt_开头的事件
---@param _handler handler Lua回调函数（handler包装）
function AddUIListenerFunction(_injectionObj, _injectionCmd, _handler)
    -- 全局注册容器初始化
    if addUIListenerInfos == nil then
        addUIListenerInfos = {}
    end
    -- 为当前对象初始化注册信息
    if addUIListenerInfos[_injectionObj] == nil then
        addUIListenerInfos[_injectionObj] = {}
        addUIListenerInfos[_injectionObj]._handlerList = {}
        addUIListenerInfos[_injectionObj]._interactTargetTimes = {}
    end

    -- 防重复注册：相同target+method不再注册
    for _, v in pairs(addUIListenerInfos[_injectionObj]._handlerList) do
        if v.target == _handler.target and v.method == _handler.method then
            return
        end
    end

    -- 保存回调与冷却时间
    table.insert(addUIListenerInfos[_injectionObj]._handlerList, _handler)
    table.insert(addUIListenerInfos[_injectionObj]._interactTargetTimes, 0)

    -- 【核心】事件派发闭包：统一转发所有UI事件
    local _dispatchFunction = function(...)
        for _, v in pairs(addUIListenerInfos[_injectionObj]._handlerList) do
            -- 按钮防点击冷却
            if Unity.Time.realtimeSinceStartup >= addUIListenerInfos[_injectionObj]._interactTargetTimes[_] then
                if v.target == _handler.target and v.method == _handler.method then
                    local finalArgsList = {...}

                    -- 按钮点击：自动注入配置的参数
                    if _injectionCmd == 'Clicked' then
                        local injections = _injectionObj:GetComponentInParent(typeof(Honor.LuaBehaviour), true).Injections
                        for i = 0, injections.Count - 1 do
                            if injections[i].InjectionTypeName == Honor.LuaInjection.InjectionType.UI_Button then
                                if injections[i].IsArray then
                                    for k, vObj in pairs(injections[i].ElementsObjs) do
                                        if injections[i].ElementsExtendsEnableds[k] and injections[i]:GetExtendsCountAtElementIndex(k) == 3 then
                                            if vObj == _injectionObj then
                                                table.insert(finalArgsList, injections[i]:GetExtendsInfoAtElementIndex(k, 2))
                                                break
                                            end
                                        end
                                    end
                                elseif injections[i].ExtendsEnabled and injections[i]:GetExtendsCount() == 3 then
                                    if injections[i].Obj == _injectionObj then
                                        table.insert(finalArgsList, injections[i]:GetExtendsInfo(2))
                                        break
                                    end
                                end
                            end
                        end
                        -- 设置按钮冷却时间
                        addUIListenerInfos[_injectionObj]._interactTargetTimes[_] = Unity.Time.realtimeSinceStartup + GameMainRoot.UI.ButtonInteractDuration
                    end

                    -- 执行真正的Lua回调
                    return v.func(table.unpack(finalArgsList))
                end
            end
        end
    end

    -- 根据事件类型注册到C#组件
    if _injectionCmd == 'Clicked' then
        _injectionObj.onClick:AddListener(_dispatchFunction)
    elseif _injectionCmd == 'Changed' then
        _injectionObj.onValueChanged:AddListener(_dispatchFunction)
    elseif _injectionCmd == 'End' then
        _injectionObj.onEndEdit:AddListener(_dispatchFunction)
    elseif _injectionCmd == 'Chosen' then
        _injectionObj.onChosen = _dispatchFunction
    elseif _injectionCmd == 'GettingItem' then
        _injectionObj:InitListView(_injectionObj.MaxItemNum, function(index)
            return _dispatchFunction(index)
        end)
    elseif _injectionCmd == 'Start' then
        _injectionObj.OnListViewStart = _dispatchFunction
    elseif _injectionCmd == 'SnapItemFinished' then
        _injectionObj.OnListViewSnapItemFinished = _dispatchFunction
    elseif _injectionCmd == 'SnapNearestChanged' then
        _injectionObj.OnListViewSnapNearestChanged = _dispatchFunction
    else
        -- 通用UI事件：PointerDown/Up/Enter/Exit 等 Evt_ 开头事件
        if string.starts(_injectionCmd, "Evt_") then
            local entry = nil
            local eventTrigger = _injectionObj.gameObject:GetOrAddComponent(typeof(Unity.EventSystems.EventTrigger))
            for _, existingEntry in pairs(eventTrigger.triggers) do
                if existingEntry.eventID == Unity.EventSystems.EventTriggerType.__CastFrom(string.sub(_injectionCmd, 5)) then
                    entry = existingEntry
                    break
                end
            end
            if entry == nil then
                entry = Unity.EventSystems.EventTrigger.Entry()
                entry.eventID = Unity.EventSystems.EventTriggerType.__CastFrom(string.sub(_injectionCmd, 5))
                eventTrigger.triggers:Add(entry)
            end
            entry.callback:AddListener(_dispatchFunction)
        end
    end
end

--- 移除UI注入对象的监听
---@type fun(_injectionObj:userdata, _injectionCmd:string, _handler:handler):void
---@param _injectionObj userdata UI对象
---@param _injectionCmd string 事件类型
---@param _handler handler 要移除的回调
function OnRemoveListener(_injectionObj, _injectionCmd, _handler)
    if addUIListenerInfos ~= nil and addUIListenerInfos[_injectionObj] ~= nil and addUIListenerInfos[_injectionObj]._handlerList ~= nil and addUIListenerInfos[_injectionObj]._interactTargetTimes ~= nil then
        for __, __v in pairs(addUIListenerInfos[_injectionObj]._handlerList) do
            if __v.target == _handler.target and __v.method == _handler.method then
                table.remove(addUIListenerInfos[_injectionObj]._handlerList, __)
                table.remove(addUIListenerInfos[_injectionObj]._interactTargetTimes, __)
                break
            end
        end
    end
    -- 清理空表
    if addUIListenerInfos[_injectionObj]._handlerList and #addUIListenerInfos[_injectionObj]._handlerList == 0 then
        addUIListenerInfos[_injectionObj]._handlerList = nil
    end
    if addUIListenerInfos[_injectionObj]._interactTargetTimes and #addUIListenerInfos[_injectionObj]._interactTargetTimes == 0 then
        addUIListenerInfos[_injectionObj]._interactTargetTimes = nil
    end
end
