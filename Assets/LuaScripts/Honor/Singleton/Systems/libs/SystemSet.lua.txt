-- Description: 无序集合（SystemSet）
--              特点：元素唯一、O(1) 查找、O(1) 插入删除、不保证顺序
-- Author:     Unity Standard
-- @date:      2025
--=====================================================================================================

--region 无序集合（元素唯一、O(1) 增删查）
local SystemSet = {}
SystemSet.__index = SystemSet

--- 创建一个无序集合
---@param isweak boolean 是否使用弱引用表（自动回收无引用对象）
---@return SystemSet 返回集合实例
function SystemSet.new(isweak)
    ---@class SystemSet 无序集合类
    ---@field _data table 存储集合元素的表（key=元素，value=true）
    ---@field _size number 集合元素数量
    local t = {}
    t._data = {}        -- 实际存储数据
    t._size = 0         -- 元素个数
    if isweak then
        -- 弱引用key，方便自动释放对象
        setmetatable(t._data, { __mode = "k" })
    end
    setmetatable(t, SystemSet)
    return t
end

--- 向集合插入数据（自动去重）
---@param self SystemSet 集合实例
---@param value any 要插入的数据
---@return boolean true=插入成功，false=已存在
function SystemSet.insert(self, value)
    if not self._data[value] then
        self._data[value] = true
        self._size = self._size + 1
        return true
    end
    return false
end

--- 从集合移除指定数据
---@param self SystemSet 集合实例
---@param value any 要移除的数据
---@return any|false 成功返回被移除的值，失败返回false
function SystemSet.remove(self, value)
    if self._data[value] then
        self._data[value] = nil
        self._size = self._size - 1
        return value
    end
    return false
end

--- 获取集合元素数量
---@param self SystemSet 集合实例
---@return number 元素个数
function SystemSet.size(self)
    return self._size
end

--- 判断集合是否包含某个值
---@param self SystemSet 集合实例
---@param value any 要检查的值
---@return boolean true=包含，false=不包含
function SystemSet.has(self, value)
    return self._data[value]
end

--- 遍历集合所有元素
---@param self SystemSet 集合实例
---@param f fun 遍历回调函数，参数为元素值
---@param ... any 传递给回调的额外参数
function SystemSet.foreach(self, f, ...)
    for k, v in pairs(self._data) do
        if v then
            f(k, ...)
        end
    end
end

--- 根据位置获取元素（无序，不推荐频繁使用）
---@param self SystemSet 集合实例
---@param pos number 位置索引（从1开始）
---@return any 返回对应位置的元素，不存在则返回nil
function SystemSet.at(self, pos)
    local n = 0
    for k, _ in pairs(self._data) do
        n = n + 1
        if n == pos then
            return k
        end
    end
    return nil
end

--- 清空集合所有元素
---@param self SystemSet 集合实例
function SystemSet.clear(self)
    for k, _ in pairs(self._data) do
        self._data[k] = nil
    end
    self._size = 0
end
--endregion

return SystemSet
