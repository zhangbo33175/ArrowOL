---@class EventCmd 全局事件枚举（全局唯一）

    ----------------------------------------------------------------------------------------------------
    -- UI 通用事件
    ----------------------------------------------------------------------------------------------------
    ---@type string 游戏关卡主界面关闭
    UIGameViewClose                      = "UIGameViewClose",
    ---@type string 游戏关卡主界面打开
    UIGameViewOpen                       = "UIGameViewOpen",

    ----------------------------------------------------------------------------------------------------

    ----------------------------------------------------------------------------------------------------
    ---@type string 账号绑定结果回调，参数(isSuccess, platform, aid, position)
    OnAidBindingResult                   = "OnAidBindingResult",
    ---@type string 账号解绑结果回调，参数(isSuccess, platform, aid, position)

    ---@type string 玩家登录完成结果，参数(isSuccess)
    OnLoginResult                        = "OnLoginResult",

    ----------------------------------------------------------------------------------------------------

    NotificationMessage                  = "NotificationMessage",

    OnGenerateUserType                   = "OnGenerateUserType",

    ----------------------------------------------------------------------------------------------------

    ----------------------------------------------------------------------------------------------------
    ---@type string 关卡挑战失败，参数(chapterId, levelId)
    OnLevelFailure                       = "OnLevelFailure",
    ---@type string 关卡挑战成功，用于解锁系统、开启活动、结算奖励
    OnLevelSuccess                       = "OnLevelSuccess",

    OnLevelUnusualExit                   = "OnLevelUnusualExit",
    ---@type string 玩家首次进入主城镇主状态，触发新手引导、首登福利
    OnFirstIntoMainState                 = "OnFirstIntoMainState",
    ---@type string 章节地图切换动画、加载完成回调
    OnChapterMapChangeComplete           = "OnChapterMapChangeComplete",

    ----------------------------------------------------------------------------------------------------
    -- 地图 / 区域
    ----------------------------------------------------------------------------------------------------
    ---@type string 游戏运行时玩家所在地图区域发生切换
    OnRuntimeAreaChanged                 = "OnRuntimeAreaChanged",

    ----------------------------------------------------------------------------------------------------
    -- 时间 / 服务器同步校正
    ----------------------------------------------------------------------------------------------------

    OnTimeCorrectionByServer             = "OnTimeCorrectionByServer",

    ----------------------------------------------------------------------------------------------------
    -- 道具 / 技能 / 限时增益道具
    ----------------------------------------------------------------------------------------------------
    ---@type string 玩家主动使用道具，参数(boosterId, num)
    OnBoosterUseEvent                    = "OnBoosterUseEvent",
    ---@type string 时间冻结道具效果倒计时结束

    ---@type string 道具功能全局启用开关变更
    OnBoostersEnable                     = "OnBoostersEnable",
    ---@type string 单道具持续效果到期结束
    OnBoosterEnd                         = "OnBoosterEnd",
    ---@type string 弹出道具解锁介绍弹窗
    OnBoosterUnlockDescShowEvent         = "OnBoosterUnlockDescShowEvent",
    ---@type string 背包内道具数量发生增减变更
    OnBoosterNumModifyEvent              = "OnBoosterNumModifyEvent",

    ---@type string HUD显示动画播放完毕回调
    OnShowMainHudUIOver                  = "OnShowMainHudUIOver",
    ---@type string HUD隐藏动画播放完毕回调
    OnHideMainHudUIOver                  = "OnHideMainHudUIOver",

    ----------------------------------------------------------------------------------------------------

    ---@type string 请求弹出成就达成提示弹窗
    OnRequestShowAchievementTipsUI       = "OnRequestShowAchievementTipsUI",

    ChangeOnTheNumberOfRemainingSearchItems = "ChangeOnTheNumberOfRemainingSearchItems",

    ----------------------------------------------------------------------------------------------------
    -- Buff / 全局限时效果
    ----------------------------------------------------------------------------------------------------
    ---@type string Buff剩余时长数值修改
    OnBufferTimeModifyEvent              = "OnBufferTimeModifyEvent",
    ---@type string Buff持续时间完全到期失效
    OnBufferTimeEndEvent                 = "OnBufferTimeEndEvent",
    ---@type string 内部帧驱动Buff倒计时Tick，每秒执行一次
    OnBufferDeltaTimeTickEvent           = "OnBufferDeltaTimeTickEvent",
    ---@type string Buff倒计时定时回调，用于刷新UI倒计时文本
    OnBufferTimeTickEvent                = "OnBufferTimeTickEvent",

    -- BGM / 音效管理
    ----------------------------------------------------------------------------------------------------
    ---@type string 切换全局背景音乐资源，交由SoundManager执行播放
    OnPlayBGMKeyChanged                  = "OnPlayBGMKeyChanged",

    ----------------------------------------------------------------------------------------------------
    ---@type string 关卡内连击数值发生增减变化
    OnComboCountChange = "OnComboCountChange",

    ----------------------------------------------------------------------------------------------------

    ----------------------------------------------------------------------------------------------------
    ---@type string 主城镇/地图内角色实体被玩家点击
    OnClickCharacter = 'OnClickCharacter',

    ----------------------------------------------------------------------------------------------------

    OnNewChapterUnlockEvent = "OnNewChapterUnlockEvent",

    ----------------------------------------------------------------------------------------------------
    -- 关卡完整生命周期事件
    ----------------------------------------------------------------------------------------------------
    ---@type string 关卡资源、数据初始化完成事件
    InitGameEvent = "InitGameEvent",
    ---@type string 正式开启关卡游玩，开放触摸、操作交互
    BeginGameEvent = "BeginGameEvent",
    ---@type string 关卡结束（胜利/失败统一触发）
    OverGameEvent = "OverGameEvent",
    ---@type string 玩家主动退出当前关卡返回城镇
    ExitGameEvent = "ExitGameEvent",
    ---@type string 结算界面点击重玩，重置关卡所有数据重新开始
